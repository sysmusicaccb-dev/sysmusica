"use client";

/**
 * SysMúsica — Vínculos organizacionais de um usuário (painel lateral).
 * O usuário pode ser ligado a um estado, região, setor ou casa de oração: o acesso desce em cascata (um setor
 * inclui todas as casas de oração dele). Sem nenhum vínculo, ele não vê nenhum dado (membros, eventos,
 * presenças e portarias). Quem tem acesso total não precisa de vínculo. Fica na auditoria.
 */
import { useState, type FormEvent, type ReactNode } from "react";
import Button from "@/components/ui/Button";
import { chaveVinculo, FILHO_DE, filhosDe, lerChaveVinculo, ROTULO_NIVEL } from "@/modules/organizacao/regras";
import { definirVinculos } from "@/modules/organizacao/services/organizacao";
import type { NoOrganizacao, VinculoRotulado, VinculoUsuario } from "@/modules/organizacao/tipos";
import type { UsuarioSistema } from "../tipos";

interface FormularioVinculosProps {
  usuario: UsuarioSistema;
  arvore: NoOrganizacao[];
  vinculosIniciais: VinculoRotulado[];
  onSalvo: (mensagem: string) => void | Promise<void>;
}

export default function FormularioVinculos({ usuario, arvore, vinculosIniciais, onSalvo }: FormularioVinculosProps) {
  const [marcados, setMarcados] = useState<Set<string>>(
    () => new Set(vinculosIniciais.map((vinculo) => chaveVinculo(vinculo.nivel, vinculo.alvo_id)))
  );
  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  function alternar(chave: string) {
    setMarcados((anterior) => {
      const novos = new Set(anterior);
      if (novos.has(chave)) novos.delete(chave);
      else novos.add(chave);
      return novos;
    });
  }

  // Item com caixa de seleção e os filhos dele (inativos só aparecem se já estiverem marcados)
  function renderNo(no: NoOrganizacao, profundidade: number): ReactNode {
    const chave = chaveVinculo(no.nivel, no.id);
    if (no.ativo === false && !marcados.has(chave)) return null;
    const nivelFilho = FILHO_DE[no.nivel];
    const filhos = nivelFilho ? filhosDe(arvore, no.id, nivelFilho) : [];

    return (
      <li key={chave}>
        <label
          className="flex items-center gap-2 rounded py-1 text-sm text-slate-700 hover:bg-slate-50"
          style={{ paddingLeft: `${profundidade * 1.1}rem` }}
        >
          <input
            type="checkbox"
            checked={marcados.has(chave)}
            onChange={() => alternar(chave)}
            disabled={salvando}
            className="h-4 w-4 rounded border-slate-300 text-brand-600 focus:ring-brand-300"
          />
          <span className="text-xs uppercase tracking-wide text-slate-400">{ROTULO_NIVEL[no.nivel]}</span>
          <span className={no.ativo === false ? "text-slate-400" : "font-medium text-slate-900"}>
            {no.nome}
            {no.ativo === false ? " (inativo)" : ""}
          </span>
        </label>
        {filhos.length > 0 && <ul>{filhos.map((filho) => renderNo(filho, profundidade + 1))}</ul>}
      </li>
    );
  }

  async function salvar(submissao: FormEvent<HTMLFormElement>) {
    submissao.preventDefault();
    if (salvando) return; // evita envio duplicado
    setErro(null);
    setSalvando(true);
    try {
      const lista: VinculoUsuario[] = [...marcados].flatMap((chave) => {
        const vinculo = lerChaveVinculo(chave);
        return vinculo ? [vinculo] : [];
      });
      await definirVinculos(usuario.id, lista);
      await onSalvo(`Vínculos de "${usuario.nome}" atualizados.`);
    } catch (e) {
      setErro(e instanceof Error ? e.message : "Não foi possível salvar os vínculos.");
    } finally {
      setSalvando(false);
    }
  }

  const estados = filhosDe(arvore, null, "ESTADO");
  const semSetor = filhosDe(arvore, null, "CASA_ORACAO");

  return (
    <form onSubmit={salvar} className="space-y-5">
      <p className="rounded-lg bg-slate-50 px-3 py-2 text-sm text-slate-600">
        <strong className="text-slate-900">{usuario.nome}</strong> só vê e usa os dados das áreas marcadas.
        Marcar um setor inclui todas as casas de oração dele; marcar uma região inclui todos os setores. Sem nenhuma
        marcação, não vê nada.
      </p>

      {arvore.length === 0 ? (
        <p className="text-sm text-slate-500">
          A organização ainda não foi carregada ou está vazia. Cadastre em Administração &gt; Organização.
        </p>
      ) : (
        <div className="max-h-96 overflow-y-auto rounded-lg border border-slate-200 p-3">
          <ul>{estados.map((estado) => renderNo(estado, 0))}</ul>
          {semSetor.length > 0 && (
            <div className="mt-3 border-t border-slate-100 pt-2">
              <p className="mb-1 text-xs font-semibold text-slate-500">Casas de Oração sem setor</p>
              <ul>{semSetor.map((casaOracao) => renderNo(casaOracao, 0))}</ul>
            </div>
          )}
        </div>
      )}

      {erro && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erro}
        </p>
      )}

      <Button type="submit" disabled={salvando} className="w-full">
        {salvando ? "Salvando..." : "Salvar vínculos"}
      </Button>
    </form>
  );
}

################ src/modules/administracao/components/FormularioAbrangencia.tsx
"use client";

/**
 * SysMúsica — Abrangência de atuação de um usuário (painel lateral).
 * REGRA PROVISÓRIA (aguarda definição administrativa): aprovar portarias e registrar presenças só
 * vale em portarias das cidades marcadas aqui. Sem nenhuma cidade, o usuário não atua em nenhuma.
 */
import { useState, type FormEvent } from "react";
import Button from "@/components/ui/Button";
import { valoresDistintos } from "@/lib/listas";
import { normalizarTexto } from "@/lib/texto";
import { definirAbrangencia } from "../services/acesso";
import type { UsuarioSistema } from "../tipos";

interface FormularioAbrangenciaProps {
  usuario: UsuarioSistema;
  cidadesConhecidas: string[];
  selecionadasIniciais: string[];
  onSalvo: (mensagem: string) => void | Promise<void>;
}

export default function FormularioAbrangencia({
  usuario,
  cidadesConhecidas,
  selecionadasIniciais,
  onSalvo,
}: FormularioAbrangenciaProps) {
  // Cidades exibidas (conhecidas + já marcadas) e as marcadas (pela forma normalizada)
  const [lista, setLista] = useState<string[]>(() => valoresDistintos([...cidadesConhecidas, ...selecionadasIniciais]));
  const [marcadas, setMarcadas] = useState<Set<string>>(
    () => new Set(selecionadasIniciais.map((cidade) => normalizarTexto(cidade)))
  );
  const [novaCidade, setNovaCidade] = useState("");
  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  function alternar(cidade: string) {
    const chave = normalizarTexto(cidade);
    setMarcadas((anterior) => {
      const novas = new Set(anterior);
      if (novas.has(chave)) novas.delete(chave);
      else novas.add(chave);
      return novas;
    });
  }

  // Inclui uma cidade nova na lista, já marcada
  function adicionar() {
    const limpa = novaCidade.trim().replace(/\s+/g, " ");
    if (!limpa) return;
    setLista((anterior) => valoresDistintos([...anterior, limpa]));
    setMarcadas((anterior) => new Set(anterior).add(normalizarTexto(limpa)));
    setNovaCidade("");
  }

  async function salvar(submissao: FormEvent<HTMLFormElement>) {
    submissao.preventDefault();
    if (salvando) return; // evita envio duplicado
    setErro(null);
    setSalvando(true);
    try {
      await definirAbrangencia(usuario.id, lista.filter((cidade) => marcadas.has(normalizarTexto(cidade))));
      await onSalvo(`Abrangência de "${usuario.nome}" atualizada.`);
    } catch (e) {
      setErro(e instanceof Error ? e.message : "Não foi possível salvar a abrangência.");
    } finally {
      setSalvando(false);
    }
  }

  return (
    <form onSubmit={salvar} className="space-y-5">
      <p className="rounded-lg bg-slate-50 px-3 py-2 text-sm text-slate-600">
        <strong className="text-slate-900">{usuario.nome}</strong> só aprova portarias e registra presenças em
        portarias das cidades marcadas. Sem nenhuma cidade, não atua em nenhuma.
      </p>

      {lista.length === 0 ? (
        <p className="text-sm text-slate-500">
          Ainda não há cidades cadastradas. Informe a cidade abaixo (ela também aparece nos eventos).
        </p>
      ) : (
        <fieldset className="space-y-2">
          <legend className="text-sm font-medium text-slate-700">Cidades</legend>
          {lista.map((cidade) => (
            <label key={cidade} className="flex items-center gap-2 text-sm text-slate-700">
              <input
                type="checkbox"
                checked={marcadas.has(normalizarTexto(cidade))}
                onChange={() => alternar(cidade)}
                disabled={salvando}
                className="h-4 w-4 rounded border-slate-300 text-brand-600 focus:ring-brand-300"
              />
              {cidade}
            </label>
          ))}
        </fieldset>
      )}

      <div className="space-y-1.5">
        <label htmlFor="nova-cidade" className="text-sm font-medium text-slate-700">
          Adicionar cidade
        </label>
        <div className="flex gap-2">
          <input
            id="nova-cidade"
            type="text"
            maxLength={80}
            value={novaCidade}
            onChange={(e) => setNovaCidade(e.target.value)}
            onKeyDown={(e) => {
              // Enter adiciona a cidade (e não envia o formulário)
              if (e.key === "Enter") {
                e.preventDefault();
                adicionar();
              }
            }}
            disabled={salvando}
            placeholder="Ex: Lauro de Freitas"
            className="w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 shadow-sm placeholder:text-slate-400 focus:border-brand-500 focus:outline-none focus:ring-2 focus:ring-brand-200 disabled:opacity-60"
          />
          <Button type="button" variante="secundario" onClick={adicionar} disabled={salvando || novaCidade.trim() === ""}>
            Adicionar
          </Button>
        </div>
        <p className="text-xs text-slate-500">
          Escreva a cidade como nos eventos (acentos e maiúsculas não importam).
        </p>
      </div>

      {erro && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erro}
        </p>
      )}

      <Button type="submit" disabled={salvando} className="w-full">
        {salvando ? "Salvando..." : "Salvar abrangência"}
      </Button>
    </form>
  );
}

import { describe, expect, it } from "vitest";
import {
  agruparCasasOracaoParaSelecao,
  agruparSetoresParaSelecao,
  chaveVinculo,
  filhosDe,
  lerChaveVinculo,
  motivoBloqueioExclusao,
  rotulosPorUsuario,

} from "@/modules/organizacao/regras";
import type { NoOrganizacao } from "@/modules/organizacao/tipos";

// Bahia > Salvador (Setor 1, Setor 2, Setor 10) e Jequié (Setor 1, Setor 9 inativo); duas casas de oração em
// Salvador/Setor 1, uma em Jequié/Setor 1 e uma sem setor
const nos: NoOrganizacao[] = [
  { nivel: "ESTADO", id: "e1", parent_id: null, nome: "Bahia", ativo: true },
  { nivel: "REGIAO", id: "r1", parent_id: "e1", nome: "Salvador", ativo: true },
  { nivel: "REGIAO", id: "r2", parent_id: "e1", nome: "Jequié", ativo: true },
  { nivel: "SETOR", id: "s1", parent_id: "r1", nome: "Setor 1", ativo: true },
  { nivel: "SETOR", id: "s2", parent_id: "r1", nome: "Setor 10", ativo: true },
  { nivel: "SETOR", id: "s3", parent_id: "r1", nome: "Setor 2", ativo: true },
  { nivel: "SETOR", id: "s4", parent_id: "r2", nome: "Setor 1", ativo: true },
  { nivel: "SETOR", id: "s5", parent_id: "r2", nome: "Setor 9", ativo: false },
  { nivel: "CASA_ORACAO", id: "c1", parent_id: "s1", nome: "Itinga" },
  { nivel: "CASA_ORACAO", id: "c2", parent_id: "s1", nome: "Caique" },
  { nivel: "CASA_ORACAO", id: "c3", parent_id: "s4", nome: "Central" },
  { nivel: "CASA_ORACAO", id: "c4", parent_id: null, nome: "Solta" },
];

describe("filhosDe", () => {
  it("traz só os itens do pai informado, em ordem natural", () => {
    expect(filhosDe(nos, "r1", "SETOR").map((n) => n.nome)).toEqual(["Setor 1", "Setor 2", "Setor 10"]);
  });

  it("separa 'Setor 1' de regiões diferentes", () => {
    expect(filhosDe(nos, "r2", "SETOR").map((n) => n.id)).toEqual(["s4", "s5"]);
    expect(filhosDe(nos, "r1", "SETOR").map((n) => n.id)).not.toContain("s4");
  });

  it("devolve as casas de oração sem setor quando o pai é nulo", () => {
    expect(filhosDe(nos, null, "CASA_ORACAO").map((n) => n.id)).toEqual(["c4"]);
  });
});

describe("agruparCasasOracaoParaSelecao", () => {
  const grupos = agruparCasasOracaoParaSelecao(nos);

  it("agrupa por Região › Setor e deixa 'sem setor' por último", () => {
    expect(grupos.map((g) => g.rotulo)).toEqual([
      "Jequié › Setor 1",
      "Salvador › Setor 1",
      "Sem setor definido",
    ]);
  });

  it("ordena as casas de oração de cada grupo pelo nome", () => {
    expect(grupos[1].casasOracao.map((c) => c.nome)).toEqual(["Caique", "Itinga"]);
  });
});

describe("agruparSetoresParaSelecao", () => {
  it("agrupa os setores ativos por região e ignora os inativos", () => {
    const grupos = agruparSetoresParaSelecao(nos);
    expect(grupos.map((g) => g.rotulo)).toEqual(["Jequié", "Salvador"]);
    expect(grupos[0].setores.map((s) => s.id)).toEqual(["s4"]);
  });
});

describe("vínculos", () => {
  it("monta e desfaz a chave de um vínculo", () => {
    const chave = chaveVinculo("SETOR", "abc-123");
    expect(chave).toBe("SETOR:abc-123");
    expect(lerChaveVinculo(chave)).toEqual({ nivel: "SETOR", alvo_id: "abc-123" });
  });

  it("recusa chaves malformadas", () => {
    expect(lerChaveVinculo("sem-separador")).toBeNull();
    expect(lerChaveVinculo("OUTRO:abc")).toBeNull();
    expect(lerChaveVinculo("SETOR:")).toBeNull();
  });

  it("lista só os rótulos por usuário", () => {
    expect(
      rotulosPorUsuario({ u1: [{ nivel: "SETOR", alvo_id: "s1", rotulo: "Salvador › Setor 1" }], u2: [] })
    ).toEqual({ u1: ["Salvador › Setor 1"], u2: [] });
  });
});

describe("motivoBloqueioExclusao", () => {
  const base = { parent_id: null, ativo: true };

  it("bloqueia o item que ainda tem itens dentro e explica o motivo", () => {
    expect(motivoBloqueioExclusao({ ...base, nivel: "ESTADO", id: "e", nome: "Bahia", qtd_filhos: 2 })).toMatch(/2 regiões/);
    expect(motivoBloqueioExclusao({ ...base, nivel: "REGIAO", id: "r", nome: "Salvador", qtd_filhos: 1 })).toMatch(/1 setor\b/);
    expect(motivoBloqueioExclusao({ ...base, nivel: "SETOR", id: "s", nome: "Setor 1", qtd_filhos: 3 })).toMatch(/3 casas de oração/);
    expect(motivoBloqueioExclusao({ ...base, nivel: "CASA_ORACAO", id: "c", nome: "Itinga", qtd_membros: 4 })).toMatch(/4 membros/);
  });

  it("libera o item vazio", () => {
    expect(motivoBloqueioExclusao({ ...base, nivel: "SETOR", id: "s", nome: "Setor 9", qtd_filhos: 0 })).toBeNull();
    expect(motivoBloqueioExclusao({ ...base, nivel: "CASA_ORACAO", id: "c", nome: "Vazia", qtd_membros: 0 })).toBeNull();
  });
});

describe("validarNomeNo", () => {
  it("exige de 2 a 80 caracteres", () => {
    expect(validarNomeNo("A")).toMatch(/entre 2 e 80/);
    expect(validarNomeNo("  Setor 1  ")).toBeNull();
    expect(validarNomeNo("x".repeat(81))).toMatch(/entre 2 e 80/);
  });
});
