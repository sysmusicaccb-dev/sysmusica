"use client";

/**
 * SysMúsica — Página /eventos: listagem em tabela (largura total), do evento mais recente para o
 * mais antigo, com novo, editar e excluir. Cada botão só aparece para quem tem a permissão
 * (grupo do usuário); o banco confere de novo em toda operação.
 * Excluir é exclusão lógica: o evento some das telas e as presenças registradas são mantidas.
 */
import { useCallback, useEffect, useState } from "react";
import { Plus } from "lucide-react";
import {
  buscarEventosEFinalidades,
  excluirEvento,
} from "@/modules/eventos/services/eventos";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";
import Button from "@/components/ui/Button";
import ConfirmDialog from "@/components/ui/ConfirmDialog";
import Drawer from "@/components/ui/Drawer";
import Notificacao from "@/components/ui/Notificacao";
import PageHeader from "@/components/ui/PageHeader";
import FormularioEvento from "@/components/FormularioEvento";
import ListaEventos from "@/components/ListaEventos";
import AbasDeLista from "@/components/AbasDeLista";
import FormularioCancelamento from "@/modules/eventos/components/FormularioCancelamento";
import { abaDoEvento, type AbaEvento } from "@/modules/eventos/ciclo";
import { formatarDataHora } from "@/lib/datas";
import { valoresDistintos } from "@/lib/listas";
import type { Evento, FinalidadeEvento } from "@/types/database";

// Qual painel lateral está aberto: novo evento, ou edição de um evento
type PainelEvento = { tipo: "novo" } | { tipo: "editar"; evento: Evento } | { tipo: "cancelar"; evento: Evento };

export default function EventosPage() {
  const { pode } = useSessao();
  const [eventos, setEventos] = useState<Evento[]>([]);
  const [finalidades, setFinalidades] = useState<FinalidadeEvento[]>([]);
  const [carregando, setCarregando] = useState(true); // inicia true: primeira busca no mount
  const [erro, setErro] = useState<string | null>(null);
  const [painel, setPainel] = useState<PainelEvento | null>(null);
  const [alvoExclusao, setAlvoExclusao] = useState<Evento | null>(null);
  const [excluindo, setExcluindo] = useState(false);
  const [erroExclusao, setErroExclusao] = useState<string | null>(null);
  const [aviso, setAviso] = useState<string | null>(null);
  const [aba, setAba] = useState<AbaEvento>("proximos");
  const [agora, setAgora] = useState(0); // instante atual (ms); 0 até a primeira carga

  // Busca eventos e finalidades
  const carregar = useCallback(async () => {
    try {
      const dados = await buscarEventosEFinalidades();
      setEventos(dados.eventos);
      setAgora(Date.now());
      setFinalidades(dados.finalidades);
      setErro(null);
    } catch (e) {
      // O detalhe técnico vai só para o console; a tela mostra uma mensagem simples
      console.error("Falha ao carregar eventos:", e);
      setErro("Não foi possível carregar os eventos. Verifique a conexão e tente novamente.");
    } finally {
      setCarregando(false);
    }
  }, []);

  // Primeira carga ao abrir a página
  useEffect(() => {
    // O setState só roda DEPOIS do await (nunca de forma síncrona), então o aviso da regra
    // set-state-in-effect não se aplica a esta carga inicial. Ignorado apenas nesta linha.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    void carregar();
  }, [carregar]);

  // Mantém a situação dos eventos em dia: um evento que termina sai de "Próximos" sozinho
  useEffect(() => {
    const id = setInterval(() => setAgora(Date.now()), 60_000);
    return () => clearInterval(id);
  }, []);

  // Botão "Tentar novamente": volta ao estado de loading e busca de novo
  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  // Depois de salvar: recarrega a lista, fecha o painel e mostra o aviso de sucesso
  async function aoSalvar(mensagem: string) {
    await carregar();
    setPainel(null);
    setAviso(mensagem);
  }

  // Confirma a exclusão (lógica) do evento escolhido
  async function confirmarExclusao() {
    if (!alvoExclusao) return;
    setExcluindo(true);
    setErroExclusao(null);
    try {
      await excluirEvento(alvoExclusao.id);
      await carregar();
      setAviso("Evento excluído.");
      setAlvoExclusao(null);
    } catch (e) {
      setErroExclusao(e instanceof Error ? e.message : "Não foi possível excluir o evento.");
    } finally {
      setExcluindo(false);
    }
  }

  // Cada evento aparece em UMA aba. Próximos: do mais perto para o mais longe; as demais: do mais recente.
  const eventosDaAba = (alvo: AbaEvento) =>
    eventos.filter((evento) => abaDoEvento(evento.data_evento, Boolean(evento.cancelado_em), agora) === alvo);
  const proximos = eventosDaAba("proximos").reverse(); // a busca vem do mais recente para o mais antigo
  const encerrados = eventosDaAba("encerrados");
  const arquivo = eventosDaAba("arquivo");
  const visiveis = aba === "proximos" ? proximos : aba === "encerrados" ? encerrados : arquivo;
  const textosVazios: Record<AbaEvento, { titulo: string; descricao: string }> = {
    proximos: {
      titulo: "Nenhum evento em andamento ou agendado.",
      descricao: "Clique em “Adicionar” para cadastrar o próximo. Os eventos já encerrados estão na aba Encerrados.",
    },
    encerrados: {
      titulo: "Nenhum evento encerrado recentemente.",
      descricao: "Quando um evento terminar (ou for cancelado), ele passa a aparecer aqui.",
    },
    arquivo: {
      titulo: "O arquivo está vazio.",
      descricao: "Eventos encerrados há mais de 30 dias ficam guardados aqui para consulta.",
    },
  };

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Eventos"
          descricao={`Cadastre e acompanhe os ensaios, cultos e reuniões do SysMúsica.${
            !carregando && !erro ? ` ${eventos.length} cadastrado(s).` : ""
          }`}
          acoes={
            pode("eventos.criar") ? (
              <Button icone={Plus} onClick={() => setPainel({ tipo: "novo" })} disabled={carregando || !!erro}>
                Adicionar
              </Button>
            ) : undefined
          }
        />

        {!carregando && !erro && (
          <AbasDeLista
            descricao="Situação dos eventos"
            atual={aba}
            onTrocar={setAba}
            abas={[
              { id: "proximos", rotulo: "Próximos", total: proximos.length },
              { id: "encerrados", rotulo: "Encerrados", total: encerrados.length },
              { id: "arquivo", rotulo: "Arquivo", total: arquivo.length },
            ]}
          />
        )}

        <ListaEventos
          eventos={eventos.length === 0 ? eventos : visiveis}
          agora={agora}
          tituloVazio={eventos.length === 0 ? undefined : textosVazios[aba].titulo}
          descricaoVazia={eventos.length === 0 ? undefined : textosVazios[aba].descricao}
          finalidades={finalidades}
          carregando={carregando}
          erro={erro}
          onTentarNovamente={tentarNovamente}
          podeEditar={pode("eventos.editar")}
          podeExcluir={pode("eventos.excluir")}
          podeCancelar={pode("eventos.cancelar")}
          onCancelar={(evento) => setPainel({ tipo: "cancelar", evento })}
          onEditar={(evento) => setPainel({ tipo: "editar", evento })}
          onExcluir={(evento) => {
            setErroExclusao(null);
            setAlvoExclusao(evento);
          }}
        />
      </div>

      {/* Painel lateral: novo evento ou edição */}
      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={
          painel?.tipo === "editar" ? "Editar evento" : painel?.tipo === "cancelar" ? "Cancelar evento" : "Novo evento"
        }
        descricao={
          painel?.tipo === "cancelar"
            ? "Informe o motivo: ele fica guardado no histórico."
            : "Preencha os dados do evento."
        }
      >
        {painel?.tipo === "cancelar" && <FormularioCancelamento evento={painel.evento} onSalvo={aoSalvar} />}
        {painel && painel.tipo !== "cancelar" && (
          <FormularioEvento
            evento={painel.tipo === "editar" ? painel.evento : null}
            sugestoesCidade={valoresDistintos(eventos.flatMap((item) => (item.cidade ? [item.cidade] : [])))}
            finalidades={finalidades}
            onSalvo={aoSalvar}
          />
        )}
      </Drawer>

      {/* Confirmação de exclusão */}
      <ConfirmDialog
        aberto={alvoExclusao !== null}
        titulo="Excluir evento?"
        mensagem={
          <>
            <p>
              O evento de{" "}
              <strong>{alvoExclusao ? formatarDataHora(alvoExclusao.data_evento) : ""}</strong> em{" "}
              <strong>{alvoExclusao?.local}</strong> deixará de aparecer nas telas. Eventos que já têm
              presenças registradas não podem ser excluídos: nesse caso o Administrador deve cancelá-lo (o histórico
              fica guardado).
            </p>
            {erroExclusao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroExclusao}
              </p>
            )}
          </>
        }
        textoConfirmar="Excluir"
        processando={excluindo}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExclusao(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

/**
 * SysMúsica — Abas da tela de Eventos (regra pura): Eventos | Gestão de portarias | Portaria.
 * Cada aba é uma tela que já existe (mesma rota e mesma permissão); aqui só se decide QUAIS abas a pessoa vê.
 * As abas existem só para quem tem a permissão de Eventos e acesso a pelo menos mais uma tela: quem não tem
 * Eventos (ex.: grupo Portaria) continua com os itens próprios no menu e não vê barra de abas.
 */
import type { CodigoPermissao } from "@/lib/permissoes";

export interface AbaEvento {
  href: string;
  rotulo: string;
}

export function abasDeEventos(pode: (permissao: CodigoPermissao) => boolean): AbaEvento[] {
  if (!pode("eventos.visualizar")) return [];

  const abas: AbaEvento[] = [{ href: "/eventos", rotulo: "Eventos" }];
  // Mesma regra do layout de /portarias: visualiza, cria ou aprova
  if (pode("portarias.visualizar") || pode("portarias.criar") || pode("portarias.aprovar")) {
    abas.push({ href: "/portarias", rotulo: "Gestão de portarias" });
  }
  if (pode("portaria.visualizar")) abas.push({ href: "/portaria", rotulo: "Portaria" });

  // Uma aba só não precisa de barra
  return abas.length > 1 ? abas : [];
}

/** O usuário pode ver o item? (a permissão principal ou qualquer uma das alternativas) */
function itemPermitido(item: ItemMenu, pode: (permissao: CodigoPermissao) => boolean): boolean {
  // Itens que viraram aba de outro item somem do menu de quem já tem esse outro item
  if (item.ocultoSe && pode(item.ocultoSe)) return false;
  return pode(item.permissao) || (item.qualquer ?? []).some((permissao) => pode(permissao));
}

/** Filtra o menu pelas permissões do usuário, removendo também os grupos que ficarem vazios. */
export function filtrarMenu(
  grupos: GrupoMenu[],
  pode: (permissao: CodigoPermissao) => boolean
): GrupoMenu[] {
  return grupos
    .map((grupo) => ({ ...grupo, itens: grupo.itens.filter((item) => itemPermitido(item, pode)) }))
    .filter((grupo) => grupo.itens.length > 0);
}

/** Primeira tela (fora do dashboard) que o usuário pode acessar; null se não puder nenhuma. */
export function primeiraRotaPermitida(pode: (permissao: CodigoPermissao) => boolean): string | null {
  for (const grupo of GRUPOS_MENU) {
    for (const item of grupo.itens) {
      if (item.href !== "/" && itemPermitido(item, pode)) return item.href;
    }
  }
  return null;
}

/** O item está ativo na rota atual? A raiz "/" só é ativa exatamente em "/". */
export function itemAtivo(href: string, caminho: string): boolean {
  if (href === "/") return caminho === "/";
  return caminho === href || caminho.startsWith(`${href}/`);
}

/** O item do menu está ativo? Vale a rota dele e as rotas das suas abas (ex.: Eventos > Portaria). */
export function itemMenuAtivo(item: ItemMenu, caminho: string): boolean {
  return itemAtivo(item.href, caminho) || (item.tambemAtivoEm ?? []).some((rota) => itemAtivo(rota, caminho));
}

export interface Migalha {
  rotulo: string;
  href?: string; // sem href = apenas texto (grupo ou página atual)
}

/** Monta o caminho exibido nos breadcrumbs: Início > Grupo > Página. */
export function montarMigalhas(caminho: string): Migalha[] {
  for (const grupo of GRUPOS_MENU) {
    for (const item of grupo.itens) {
      if (!itemAtivo(item.href, caminho)) continue;

      // Na raiz, o próprio Dashboard é o único nível
      if (item.href === "/") return [{ rotulo: item.rotulo }];

      const migalhas: Migalha[] = [{ rotulo: "Início", href: "/" }];
      if (grupo.titulo) migalhas.push({ rotulo: grupo.titulo });
      migalhas.push({ rotulo: item.rotulo });
      return migalhas;
    }
  }
  // Rota fora do menu: mostra só o caminho de volta ao início
  return [{ rotulo: "Início", href: "/" }];
}

/**
 * Remove dos breadcrumbs o link "Início" para quem não acessa o dashboard (ex: grupo Portaria):
 * para essa pessoa o menu mostra somente o que o grupo permite, e nada além disso.
 */
export function migalhasVisiveis(
  migalhas: Migalha[],
  pode: (permissao: CodigoPermissao) => boolean
): Migalha[] {
  return migalhas.filter((migalha) => migalha.href !== "/" || pode("dashboard.visualizar"));
}

-- C) Cancelar evento: as portarias pendentes ou aprovadas dele deixam de valer (ficam como rejeitadas, no histórico)
create or replace function public.cancelar_evento(p_id uuid, p_motivo text)
returns void
language plpgsql
security definer
set search_path to 'public'
as $function$
declare
  v_uid    uuid := (select auth.uid());
  v_motivo text := nullif(btrim(coalesce(p_motivo, '')), '');
  v_e      public.eventos;
  r        record;
begin
  perform public.exigir_permissao('eventos.cancelar');
  if v_motivo is null or char_length(v_motivo) < 5 then
    raise exception 'Informe o motivo do cancelamento (mínimo de 5 caracteres).';
  end if;
  if char_length(v_motivo) > 250 then
    raise exception 'O motivo pode ter no máximo 250 caracteres.';
  end if;

  select * into v_e from public.eventos e where e.id = p_id and e.excluido_em is null for update;
  if not found then
    raise exception 'Evento não encontrado.';
  end if;
  if v_e.cancelado_em is not null then
    raise exception 'Este evento já está cancelado.';
  end if;

  update public.eventos
     set cancelado_em = now(), cancelado_por = v_uid, motivo_cancelamento = v_motivo
   where id = p_id;

  for r in select p.id, p.status from public.portarias p
            where p.evento_id = p_id and p.excluido_em is null and p.status in ('PENDENTE_APROVACAO', 'APROVADA') loop
    update public.portarias
       set status = 'REJEITADA', decidida_por = v_uid, decidida_em = now(),
           motivo_rejeicao = 'Evento cancelado: ' || v_motivo, decisao_controle_total = true
     where id = r.id;
    insert into public.portarias_historico (portaria_id, acao, status_anterior, status_novo, usuario_id, motivo)
    values (r.id, 'REJEITADA', r.status, 'REJEITADA', v_uid, 'Evento cancelado: ' || v_motivo);
  end loop;

  perform public.registrar_auditoria('evento_cancelado', 'evento', p_id, jsonb_build_object('motivo', v_motivo));
end;
$function$;

-- D) Excluir: só evento SEM presenças (com presenças, o Administrador cancela)
CREATE OR REPLACE FUNCTION public.excluir_evento(p_id uuid)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  perform public.exigir_permissao('eventos.excluir');

  -- Evento com presenças é parte do histórico: não se exclui, só o Administrador cancela
  if exists (select 1 from public.presencas pr where pr.evento_id = p_id and pr.excluido_em is null)
     or exists (select 1 from public.presencas_avulsas a where a.evento_id = p_id and a.excluido_em is null) then
    raise exception 'Este evento já tem presenças registradas e não pode ser excluído. O Administrador pode cancelá-lo (o histórico fica guardado).';
  end if;

  update public.eventos set excluido_em = now() where id = p_id and excluido_em is null;
  if not found then
    raise exception 'Evento não encontrado.';
  end if;
end;
$function$;

-- E) Portaria: evento cancelado ou já encerrado não recebe portaria nova
CREATE OR REPLACE FUNCTION public.criar_portaria(p_evento_id uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_uid    uuid := (select auth.uid());
  v_dia    date := (now() at time zone 'America/Sao_Paulo')::date;
  v_evento public.eventos;
  v_usadas int;
  v_total  boolean;
  v_setor  uuid;
  v_id     uuid;
begin
  perform public.exigir_permissao('portarias.criar');
  v_total := public.tem_permissao('portarias.controle_total');

  -- Serializa as criações do MESMO usuário no MESMO dia (a 2ª requisição espera e recota)
  perform pg_advisory_xact_lock(hashtextextended('portarias:' || v_uid::text || ':' || v_dia::text, 0));

  select count(*)::int into v_usadas
  from public.portarias p
  where p.criada_por = v_uid and p.dia_criacao = v_dia;   -- inclui rejeitadas e excluídas

  if not v_total and v_usadas >= public.limite_diario_portarias() then
    raise exception 'Você atingiu o limite máximo de % portarias criadas hoje. Novas criações estarão disponíveis no próximo dia.',
      public.limite_diario_portarias();
  end if;

  select * into v_evento from public.eventos e where e.id = p_evento_id and e.excluido_em is null;
  if not found then
    raise exception 'Evento não encontrado.';
  end if;
  if v_evento.cancelado_em is not null then
    raise exception 'Este evento foi cancelado e não pode receber portaria.';
  end if;
  if public.portaria_encerrada(v_evento.data_evento) then
    raise exception 'Este evento já foi encerrado e não pode receber uma nova portaria.';
  end if;
  if v_evento.casa_oracao_id is null then
    raise exception 'Escolha a casa de oração do evento (Eventos > Editar) antes de criar a portaria.';
  end if;

  select c.setor_id into v_setor from public.casas_oracao c where c.id = v_evento.casa_oracao_id;
  if v_setor is null then
    raise exception 'A casa de oração deste evento ainda não tem setor. Peça a um administrador para vinculá-la a um setor (Administração > Organização).';
  end if;
  if not public.casa_oracao_na_abrangencia(v_evento.casa_oracao_id) then
    raise exception 'Este evento está fora da sua abrangência (casa de oração de outro setor).' using errcode = '42501';
  end if;

  if exists (select 1 from public.portarias p
             where p.evento_id = p_evento_id and p.excluido_em is null
               and p.status in ('PENDENTE_APROVACAO', 'APROVADA')) then
    raise exception 'Já existe uma portaria pendente ou aprovada para este evento.';
  end if;

  insert into public.portarias (evento_id, casa_oracao_id, cidade, criada_por, dia_criacao)
  values (p_evento_id, v_evento.casa_oracao_id, public.rotulo_setor(v_evento.casa_oracao_id), v_uid, v_dia)
  returning id into v_id;

  insert into public.portarias_historico (portaria_id, acao, status_novo, usuario_id)
  values (v_id, 'CRIADA', 'PENDENTE_APROVACAO', v_uid);

  perform public.registrar_auditoria('portaria_criada', 'portaria', v_id,
            jsonb_build_object('evento_id', p_evento_id,
                               'acima_do_limite', v_usadas >= public.limite_diario_portarias()));
  return v_id;
end;
$function$;

CREATE OR REPLACE FUNCTION public.listar_eventos_para_portaria()
 RETURNS TABLE(id uuid, tipo text, data_evento timestamp with time zone, local text, cidade text, casa_oracao_id uuid)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  perform public.exigir_permissao('portarias.criar');
  return query
    select e.id, e.tipo, e.data_evento, e.local,
           case when c.setor_id is not null then public.rotulo_setor(e.casa_oracao_id) end,
           e.casa_oracao_id
    from public.eventos e
    join public.casas_oracao c on c.id = e.casa_oracao_id

      and not public.portaria_encerrada(e.data_evento)
      and public.casa_oracao_na_abrangencia(e.casa_oracao_id)
    order by e.data_evento desc
    limit 300;
end;
$function$;
