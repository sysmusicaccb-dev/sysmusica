################ src/app/(painel)/eventos/page.tsx
"use client";

/**
 * SysMúsica — Página /eventos: listagem em tabela (largura total), do evento mais recente para o
 * mais antigo, com novo, editar e excluir. Cada botão só aparece para quem tem a permissão
 * (grupo do usuário); o banco confere de novo em toda operação.
 * Excluir é exclusão lógica: o evento some das telas e as presenças registradas são mantidas.
 */
import { useCallback, useEffect, useState } from "react";
import { Plus } from "lucide-react";
import {
  buscarEventosEFinalidades,
  excluirEvento,
} from "@/modules/eventos/services/eventos";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";
import Button from "@/components/ui/Button";
import ConfirmDialog from "@/components/ui/ConfirmDialog";
import Drawer from "@/components/ui/Drawer";
import Notificacao from "@/components/ui/Notificacao";
import PageHeader from "@/components/ui/PageHeader";
import FormularioEvento from "@/components/FormularioEvento";
import ListaEventos from "@/components/ListaEventos";
import { formatarDataHora } from "@/lib/datas";
import { valoresDistintos } from "@/lib/listas";
import type { Evento, FinalidadeEvento } from "@/types/database";

// Qual painel lateral está aberto: novo evento, ou edição de um evento
type PainelEvento = { tipo: "novo" } | { tipo: "editar"; evento: Evento };

export default function EventosPage() {
  const { pode } = useSessao();
  const [eventos, setEventos] = useState<Evento[]>([]);
  const [finalidades, setFinalidades] = useState<FinalidadeEvento[]>([]);
  const [carregando, setCarregando] = useState(true); // inicia true: primeira busca no mount
  const [erro, setErro] = useState<string | null>(null);
  const [painel, setPainel] = useState<PainelEvento | null>(null);
  const [alvoExclusao, setAlvoExclusao] = useState<Evento | null>(null);
  const [excluindo, setExcluindo] = useState(false);
  const [erroExclusao, setErroExclusao] = useState<string | null>(null);
  const [aviso, setAviso] = useState<string | null>(null);

  // Busca eventos e finalidades
  const carregar = useCallback(async () => {
    try {
      const dados = await buscarEventosEFinalidades();
      setEventos(dados.eventos);
      setFinalidades(dados.finalidades);
      setErro(null);
    } catch (e) {
      // O detalhe técnico vai só para o console; a tela mostra uma mensagem simples
      console.error("Falha ao carregar eventos:", e);
      setErro("Não foi possível carregar os eventos. Verifique a conexão e tente novamente.");
    } finally {
      setCarregando(false);
    }
  }, []);

  // Primeira carga ao abrir a página
  useEffect(() => {
    // O setState só roda DEPOIS do await (nunca de forma síncrona), então o aviso da regra
    // set-state-in-effect não se aplica a esta carga inicial. Ignorado apenas nesta linha.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    void carregar();
  }, [carregar]);

  // Botão "Tentar novamente": volta ao estado de loading e busca de novo
  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  // Depois de salvar: recarrega a lista, fecha o painel e mostra o aviso de sucesso
  async function aoSalvar(mensagem: string) {
    await carregar();
    setPainel(null);
    setAviso(mensagem);
  }

  // Confirma a exclusão (lógica) do evento escolhido
  async function confirmarExclusao() {
    if (!alvoExclusao) return;
    setExcluindo(true);
    setErroExclusao(null);
    try {
      await excluirEvento(alvoExclusao.id);
      await carregar();
      setAviso("Evento excluído.");
      setAlvoExclusao(null);
    } catch (e) {
      setErroExclusao(e instanceof Error ? e.message : "Não foi possível excluir o evento.");
    } finally {
      setExcluindo(false);
    }
  }

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Eventos"
          descricao={`Cadastre e acompanhe os ensaios, cultos e reuniões do SysMúsica.${
            !carregando && !erro ? ` ${eventos.length} cadastrado(s).` : ""
          }`}
          acoes={
            pode("eventos.criar") ? (
              <Button icone={Plus} onClick={() => setPainel({ tipo: "novo" })} disabled={carregando || !!erro}>
                Novo evento
              </Button>
            ) : undefined
          }
        />

        <ListaEventos
          eventos={eventos}
          finalidades={finalidades}
          carregando={carregando}
          erro={erro}
          onTentarNovamente={tentarNovamente}
          podeEditar={pode("eventos.editar")}
          podeExcluir={pode("eventos.excluir")}
          onEditar={(evento) => setPainel({ tipo: "editar", evento })}
          onExcluir={(evento) => {
            setErroExclusao(null);
            setAlvoExclusao(evento);
          }}
        />
      </div>

      {/* Painel lateral: novo evento ou edição */}
      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={painel?.tipo === "editar" ? "Editar evento" : "Novo evento"}
        descricao="Preencha os dados do evento."
      >
        {painel && (
          <FormularioEvento
            evento={painel.tipo === "editar" ? painel.evento : null}
            sugestoesCidade={valoresDistintos(eventos.flatMap((item) => (item.cidade ? [item.cidade] : [])))}
            finalidades={finalidades}
            onSalvo={aoSalvar}
          />
        )}
      </Drawer>

      {/* Confirmação de exclusão */}
      <ConfirmDialog
        aberto={alvoExclusao !== null}
        titulo="Excluir evento?"
        mensagem={
          <>
            <p>
              O evento de{" "}
              <strong>{alvoExclusao ? formatarDataHora(alvoExclusao.data_evento) : ""}</strong> em{" "}
              <strong>{alvoExclusao?.local}</strong> deixará de aparecer nas telas. As presenças já
              registradas nele são mantidas no histórico.
            </p>
            {erroExclusao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroExclusao}
              </p>
            )}
          </>
        }
        textoConfirmar="Excluir"
        processando={excluindo}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExclusao(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

################ src/app/(painel)/portarias/page.tsx
"use client";

/**
 * SysMúsica — Página /portarias (Gestão de portarias): criar, aprovar, rejeitar, histórico, excluir.
 * Quem pode criar escolhe um evento e cria a portaria (PENDENTE); um aprovador autorizado, de outro
 * usuário e dentro da abrangência, aprova ou rejeita (com motivo). Só portaria APROVADA registra
 * presenças. Cada botão só aparece quando o banco diz que o usuário pode; o banco confere de novo.
 */
import { useCallback, useEffect, useState } from "react";
import { Plus } from "lucide-react";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";
import CotaPortariasIndicador from "@/modules/portarias/components/CotaPortarias";
import FormularioNovaPortaria from "@/modules/portarias/components/FormularioNovaPortaria";
import FormularioRejeicao from "@/modules/portarias/components/FormularioRejeicao";
import HistoricoPortaria from "@/modules/portarias/components/HistoricoPortaria";
import TabelaPortarias from "@/modules/portarias/components/TabelaPortarias";
import {
  buscarEventosParaPortaria,
  decidirPortaria,
  excluirPortaria,
  historicoDaPortaria,
  listarPortarias,
  minhaCota,
  minhasCidades,
} from "@/modules/portarias/services/portarias";
import type {
  CotaPortarias,
  EventoParaPortaria,
  HistoricoPortaria as ItemHistorico,
  PortariaDetalhada,
} from "@/modules/portarias/tipos";
import Button from "@/components/ui/Button";
import ConfirmDialog from "@/components/ui/ConfirmDialog";
import Drawer from "@/components/ui/Drawer";
import EmptyState from "@/components/ui/EmptyState";
import ErrorState from "@/components/ui/ErrorState";
import Notificacao from "@/components/ui/Notificacao";
import PageHeader from "@/components/ui/PageHeader";
import Skeleton from "@/components/ui/Skeleton";
import { formatarDataHora } from "@/lib/datas";

// Qual painel lateral está aberto
type Painel =
  | { tipo: "nova" }
  | { tipo: "rejeitar"; portaria: PortariaDetalhada }
  | { tipo: "historico"; portaria: PortariaDetalhada; itens: ItemHistorico[] };

export default function PortariasPage() {
  const { pode } = useSessao();
  const podeCriar = pode("portarias.criar");
  const podeAprovar = pode("portarias.aprovar");
  const controleTotal = pode("portarias.controle_total"); // atua em qualquer cidade: não precisa de abrangência

  const [portarias, setPortarias] = useState<PortariaDetalhada[]>([]);
  const [eventos, setEventos] = useState<EventoParaPortaria[]>([]);
  const [cota, setCota] = useState<CotaPortarias | null>(null);
  const [cidades, setCidades] = useState<string[] | null>(null); // abrangência (só aprovadores)
  const [carregando, setCarregando] = useState(true); // inicia true: primeira busca no mount
  const [erro, setErro] = useState<string | null>(null);
  const [painel, setPainel] = useState<Painel | null>(null);
  const [alvoAprovar, setAlvoAprovar] = useState<PortariaDetalhada | null>(null);
  const [alvoExcluir, setAlvoExcluir] = useState<PortariaDetalhada | null>(null);
  const [processando, setProcessando] = useState(false);
  const [erroConfirmacao, setErroConfirmacao] = useState<string | null>(null);
  const [erroAcao, setErroAcao] = useState<string | null>(null);
  const [aviso, setAviso] = useState<string | null>(null);

  // Busca as portarias e, conforme as permissões, os eventos, a cota e a abrangência
  const carregar = useCallback(async () => {
    try {
      const [lista, listaEventos, cotaAtual, cidadesAtuais] = await Promise.all([
        listarPortarias(),
        podeCriar ? buscarEventosParaPortaria() : Promise.resolve([] as EventoParaPortaria[]),
        podeCriar ? minhaCota() : Promise.resolve(null),
        podeAprovar ? minhasCidades() : Promise.resolve(null),
      ]);
      setPortarias(lista);
      setEventos(listaEventos);
      setCota(cotaAtual);
      setCidades(cidadesAtuais);
      setErro(null);
    } catch (e) {
      console.error("Falha ao carregar as portarias:", e);
      setErro(e instanceof Error ? e.message : "Não foi possível carregar as portarias.");
    } finally {
      setCarregando(false);
    }
  }, [podeCriar, podeAprovar]);

  // Primeira carga ao abrir a página
  useEffect(() => {
    // O setState só roda DEPOIS do await (nunca de forma síncrona), então o aviso da regra
    // set-state-in-effect não se aplica a esta carga inicial. Ignorado apenas nesta linha.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    void carregar();
  }, [carregar]);

  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  // Depois de uma ação: recarrega, fecha o painel e mostra o aviso de sucesso
  async function aoSalvar(mensagem: string) {
    await carregar();
    setPainel(null);
    setAviso(mensagem);
  }

  async function abrirHistorico(portaria: PortariaDetalhada) {
    setErroAcao(null);
    try {
      setPainel({ tipo: "historico", portaria, itens: await historicoDaPortaria(portaria.id) });
    } catch (e) {
      setErroAcao(e instanceof Error ? e.message : "Não foi possível abrir o histórico.");
    }
  }

  async function confirmarAprovacao() {
    if (!alvoAprovar) return;
    setProcessando(true);
    setErroConfirmacao(null);
    try {
      await decidirPortaria(alvoAprovar.id, true, null);
      await carregar();
      setAviso("Portaria aprovada. Ela já pode ser usada para registrar presenças.");
      setAlvoAprovar(null);
    } catch (e) {
      setErroConfirmacao(e instanceof Error ? e.message : "Não foi possível aprovar a portaria.");
    } finally {
      setProcessando(false);
    }
  }

  async function confirmarExclusao() {
    if (!alvoExcluir) return;
    setProcessando(true);
    setErroConfirmacao(null);
    try {
      await excluirPortaria(alvoExcluir.id);
      await carregar();
      setAviso("Portaria excluída. A cota do dia não é devolvida.");
      setAlvoExcluir(null);
    } catch (e) {
      setErroConfirmacao(e instanceof Error ? e.message : "Não foi possível excluir a portaria.");
    } finally {
      setProcessando(false);
    }
  }

  // Eventos que ainda podem receber portaria (sem uma pendente ou aprovada)
  const eventosDisponiveis = eventos.filter(
    (evento) => !portarias.some((portaria) => portaria.evento_id === evento.id && portaria.status !== "REJEITADA")
  );

  const descricaoEvento = (portaria: PortariaDetalhada) =>
    `${portaria.evento_data ? formatarDataHora(portaria.evento_data) : "Evento removido"} · ${portaria.evento_local ?? ""}`;

  const tituloPainel =
    painel?.tipo === "rejeitar" ? "Rejeitar portaria" : painel?.tipo === "historico" ? "Histórico da portaria" : "Nova portaria";

  // Mensagem de erro reaproveitada pelos dois diálogos de confirmação
  const erroDialogo = erroConfirmacao && (
    <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
      {erroConfirmacao}
    </p>
  );

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Portarias"
          descricao={`Criação e aprovação das portarias dos eventos.${
            !carregando && !erro ? ` ${portarias.length} portaria(s).` : ""
          }`}
          acoes={
            podeCriar ? (
              <>
                {cota && <CotaPortariasIndicador cota={cota} />}
                <Button icone={Plus} onClick={() => setPainel({ tipo: "nova" })} disabled={carregando || !!erro}>
                  Nova portaria
                </Button>
              </>
            ) : undefined
          }
        />

        {/* Aprovador sem cidade de abrangência: avisa o que fazer */}
        {podeAprovar && !controleTotal && cidades !== null && cidades.length === 0 && (
          <p className="mb-4 rounded-lg bg-amber-50 px-3 py-2 text-sm text-amber-700">
            Você ainda não tem vínculo com nenhum setor, então não consegue aprovar nem rejeitar portarias. Peça a
            um administrador para vinculá-lo a um setor em Administração &gt; Usuários.
          </p>
        )}

        {erroAcao && (
          <p role="alert" className="mb-4 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
            {erroAcao}
          </p>
        )}

        {carregando ? (
          <div role="status" aria-live="polite" className="space-y-3">
            <span className="sr-only">Carregando portarias...</span>
            <Skeleton className="h-14 rounded-2xl" />
            {[0, 1, 2, 3].map((i) => (
              <Skeleton key={i} className="h-12" />
            ))}
          </div>
        ) : erro ? (
          <ErrorState titulo="Não foi possível carregar as portarias." mensagem={erro} onTentarNovamente={tentarNovamente} />
        ) : portarias.length === 0 ? (
          <EmptyState
            titulo="Nenhuma portaria por aqui ainda."
            descricao={
              podeCriar
                ? "Clique em “Nova portaria” para solicitar a portaria de um evento."
                : "Quando houver portarias na sua abrangência, elas aparecem aqui."
            }
          />
        ) : (
          <TabelaPortarias
            portarias={portarias}
            onAprovar={(portaria) => {
              setErroConfirmacao(null);
              setAlvoAprovar(portaria);
            }}
            onRejeitar={(portaria) => setPainel({ tipo: "rejeitar", portaria })}
            onHistorico={(portaria) => void abrirHistorico(portaria)}
            onExcluir={(portaria) => {
              setErroConfirmacao(null);
              setAlvoExcluir(portaria);
            }}
          />
        )}
      </div>

      {/* Painel lateral: nova portaria, rejeição ou histórico */}
      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={tituloPainel}
        descricao={
          painel?.tipo === "nova"
            ? "A portaria só poderá ser usada depois de aprovada por outro usuário."
            : painel?.tipo === "rejeitar"
              ? "Informe o motivo: ele fica no histórico."
              : painel?.tipo === "historico"
                ? descricaoEvento(painel.portaria)
                : undefined
        }
      >
        {painel?.tipo === "nova" && <FormularioNovaPortaria eventos={eventosDisponiveis} cota={cota} onSalvo={aoSalvar} />}
        {painel?.tipo === "rejeitar" && <FormularioRejeicao portaria={painel.portaria} onSalvo={aoSalvar} />}
        {painel?.tipo === "historico" && <HistoricoPortaria itens={painel.itens} />}
      </Drawer>

      <ConfirmDialog
        aberto={alvoAprovar !== null}
        titulo="Aprovar portaria?"
        mensagem={
          <>
            <p>
              A portaria de <strong>{alvoAprovar ? descricaoEvento(alvoAprovar) : ""}</strong> poderá ser usada para
              registrar presenças pelos atendentes autorizados. A decisão não pode ser desfeita.
            </p>
            {erroDialogo}
          </>
        }
        textoConfirmar="Aprovar" variante="primario"
        processando={processando}
        onConfirmar={() => void confirmarAprovacao()}
        onCancelar={() => setAlvoAprovar(null)}
      />

      <ConfirmDialog
        aberto={alvoExcluir !== null}
        titulo="Excluir portaria?"
        mensagem={
          <>
            <p>
              A portaria de <strong>{alvoExcluir ? descricaoEvento(alvoExcluir) : ""}</strong> deixará de aparecer.
              Atenção: a cota diária de criação <strong>não é devolvida</strong>.
            </p>
            {erroDialogo}
          </>
        }
        textoConfirmar="Excluir"
        processando={processando}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExcluir(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

################ src/components/FormularioEvento.tsx
"use client";

import SeletorCasaOracao from "@/modules/organizacao/components/SeletorCasaOracao";

/**
 * SysMúsica — Formulário de evento (cadastro e edição) — conteúdo do painel lateral.
 * evento = null -> cadastrar; preenchido -> editar.
 *
 * Finalidade (combobox): sugere as finalidades já cadastradas. Se a pessoa digita uma que ainda não
 * existe, ela é salva automaticamente e passa a aparecer como sugestão nos próximos eventos.
 */
import { useState, type FormEvent } from "react";
import Button from "@/components/ui/Button";
import { datetimeLocalParaIso, isoParaDatetimeLocal } from "@/lib/dataLocal";
import { normalizarTexto } from "@/lib/texto";
import type { Evento, FinalidadeEvento } from "@/types/database";
import {
  FINALIDADE_MAXIMO,
  formatarNomeFinalidade,
  validarFinalidade,
} from "@/modules/eventos/regras";
import { validarDataDoEvento } from "@/modules/eventos/dataEvento";
import {
  camposFaltando,
  mensagemCamposFaltando,
  ROTULO_CIDADE,
  ROTULO_CASA_ORACAO,
  ROTULO_DATA,
  ROTULO_FINALIDADE,
  ROTULO_LOCAL,
  ROTULO_TIPO,

import { atualizarEvento, criarEvento } from "@/modules/eventos/services/eventos";
import { obterOuCriarFinalidade } from "@/modules/eventos/services/finalidades";

interface FormularioEventoProps {
  evento: Evento | null;
  finalidades: FinalidadeEvento[];
  sugestoesCidade: string[]; // cidades já cadastradas em outros eventos
  /** Chamado após salvar com sucesso, com a mensagem para o aviso da tela. */
  onSalvo: (mensagem: string) => void | Promise<void>;
}

// Mesma identidade visual dos demais formulários.
// [color-scheme:light] mantém o ícone do seletor de data visível mesmo com o navegador em tema escuro.
const classeInput =
  "w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 " +
  "placeholder:text-slate-400 shadow-sm focus:border-brand-500 focus:outline-none " +
  "focus:ring-2 focus:ring-brand-200 disabled:cursor-not-allowed disabled:opacity-60 " +
  "[color-scheme:light]";

export default function FormularioEvento({
  evento,
  finalidades,
  sugestoesCidade,
  onSalvo,
}: FormularioEventoProps) {
  const editando = evento !== null;

  const finalidadeInicial = evento?.finalidade_id
    ? (finalidades.find((finalidade) => finalidade.id === evento.finalidade_id)?.nome ?? "")
    : "";

  const [tipo, setTipo] = useState(evento?.tipo ?? "");
  const [finalidadeTexto, setFinalidadeTexto] = useState(finalidadeInicial);
  const [dataEvento, setDataEvento] = useState(evento ? isoParaDatetimeLocal(evento.data_evento) : "");
  const [local, setLocal] = useState(evento?.local ?? "");
  const [cidade, setCidade] = useState(evento?.cidade ?? "");
  const [casaOracaoId, setCasaOracaoId] = useState(evento?.casa_oracao_id ?? "");
  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);
  const [faltando, setFaltando] = useState<string[]>([]); // campos obrigatórios vazios (borda vermelha)

  // Borda vermelha no campo obrigatório que ficou vazio
  const classeCampo = (rotulo: string) =>
    classeInput + (faltando.includes(rotulo) ? " !border-red-500 !ring-2 !ring-red-200" : "");

  // A finalidade digitada já existe? (comparação sem acento e sem diferenciar maiúsculas)
  const finalidadeDigitada = finalidadeTexto.trim();
  const finalidadeExistente = finalidadeDigitada
    ? finalidades.find(
        (finalidade) => normalizarTexto(finalidade.nome) === normalizarTexto(finalidadeDigitada)
      )
    : undefined;

  async function salvar(submissao: FormEvent<HTMLFormElement>) {
    submissao.preventDefault();
    if (salvando) return; // evita envio duplicado
    setErro(null);

    // Todos os campos são obrigatórios: o alerta lista os que faltam e eles ficam com borda vermelha
    const vazios = camposFaltando({ tipo, finalidade: finalidadeTexto, dataEvento, local, cidade, casaOracaoId });
    setFaltando(vazios);
    if (vazios.length > 0) return setErro(mensagemCamposFaltando(vazios));

    const erroFinalidade = validarFinalidade(finalidadeTexto);
    if (erroFinalidade) return setErro(erroFinalidade);

    // O datetime-local não traz fuso: new Date() o lê como hora LOCAL do navegador e a conversão
    // para ISO (UTC) é o que o campo timestamptz do banco espera.
    const dataIso = datetimeLocalParaIso(dataEvento);
    if (!dataIso) return setErro("Informe uma data e hora válidas.");

    // Não se agenda evento em data e hora que já passaram (ao editar, só vale se a data for alterada)
    const erroData = validarDataDoEvento(dataIso, Date.now(), evento?.data_evento ?? null);
    if (erroData) return setErro(erroData);

    setSalvando(true);
    try {
      // Passo 1: finalidade (usa a existente ou salva uma nova para os próximos eventos)
      const finalidade = await obterOuCriarFinalidade(finalidadeTexto, finalidades);

      // Passo 2: grava o evento
      const dados = {
        tipo: tipo.trim(),
        data_evento: dataIso,
        local: local.trim(),
        finalidade_id: finalidade.id,
        cidade: cidade.trim() || null,
        casa_oracao_id: casaOracaoId || null,
      };
      if (evento) await atualizarEvento(evento.id, dados);
      else await criarEvento(dados);

      const complemento = finalidade.criada
        ? ` A finalidade "${finalidade.nome}" foi salva para os próximos eventos.`
        : "";
      await onSalvo(
        editando ? `Evento atualizado.${complemento}` : `Evento cadastrado com sucesso!${complemento}`
      );
    } catch (e) {
      setErro(e instanceof Error ? e.message : "Não foi possível salvar o evento.");
    } finally {
      setSalvando(false);
    }
  }

  return (
    <form onSubmit={salvar} noValidate className="space-y-5">
      <p className="text-xs text-slate-500">Todos os campos são obrigatórios (marcados com *).</p>

      <div className="space-y-1.5">
        <label htmlFor="tipo" className="text-sm font-medium text-slate-700">
          Tipo <span className="text-red-600"> *</span>
        </label>
        {/* Campo livre com sugestões (datalist): aceita Ensaio, Culto ou qualquer outro tipo */}
        <input
          id="tipo"
          type="text"
          list="sugestoes-tipo-evento"
          required
          value={tipo}
          onChange={(e) => setTipo(e.target.value)}
          disabled={salvando}
          placeholder="Ex: Ensaio, Culto"
          className={classeCampo(ROTULO_TIPO)}
        />
        <datalist id="sugestoes-tipo-evento">
          <option value="Ensaio" />
          <option value="Culto" />
        </datalist>
      </div>

      {/* Finalidade (combobox): escolha uma cadastrada ou digite uma nova */}
      <div className="space-y-1.5">
        <label htmlFor="finalidade" className="text-sm font-medium text-slate-700">
          Finalidade do evento <span className="text-red-600"> *</span>
        </label>
        <input
          id="finalidade"
          type="text"
          list="lista-finalidades"
          autoComplete="off"
          required
          maxLength={FINALIDADE_MAXIMO}
          value={finalidadeTexto}
          onChange={(e) => setFinalidadeTexto(e.target.value)}
          disabled={salvando}
          placeholder="Digite ou escolha uma finalidade"
          className={classeCampo(ROTULO_FINALIDADE)}
        />
        <datalist id="lista-finalidades">
          {finalidades.map((finalidade) => (
            <option key={finalidade.id} value={finalidade.nome} />
          ))}
        </datalist>
        {/* Feedback ao vivo: finalidade já cadastrada ou nova (será salva para os próximos eventos) */}
        {finalidadeDigitada && (
          <p className={"text-xs " + (finalidadeExistente ? "text-slate-500" : "text-brand-700")}>
            {finalidadeExistente
              ? "Finalidade já cadastrada."
              : `Nova finalidade: "${formatarNomeFinalidade(finalidadeDigitada)}" será salva para os próximos eventos.`}
          </p>
        )}
      </div>

      <div className="space-y-1.5">
        <label htmlFor="data_evento" className="text-sm font-medium text-slate-700">
          Data e hora <span className="text-red-600"> *</span>
        </label>
        <input
          id="data_evento"
          type="datetime-local"
          required
          value={dataEvento}
          onChange={(e) => setDataEvento(e.target.value)}
          disabled={salvando}
          className={classeCampo(ROTULO_DATA)}
        />
      </div>

      <div className="space-y-1.5">

        if (atual && liberadas.some((portaria) => portaria.id === atual)) return atual;
        const guardada = lerPortariaGuardada();
        if (guardada && liberadas.some((portaria) => portaria.id === guardada)) return guardada;
        return liberadas.length === 1 ? liberadas[0].id : null;
      });
      setErro(null);
    } catch (e) {
      console.error("Falha ao carregar as portarias do atendimento:", e);
      setErro(e instanceof Error ? e.message : "Não foi possível carregar as portarias.");
    } finally {
      setCarregando(false);
    }
  }, [controleTotal]);

  const carregarEntradas = useCallback(async (id: string) => {
    try {
      const [lista, semCadastroLista] = await Promise.all([entradasDaPortaria(id), entradasAvulsasDaPortaria(id)]);
      setEntradas(lista);
      setAvulsas(semCadastroLista);
    } catch (e) {
      // A lista é apenas informativa: se falhar, o atendimento continua
      console.error("Falha ao carregar as entradas:", e);
    }
  }, []);

  /* eslint-disable react-hooks/set-state-in-effect */
  useEffect(() => {
    void carregar();
  }, [carregar]);

  // Ao ter uma portaria escolhida, busca as últimas entradas dela
  useEffect(() => {
    if (portariaId) void carregarEntradas(portariaId);
  }, [portariaId, carregarEntradas]);

  // Linhas do painel e instrumentos (uma vez só): usados na pergunta "Vai tocar?" e na entrada sem cadastro
  useEffect(() => {
    opcoesEntrada()
      .then(setOpcoes)
      .catch((e: unknown) => console.error("Falha ao carregar as opções da entrada:", e));
  }, []);

  // Se a portaria em atendimento encerrar, sai da tela na hora: fecha a conferência e a entrada sem cadastro
  useEffect(() => {
    if (portariaId && portarias.length > 0 && !disponiveis.some((portaria) => portaria.id === portariaId)) {
      setCartao(null);
      setSemCadastro(false);
      setPortariaId(null);
      guardarPortaria(null);
      setAviso("Esta portaria foi encerrada e saiu da tela. Só o Administrador pode corrigir registros.");
    }
  }, [portariaId, portarias, disponiveis]);
  /* eslint-enable react-hooks/set-state-in-effect */

  // O aviso de leitura some sozinho depois de alguns segundos
  useEffect(() => {
    if (!erroLeitura) return;
    const temporizador = setTimeout(() => setErroLeitura(null), 5000);
    return () => clearTimeout(temporizador);
  }, [erroLeitura]);

  // Relógio da tela: a cada 30 segundos recalcula quais portarias estão com a janela aberta
  useEffect(() => {
    const temporizador = setInterval(() => setAgora(Date.now()), 30_000);
    return () => clearInterval(temporizador);
  }, []);

  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  function escolherPortaria(id: string | null) {
    setEntradas([]);
    setAvulsas([]);
    setCartao(null);
    setErroLeitura(null);
    setPortariaId(id);
    guardarPortaria(id);
  }

  // QR Code lido: consulta o membro e abre a conferência
  async function aoLerQr(texto: string) {
    if (!portariaId || cartao || consultando) return;
    setErroLeitura(null);

    if (!pareceQrSysMusica(texto)) {
      setErroLeitura("Este QR Code não é do SysMúsica.");
      return;
    }

    setConsultando(true);
    try {
      const membro = await buscarMembroPorQr(portariaId, texto);
      await abrirCartao(membro, "QR");
    } catch (e) {
      setErroLeitura(e instanceof Error ? e.message : "Não foi possível ler o QR Code.");
    } finally {
      setConsultando(false);
    }
  }

  // Foto tirada na conferência: recarrega os dados do cartão (a foto já aparece)
  async function aoSalvarFotoNoCartao() {
    if (!portariaId || !cartao) return;
    const atualizado = await recarregarCartao(portariaId, cartao.membro.membro_id);
    setCartao({ membro: atualizado, origem: cartao.origem });
  }

  // Abre a conferência: antes, consulta se a pessoa precisa responder "Vai tocar?"
  async function abrirCartao(membro: CartaoMembro, origem: OrigemEntrada) {
    if (!portariaId) return;
    try {
      setInfo(await infoEntrada(portariaId, membro.membro_id));
      setEscolha({ toca: null, instrumentoId: "" });
      setErroCartao(null);
      setCartao({ membro, origem });
    } catch (e) {
      setErroLeitura(e instanceof Error ? e.message : "Não foi possível abrir a conferência.");
    }
  }

  // Membro localizado pelo nome: abre a conferência
  function aoEscolherPorNome(membro: CartaoMembro) {
    void abrirCartao(membro, "MANUAL");
  }

  // Entrada sem cadastro registrada: fecha a janela e atualiza as listas
  async function aoRegistrarSemCadastro(descricao: string) {
    setSemCadastro(false);
    setAviso(`Entrada sem cadastro registrada: ${descricao}`);
    if (portariaId) await carregarEntradas(portariaId);
  }

  // Desfaz uma entrada sem cadastro (pede confirmação)
  async function desfazerSemCadastro(id: string) {
    if (!portariaId || !window.confirm("Desfazer esta entrada sem cadastro?")) return;
    try {
      await desfazerEntradaAvulsa(id);
      setAviso("Entrada desfeita.");
      await carregarEntradas(portariaId);
    } catch (e) {
      setErroLeitura(e instanceof Error ? e.message : "Não foi possível desfazer a entrada.");
    }
  }

  async function confirmar() {
    if (!portariaId || !cartao || registrando) return;
    setRegistrando(true);
    setErroCartao(null);
    try {
      const valores = valoresDaEntrada(info, escolha);
      await registrarEntrada(portariaId, cartao.membro.membro_id, cartao.origem, valores.toca, valores.instrumentoId);
      setAviso(`Entrada registrada: ${cartao.membro.nome}`);
      setCartao(null);
      await carregarEntradas(portariaId);
    } catch (e) {
      setErroCartao(e instanceof Error ? e.message : "Não foi possível registrar a entrada.");
    } finally {
      setRegistrando(false);
    }
  }

  const portaria = disponiveis.find((item) => item.id === portariaId) ?? null;
  const janela = portaria ? janelaDaPortaria(portaria.evento_data, agora) : null;

  return (
    <div className="space-y-6">
      {carregando ? (
        <div role="status" aria-live="polite" className="space-y-3">
          <span className="sr-only">Carregando portarias...</span>
          <Skeleton className="h-20 rounded-2xl" />
          <Skeleton className="h-64 rounded-2xl" />
        </div>
      ) : erro ? (
        <ErrorState titulo="Não foi possível carregar as portarias." mensagem={erro} onTentarNovamente={tentarNovamente} />
      ) : disponiveis.length === 0 ? (
        <EmptyState
          titulo="Nenhuma portaria aprovada disponível para você."
          descricao="Só é possível atender em portarias aprovadas, dentro da sua abrangência. Peça a criação e a aprovação da portaria do evento."
        />
      ) : !portaria ? (
        <SelecaoPortaria
          portarias={disponiveis}
          agora={agora}
          controleTotal={controleTotal}
          onEscolher={(id) => escolherPortaria(id)}
        />
      ) : (
        <>
          {/* Portaria em atendimento */}
          <div className="flex flex-wrap items-center justify-between gap-3 rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
            <div className="min-w-0">
              <p className="text-xs font-semibold uppercase tracking-wide text-emerald-700">Portaria em atendimento</p>
              <p className="text-base font-semibold text-slate-900">
                {portaria.evento_data ? formatarDataHora(portaria.evento_data) : "Evento removido"}
              </p>
              <p className="text-sm text-slate-600">
                {[portaria.evento_tipo, portaria.evento_local, portaria.cidade].filter(Boolean).join(" · ")}
              </p>
              {janela && agora > 0 && (
                <p className={"mt-1 text-xs font-medium " + (janela.situacao === "aberta" ? "text-emerald-700" : "text-amber-700")}>
                  {descreverJanela(janela, agora)}
                  {janela.situacao !== "aberta" && " — fora do horário: o controle total permite o registro e ele fica na Auditoria."}
                </p>
              )}
            </div>
            {disponiveis.length > 1 && (
              <button
                type="button"
                onClick={() => escolherPortaria(null)}
                className="rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm font-medium text-slate-700 transition hover:bg-slate-50"
              >
                Trocar portaria
              </button>
            )}
          </div>

          <div className="grid gap-6 lg:grid-cols-[minmax(0,1fr)_22rem]">
            <div className="space-y-6">
              <LeitorQr ativo={!cartao && !consultando} onLeitura={(texto) => void aoLerQr(texto)} />

              {(erroLeitura || consultando) && (
                <p
                  role={erroLeitura ? "alert" : "status"}
                  className={
                    "rounded-lg px-3 py-2 text-sm " +
                    (erroLeitura ? "bg-red-50 text-red-700" : "bg-slate-50 text-slate-600")
                  }
                >
                  {erroLeitura ?? "Consultando o QR Code..."}
                </p>
              )}

              <BuscaPorNome portariaId={portaria.id} desabilitado={cartao !== null} onEscolher={aoEscolherPorNome} />

              <section className="rounded-2xl border border-dashed border-slate-300 bg-white p-4 shadow-sm">
                <h2 className="text-sm font-semibold text-slate-900">Candidato, visitante ou irmão sem cadastro?</h2>
                <p className="mt-1 text-xs text-slate-500">
                  Registre a entrada sem criar cadastro: informe o cargo ou categoria, o instrumento e de onde a
                  pessoa vem. Ela entra na contagem do evento.
                </p>
                <button
                  type="button"
                  onClick={() => setSemCadastro(true)}
                  disabled={cartao !== null || opcoes === null}
                  className="mt-3 rounded-lg border border-slate-300 bg-white px-4 py-2 text-sm font-medium text-slate-700 transition hover:bg-slate-50 disabled:cursor-not-allowed disabled:opacity-60"
                >
                  Entrada sem cadastro
                </button>
              </section>
            </div>

            <aside>
              <EntradasRecentes entradas={entradas} avulsas={avulsas} onDesfazerAvulsa={(id) => void desfazerSemCadastro(id)} />
            </aside>
          </div>
        </>
      )}

      <div className={podeCorrigir ? "text-sm" : "hidden"}>
        <Link href="/portaria/registros" className="font-semibold text-brand-700 hover:underline">
          Registros e correções
        </Link>
      </div>

      {cartao && (
        <CartaoConferencia
          // A chave reinicia a caixa "conferi a identidade" a cada membro
          key={cartao.membro.membro_id}
          membro={cartao.membro}
          origem={cartao.origem}
          registrando={registrando}
          erro={erroCartao}
          onConfirmar={() => void confirmar()}
          onCancelar={() => setCartao(null)}
          podeFoto={pode("membros.foto")}
          onFotoSalva={aoSalvarFotoNoCartao}
          info={info}
          instrumentos={opcoes?.instrumentos ?? []}
          escolha={escolha}
          onEscolha={setEscolha}
        />
      )}

      <Drawer
        aberto={semCadastro}
        onFechar={() => setSemCadastro(false)}
        titulo="Entrada sem cadastro"
        descricao="Candidato, visitante ou irmão sem cadastro: não cria cadastro, só conta na presença do evento."
      >
        {semCadastro && portaria && opcoes && (
          <EntradaSemCadastro portariaId={portaria.id} opcoes={opcoes} onRegistrada={(d) => void aoRegistrarSemCadastro(d)} />
        )}
      </Drawer>

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </div>
  );
}

################ src/modules/eventos/dataEvento.ts
/**
 * SysMúsica — Regra da data do evento (pura, sem banco): não se agenda evento em data e hora que já passaram.
 * Uma tolerância de 5 minutos evita falso alarme (o campo de data só tem precisão de minutos e o usuário leva
 * alguns segundos para salvar). Ao EDITAR, só vale se a data/hora for alterada: um evento antigo continua
 * podendo ter outros dados corrigidos. A mesma regra é aplicada no banco (migração 0025).
 */
export const TOLERANCIA_MS = 5 * 60 * 1000;

export const MENSAGEM_DATA_PASSADA =
  "Não é possível agendar um evento em uma data e hora que já passaram. Escolha uma data e hora futuras.";

/**
 * Devolve a mensagem de erro ou null se a data está correta.
 * dataOriginal = data já gravada do evento (null ao cadastrar um evento novo).
 */
export function validarDataDoEvento(dataIso: string, agora: number, dataOriginal: string | null): string | null {
  const nova = new Date(dataIso).getTime();
  if (Number.isNaN(nova)) return "Informe uma data e hora válidas.";

  // Edição sem mudar a data e a hora (no minuto): não reavalia
  if (dataOriginal !== null) {
    const original = new Date(dataOriginal).getTime();
    if (!Number.isNaN(original) && Math.floor(original / 60000) === Math.floor(nova / 60000)) return null;
  }

  return nova < agora - TOLERANCIA_MS ? MENSAGEM_DATA_PASSADA : null;
}

################ src/modules/eventos/regras.ts
/**
 * SysMúsica — Regras puras de eventos (sem acesso ao banco). Testadas em regras.test.ts.
 */
export const FINALIDADE_MINIMO = 3;
export const FINALIDADE_MAXIMO = 150;

/** Ajusta a finalidade antes de salvar: tira espaços extras e põe a 1ª letra em maiúscula. */
export function formatarNomeFinalidade(texto: string): string {
  const limpo = texto.trim().replace(/\s+/g, " ");
  return limpo.charAt(0).toUpperCase() + limpo.slice(1);
}

/** Valida o texto da finalidade; devolve a mensagem de erro ou null quando está tudo certo. */
export function validarFinalidade(texto: string): string | null {
  const tamanho = texto.trim().replace(/\s+/g, " ").length;
  if (tamanho < FINALIDADE_MINIMO) {
    return `Informe a finalidade do evento (mínimo de ${FINALIDADE_MINIMO} caracteres).`;
  }
  if (tamanho > FINALIDADE_MAXIMO) {
    return `A finalidade pode ter no máximo ${FINALIDADE_MAXIMO} caracteres.`;
  }
  return null;
}

/**
 * SysMúsica — Serviço de dados de eventos.
 * Excluir é EXCLUSÃO LÓGICA: o evento some das telas e as presenças registradas são mantidas.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import type { Evento, EventoInsert, FinalidadeEvento } from "@/types/database";

/** Busca eventos (do mais recente para o mais antigo) e finalidades. Lança erro se a consulta falhar. */
export async function buscarEventosEFinalidades(): Promise<{
  eventos: Evento[];
  finalidades: FinalidadeEvento[];
}> {
  const [resEventos, resFinalidades] = await Promise.all([
    supabase.from("eventos").select("*").order("data_evento", { ascending: false }),
    supabase.from("finalidades_evento").select("*").order("nome", { ascending: true }),
  ]);

  const erroBanco = resEventos.error ?? resFinalidades.error;
  if (erroBanco) throw new Error(erroBanco.message);

  return {
    eventos: (resEventos.data ?? []) as Evento[],
    finalidades: (resFinalidades.data ?? []) as FinalidadeEvento[],
  };
}

/**
 * SysMúsica — Serviço do catálogo de finalidades de evento.
 * O formulário de evento usa obterOuCriarFinalidade: se a pessoa digita uma finalidade que ainda
 * não existe, ela é salva e fica disponível para os próximos eventos.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import { normalizarTexto } from "@/lib/texto";
import type { FinalidadeEvento, FinalidadeEventoInsert } from "@/types/database";
import { formatarNomeFinalidade } from "../regras";

export interface ResultadoFinalidade {
  id: string;
  nome: string;
  criada: boolean; // true = acabou de ser criada
}

/**
 * Devolve a finalidade digitada, criando-a se ainda não existir.
 * A comparação ignora acento, maiúsculas e espaços repetidos.
 */
export async function obterOuCriarFinalidade(
  texto: string,
  existentes: FinalidadeEvento[]
): Promise<ResultadoFinalidade> {
  const chave = normalizarTexto(texto);

  // 1) Já existe na lista carregada? Então só reaproveita o id
  const existente = existentes.find((finalidade) => normalizarTexto(finalidade.nome) === chave);
  if (existente) return { id: existente.id, nome: existente.nome, criada: false };

  // 2) Não existe: salva e resgata o registro
  const nova: FinalidadeEventoInsert = { nome: formatarNomeFinalidade(texto) };
  const { data, error } = await supabase.from("finalidades_evento").insert(nova).select().single();
  if (!error && data) {
    const criada = data as FinalidadeEvento;
    return { id: criada.id, nome: criada.nome, criada: true };
  }

  // 3) 23505 = repetida: outra pessoa criou a mesma finalidade agora há pouco. Busca de novo.
  if (error?.code === "23505") {
    const { data: lista, error: erroBusca } = await supabase.from("finalidades_evento").select("*");
    if (erroBusca) throw new Error(erroBusca.message);
    const achada = ((lista ?? []) as FinalidadeEvento[]).find(
      (finalidade) => normalizarTexto(finalidade.nome) === chave
    );
    if (achada) return { id: achada.id, nome: achada.nome, criada: false };
  }

  throw new Error(
    error
      ? mensagemDoErro(error, "Não foi possível salvar a finalidade do evento.")
      : "Não foi possível salvar a finalidade do evento."
  );
}

################ src/modules/portarias/janela.ts
/**
 * SysMúsica — Janela de atendimento da portaria (regras puras, sem banco).
 * A portaria abre 2 horas ANTES do início do evento e encerra 3 horas DEPOIS do início (horário de Salvador).
 * Ex.: ensaio às 14h -> aberta das 12h às 17h. A regra de verdade é aplicada pelo banco (migração 0024);
 * aqui ficam só os cálculos para a tela mostrar quando abre, até quando vale e se está liberada.
 */

export const HORAS_ANTES = 2;
export const HORAS_DEPOIS = 3; // encerra 3 horas depois do início (igual ao banco, migração 0027)
const MS_HORA = 60 * 60 * 1000;
const FUSO = "America/Sao_Paulo";

export type SituacaoJanela = "antes" | "aberta" | "encerrada";

export interface JanelaPortaria {
  abre: Date;
  fecha: Date;
  situacao: SituacaoJanela;
}

/** Janela do evento no instante "agora" (ms). Nulo se o evento não tem data válida. Os limites são inclusivos. */
export function janelaDaPortaria(dataEvento: string | null, agora: number): JanelaPortaria | null {
  if (!dataEvento) return null;
  const inicio = new Date(dataEvento).getTime();
  if (Number.isNaN(inicio)) return null;

  const abre = inicio - HORAS_ANTES * MS_HORA;
  const fecha = inicio + HORAS_DEPOIS * MS_HORA;
  const situacao: SituacaoJanela = agora < abre ? "antes" : agora > fecha ? "encerrada" : "aberta";
  return { abre: new Date(abre), fecha: new Date(fecha), situacao };
}

// Dia (AAAA-MM-DD) e hora no fuso de Salvador
function diaSalvador(data: Date): string {
  return new Intl.DateTimeFormat("en-CA", { timeZone: FUSO, year: "numeric", month: "2-digit", day: "2-digit" }).format(data);
}

function horaSalvador(data: Date): string {
  const [h, m] = new Intl.DateTimeFormat("pt-BR", { timeZone: FUSO, hour: "2-digit", minute: "2-digit", hour12: false })
    .format(data)
    .split(":");
  return `${h}h${m}`;
}

/** "12h00" quando é hoje; "19/10 às 12h00" quando é em outro dia. */
export function rotuloDoHorario(data: Date, agora: number): string {
  if (diaSalvador(data) === diaSalvador(new Date(agora))) return horaSalvador(data);
  const [, mes, dia] = diaSalvador(data).split("-");
  return `${dia}/${mes} às ${horaSalvador(data)}`;
}

/** Texto curto para a tela: "Abre às 12h00", "Aberta até 16h00" ou "Encerrada às 16h00" (com a data se não for hoje). */
export function descreverJanela(janela: JanelaPortaria, agora: number): string {
  if (janela.situacao === "antes") return `Abre ${comPreposicao(janela.abre, agora)}`;
  if (janela.situacao === "aberta") return `Aberta até ${rotuloDoHorario(janela.fecha, agora)}`;
  return `Encerrada ${comPreposicao(janela.fecha, agora)}`;
}

// "às 12h00" (hoje) ou "19/10 às 12h00" (outro dia)
function comPreposicao(data: Date, agora: number): string {
  const rotulo = rotuloDoHorario(data, agora);
  return rotulo.includes("às") ? rotulo : `às ${rotulo}`;
}

export const MOTIVO_MINIMO = 5;
export const MOTIVO_MAXIMO = 300;

/** Valida o motivo da rejeição: devolve a mensagem de erro ou null se estiver certo. */
export function validarMotivoRejeicao(texto: string): string | null {
  const tamanho = texto.trim().length;
  if (tamanho < MOTIVO_MINIMO) return `Informe o motivo da rejeição (mínimo de ${MOTIVO_MINIMO} caracteres).`;
  if (tamanho > MOTIVO_MAXIMO) return `O motivo pode ter no máximo ${MOTIVO_MAXIMO} caracteres.`;
  return null;
}

/** Texto da cota do dia: "Hoje: 2 de 4 portarias criadas · restam 2". */
export function descreverCota(cota: CotaPortarias): string {
  // Controle total: não há limite diário
  if (cota.ilimitada) return `Hoje: ${cota.usadas} portarias criadas · sem limite (controle total)`;
  const sobra =
    cota.restantes === 0 ? "limite atingido" : cota.restantes === 1 ? "resta 1" : `restam ${cota.restantes}`;
  return `Hoje: ${cota.usadas} de ${cota.limite} portarias criadas · ${sobra}`;
}

/** A cidade está na lista? (ignora acento e maiúsculas, como o banco) */
export function cidadeNaAbrangencia(cidade: string | null, cidades: string[]): boolean {
  if (!cidade) return false;
  const chave = normalizarTexto(cidade);
  return cidades.some((item) => normalizarTexto(item) === chave);
}

/**
 * SysMúsica — Regras puras da entrada na Portaria (sem banco). Testadas em regrasEntrada.test.ts.
 * Quem decide de verdade é o banco (registrar_entrada e registrar_entrada_avulsa conferem tudo de novo);

 */
import type {
  BlocoEntrada,
  EscolhaToca,
  InfoEntrada,
  InstrumentoEntrada,
  LinhaEntrada,
} from "./tiposEntrada";

export const ROTULO_BLOCO_ENTRADA: Record<BlocoEntrada, string> = {
  MINISTERIO: "Ministério",
  MUSICOS: "Músicos",
  ORGANISTAS: "Organistas",
};

const ORDEM_BLOCOS: BlocoEntrada[] = ["MINISTERIO", "MUSICOS", "ORGANISTAS"];

// ---------- Membro cadastrado: "Vai tocar?" ----------

/** Confere a resposta da conferência; devolve a mensagem de erro ou null. */
export function validarEscolhaMembro(info: InfoEntrada | null, escolha: EscolhaToca): string | null {
  if (!info || !info.pergunta_toca) return null;
  if (escolha.toca === null) return "Informe se a pessoa vai tocar.";
  if (escolha.toca && !escolha.instrumentoId && !info.instrumento_id) {
    return "Escolha o instrumento que a pessoa vai tocar.";
  }
  return null;
}

/** Valores a enviar ao banco na entrada de um membro (sem pergunta: nada é enviado). */
export function valoresDaEntrada(
  info: InfoEntrada | null,
  escolha: EscolhaToca
): { toca: boolean | null; instrumentoId: string | null } {
  if (!info || !info.pergunta_toca) return { toca: null, instrumentoId: null };
  return {
    toca: escolha.toca,
    instrumentoId: escolha.toca && escolha.instrumentoId ? escolha.instrumentoId : null,
  };
}

// ---------- Entrada sem cadastro ----------

/** A linha escolhida pede o instrumento? (Músicos sempre; os demais só se a pessoa for tocar) */
export function precisaInstrumento(linha: LinhaEntrada | null, toca: boolean | null): boolean {
  if (!linha || linha.instrumento !== "LIVRE") return false;
  return linha.toca_sempre || (linha.pergunta_toca && toca === true);
}

/** A tela deve perguntar "Vai tocar?" para esta linha? */
export function precisaPerguntar(linha: LinhaEntrada | null): boolean {
  return linha?.pergunta_toca === true;
}

/** Confere a entrada sem cadastro; devolve a mensagem de erro ou null. */
export function validarEntradaAvulsa(dados: {
  linha: LinhaEntrada | null;
  toca: boolean | null;
  instrumentoId: string;
  outraLocalidade: boolean;
  localidadeTexto: string;
}): string | null {
  if (!dados.linha) return "Escolha o cargo ou a categoria da pessoa.";
  if (precisaPerguntar(dados.linha) && dados.toca === null) return "Informe se a pessoa vai tocar.";
  if (precisaInstrumento(dados.linha, dados.toca) && !dados.instrumentoId) {
    return "Escolha o instrumento que a pessoa vai tocar.";
  }
  if (dados.outraLocalidade) {
    const tamanho = dados.localidadeTexto.trim().replace(/\s+/g, " ").length;
    if (tamanho < 2 || tamanho > 120) return "Escreva a localidade (de 2 a 120 letras) ou desmarque a opção.";
  }
  return null;
}

/** Texto da localidade a enviar: nulo quando "Outra" não está marcada ou o campo está vazio. */
export function textoDaLocalidade(outra: boolean, texto: string): string | null {
  if (!outra) return null;
  const limpo = texto.trim().replace(/\s+/g, " ");
  return limpo === "" ? null : limpo;
}

// ---------- Listas para os campos ----------

/** Linhas agrupadas por bloco, na ordem Ministério, Músicos, Organistas. */
export function agruparLinhas(linhas: LinhaEntrada[]): { bloco: BlocoEntrada; rotulo: string; linhas: LinhaEntrada[] }[] {
  return ORDEM_BLOCOS.map((bloco) => ({
    bloco,
    rotulo: ROTULO_BLOCO_ENTRADA[bloco],
    linhas: linhas.filter((linha) => linha.bloco === bloco),
  })).filter((grupo) => grupo.linhas.length > 0);
}

/** Instrumentos agrupados por família (a ordem de chegada do banco é mantida). */
export function agruparInstrumentos(
  instrumentos: InstrumentoEntrada[]
): { familia: string; itens: InstrumentoEntrada[] }[] {
  const grupos = new Map<string, InstrumentoEntrada[]>();
  for (const item of instrumentos) {
    const lista = grupos.get(item.familia) ?? [];
    lista.push(item);
    grupos.set(item.familia, lista);
  }
  return [...grupos.entries()].map(([familia, itens]) => ({ familia, itens }));
}

/** Como a entrada sem cadastro aparece na lista: "Candidato · tocando Clarinete". */
export function descreverAvulsa(entrada: {
  linha_rotulo: string;
  toca: boolean | null;
  instrumento: string | null;
}): string {
  if (entrada.toca === false) return `${entrada.linha_rotulo} · não vai tocar`;
  if (entrada.instrumento) return `${entrada.linha_rotulo} · ${entrada.instrumento}`;
  return entrada.linha_rotulo;
}
