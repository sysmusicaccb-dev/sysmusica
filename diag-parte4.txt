################ src/components/FormularioMembro.tsx
"use client";

/**
 * SysMúsica — Formulário de membro (cadastro e edição) — conteúdo do painel lateral.
 * membro = null -> cadastrar; preenchido -> editar (inclui o status Ativo/Inativo).
 *
 * Casa de Oração: lista agrupada por Região › Setor (cadastrada em Organização). O nome da congregação sai da casa de oração.
 * Cidade e bairro: texto com sugestões dos valores já cadastrados.
 * CPF: só aparece para quem tem a permissão membros.cpf (o banco também confere). É opcional,
 *      validado pelos dígitos verificadores e não pode repetir entre os membros.
 *
 * Instrumento: lista suspensa agrupada por família, igual em todo o sistema (só instrumentos cadastrados).
 *  - O campo só aparece para cargos com permite_instrumento = true.
 *  - É OBRIGATÓRIO nos cargos que exigem instrumento (o Músico); o banco também recusa sem ele.
 *  - A FAMÍLIA é automática e somente leitura (vem do cadastro do instrumento). Instrumento novo e
 *    correção de família: tela Instrumentos e famílias.
 */
import { useState, type FormEvent } from "react";
import Button from "@/components/ui/Button";
import { formatarCpf, limparCpf, validarCpf } from "@/lib/cpf";
import SeletorInstrumento, { FamiliaAutomatica } from "@/modules/instrumentos/components/SeletorInstrumento";
import { validarInstrumentoDoCargo } from "@/modules/instrumentos/selecao";
import { atualizarMembro, criarMembro } from "@/modules/membros/services/membros";
import { CATEGORIAS, cargoUsaCategoria, validarCategoria } from "@/modules/painel/regras";
import SeletorCasaOracao from "@/modules/organizacao/components/SeletorCasaOracao";
import CampoFoto from "@/modules/membros/components/CampoFoto";
import FotoMembro from "@/modules/membros/components/FotoMembro";
import { descartarFoto, enviarFoto, subirFotoNova } from "@/modules/membros/services/fotos";
import type { Cargo, Familia, Instrumento, MembroPublico } from "@/types/database";

/** Listas de sugestões para os campos de localidade. */
export interface SugestoesLocalidade {
  congregacoes: string[];
  cidades: string[];
  bairros: string[];
}

interface FormularioMembroProps {
  membro: MembroPublico | null; // null = cadastrar; preenchido = editar
  cpfInicial: string | null; // CPF completo (só na edição e só para quem tem membros.cpf)
  podeCpf: boolean; // permissão membros.cpf
  podeFoto: boolean; // permissão membros.foto (trocar a fotografia de um membro já cadastrado)
  cargos: Cargo[];
  instrumentos: Instrumento[];
  familias: Familia[];
  sugestoes: SugestoesLocalidade;
  carregando: boolean;
  erro: string | null;
  /** Chamado após salvar com sucesso, com a mensagem para o aviso da tela. */
  onSalvo: (mensagem: string) => void | Promise<void>;
}

// Classe compartilhada pelos campos (mesma identidade visual dos demais módulos).
// [color-scheme:light] mantém o ícone do seletor de data visível mesmo com o navegador em tema escuro.
const classeInput =
  "w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 " +
  "placeholder:text-slate-400 shadow-sm focus:border-brand-500 focus:outline-none " +
  "focus:ring-2 focus:ring-brand-200 disabled:cursor-not-allowed disabled:opacity-60 " +
  "[color-scheme:light]";

export default function FormularioMembro({
  membro,
  cpfInicial,
  podeCpf,
  podeFoto,
  cargos,
  instrumentos,
  familias,
  sugestoes,
  carregando,
  erro,
  onSalvo,
}: FormularioMembroProps) {
  const editando = membro !== null;

  // Valores iniciais: vazios no cadastro; os dados do membro na edição
  const [nome, setNome] = useState(membro?.nome ?? "");
  const [congregacao, setCongregacao] = useState(membro?.congregacao ?? "");
  const [casaOracaoId, setCasaOracaoId] = useState(membro?.casa_oracao_id ?? ""); // a casa de oração define o nome da congregação
  const [cidade, setCidade] = useState(membro?.cidade ?? "");
  const [bairro, setBairro] = useState(membro?.bairro ?? "");
  const [cpf, setCpf] = useState(cpfInicial ? formatarCpf(cpfInicial) : "");
  const [cargoId, setCargoId] = useState(membro?.cargo_id ?? "");
  const [dataNascimento, setDataNascimento] = useState(membro?.data_nascimento ?? ""); // "AAAA-MM-DD"
  const [instrumentoId, setInstrumentoId] = useState(membro?.instrumento_id ?? "");
  const [categoria, setCategoria] = useState(membro?.categoria ?? ""); // Oficializado, Culto Oficial ou RJM
  const [status, setStatus] = useState(membro?.status ?? "Ativo");
  const [foto, setFoto] = useState<Blob | null>(null); // foto nova já preparada (obrigatória no cadastro novo)
  const [salvando, setSalvando] = useState(false);
  const [erroForm, setErroForm] = useState<string | null>(null);

  // O cargo escolhido define se o campo de instrumento aparece e se ele é obrigatório
  const cargoSelecionado = cargos.find((cargo) => cargo.id === cargoId);
  const permiteInstrumento = cargoSelecionado?.permite_instrumento ?? false;
  const exigeInstrumento =
    (cargoSelecionado as (Cargo & { exige_instrumento?: boolean }) | undefined)?.exige_instrumento === true;
  // Só Músico e Organista têm categoria (as demais pessoas contam no painel pelo próprio cargo)
  const tipoCategoria = cargoUsaCategoria(cargoSelecionado?.nome);

  // Ao trocar de cargo, limpa o instrumento se o novo cargo não usar instrumento
  function trocarCargo(novoCargoId: string) {
    setCargoId(novoCargoId);
    const novoCargo = cargos.find((cargo) => cargo.id === novoCargoId);
    if (!novoCargo?.permite_instrumento) setInstrumentoId("");
    if (!cargoUsaCategoria(novoCargo?.nome)) setCategoria("");
  }

  async function salvar(submissao: FormEvent<HTMLFormElement>) {
    submissao.preventDefault();
    if (salvando) return; // evita envio duplicado
    setErroForm(null);

    // Barra campos obrigatórios vazios ou preenchidos só com espaços
    if (!nome.trim() || !casaOracaoId || !cargoId) {
      return setErroForm("Preencha nome, cargo e casa de oração.");
    }

    // CPF (opcional): se preenchido, precisa ser válido
    const cpfDigitos = limparCpf(cpf);
    if (podeCpf && cpfDigitos !== "" && !validarCpf(cpfDigitos)) {
      return setErroForm("CPF inválido. Confira os números digitados.");
    }

    // Instrumento obrigatório nos cargos que exigem (o banco confere de novo)
    const erroInstrumento = validarInstrumentoDoCargo(exigeInstrumento, instrumentoId, cargoSelecionado?.nome);
    if (erroInstrumento) return setErroForm(erroInstrumento);

    // Categoria: obrigatória para Músico e Organista (define em qual linha do painel ao vivo ele conta)
    const erroCategoria = validarCategoria(cargoSelecionado?.nome, categoria);
    if (erroCategoria) return setErroForm(erroCategoria);

    // Fotografia: obrigatória no cadastro novo (o banco também recusa sem ela)
    if (!editando && !foto) {
      return setErroForm("A fotografia é obrigatória. Tire a foto ou escolha uma imagem.");
    }

    setSalvando(true);
    try {
      const dados = {
        nome: nome.trim(),
        congregacao: congregacao.trim(),
        casa_oracao_id: casaOracaoId,
        cidade: cidade.trim() || null,
        bairro: bairro.trim() || null,
        cargo_id: cargoId,
        data_nascimento: dataNascimento || null,
        instrumento_id: permiteInstrumento && instrumentoId ? instrumentoId : null,
        categoria: tipoCategoria && categoria ? categoria : null,
      };
      if (membro) {
        // O CPF só é enviado por quem tem a permissão (os demais nem tocam nele)
        await atualizarMembro(membro.id, {
          ...dados,
          status,
          ...(podeCpf ? { cpf: cpfDigitos || null } : {}),
        });
        // Foto nova (opcional na edição): só quem tem a permissão de fotografia a troca
        if (foto && podeFoto) await enviarFoto(membro.id, foto);
      } else {
        // Cadastro novo: o identificador é gerado aqui para a foto subir para a pasta do próprio membro
        // antes do cadastro; o banco confere o arquivo e recusa o cadastro sem foto.
        const novoId = crypto.randomUUID();
        const caminho = await subirFotoNova(novoId, foto as Blob);
        try {
          await criarMembro({
            ...dados,
            id: novoId,
            foto_path: caminho,
            ...(podeCpf && cpfDigitos ? { cpf: cpfDigitos } : {}),
          });
        } catch (e) {
          await descartarFoto(caminho); // o cadastro falhou: não deixa a foto solta
          throw e;
        }
      }

      await onSalvo(editando ? `Membro "${dados.nome}" atualizado.` : `Membro "${dados.nome}" cadastrado.`);
    } catch (e) {
      setErroForm(e instanceof Error ? e.message : "Não foi possível salvar o membro.");
    } finally {
      setSalvando(false);
    }
  }

  // ---------- Estados da tela ----------

  if (carregando) {
    return <p className="text-sm text-slate-600">Carregando os dados do formulário...</p>;
  }

  if (erro) {
    return (
      <p className="text-sm text-slate-600">
        Formulário indisponível porque os dados não puderam ser carregados.
      </p>
    );
  }

  // Sem cargos: a migração do banco ainda não foi aplicada
  if (cargos.length === 0) {
    return (
      <p className="text-sm text-slate-600">
        Nenhum cargo encontrado. Aplique as migrações do banco (pasta supabase/migrations) para criar
        os cargos padrão.
      </p>
    );
  }

  // Data máxima do campo de nascimento = hoje (formato AAAA-MM-DD, no fuso local)
  const hoje = new Date().toLocaleDateString("sv-SE");

  return (
    <form onSubmit={salvar} className="space-y-5">
      <div className="space-y-1.5">
        <label htmlFor="nome" className="text-sm font-medium text-slate-700">
          Nome completo
        </label>
        <input
          id="nome"
          type="text"
          required
          value={nome}
          onChange={(e) => setNome(e.target.value)}
          disabled={salvando}
          placeholder="Ex: João da Silva"
          className={classeInput}
        />
      </div>

      {/* Fotografia: obrigatória no cadastro novo; na edição só troca quem tem a permissão de fotografia */}
      {!editando || podeFoto ? (
        <div className="space-y-1.5">
          <span className="text-sm font-medium text-slate-700">
            Fotografia{" "}
            {editando ? (
              <span className="font-normal text-slate-400">(opcional na edição)</span>
            ) : (
              <span className="font-normal text-red-600">(obrigatória)</span>
            )}
          </span>
          <CampoFoto
            nome={nome || "Membro"}
            fotoAtualCaminho={membro?.foto_path ?? null}
            obrigatoria={!editando}
            onChange={setFoto}
            disabled={salvando}
          />
        </div>
      ) : (
        <div className="space-y-1.5">
          <span className="text-sm font-medium text-slate-700">Fotografia</span>
          <FotoMembro caminho={membro?.foto_path ?? null} nome={nome || "Membro"} tamanho="grande" />
          <p className="text-xs text-slate-500">A fotografia só pode ser trocada por quem tem a permissão Fotografia.</p>
        </div>
      )}

      {/* CPF: só para quem tem a permissão membros.cpf */}
      {podeCpf && (
        <div className="space-y-1.5">
          <label htmlFor="cpf" className="text-sm font-medium text-slate-700">
            CPF <span className="font-normal text-slate-400">(opcional)</span>
          </label>
          <input
            id="cpf"
            type="text"
            inputMode="numeric"
            autoComplete="off"
            value={cpf}
            onChange={(e) => setCpf(formatarCpf(e.target.value))}
            disabled={salvando}
            placeholder="000.000.000-00"
            className={classeInput}
          />
          <p className="text-xs text-slate-500">
            Evita cadastros duplicados. O CPF é protegido: só quem tem permissão o vê por completo.
          </p>
        </div>
      )}

      <div className="space-y-1.5">
        <label htmlFor="cargo" className="text-sm font-medium text-slate-700">
          Cargo
        </label>
        <select
          id="cargo"
          required
          value={cargoId}
          onChange={(e) => trocarCargo(e.target.value)}
          disabled={salvando}
          className={classeInput}
        >
          <option value="">Selecione um cargo</option>
          {cargos.map((cargo) => (
            <option key={cargo.id} value={cargo.id}>
              {cargo.nome}
            </option>
          ))}
        </select>
      </div>

      {/* Instrumento (lista por família; obrigatório para Músico) + Família automática */}
      {permiteInstrumento && (
        <>
          <div className="space-y-1.5">
            <label htmlFor="instrumento" className="text-sm font-medium text-slate-700">
              Instrumento{" "}
              {exigeInstrumento ? (
                <span className="font-normal text-red-600">(obrigatório)</span>
              ) : (
                <span className="font-normal text-slate-400">(opcional)</span>
              )}
            </label>
            <SeletorInstrumento
              id="instrumento"
              valor={instrumentoId}
              onChange={setInstrumentoId}
              instrumentos={instrumentos}
              familias={familias}
              obrigatorio={exigeInstrumento}
              disabled={salvando}
              className={classeInput}
            />
            <p className="text-xs text-slate-500">
              Escolha na lista. Para incluir um instrumento novo, cadastre em Instrumentos e famílias.
            </p>
          </div>

          <FamiliaAutomatica instrumentoId={instrumentoId} instrumentos={instrumentos} familias={familias} />
        </>
      )}

      {/* Categoria: só para Músico e Organista (define em qual linha do painel ao vivo a pessoa conta) */}
      {tipoCategoria && (
        <div className="space-y-1.5">
          <label htmlFor="categoria" className="text-sm font-medium text-slate-700">
            Categoria <span className="font-normal text-red-600">(obrigatória)</span>
          </label>
          <select
            id="categoria"
            required
            value={categoria}
            onChange={(e) => setCategoria(e.target.value)}
            disabled={salvando}
            className={classeInput}
          >
            <option value="">Selecione a categoria</option>
            {CATEGORIAS.map((item) => (
              <option key={item.valor} value={item.valor}>
                {item[tipoCategoria]}
              </option>
            ))}
          </select>
        </div>
      )}

      {/* Localidade: casa de oração (lista do banco), cidade e bairro (com sugestões dos já cadastrados) */}
      <div className="space-y-1.5">
        <label htmlFor="casa-oracao-membro" className="text-sm font-medium text-slate-700">
          Casa de Oração
        </label>
        <SeletorCasaOracao
          id="casa-oracao-membro"
          valor={casaOracaoId}
          onChange={setCasaOracaoId}
          onEscolherNome={setCongregacao}
          disabled={salvando}
          className={classeInput}
        />
        <p className="text-xs text-slate-500">
          A casa de oração define o setor e quem enxerga o cadastro. Se a casa de oração não aparecer, peça a um administrador para
          cadastrá-la em Administração &gt; Organização.
        </p>
      </div>

      <div className="space-y-1.5">
        <label htmlFor="cidade" className="text-sm font-medium text-slate-700">
          Cidade <span className="font-normal text-slate-400">(opcional)</span>
        </label>
        <input
          id="cidade"
          type="text"
          list="sugestoes-cidade"
          autoComplete="off"
          value={cidade}
          onChange={(e) => setCidade(e.target.value)}
          disabled={salvando}
          placeholder="Ex: Salvador, Lauro de Freitas"
          className={classeInput}
        />
        <datalist id="sugestoes-cidade">
          {sugestoes.cidades.map((valor) => (
            <option key={valor} value={valor} />
          ))}
        </datalist>
      </div>

      <div className="space-y-1.5">
        <label htmlFor="bairro" className="text-sm font-medium text-slate-700">
          Bairro / igreja de referência <span className="font-normal text-slate-400">(opcional)</span>
        </label>
        <input
          id="bairro"
          type="text"
          list="sugestoes-bairro"
          autoComplete="off"
          value={bairro}
          onChange={(e) => setBairro(e.target.value)}
          disabled={salvando}
          placeholder="Ex: Itinga"
          className={classeInput}
        />
        <datalist id="sugestoes-bairro">
          {sugestoes.bairros.map((valor) => (
            <option key={valor} value={valor} />
          ))}
        </datalist>
      </div>

      <div className="space-y-1.5">
        <label htmlFor="data_nascimento" className="text-sm font-medium text-slate-700">
          Data de nascimento <span className="font-normal text-slate-400">(opcional)</span>
        </label>
        <input
          id="data_nascimento"
          type="date"
          max={hoje}
          value={dataNascimento}
          onChange={(e) => setDataNascimento(e.target.value)}
          disabled={salvando}
          className={classeInput}
        />
      </div>

      {/* Situação: só na edição */}
      {editando && (
        <div className="space-y-1.5">
          <label htmlFor="status" className="text-sm font-medium text-slate-700">
            Situação
          </label>
          <select
            id="status"
            value={status}
            onChange={(e) => setStatus(e.target.value)}
            disabled={salvando}
            className={classeInput}
          >
            <option value="Ativo">Ativo</option>
            <option value="Inativo">Inativo</option>
          </select>
        </div>
      )}

      {erroForm && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erroForm}
        </p>
      )}

      <Button type="submit" disabled={salvando} className="w-full">
        {salvando ? "Salvando..." : editando ? "Salvar alterações" : "Cadastrar membro"}
      </Button>
    </form>
  );
}

################ src/components/ListaMembros.tsx
"use client";

/**
 * SysMúsica — Lista de membros em tabela, com barra de ferramentas de filtros.
 *
 * Não acessa o banco: recebe os dados e os estados (loading/erro) da página.
 * Os filtros (busca por nome, cargo e família) são aplicados no navegador sobre a lista recebida.
 * Cruza cada membro com seu cargo e instrumento; a FAMÍLIA do membro é a do seu instrumento
 * (instrumento -> familia_id), então corrigir a família de um instrumento atualiza todos os membros.
 *
 * Responsividade:
 *  - telas >= 768px: colunas Nome, Cargo, Instrumento, Família, Casa de Oração e Status;
 *  - telas menores: apenas Nome (com os demais dados em uma linha abaixo) e Status.
 *
 * Trata quatro cenários: carregando, erro, nenhum membro cadastrado e tabela com dados.
 */
import { useState } from "react";
import { Pencil, QrCode, Trash2 } from "lucide-react";
import BotaoIcone from "@/components/ui/BotaoIcone";
import { normalizarTexto } from "@/lib/texto";
import type { Cargo, Familia, Instrumento, MembroPublico } from "@/types/database";

interface ListaMembrosProps {
  membros: MembroPublico[];
  cargos: Cargo[];
  instrumentos: Instrumento[];
  familias: Familia[];
  carregando: boolean;
  erro: string | null;
  onTentarNovamente: () => void;
  podeEditar: boolean; // mostra o botão Editar em cada linha
  podeExcluir: boolean; // mostra o botão Excluir em cada linha
  onEditar: (membro: MembroPublico) => void;
  onExcluir: (membro: MembroPublico) => void;
  podeQr: boolean; // mostra o botão QR Code em cada linha
  onQr: (membro: MembroPublico) => void;
}

// Valor especial do filtro de família: membros sem instrumento ou com instrumento não classificado
const FILTRO_SEM_FAMILIA = "sem-familia";

// Classe compartilhada pelos cabeçalhos das colunas
const classeCabecalho =
  "px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500";

// Classe compartilhada pelos selects da barra de ferramentas
const classeSelect =
  "w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 shadow-sm focus:border-indigo-500 focus:outline-none focus:ring-2 focus:ring-indigo-200";

export default function ListaMembros({
  membros,
  cargos,
  instrumentos,
  familias,
  carregando,
  erro,
  onTentarNovamente,
  podeEditar,
  podeExcluir,
  onEditar,
  onExcluir,
  podeQr,
  onQr,
}: ListaMembrosProps) {
  const [busca, setBusca] = useState("");
  const [cargoFiltroId, setCargoFiltroId] = useState(""); // "" = todos os cargos
  const [familiaFiltroId, setFamiliaFiltroId] = useState(""); // "" = todas as famílias
  const mostrarAcoes = podeEditar || podeExcluir || podeQr; // a coluna Ações só existe para quem pode algo

  // 1) Carregando: skeleton (barra de ferramentas + linhas)
  if (carregando) {
    return (
      <div
        role="status"
        aria-live="polite"
        className="overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-sm"
      >
        <span className="sr-only">Carregando membros...</span>
        <div className="flex flex-col gap-3 border-b border-slate-200 p-4 sm:flex-row">
          <div className="h-10 flex-1 animate-pulse rounded-lg bg-slate-200" />
          <div className="h-10 animate-pulse rounded-lg bg-slate-200 sm:w-48" />
          <div className="h-10 animate-pulse rounded-lg bg-slate-200 sm:w-44" />
        </div>
        <div className="space-y-3 p-4">
          {[0, 1, 2, 3, 4].map((i) => (
            <div key={i} className="h-10 animate-pulse rounded bg-slate-100" />
          ))}
        </div>
      </div>
    );
  }

  // 2) Erro na comunicação com o banco
  if (erro) {
    return (
      <div role="alert" className="rounded-2xl border border-red-200 bg-red-50 p-6">
        <p className="font-semibold text-red-800">Não foi possível carregar os membros.</p>
        <p className="mt-1 text-sm text-red-700">{erro}</p>
        <button
          type="button"
          onClick={onTentarNovamente}
          className="mt-4 rounded-lg bg-red-600 px-4 py-2 text-sm font-semibold text-white transition hover:bg-red-500"
        >
          Tentar novamente
        </button>
      </div>
    );
  }

  // 3) Nenhum membro cadastrado
  if (membros.length === 0) {
    return (
      <div className="rounded-2xl border border-dashed border-slate-300 bg-white p-10 text-center">
        <p className="font-medium text-slate-700">Nenhum membro cadastrado ainda.</p>
        <p className="mt-1 text-sm text-slate-500">Clique em “Adicionar” para cadastrar o primeiro.</p>
      </div>
    );
  }

  // Índices por id (cruzamento em memória)
  const cargoPorId = new Map(cargos.map((cargo): [string, Cargo] => [cargo.id, cargo]));
  const familiaPorId = new Map(familias.map((familia): [string, Familia] => [familia.id, familia]));
  const instrumentoPorId = new Map(
    instrumentos.map((instrumento): [string, Instrumento] => [instrumento.id, instrumento])
  );

  // Família do membro = família do instrumento dele (null se não tiver instrumento ou classificação)
  const familiaIdDoMembro = (membro: MembroPublico): string | null =>
    membro.instrumento_id
      ? (instrumentoPorId.get(membro.instrumento_id)?.familia_id ?? null)
      : null;

  // Filtros: nome (sem acento/maiúsculas, "contém"), cargo e família (igualdade)
  const termo = normalizarTexto(busca);
  const membrosFiltrados = membros.filter((membro) => {
    const familiaId = familiaIdDoMembro(membro);
    const passaNome = termo === "" || normalizarTexto(membro.nome).includes(termo);
    const passaCargo = cargoFiltroId === "" || membro.cargo_id === cargoFiltroId;
    const passaFamilia =
      familiaFiltroId === "" ||
      (familiaFiltroId === FILTRO_SEM_FAMILIA ? familiaId === null : familiaId === familiaFiltroId);
    return passaNome && passaCargo && passaFamilia;
  });
  const filtrosAtivos = termo !== "" || cargoFiltroId !== "" || familiaFiltroId !== "";

  function limparFiltros() {
    setBusca("");
    setCargoFiltroId("");
    setFamiliaFiltroId("");
  }

  // 4) Tabela com barra de ferramentas
  return (
    <div className="overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-sm">
      {/* Barra de ferramentas: pesquisa por nome + filtro por cargo + filtro por família */}
      <div className="flex flex-col gap-3 border-b border-slate-200 p-4 sm:flex-row sm:items-center">
        <div className="relative min-w-0 flex-1">
          <label htmlFor="busca-nome" className="sr-only">
            Pesquisar por nome
          </label>
          <svg
            className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400"
            viewBox="0 0 20 20"
            fill="currentColor"
            aria-hidden="true"
          >
            <path
              fillRule="evenodd"
              d="M8 4a4 4 0 100 8 4 4 0 000-8zM2 8a6 6 0 1110.89 3.476l4.817 4.817a1 1 0 01-1.414 1.414l-4.816-4.816A6 6 0 012 8z"
              clipRule="evenodd"
            />
          </svg>
          <input
            id="busca-nome"
            type="search"
            autoComplete="off"
            value={busca}
            onChange={(e) => setBusca(e.target.value)}
            placeholder="Pesquisar por nome..."
            className="w-full rounded-lg border border-slate-300 bg-white py-2 pl-9 pr-3 text-sm text-slate-900 placeholder:text-slate-400 shadow-sm focus:border-indigo-500 focus:outline-none focus:ring-2 focus:ring-indigo-200"
          />
        </div>

        <div className="sm:w-48">
          <label htmlFor="filtro-cargo" className="sr-only">
            Filtrar por cargo
          </label>
          <select
            id="filtro-cargo"
            value={cargoFiltroId}
            onChange={(e) => setCargoFiltroId(e.target.value)}
            className={classeSelect}
          >
            <option value="">Todos os cargos</option>
            {cargos.map((cargo) => (
              <option key={cargo.id} value={cargo.id}>
                {cargo.nome}
              </option>
            ))}
          </select>
        </div>

        <div className="sm:w-44">
          <label htmlFor="filtro-familia" className="sr-only">
            Filtrar por família do instrumento
          </label>
          <select
            id="filtro-familia"
            value={familiaFiltroId}
            onChange={(e) => setFamiliaFiltroId(e.target.value)}
            className={classeSelect}
          >
            <option value="">Todas as famílias</option>
            {familias.map((familia) => (
              <option key={familia.id} value={familia.id}>
                {familia.nome}
              </option>
            ))}
            <option value={FILTRO_SEM_FAMILIA}>Sem família</option>
          </select>
        </div>
      </div>

      {/* Tabela de dados (rola na horizontal só se faltar espaço) */}
      <div className="overflow-x-auto">
        <table className="min-w-full divide-y divide-slate-200 text-sm">
          <thead className="bg-slate-50">
            <tr>
              <th scope="col" className={classeCabecalho}>
                Nome
              </th>
              <th scope="col" className={classeCabecalho + " hidden md:table-cell"}>
                Cargo
              </th>
              <th scope="col" className={classeCabecalho + " hidden md:table-cell"}>
                Instrumento
              </th>
              <th scope="col" className={classeCabecalho + " hidden md:table-cell"}>
                Família
              </th>
              <th scope="col" className={classeCabecalho + " hidden md:table-cell"}>
                Casa de Oração
              </th>
              <th scope="col" className={classeCabecalho}>
                Status
              </th>
              {mostrarAcoes && (
                <th scope="col" className={classeCabecalho + " text-right"}>
                  Ações
                </th>
              )}
            </tr>
          </thead>

          <tbody className="divide-y divide-slate-100">
            {membrosFiltrados.length === 0 ? (
              // Filtros sem resultado
              <tr>
                <td colSpan={mostrarAcoes ? 7 : 6} className="px-4 py-10 text-center text-sm text-slate-500">
                  Nenhum membro encontrado com os filtros atuais.
                </td>
              </tr>
            ) : (
              membrosFiltrados.map((membro) => {
                const nomeCargo = cargoPorId.get(membro.cargo_id)?.nome ?? "Cargo não encontrado";
                const nomeInstrumento = membro.instrumento_id
                  ? (instrumentoPorId.get(membro.instrumento_id)?.nome ?? null)
                  : null;
                const familiaId = familiaIdDoMembro(membro);
                const nomeFamilia = familiaId ? (familiaPorId.get(familiaId)?.nome ?? null) : null;

                return (
                  <tr key={membro.id} className="transition hover:bg-slate-50">
                    <td className="px-4 py-3">
                      <div className="flex items-center gap-3">
                        {/* Avatar com a inicial do nome */}
                        <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-indigo-100 text-sm font-bold text-indigo-700">
                          {membro.nome.charAt(0).toUpperCase()}
                        </div>
                        <div className="min-w-0">
                          <p className="truncate font-medium text-slate-900">{membro.nome}</p>
                          {/* Telas pequenas: os dados das colunas ocultas aparecem aqui */}
                          <p className="truncate text-xs text-slate-500 md:hidden">
                            {[nomeCargo, nomeInstrumento, nomeFamilia, membro.congregacao]
                              .filter(Boolean)
                              .join(" · ")}
                          </p>
                        </div>
                      </div>
                    </td>
                    <td className="hidden px-4 py-3 text-slate-700 md:table-cell">{nomeCargo}</td>
                    <td className="hidden px-4 py-3 text-slate-700 md:table-cell">
                      {nomeInstrumento ?? "—"}
                    </td>
                    <td className="hidden px-4 py-3 md:table-cell">
                      {nomeFamilia ? (
                        <span className="rounded-full bg-indigo-50 px-2 py-0.5 text-xs font-medium text-indigo-700">
                          {nomeFamilia}
                        </span>
                      ) : (
                        <span className="text-slate-400">—</span>
                      )}
                    </td>
                    <td className="hidden px-4 py-3 text-slate-700 md:table-cell">
                      {membro.congregacao}
                      {(membro.bairro || membro.cidade) && (
                        <span className="block text-xs text-slate-500">
                          {[membro.bairro, membro.cidade].filter(Boolean).join(" · ")}
                        </span>
                      )}
                    </td>
                    <td className="px-4 py-3">
                      <span
                        className={
                          "rounded-full px-2 py-0.5 text-xs font-medium " +
                          (membro.status === "Ativo"
                            ? "bg-emerald-100 text-emerald-700"
                            : "bg-slate-100 text-slate-600")
                        }
                      >
                        {membro.status}
                      </span>
                    </td>
                    {mostrarAcoes && (
                      <td className="px-4 py-3">
                        <div className="flex justify-end gap-1">
                          {podeQr && (
                            <BotaoIcone
                              icone={QrCode}
                              rotulo={`QR Code de ${membro.nome}`}
                              onClick={() => onQr(membro)}
                            />
                          )}
                          {podeEditar && (
                            <BotaoIcone
                              icone={Pencil}
                              rotulo={`Editar ${membro.nome}`}
                              onClick={() => onEditar(membro)}
                            />
                          )}
                          {podeExcluir && (
                            <BotaoIcone
                              icone={Trash2}
                              variante="perigo"
                              rotulo={`Excluir ${membro.nome}`}
                              onClick={() => onExcluir(membro)}
                            />
                          )}
                        </div>
                      </td>
                    )}
                  </tr>
                );
              })
            )}
          </tbody>
        </table>
      </div>

      {/* Rodapé: contagem e atalho para limpar os filtros */}
      <div className="flex items-center justify-between gap-2 border-t border-slate-200 bg-slate-50 px-4 py-3 text-xs text-slate-500">
        <span>
          Exibindo {membrosFiltrados.length} de {membros.length} membro(s)
        </span>
        {filtrosAtivos && (
          <button
            type="button"
            onClick={limparFiltros}
            className="font-semibold text-indigo-600 hover:underline"
          >
            Limpar filtros
          </button>
        )}
      </div>
    </div>
  );
}

################ src/modules/membros/components/CampoFoto.tsx
"use client";

/**
 * SysMúsica — Campo de fotografia: "Tirar foto" e "Escolher da galeria", com pré-visualização.
 * No celular, "Tirar foto" abre a câmera do aparelho; no computador abre o seletor de arquivos.
 * A imagem é reduzida e convertida para JPEG no navegador (leve e rápida de enviar). O arquivo só é
 * enviado quando o formulário que usa este campo salvar. Sem foto nova, mostra a foto atual (se houver).
 */
import { useEffect, useRef, useState, type ChangeEvent } from "react";
import { Camera, ImagePlus, Trash2 } from "lucide-react";
import { prepararFoto } from "../fotoImagem";
import FotoMembro from "./FotoMembro";

interface CampoFotoProps {
  nome: string; // para o texto alternativo da imagem
  fotoAtualCaminho?: string | null; // foto já cadastrada (edição)
  obrigatoria?: boolean;
  onChange: (foto: Blob | null) => void; // a foto nova já preparada, ou nulo ao remover a escolha
  disabled?: boolean;
}

const classeBotao =
  "inline-flex items-center gap-2 rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm font-medium " +
  "text-slate-700 transition hover:bg-slate-50 disabled:cursor-not-allowed disabled:opacity-60";

export default function CampoFoto({
  nome,
  fotoAtualCaminho = null,
  obrigatoria = false,
  onChange,
  disabled = false,
}: CampoFotoProps) {
  const cameraRef = useRef<HTMLInputElement>(null);
  const galeriaRef = useRef<HTMLInputElement>(null);
  const [previa, setPrevia] = useState<string | null>(null); // endereço temporário da foto nova
  const [processando, setProcessando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  // Libera o endereço temporário da pré-visualização quando ela troca ou o campo sai da tela
  useEffect(() => {
    return () => {
      if (previa) URL.revokeObjectURL(previa);
    };
  }, [previa]);

  async function aoEscolher(evento: ChangeEvent<HTMLInputElement>) {
    const arquivo = evento.target.files?.[0];
    evento.target.value = ""; // permite escolher o mesmo arquivo de novo
    if (!arquivo) return;

    setErro(null);
    setProcessando(true);
    try {
      const imagem = await prepararFoto(arquivo);
      setPrevia(URL.createObjectURL(imagem));
      onChange(imagem);
    } catch (e) {
      setErro(e instanceof Error ? e.message : "Não foi possível preparar a imagem.");
    } finally {
      setProcessando(false);
    }
  }

  function removerEscolha() {
    setPrevia(null);
    onChange(null);
  }

  const bloqueado = disabled || processando;

  return (
    <div className="space-y-3">
      {/* Câmera (celular) e galeria: seletores escondidos, acionados pelos botões abaixo */}
      <input
        ref={cameraRef}
        type="file"
        accept="image/*"
        capture="environment"
        className="sr-only"
        tabIndex={-1}
        aria-hidden="true"
        onChange={(e) => void aoEscolher(e)}
      />
      <input
        ref={galeriaRef}
        type="file"
        accept="image/jpeg,image/png,image/webp"
        className="sr-only"
        tabIndex={-1}
        aria-hidden="true"
        onChange={(e) => void aoEscolher(e)}
      />

      <div className="flex items-start gap-4">
        {previa ? (
          <div
            role="img"
            aria-label={`Nova fotografia de ${nome}`}
            style={{ backgroundImage: `url(${previa})` }}
            className="h-44 w-36 shrink-0 rounded-xl bg-slate-100 bg-cover bg-center"
          />
        ) : (
          <FotoMembro caminho={fotoAtualCaminho} nome={nome} tamanho="grande" />
        )}

        <div className="flex flex-col gap-2">
          <button type="button" className={classeBotao} onClick={() => cameraRef.current?.click()} disabled={bloqueado}>
            <Camera className="h-4 w-4" aria-hidden="true" />
            Tirar foto
          </button>
          <button type="button" className={classeBotao} onClick={() => galeriaRef.current?.click()} disabled={bloqueado}>
            <ImagePlus className="h-4 w-4" aria-hidden="true" />
            Escolher da galeria
          </button>
          {previa && (
            <button type="button" className={classeBotao} onClick={removerEscolha} disabled={bloqueado}>
              <Trash2 className="h-4 w-4" aria-hidden="true" />
              Remover a nova foto
            </button>
          )}
        </div>
      </div>

      {processando && <p className="text-xs text-slate-500">Preparando a imagem...</p>}
      {erro && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erro}
        </p>
      )}
      <p className="text-xs text-slate-500">
        {obrigatoria ? "A fotografia é obrigatória: ela identifica a pessoa na portaria. " : ""}
        Foto de rosto, de frente e com boa luz. A imagem é reduzida antes de ser enviada.
      </p>
    </div>
  );
}

/**
 * SysMúsica — Fotografia do membro (miniatura ou retrato grande).
 * Busca a URL temporária da foto; sem foto (ou se falhar), mostra um espaço reservado.
 * Usa um fundo de imagem (e não <img>) com texto alternativo para leitores de tela.
 */
import { useEffect, useState } from "react";
import { User } from "lucide-react";
import { cn } from "@/lib/cn";
import { urlDaFoto } from "../services/fotos";

interface FotoMembroProps {
  caminho: string | null;
  nome: string;
  tamanho?: "pequena" | "grande";
}

export default function FotoMembro({ caminho, nome, tamanho = "pequena" }: FotoMembroProps) {
  // Guarda junto o caminho a que a URL pertence (evita mostrar a foto de outro membro por um instante)
  const [carregada, setCarregada] = useState<{ caminho: string; url: string | null } | null>(null);

  /* eslint-disable react-hooks/set-state-in-effect */
  useEffect(() => {
    if (!caminho) return;
    let cancelado = false;
    void urlDaFoto(caminho).then((url) => {
      if (!cancelado) setCarregada({ caminho, url });
    });
    return () => {
      cancelado = true;
    };
  }, [caminho]);
  /* eslint-enable react-hooks/set-state-in-effect */

  const url = caminho && carregada?.caminho === caminho ? carregada.url : null;
  const medidas = tamanho === "grande" ? "h-44 w-36" : "h-12 w-10";

  return (
    <div
      role="img"
      aria-label={url ? `Fotografia de ${nome}` : `${nome}: sem fotografia`}
      style={url ? { backgroundImage: `url(${url})` } : undefined}
      className={cn(
        "flex shrink-0 items-center justify-center overflow-hidden rounded-xl bg-slate-100 bg-cover bg-center text-slate-400",
        medidas
      )}
    >
      {!url && <User className={tamanho === "grande" ? "h-14 w-14" : "h-5 w-5"} aria-hidden="true" />}
    </div>
  );
}

    return passaFiltro && (termo === "" || texto.includes(termo));

            const ocupado = ocupadoId === membro.id;
            return (

                  aria-label={url ? `Fotografia de ${membro.nome}` : `${membro.nome}: sem fotografia`}
                  style={url ? { backgroundImage: `url(${url})` } : undefined}
                  className="flex h-14 w-12 shrink-0 items-center justify-center overflow-hidden rounded-lg bg-slate-100 bg-cover bg-center text-slate-400"

                  {!url && <User className="h-5 w-5" aria-hidden="true" />}

/**
 * SysMúsica — Preparo da fotografia no navegador antes do envio.
 * Reduz a imagem (lado maior de 800 px) e converte para JPEG: a foto fica leve (poucas centenas de KB)
 * e o envio é rápido, inclusive em rede móvel. A conferência de tipo e tamanho final também é feita
 * pelo armazenamento do banco (limite de 2 MB; JPEG, PNG ou WebP).
 */

const LADO_MAXIMO = 800;
const QUALIDADE_JPEG = 0.82;
const TAMANHO_MAXIMO_ORIGINAL = 15 * 1024 * 1024; // 15 MB de entrada (a saída é bem menor)

export async function prepararFoto(arquivo: File): Promise<Blob> {
  if (!arquivo.type.startsWith("image/")) {
    throw new Error("Escolha um arquivo de imagem (JPG, PNG ou WebP).");
  }
  if (arquivo.size > TAMANHO_MAXIMO_ORIGINAL) {
    throw new Error("A imagem é grande demais (máximo de 15 MB).");
  }

  // "from-image" respeita a rotação gravada pela câmera do celular; se o navegador não aceitar a opção, tenta sem ela
  let bitmap: ImageBitmap;
  try {
    bitmap = await createImageBitmap(arquivo, { imageOrientation: "from-image" });
  } catch {
    try {
      bitmap = await createImageBitmap(arquivo);
    } catch {
      throw new Error("Não foi possível ler esta imagem. Tente outra foto.");
    }
  }

  const escala = Math.min(1, LADO_MAXIMO / Math.max(bitmap.width, bitmap.height));
  const largura = Math.max(1, Math.round(bitmap.width * escala));
  const altura = Math.max(1, Math.round(bitmap.height * escala));

  const canvas = document.createElement("canvas");
  canvas.width = largura;
  canvas.height = altura;
  const contexto = canvas.getContext("2d");
  if (!contexto) {
    bitmap.close();
    throw new Error("Não foi possível processar a imagem neste navegador.");
  }
  contexto.drawImage(bitmap, 0, 0, largura, altura);
  bitmap.close();

  const imagem = await new Promise<Blob | null>((resolver) =>
    canvas.toBlob(resolver, "image/jpeg", QUALIDADE_JPEG)
  );
  if (!imagem) throw new Error("Não foi possível processar a imagem.");
  return imagem;
}

/**
 * SysMúsica — Serviço das fotografias dos membros.
 * As fotos ficam em um bucket PRIVADO: para exibir, gera-se uma URL temporária (assinada).
 * Quem pode enviar, trocar ou remover é controlado pela permissão membros.foto (banco + armazenamento).
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";

export const BUCKET_FOTOS = "fotos-membros";

export interface MembroFoto {
  id: string;
  nome: string;
  congregacao: string | null;
  foto_path: string | null;
}

/** Membros (não excluídos) com o caminho da foto, para a tela de fotografias. */
export async function listarMembrosParaFoto(): Promise<MembroFoto[]> {
  const { data, error } = await supabase.rpc("listar_membros_para_foto");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar os membros."));
  return (data ?? []) as MembroFoto[];
}

// Guarda as URLs temporárias por alguns minutos para não pedir a mesma foto várias vezes
const cacheUrls = new Map<string, { url: string; expira: number }>();
const VALIDADE_URL_SEGUNDOS = 600;

/** URL temporária de uma foto (ou null se não for possível gerar). */
export async function urlDaFoto(caminho: string): Promise<string | null> {
  const agora = Date.now();
  const guardada = cacheUrls.get(caminho);
  if (guardada && guardada.expira > agora) return guardada.url;

  const { data, error } = await supabase.storage
    .from(BUCKET_FOTOS)
    .createSignedUrl(caminho, VALIDADE_URL_SEGUNDOS);
  if (error || !data) return null;

  cacheUrls.set(caminho, { url: data.signedUrl, expira: agora + (VALIDADE_URL_SEGUNDOS - 60) * 1000 });
  return data.signedUrl;
}

/** URLs temporárias de várias fotos de uma vez: { caminho: url }. */
export async function urlsDasFotos(caminhos: string[]): Promise<Record<string, string>> {
  if (caminhos.length === 0) return {};
  const { data, error } = await supabase.storage
    .from(BUCKET_FOTOS)
    .createSignedUrls(caminhos, VALIDADE_URL_SEGUNDOS);
  if (error || !data) return {};

  const mapa: Record<string, string> = {};
  for (const item of data) {
    if (item.path && item.signedUrl) mapa[item.path] = item.signedUrl;
  }
  return mapa;
}

/** Envia a foto (já reduzida) e a vincula ao membro; apaga o arquivo anterior. */
export async function enviarFoto(membroId: string, imagem: Blob): Promise<void> {
  const caminho = `${membroId}/${crypto.randomUUID()}.jpg`;

  const { error: erroEnvio } = await supabase.storage
    .from(BUCKET_FOTOS)
    .upload(caminho, imagem, { contentType: "image/jpeg", upsert: false });
  if (erroEnvio) {
    throw new Error("Não foi possível enviar a fotografia. Verifique a conexão e tente novamente.");
  }

  const { data, error } = await supabase.rpc("definir_foto_membro", { p_membro_id: membroId, p_path: caminho });
  if (error) {
    // Desfaz o envio: não deixa arquivo solto no armazenamento
    await supabase.storage.from(BUCKET_FOTOS).remove([caminho]);
    throw new Error(mensagemDoErro(error, "Não foi possível salvar a fotografia."));
  }

  // Arquivo antigo: apagar é um cuidado extra; se falhar, a foto nova já está valendo
  const anterior = data as string | null;
  if (anterior) await supabase.storage.from(BUCKET_FOTOS).remove([anterior]);
}

/**
 * Envia a foto para a pasta do membro e devolve o caminho, SEM ligar a foto a ninguém ainda.
 * Usado no cadastro novo (membro e visitante): a foto sobe antes e o caminho vai junto com o cadastro,
 * onde o banco confere se o arquivo existe e se está na pasta certa.
 */
export async function subirFotoNova(membroId: string, imagem: Blob): Promise<string> {
  const caminho = `${membroId}/${crypto.randomUUID()}.jpg`;
  const { error } = await supabase.storage
    .from(BUCKET_FOTOS)
    .upload(caminho, imagem, { contentType: "image/jpeg", upsert: false });
  if (error) {
    throw new Error("Não foi possível enviar a fotografia. Verifique a conexão e tente novamente.");
  }
  return caminho;
}

/**
 * Tenta apagar uma foto enviada que não chegou a ser usada (o cadastro falhou). Se o usuário não tiver
 * permissão para apagar, o arquivo apenas fica sem uso no armazenamento privado.
 */
export async function descartarFoto(caminho: string): Promise<void> {
  await supabase.storage.from(BUCKET_FOTOS).remove([caminho]);
}

/** Remove a fotografia do membro (referência no banco e arquivo). */
export async function removerFoto(membroId: string): Promise<void> {
  const { data, error } = await supabase.rpc("remover_foto_membro", { p_membro_id: membroId });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível remover a fotografia."));

  if (anterior) await supabase.storage.from(BUCKET_FOTOS).remove([anterior]);
}

/**

 * Em "membros" só se lêem colunas liberadas (nunca "*": o qr_secret e o CPF completo são
 * protegidos). O CPF completo e o código do QR Code vêm de funções do banco que conferem a
 * permissão e registram a consulta na auditoria.
 * Excluir é EXCLUSÃO LÓGICA: o membro some das telas e o histórico de presenças é mantido.

import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import {
  COLUNAS_MEMBRO_PUBLICO,
  type Cargo,
  type Familia,
  type Instrumento,
  type MembroInsert,
  type MembroPublico,
} from "@/types/database";

const MSG_CPF_DUPLICADO = "Já existe um membro cadastrado com este CPF.";

export interface DadosMembros {
  cargos: Cargo[];
  instrumentos: Instrumento[];
  familias: Familia[];
  membros: MembroPublico[];
}

/** Busca cargos, instrumentos, famílias e membros em paralelo. Lança erro se qualquer consulta falhar. */
export async function buscarDadosMembros(): Promise<DadosMembros> {
  const [resCargos, resInstrumentos, resFamilias, resMembros] = await Promise.all([
    supabase.from("cargos").select("*").order("nome", { ascending: true }),
    supabase.from("instrumentos").select("*").order("nome", { ascending: true }),
    supabase.from("familias").select("*").order("nome", { ascending: true }),
    supabase.from("membros").select(COLUNAS_MEMBRO_PUBLICO).order("nome", { ascending: true }),
  ]);

  const erroBanco = resCargos.error ?? resInstrumentos.error ?? resFamilias.error ?? resMembros.error;
  if (erroBanco) throw new Error(erroBanco.message);

  return {
    cargos: (resCargos.data ?? []) as Cargo[],
    instrumentos: (resInstrumentos.data ?? []) as Instrumento[],
    familias: (resFamilias.data ?? []) as Familia[],
    // Cast duplo: com lista de colunas, o cliente (sem tipagem gerada) não infere a linha
    membros: (resMembros.data ?? []) as unknown as MembroPublico[],
  };
}

export async function criarMembro(dados: MembroInsert): Promise<void> {

  if (error) {
    throw new Error(mensagemDoErro(error, "Não foi possível cadastrar o membro.", MSG_CPF_DUPLICADO));
  }
}

  cpf?: string | null;
}

export async function atualizarMembro(id: string, dados: DadosAtualizacaoMembro): Promise<void> {
  // .select("id") devolve as linhas alteradas: se o RLS negar, o update não dá erro, só não altera nada
  const { data, error } = await supabase.from("membros").update(dados).eq("id", id).select("id");
  if (error) {
    throw new Error(
      mensagemDoErro(error, "Não foi possível salvar as alterações do membro.", MSG_CPF_DUPLICADO)
    );
  }
  if (!data || data.length === 0) {
    throw new Error("Não foi possível salvar: você não tem permissão ou o membro não existe mais.");
  }
}

  }

  const [alvoExclusao, setAlvoExclusao] = useState<MembroPublico | null>(null);
  const [excluindo, setExcluindo] = useState(false);
  const [erroExclusao, setErroExclusao] = useState<string | null>(null);
  const [aviso, setAviso] = useState<string | null>(null);

  // Busca cargos, instrumentos, famílias e membros
  const carregar = useCallback(async () => {
    try {
      const dados = await buscarDadosMembros();
      setCargos(dados.cargos);
      setInstrumentos(dados.instrumentos);
      setFamilias(dados.familias);
      setMembros(dados.membros);
      setErro(null);
    } catch (e) {
      // O detalhe técnico vai só para o console; a tela mostra uma mensagem simples
      console.error("Falha ao carregar membros:", e);
      setErro("Não foi possível carregar os dados. Verifique a conexão e tente novamente.");
    } finally {
      setCarregando(false);
    }
  }, []);

  // Primeira carga ao abrir a página
  useEffect(() => {
    // O setState só roda DEPOIS do await (nunca de forma síncrona), então o aviso da regra
    // set-state-in-effect não se aplica a esta carga inicial. Ignorado apenas nesta linha.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    void carregar();
  }, [carregar]);

  // Botão "Tentar novamente": volta ao estado de loading e busca de novo
  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  // Depois de salvar: recarrega a lista, fecha o painel e mostra o aviso de sucesso
  async function aoSalvar(mensagem: string) {
    await carregar();
    setPainel(null);
    setAviso(mensagem);
  }

  // Abre a edição já com o CPF completo (a consulta fica registrada na auditoria)
  async function abrirEdicao(membro: MembroPublico) {
    if (abrindo) return;
    setErroAcao(null);
    setAbrindo(true);
    try {
      const cpfInicial = pode("membros.cpf") ? await obterCpfMembro(membro.id) : null;
      setPainel({ tipo: "editar", membro, cpfInicial });
    } catch (e) {
      setErroAcao(e instanceof Error ? e.message : "Não foi possível abrir a edição.");
    } finally {
      setAbrindo(false);
    }
  }

  // Busca o código do QR Code no servidor (fica registrado na auditoria) e gera a imagem
  async function montarPainelQr(membro: MembroPublico, codigo: string): Promise<PainelMembro> {
    const conteudo = montarConteudoQr(codigo);
    const imagem = await gerarImagemQr(conteudo);
    return { tipo: "qr", membro, imagem, conteudo };
  }

  async function abrirQr(membro: MembroPublico) {
    if (abrindo) return;
    setErroAcao(null);
    setAbrindo(true);
    try {
      setPainel(await montarPainelQr(membro, await obterQrMembro(membro.id)));
    } catch (e) {
      setErroAcao(e instanceof Error ? e.message : "Não foi possível abrir o QR Code.");
    } finally {
      setAbrindo(false);
    }
  }

  // Substitui o QR Code (o código antigo deixa de funcionar) e mostra a nova imagem
  async function substituirQr(membro: MembroPublico, justificativa: string) {
    const novoCodigo = await regenerarQrMembro(membro.id, justificativa);
    setPainel(await montarPainelQr(membro, novoCodigo));
    setAviso(`QR Code de "${membro.nome}" substituído. O código anterior deixou de funcionar.`);
  }

  // Confirma a exclusão (lógica) do membro escolhido
  async function confirmarExclusao() {
    if (!alvoExclusao) return;
    setExcluindo(true);
    setErroExclusao(null);
    try {
      await excluirMembro(alvoExclusao.id);
      await carregar();
      setAviso(`Membro "${alvoExclusao.nome}" excluído.`);
      setAlvoExclusao(null);
    } catch (e) {
      setErroExclusao(e instanceof Error ? e.message : "Não foi possível excluir o membro.");
    } finally {
      setExcluindo(false);
    }
  }

  // Sugestões dos campos de localidade (valores já cadastrados)
  const sugestoes = {
    congregacoes: valoresDistintos(membros.map((membro) => membro.congregacao)),
    cidades: valoresDistintos(membros.map((membro) => membro.cidade)),
    bairros: valoresDistintos(membros.map((membro) => membro.bairro)),
  };

  const tituloPainel =
    painel?.tipo === "editar" ? "Editar membro" : painel?.tipo === "qr" ? "QR Code do membro" : "Novo membro";

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Músicos / Organistas"
          descricao={`Cadastre e acompanhe os membros do SysMúsica.${
            !carregando && !erro ? ` ${membros.length} cadastrado(s).` : ""
          }`}
          acoes={
            pode("membros.criar") ? (
              <Button icone={Plus} onClick={() => setPainel({ tipo: "novo" })} disabled={carregando || !!erro}>
                Adicionar
              </Button>
            ) : undefined
          }
        />

        {erroAcao && (
          <p role="alert" className="mb-4 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
            {erroAcao}
          </p>
        )}

        <ListaMembros
          membros={membros}
          cargos={cargos}
          instrumentos={instrumentos}
          familias={familias}
          carregando={carregando}
          erro={erro}
          onTentarNovamente={tentarNovamente}
          podeEditar={pode("membros.editar")}
          podeExcluir={pode("membros.excluir")}
          podeQr={pode("membros.qr")}
          onEditar={(membro) => void abrirEdicao(membro)}
          onQr={(membro) => void abrirQr(membro)}
          onExcluir={(membro) => {
            setErroExclusao(null);
            setAlvoExclusao(membro);
          }}
        />
      </div>

      {/* Painel lateral: novo membro, edição ou QR Code */}
      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={tituloPainel}
        descricao={
          painel?.tipo === "qr" ? "Identificação individual e permanente do membro." : "Preencha os dados do membro."
        }
      >
        {(painel?.tipo === "novo" || painel?.tipo === "editar") && (
          <FormularioMembro
            membro={painel.tipo === "editar" ? painel.membro : null}
            cpfInicial={painel.tipo === "editar" ? painel.cpfInicial : null}
            podeCpf={pode("membros.cpf")}
            podeFoto={pode("membros.foto")}
            cargos={cargos}
            instrumentos={instrumentos}
            familias={familias}
            sugestoes={sugestoes}
            carregando={carregando}
            erro={erro}
            onSalvo={aoSalvar}
          />
        )}
        {painel?.tipo === "qr" && (
          <QrMembro
            nome={painel.membro.nome}
            congregacao={painel.membro.congregacao}
            imagem={painel.imagem}
            conteudo={painel.conteudo}
            podeSubstituir={pode("membros.qr_revogar")}
            onSubstituir={(justificativa) => substituirQr(painel.membro, justificativa)}
          />
        )}
      </Drawer>

      {/* Confirmação de exclusão */}
      <ConfirmDialog
        aberto={alvoExclusao !== null}
        titulo="Excluir membro?"
        mensagem={
          <>
            <p>
              <strong>{alvoExclusao?.nome}</strong> deixará de aparecer nas telas. O histórico de
              presenças dele é mantido. Para apenas impedir novos registros sem excluir, altere a situação
              cadastral para Inativo na edição.
            </p>
            {erroExclusao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroExclusao}
              </p>
            )}
          </>
        }
        textoConfirmar="Excluir"
        processando={excluindo}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExclusao(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

import { useCallback, useEffect, useState } from "react";
import ConfirmDialog from "@/components/ui/ConfirmDialog";
import ErrorState from "@/components/ui/ErrorState";
import Notificacao from "@/components/ui/Notificacao";
import PageHeader from "@/components/ui/PageHeader";
import Skeleton from "@/components/ui/Skeleton";

import { prepararFoto } from "@/modules/membros/fotoImagem";
import {
  enviarFoto,
  listarMembrosParaFoto,
  removerFoto,
  urlsDasFotos,
  type MembroFoto,

export default function FotografiasPage() {
  const [membros, setMembros] = useState<MembroFoto[]>([]);
  const [urls, setUrls] = useState<Record<string, string>>({});
  const [carregando, setCarregando] = useState(true); // inicia true: primeira busca no mount
  const [erro, setErro] = useState<string | null>(null);
  const [ocupadoId, setOcupadoId] = useState<string | null>(null);

  const [aviso, setAviso] = useState<string | null>(null);

  // Busca os membros e as URLs temporárias das fotos já cadastradas
  const carregar = useCallback(async () => {
    try {
      const lista = await listarMembrosParaFoto();
      const mapa = await urlsDasFotos(lista.flatMap((membro) => (membro.foto_path ? [membro.foto_path] : [])));
      setMembros(lista);
      setUrls(mapa);
      setErro(null);
    } catch (e) {
      console.error("Falha ao carregar as fotografias:", e);
      setErro(e instanceof Error ? e.message : "Não foi possível carregar os membros.");
    } finally {
      setCarregando(false);
    }

    void carregar();

    setCarregando(true);
    void carregar();

    setErroAcao(null);

      const imagem = await prepararFoto(arquivo);
      await enviarFoto(membro.id, imagem);

      setAviso(`Fotografia de ${membro.nome} atualizada.`);

      await removerFoto(alvoRemocao.id);

        ) : (
          <GaleriaFotos
            membros={membros}
            urls={urls}
            ocupadoId={ocupadoId}
            onEscolherArquivo={(membro, arquivo) => void aoEscolherArquivo(membro, arquivo)}
            onRemover={setAlvoRemocao}
          />
        )}
      </div>

      <ConfirmDialog
        aberto={alvoRemocao !== null}
        titulo="Remover fotografia?"
        mensagem={
          <p>

          </p>
        }
        textoConfirmar="Remover"
        processando={removendo}
        onConfirmar={() => void confirmarRemocao()}
        onCancelar={() => setAlvoRemocao(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}
