################ src/app/(painel)/instrumentos/page.tsx
"use client";

/**
 * SysMúsica — Página /instrumentos: tabela de instrumentos e famílias com novo, editar e excluir.
 * Cada botão só aparece para quem tem a permissão (grupo do usuário); o banco confere de novo.
 * Os instrumentos também nascem no cadastro de membros (digitando um instrumento novo).
 */
import { useCallback, useEffect, useState } from "react";
import { Plus } from "lucide-react";
import {
  buscarInstrumentosEFamilias,
  excluirInstrumento,
} from "@/modules/instrumentos/services/instrumentos";
import FormularioInstrumento from "@/modules/instrumentos/components/FormularioInstrumento";
import TabelaInstrumentos from "@/modules/instrumentos/components/TabelaInstrumentos";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";
import Button from "@/components/ui/Button";
import ConfirmDialog from "@/components/ui/ConfirmDialog";
import Drawer from "@/components/ui/Drawer";
import EmptyState from "@/components/ui/EmptyState";
import ErrorState from "@/components/ui/ErrorState";
import Notificacao from "@/components/ui/Notificacao";
import PageHeader from "@/components/ui/PageHeader";
import Skeleton from "@/components/ui/Skeleton";
import type { Familia, Instrumento } from "@/types/database";

// Qual painel lateral está aberto: novo instrumento, ou edição de um instrumento
type PainelInstrumento = { tipo: "novo" } | { tipo: "editar"; instrumento: Instrumento };

export default function InstrumentosPage() {
  const { pode } = useSessao();
  const [instrumentos, setInstrumentos] = useState<Instrumento[]>([]);
  const [familias, setFamilias] = useState<Familia[]>([]);
  const [carregando, setCarregando] = useState(true); // inicia true: primeira busca no mount
  const [erro, setErro] = useState<string | null>(null);
  const [painel, setPainel] = useState<PainelInstrumento | null>(null);
  const [alvoExclusao, setAlvoExclusao] = useState<Instrumento | null>(null);
  const [excluindo, setExcluindo] = useState(false);
  const [erroExclusao, setErroExclusao] = useState<string | null>(null);
  const [aviso, setAviso] = useState<string | null>(null);

  // Busca instrumentos e famílias
  const carregar = useCallback(async () => {
    try {
      const dados = await buscarInstrumentosEFamilias();
      setInstrumentos(dados.instrumentos);
      setFamilias(dados.familias);
      setErro(null);
    } catch (e) {
      // O detalhe técnico vai só para o console; a tela mostra uma mensagem simples
      console.error("Falha ao carregar instrumentos:", e);
      setErro("Não foi possível carregar os instrumentos. Verifique a conexão e tente novamente.");
    } finally {
      setCarregando(false);
    }
  }, []);

  // Primeira carga ao abrir a página
  useEffect(() => {
    // O setState só roda DEPOIS do await (nunca de forma síncrona), então o aviso da regra
    // set-state-in-effect não se aplica a esta carga inicial. Ignorado apenas nesta linha.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    void carregar();
  }, [carregar]);

  // Botão "Tentar novamente": volta ao estado de loading e busca de novo
  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  // Depois de salvar: recarrega a lista, fecha o painel e mostra o aviso de sucesso
  async function aoSalvar(mensagem: string) {
    await carregar();
    setPainel(null);
    setAviso(mensagem);
  }

  // Confirma a exclusão do instrumento escolhido (o banco recusa se algum membro ainda usar)
  async function confirmarExclusao() {
    if (!alvoExclusao) return;
    setExcluindo(true);
    setErroExclusao(null);
    try {
      await excluirInstrumento(alvoExclusao.id);
      await carregar();
      setAviso(`Instrumento "${alvoExclusao.nome}" excluído.`);
      setAlvoExclusao(null);
    } catch (e) {
      setErroExclusao(e instanceof Error ? e.message : "Não foi possível excluir o instrumento.");
    } finally {
      setExcluindo(false);
    }
  }

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Instrumentos e famílias"
          descricao={`A família de cada instrumento é definida automaticamente pelo nome.${
            !carregando && !erro ? ` ${instrumentos.length} instrumento(s).` : ""
          }`}
          acoes={
            pode("instrumentos.criar") ? (
              <Button icone={Plus} onClick={() => setPainel({ tipo: "novo" })} disabled={carregando || !!erro}>
                Adicionar
              </Button>
            ) : undefined
          }
        />

        {carregando ? (
          <div role="status" aria-live="polite" className="space-y-3">
            <span className="sr-only">Carregando instrumentos...</span>
            <Skeleton className="h-14 rounded-2xl" />
            {[0, 1, 2, 3].map((i) => (
              <Skeleton key={i} className="h-10" />
            ))}
          </div>
        ) : erro ? (
          <ErrorState
            titulo="Não foi possível carregar os instrumentos."
            mensagem={erro}
            onTentarNovamente={tentarNovamente}
          />
        ) : instrumentos.length === 0 ? (
          <EmptyState
            titulo="Nenhum instrumento cadastrado ainda."
            descricao="Clique em “Adicionar” ou cadastre um músico ou organista com instrumento."
          />
        ) : (
          <TabelaInstrumentos
            instrumentos={instrumentos}
            familias={familias}
            podeEditar={pode("instrumentos.editar")}
            podeExcluir={pode("instrumentos.excluir")}
            onEditar={(instrumento) => setPainel({ tipo: "editar", instrumento })}
            onExcluir={(instrumento) => {
              setErroExclusao(null);
              setAlvoExclusao(instrumento);
            }}
          />
        )}
      </div>

      {/* Painel lateral: novo instrumento ou edição */}
      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={painel?.tipo === "editar" ? "Editar instrumento" : "Novo instrumento"}
        descricao="A família é sugerida automaticamente pelo nome."
      >
        {painel && (
          <FormularioInstrumento
            instrumento={painel.tipo === "editar" ? painel.instrumento : null}
            instrumentos={instrumentos}
            familias={familias}
            podeEscolherFamilia={pode("instrumentos.editar")}
            onSalvo={aoSalvar}
          />
        )}
      </Drawer>

      {/* Confirmação de exclusão */}
      <ConfirmDialog
        aberto={alvoExclusao !== null}
        titulo="Excluir instrumento?"
        mensagem={
          <>
            <p>
              O instrumento <strong>{alvoExclusao?.nome}</strong> será excluído. Esta ação não pode ser
              desfeita. Se algum membro ainda usar este instrumento, a exclusão será recusada.
            </p>
            {erroExclusao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroExclusao}
              </p>
            )}
          </>
        }
        textoConfirmar="Excluir"
        processando={excluindo}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExclusao(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

################ src/modules/instrumentos/components/FormularioInstrumento.tsx
"use client";

/**
 * SysMúsica — Formulário de instrumento (novo ou edição) — conteúdo do painel lateral.
 * instrumento = null -> criar; preenchido -> editar.
 *
 * A família é AUTOMÁTICA: enquanto ninguém a escolhe à mão, ela acompanha o nome digitado
 * (classificação por palavras-chave). Só quem tem permissão de editar instrumentos pode
 * corrigi-la manualmente; os demais veem apenas a sugestão automática.
 */
import { useState, type FormEvent } from "react";
import Button from "@/components/ui/Button";
import { inferirFamilia } from "@/lib/familias";
import type { Familia, Instrumento } from "@/types/database";
import { NOME_INSTRUMENTO_MAXIMO, validarNomeInstrumento } from "../regras";
import { atualizarInstrumento, criarInstrumento } from "../services/instrumentos";

interface FormularioInstrumentoProps {
  instrumento: Instrumento | null;
  instrumentos: Instrumento[]; // todos, para impedir nomes repetidos
  familias: Familia[];
  podeEscolherFamilia: boolean; // permissão instrumentos.editar
  onSalvo: (mensagem: string) => void | Promise<void>;
}

const classeInput =
  "w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 " +
  "placeholder:text-slate-400 shadow-sm focus:border-brand-500 focus:outline-none " +
  "focus:ring-2 focus:ring-brand-200 disabled:cursor-not-allowed disabled:opacity-60";

export default function FormularioInstrumento({
  instrumento,
  instrumentos,
  familias,
  podeEscolherFamilia,
  onSalvo,
}: FormularioInstrumentoProps) {
  const editando = instrumento !== null;

  const [nome, setNome] = useState(instrumento?.nome ?? "");
  const [familiaId, setFamiliaId] = useState(instrumento?.familia_id ?? ""); // "" = não classificada
  // Se o instrumento já tem família, respeita; se não tem, a família acompanha o nome digitado
  const [familiaManual, setFamiliaManual] = useState(instrumento?.familia_id != null);
  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  // Ao digitar o nome, a família é sugerida automaticamente (enquanto não foi escolhida à mão)
  function alterarNome(novoNome: string) {
    setNome(novoNome);
    if (!familiaManual) setFamiliaId(inferirFamilia(novoNome, familias)?.id ?? "");
  }

  async function salvar(submissao: FormEvent<HTMLFormElement>) {
    submissao.preventDefault();
    if (salvando) return; // evita envio duplicado
    setErro(null);

    const erroNome = validarNomeInstrumento(nome, instrumentos, familias, instrumento?.id);
    if (erroNome) return setErro(erroNome);

    setSalvando(true);
    try {
      if (instrumento) {
        await atualizarInstrumento({ id: instrumento.id, nome, familiaId: familiaId || null });
        await onSalvo(`Instrumento "${nome.trim()}" atualizado.`);
      } else {
        const criado = await criarInstrumento({ nome, familiaId: familiaId || null });
        await onSalvo(`Instrumento "${criado.nome}" criado.`);
      }
    } catch (e) {
      setErro(e instanceof Error ? e.message : "Não foi possível salvar o instrumento.");
    } finally {
      setSalvando(false);
    }
  }

  const familiaAtual = familias.find((familia) => familia.id === familiaId);

  return (
    <form onSubmit={salvar} className="space-y-5">
      <div className="space-y-1.5">
        <label htmlFor="nome-instrumento" className="text-sm font-medium text-slate-700">
          Nome do instrumento
        </label>
        <input
          id="nome-instrumento"
          type="text"
          required
          maxLength={NOME_INSTRUMENTO_MAXIMO}
          value={nome}
          onChange={(e) => alterarNome(e.target.value)}
          disabled={salvando}
          placeholder="Ex: Clarinete"
          className={classeInput}
        />
      </div>

      <div className="space-y-1.5">
        <label htmlFor="familia-instrumento" className="text-sm font-medium text-slate-700">
          Família {!podeEscolherFamilia && <span className="font-normal text-slate-400">(automática)</span>}
        </label>
        {podeEscolherFamilia ? (
          <select
            id="familia-instrumento"
            value={familiaId}
            onChange={(e) => {
              setFamiliaId(e.target.value);
              setFamiliaManual(true); // a partir daqui a família não acompanha mais o nome
            }}
            disabled={salvando}
            className={classeInput}
          >
            <option value="">Não classificada</option>
            {familias.map((familia) => (
              <option key={familia.id} value={familia.id}>
                {familia.nome}
              </option>
            ))}
          </select>
        ) : (
          <input
            id="familia-instrumento"
            type="text"
            readOnly
            tabIndex={-1}
            value={familiaAtual?.nome ?? ""}
            placeholder="Definida automaticamente pelo nome"
            className="w-full cursor-default rounded-lg border border-slate-200 bg-slate-50 px-3 py-2 text-sm text-slate-700 placeholder:text-slate-400 focus:outline-none"
          />
        )}
        <p className="text-xs text-slate-500">
          {podeEscolherFamilia
            ? "Sugerida automaticamente pelo nome. Você pode corrigir se estiver errada."
            : "Definida automaticamente pelo nome do instrumento."}
        </p>
      </div>

      {erro && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erro}
        </p>
      )}

      <Button type="submit" disabled={salvando} className="w-full">
        {salvando ? "Salvando..." : editando ? "Salvar alterações" : "Criar instrumento"}
      </Button>
    </form>
  );
}

"use client";

/**
 * SysMúsica — Seletor de instrumento (lista suspensa agrupada por FAMÍLIA) e campo "Família (automática)".
 * É o único jeito de escolher instrumento no sistema (cadastro de membro e de visitante), igual aos
 * demais seletores (Cargo, Casa de Oração). Só aparecem os instrumentos cadastrados no banco; instrumento novo
 * se cadastra em Instrumentos e famílias, onde a família também é definida.
 */
import { useMemo } from "react";
import {
  agruparInstrumentosPorFamilia,
  nomeDaFamiliaDoInstrumento,
  type FamiliaOpcao,
  type InstrumentoOpcao,
} from "../selecao";

interface SeletorInstrumentoProps {
  id?: string;
  valor: string; // id do instrumento escolhido ("" = nenhum)
  onChange: (instrumentoId: string) => void;
  instrumentos: InstrumentoOpcao[];
  familias: FamiliaOpcao[];
  obrigatorio?: boolean;
  disabled?: boolean;
  className?: string;
}

export default function SeletorInstrumento({
  id,
  valor,
  onChange,
  instrumentos,
  familias,
  obrigatorio,
  disabled,
  className,
}: SeletorInstrumentoProps) {
  const grupos = useMemo(() => agruparInstrumentosPorFamilia(instrumentos, familias), [instrumentos, familias]);

  return (
    <select
      id={id}
      value={valor}
      onChange={(e) => onChange(e.target.value)}
      required={obrigatorio}
      disabled={disabled}
      className={className}
    >
      <option value="">{obrigatorio ? "Selecione o instrumento" : "Sem instrumento / não informado"}</option>
      {grupos.map((grupo) => (
        <optgroup key={grupo.familiaId ?? "sem-familia"} label={grupo.rotulo}>
          {grupo.instrumentos.map((instrumento) => (
            <option key={instrumento.id} value={instrumento.id}>
              {instrumento.nome}
            </option>
          ))}
        </optgroup>
      ))}
    </select>
  );
}

interface FamiliaAutomaticaProps {
  instrumentoId: string;
  instrumentos: InstrumentoOpcao[];
  familias: FamiliaOpcao[];
}

/** Campo somente leitura: a família sai do instrumento escolhido (cadastro do banco). */
export function FamiliaAutomatica({ instrumentoId, instrumentos, familias }: FamiliaAutomaticaProps) {
  const escolhido = instrumentoId !== "" && instrumentos.some((instrumento) => instrumento.id === instrumentoId);
  const nome = nomeDaFamiliaDoInstrumento(instrumentoId, instrumentos, familias);
  const texto = !escolhido ? "" : (nome ?? "Sem família definida (classifique em Instrumentos e famílias)");

  return (
    <div className="space-y-1.5">
      <label htmlFor="familia-instrumento" className="text-sm font-medium text-slate-700">
        Família <span className="font-normal text-slate-400">(automática)</span>
      </label>
      <input
        id="familia-instrumento"
        type="text"
        readOnly
        tabIndex={-1}
        value={texto}
        placeholder="Definida automaticamente pelo instrumento"
        className="w-full cursor-default rounded-lg border border-slate-200 bg-slate-50 px-3 py-2 text-sm text-slate-700 placeholder:text-slate-400 focus:outline-none"
      />
    </div>
  );
}

################ src/modules/instrumentos/components/TabelaInstrumentos.tsx
"use client";

/**
 * SysMúsica — Tabela de instrumentos e famílias, com busca por nome.
 * Ações por linha (só aparecem para quem tem a permissão): editar e excluir.
 * A família é corrigida no formulário de edição (painel lateral).
 */
import { useState } from "react";
import { Pencil, Search, Trash2 } from "lucide-react";
import Badge from "@/components/ui/Badge";
import BotaoIcone from "@/components/ui/BotaoIcone";
import { normalizarTexto } from "@/lib/texto";
import type { Familia, Instrumento } from "@/types/database";

interface TabelaInstrumentosProps {
  instrumentos: Instrumento[];
  familias: Familia[];
  podeEditar: boolean;
  podeExcluir: boolean;
  onEditar: (instrumento: Instrumento) => void;
  onExcluir: (instrumento: Instrumento) => void;
}

const classeCabecalho =
  "px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500";

export default function TabelaInstrumentos({
  instrumentos,
  familias,
  podeEditar,
  podeExcluir,
  onEditar,
  onExcluir,
}: TabelaInstrumentosProps) {
  const [busca, setBusca] = useState("");
  const mostrarAcoes = podeEditar || podeExcluir;

  const termo = normalizarTexto(busca);
  const filtrados = instrumentos.filter(
    (instrumento) => termo === "" || normalizarTexto(instrumento.nome).includes(termo)
  );

  return (
    <div className="overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-sm">
      {/* Barra de ferramentas */}
      <div className="border-b border-slate-200 p-4">
        <div className="relative max-w-md">
          <label htmlFor="busca-instrumento" className="sr-only">
            Pesquisar instrumento
          </label>
          <Search
            className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400"
            aria-hidden="true"
          />
          <input
            id="busca-instrumento"
            type="search"
            autoComplete="off"
            value={busca}
            onChange={(e) => setBusca(e.target.value)}
            placeholder="Pesquisar instrumento..."
            className="w-full rounded-lg border border-slate-300 bg-white py-2 pl-9 pr-3 text-sm text-slate-900 placeholder:text-slate-400 shadow-sm focus:border-brand-500 focus:outline-none focus:ring-2 focus:ring-brand-200"
          />
        </div>
      </div>

      <div className="overflow-x-auto">
        <table className="min-w-full divide-y divide-slate-200 text-sm">
          <thead className="bg-slate-50">
            <tr>
              <th scope="col" className={classeCabecalho}>
                Instrumento
              </th>
              <th scope="col" className={classeCabecalho}>
                Família
              </th>
              {mostrarAcoes && (
                <th scope="col" className={classeCabecalho + " text-right"}>
                  Ações
                </th>
              )}
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {filtrados.length === 0 ? (
              <tr>
                <td colSpan={mostrarAcoes ? 3 : 2} className="px-4 py-10 text-center text-sm text-slate-500">
                  Nenhum instrumento encontrado.
                </td>
              </tr>
            ) : (
              filtrados.map((instrumento) => {
                const familia = familias.find((item) => item.id === instrumento.familia_id);

                return (
                  <tr key={instrumento.id} className="transition hover:bg-slate-50">
                    <td className="px-4 py-3 font-medium text-slate-900">{instrumento.nome}</td>
                    <td className="px-4 py-3">
                      {familia ? (
                        <Badge variante="marca">{familia.nome}</Badge>
                      ) : (
                        <Badge variante="alerta">Não classificada</Badge>
                      )}
                    </td>
                    {mostrarAcoes && (
                      <td className="px-4 py-3">
                        <div className="flex justify-end gap-1">
                          {podeEditar && (
                            <BotaoIcone
                              icone={Pencil}
                              rotulo={`Editar ${instrumento.nome}`}
                              onClick={() => onEditar(instrumento)}
                            />
                          )}
                          {podeExcluir && (
                            <BotaoIcone
                              icone={Trash2}
                              variante="perigo"
                              rotulo={`Excluir ${instrumento.nome}`}
                              onClick={() => onExcluir(instrumento)}
                            />
                          )}
                        </div>
                      </td>
                    )}
                  </tr>
                );
              })
            )}
          </tbody>
        </table>
      </div>

      <div className="border-t border-slate-200 bg-slate-50 px-4 py-3 text-xs text-slate-500">
        Exibindo {filtrados.length} de {instrumentos.length} instrumento(s)
      </div>
    </div>
  );
}

################ src/modules/instrumentos/regras.ts
/**
 * SysMúsica — Regras puras de instrumentos (sem acesso ao banco). Testadas em regras.test.ts.
 */
import { ehNomeDeFamilia } from "@/lib/familias";
import { normalizarTexto } from "@/lib/texto";
import type { Familia, Instrumento } from "@/types/database";

export const NOME_INSTRUMENTO_MAXIMO = 60;

/** Ajusta o nome antes de salvar: tira espaços extras e põe a 1ª letra em maiúscula ("sax  soprano" -> "Sax soprano"). */
export function formatarNomeInstrumento(texto: string): string {
  const limpo = texto.trim().replace(/\s+/g, " ");
  return limpo.charAt(0).toUpperCase() + limpo.slice(1);
}

/**
 * Valida o nome de um instrumento: tamanho, não ser nome de família e não repetir outro
 * (ignorando acento e maiúsculas). Em edição, informe o id do próprio instrumento.
 * Devolve a mensagem de erro ou null quando está tudo certo.
 */
export function validarNomeInstrumento(
  nome: string,
  instrumentos: Instrumento[],
  familias: Familia[],
  idAtual?: string
): string | null {
  const limpo = nome.trim().replace(/\s+/g, " ");
  if (limpo.length < 2) return "Informe o nome do instrumento (mínimo de 2 caracteres).";
  if (limpo.length > NOME_INSTRUMENTO_MAXIMO) {
    return `O nome pode ter no máximo ${NOME_INSTRUMENTO_MAXIMO} caracteres.`;
  }
  if (ehNomeDeFamilia(limpo, familias)) {
    return `"${limpo}" é uma família de instrumentos. Informe o instrumento (ex.: Clarinete).`;
  }

  const chave = normalizarTexto(limpo);
  const repetido = instrumentos.find(
    (instrumento) => instrumento.id !== idAtual && normalizarTexto(instrumento.nome) === chave
  );
  if (repetido) return `Já existe o instrumento "${repetido.nome}".`;

  return null;
}

/**
 * SysMúsica — Regras puras do seletor de instrumento (sem banco). Testadas em selecao.test.ts.
 * O mesmo seletor é usado no cadastro de membro e no cadastro de visitante, para a escolha ser sempre
 * igual: lista suspensa agrupada por família, e a família aparece sozinha pelo instrumento escolhido.
 */

export interface InstrumentoOpcao {
  id: string;
  nome: string;
  familia_id: string | null;
}

export interface FamiliaOpcao {
  id: string;
  nome: string;
}

export interface GrupoInstrumentos {
  familiaId: string | null; // nulo = instrumentos sem família
  rotulo: string;
  instrumentos: InstrumentoOpcao[];
}

const ROTULO_SEM_FAMILIA = "Sem família definida";

// Ordem natural em português
const porNome = (a: { nome: string }, b: { nome: string }) => a.nome.localeCompare(b.nome, "pt-BR", { numeric: true });

/** Agrupa os instrumentos por família (famílias em ordem de nome; os sem família vão para o fim). */
export function agruparInstrumentosPorFamilia(
  instrumentos: InstrumentoOpcao[],
  familias: FamiliaOpcao[]
): GrupoInstrumentos[] {
  const conhecidas = new Set(familias.map((familia) => familia.id));
  const porFamilia = new Map<string, InstrumentoOpcao[]>();
  const semFamilia: InstrumentoOpcao[] = [];

  for (const instrumento of instrumentos) {
    if (instrumento.familia_id && conhecidas.has(instrumento.familia_id)) {
      porFamilia.set(instrumento.familia_id, [...(porFamilia.get(instrumento.familia_id) ?? []), instrumento]);
    } else {
      semFamilia.push(instrumento);
    }
  }

  const grupos: GrupoInstrumentos[] = [...familias]
    .filter((familia) => porFamilia.has(familia.id))
    .sort(porNome)
    .map((familia) => ({
      familiaId: familia.id,
      rotulo: familia.nome,
      instrumentos: [...(porFamilia.get(familia.id) ?? [])].sort(porNome),
    }));

  if (semFamilia.length > 0) {
    grupos.push({ familiaId: null, rotulo: ROTULO_SEM_FAMILIA, instrumentos: [...semFamilia].sort(porNome) });
  }
  return grupos;
}

/** Nome da família do instrumento escolhido; nulo se não houver instrumento ou se ele estiver sem família. */
export function nomeDaFamiliaDoInstrumento(
  instrumentoId: string,
  instrumentos: InstrumentoOpcao[],
  familias: FamiliaOpcao[]
): string | null {
  const instrumento = instrumentos.find((item) => item.id === instrumentoId);
  if (!instrumento?.familia_id) return null;
  return familias.find((familia) => familia.id === instrumento.familia_id)?.nome ?? null;
}

/**
 * Regra do instrumento obrigatório: devolve a mensagem de erro ou null.
 * (O banco confere de novo: a tela é só a resposta rápida.)
 */
export function validarInstrumentoDoCargo(
  exige: boolean,
  instrumentoId: string,
  nomeCargo: string | undefined
): string | null {
  if (exige && !instrumentoId) {
    return `O cargo ${nomeCargo ?? "escolhido"} exige o instrumento. Escolha o instrumento na lista.`;
  }
  return null;
}

/**
 * SysMúsica — Serviço de dados de instrumentos e famílias.
 *
 * Criar, editar e excluir passam pelas regras do banco (permissões e RLS). Excluir usa uma função
 * do banco que recusa a exclusão enquanto algum membro usar o instrumento.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import { inferirFamilia } from "@/lib/familias";
import { normalizarTexto } from "@/lib/texto";
import type { Familia, Instrumento, InstrumentoInsert } from "@/types/database";
import { formatarNomeInstrumento } from "../regras";

const MSG_DUPLICADO = "Já existe um instrumento com este nome.";

/** Busca instrumentos e famílias, ambos em ordem alfabética. Lança erro se a consulta falhar. */
export async function buscarInstrumentosEFamilias(): Promise<{
  instrumentos: Instrumento[];
  familias: Familia[];
}> {
  const [resInstrumentos, resFamilias] = await Promise.all([
    supabase.from("instrumentos").select("*").order("nome", { ascending: true }),
    supabase.from("familias").select("*").order("nome", { ascending: true }),
  ]);

  const erroBanco = resInstrumentos.error ?? resFamilias.error;
  if (erroBanco) throw new Error(erroBanco.message);

  return {
    instrumentos: (resInstrumentos.data ?? []) as Instrumento[],
    familias: (resFamilias.data ?? []) as Familia[],
  };
}

/** Cria um instrumento (a família pode ir vazia: o banco classifica pelo nome). */
export async function criarInstrumento(dados: {
  nome: string;
  familiaId: string | null;
}): Promise<Instrumento> {
  const novo: InstrumentoInsert = {
    nome: formatarNomeInstrumento(dados.nome),
    familia_id: dados.familiaId,
  };
  const { data, error } = await supabase.from("instrumentos").insert(novo).select().single();
  if (error || !data) {
    throw new Error(
      error
        ? mensagemDoErro(error, "Não foi possível criar o instrumento.", MSG_DUPLICADO)
        : "Não foi possível criar o instrumento."
    );
  }
  return data as Instrumento;
}

/** Altera nome e família de um instrumento (exige a permissão instrumentos.editar). */
export async function atualizarInstrumento(dados: {
  id: string;
  nome: string;
  familiaId: string | null;
}): Promise<void> {
  // .select("id") devolve as linhas alteradas: se o RLS negar, o update não dá erro, só não altera nada
  const { data, error } = await supabase
    .from("instrumentos")
    .update({ nome: formatarNomeInstrumento(dados.nome), familia_id: dados.familiaId })
    .eq("id", dados.id)
    .select("id");
  if (error) {
    throw new Error(mensagemDoErro(error, "Não foi possível salvar o instrumento.", MSG_DUPLICADO));
  }
  if (!data || data.length === 0) {
    throw new Error("Não foi possível salvar: você não tem permissão ou o instrumento não existe mais.");
  }
}

/** Exclui um instrumento (função do banco: recusa se algum membro ainda usar). */
export async function excluirInstrumento(id: string): Promise<void> {
  const { error } = await supabase.rpc("excluir_instrumento", { p_id: id });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível excluir o instrumento."));
}

export interface ResultadoInstrumento {
  id: string;
  nome: string;
  criado: boolean; // true = acabou de ser criado
  familiaNome: string | null;
}

/**
 * Usado no cadastro de membros: devolve o instrumento digitado, criando-o se ainda não existir.
 * A comparação ignora acento e maiúsculas. Ao criar, a família é inferida pelo nome.
 */
export async function obterOuCriarInstrumento(
  texto: string,
  instrumentos: Instrumento[],
  familias: Familia[]
): Promise<ResultadoInstrumento> {
  const chave = normalizarTexto(texto);
  const nomeDaFamilia = (familiaId: string | null) =>
    familias.find((familia) => familia.id === familiaId)?.nome ?? null;

  // 1) Já existe na lista carregada? Então só reaproveita o id
  const existente = instrumentos.find((instrumento) => normalizarTexto(instrumento.nome) === chave);
  if (existente) {
    return {
      id: existente.id,
      nome: existente.nome,
      criado: false,
      familiaNome: nomeDaFamilia(existente.familia_id),
    };
  }

  // 2) Não existe: cria já com a família inferida e resgata o registro
  const novo: InstrumentoInsert = {
    nome: formatarNomeInstrumento(texto),
    familia_id: inferirFamilia(texto, familias)?.id ?? null,
  };
  const { data, error } = await supabase.from("instrumentos").insert(novo).select().single();
  if (!error && data) {
    const criado = data as Instrumento;
    return {
      id: criado.id,
      nome: criado.nome,
      criado: true,
      familiaNome: nomeDaFamilia(criado.familia_id),
    };
  }

  // 3) 23505 = nome duplicado: outra pessoa criou o mesmo instrumento agora há pouco.
  //    Busca a lista de novo e reaproveita o registro que já existe.
  if (error?.code === "23505") {
    const { data: lista, error: erroBusca } = await supabase.from("instrumentos").select("*");
    if (erroBusca) throw new Error(erroBusca.message);
    const achado = ((lista ?? []) as Instrumento[]).find(
      (instrumento) => normalizarTexto(instrumento.nome) === chave
    );
    if (achado) {
      return {
        id: achado.id,
        nome: achado.nome,
        criado: false,
        familiaNome: nomeDaFamilia(achado.familia_id),
      };
    }
  }

  throw new Error(
    error
      ? mensagemDoErro(error, "Não foi possível criar o instrumento.", MSG_DUPLICADO)
      : "Não foi possível criar o instrumento."
  );
}
