"use client";

/**
 * SysMúsica — Seletor de instrumento (lista suspensa agrupada por FAMÍLIA) e campo "Família (automática)".
 * É o único jeito de escolher instrumento no sistema (cadastro de membro e de visitante), igual aos
 * demais seletores (Cargo, Casa de Oração). Só aparecem os instrumentos cadastrados no banco; instrumento novo
 * se cadastra em Instrumentos e famílias, onde a família também é definida.
 */
import { useMemo } from "react";
import {
  agruparInstrumentosPorFamilia,
  nomeDaFamiliaDoInstrumento,
  type FamiliaOpcao,
  type InstrumentoOpcao,
} from "../selecao";

interface SeletorInstrumentoProps {
  id?: string;
  valor: string; // id do instrumento escolhido ("" = nenhum)
  onChange: (instrumentoId: string) => void;
  instrumentos: InstrumentoOpcao[];
  familias: FamiliaOpcao[];
  obrigatorio?: boolean;
  disabled?: boolean;
  className?: string;
}

export default function SeletorInstrumento({
  id,
  valor,
  onChange,
  instrumentos,
  familias,
  obrigatorio,
  disabled,
  className,
}: SeletorInstrumentoProps) {
  const grupos = useMemo(() => agruparInstrumentosPorFamilia(instrumentos, familias), [instrumentos, familias]);

  return (
    <select
      id={id}
      value={valor}
      onChange={(e) => onChange(e.target.value)}
      required={obrigatorio}
      disabled={disabled}
      className={className}
    >
      <option value="">{obrigatorio ? "Selecione o instrumento" : "Sem instrumento / não informado"}</option>
      {grupos.map((grupo) => (
        <optgroup key={grupo.familiaId ?? "sem-familia"} label={grupo.rotulo}>
          {grupo.instrumentos.map((instrumento) => (
            <option key={instrumento.id} value={instrumento.id}>
              {instrumento.nome}
            </option>
          ))}
        </optgroup>
      ))}
    </select>
  );
}

interface FamiliaAutomaticaProps {
  instrumentoId: string;
  instrumentos: InstrumentoOpcao[];
  familias: FamiliaOpcao[];
}

/** Campo somente leitura: a família sai do instrumento escolhido (cadastro do banco). */
export function FamiliaAutomatica({ instrumentoId, instrumentos, familias }: FamiliaAutomaticaProps) {
  const escolhido = instrumentoId !== "" && instrumentos.some((instrumento) => instrumento.id === instrumentoId);
  const nome = nomeDaFamiliaDoInstrumento(instrumentoId, instrumentos, familias);
  const texto = !escolhido ? "" : (nome ?? "Sem família definida (classifique em Instrumentos e famílias)");

  return (
    <div className="space-y-1.5">
      <label htmlFor="familia-instrumento" className="text-sm font-medium text-slate-700">
        Família <span className="font-normal text-slate-400">(automática)</span>
      </label>
      <input
        id="familia-instrumento"
        type="text"
        readOnly
        tabIndex={-1}
        value={texto}
        placeholder="Definida automaticamente pelo instrumento"
        className="w-full cursor-default rounded-lg border border-slate-200 bg-slate-50 px-3 py-2 text-sm text-slate-700 placeholder:text-slate-400 focus:outline-none"
      />
    </div>
  );
}

/**
 * SysMúsica — Preparo da fotografia no navegador antes do envio.
 * Reduz a imagem (lado maior de 800 px) e converte para JPEG: a foto fica leve (poucas centenas de KB)
 * e o envio é rápido, inclusive em rede móvel. A conferência de tipo e tamanho final também é feita
 * pelo armazenamento do banco (limite de 2 MB; JPEG, PNG ou WebP).
 */

const LADO_MAXIMO = 800;
const QUALIDADE_JPEG = 0.82;
const TAMANHO_MAXIMO_ORIGINAL = 15 * 1024 * 1024; // 15 MB de entrada (a saída é bem menor)

export async function prepararFoto(arquivo: File): Promise<Blob> {
  if (!arquivo.type.startsWith("image/")) {
    throw new Error("Escolha um arquivo de imagem (JPG, PNG ou WebP).");
  }
  if (arquivo.size > TAMANHO_MAXIMO_ORIGINAL) {
    throw new Error("A imagem é grande demais (máximo de 15 MB).");
  }

  // "from-image" respeita a rotação gravada pela câmera do celular; se o navegador não aceitar a opção, tenta sem ela
  let bitmap: ImageBitmap;
  try {
    bitmap = await createImageBitmap(arquivo, { imageOrientation: "from-image" });
  } catch {
    try {
      bitmap = await createImageBitmap(arquivo);
    } catch {
      throw new Error("Não foi possível ler esta imagem. Tente outra foto.");
    }
  }

  const escala = Math.min(1, LADO_MAXIMO / Math.max(bitmap.width, bitmap.height));
  const largura = Math.max(1, Math.round(bitmap.width * escala));
  const altura = Math.max(1, Math.round(bitmap.height * escala));

  const canvas = document.createElement("canvas");
  canvas.width = largura;
  canvas.height = altura;
  const contexto = canvas.getContext("2d");
  if (!contexto) {
    bitmap.close();
    throw new Error("Não foi possível processar a imagem neste navegador.");
  }
  contexto.drawImage(bitmap, 0, 0, largura, altura);
  bitmap.close();

  const imagem = await new Promise<Blob | null>((resolver) =>
    canvas.toBlob(resolver, "image/jpeg", QUALIDADE_JPEG)
  );
  if (!imagem) throw new Error("Não foi possível processar a imagem.");
  return imagem;
}

################ src/app/(painel)/membros/page.tsx
"use client";

/**
 * SysMúsica — Página /membros: listagem em tabela (largura total) com busca e filtros, novo,
 * editar, excluir e QR Code. Cada botão só aparece para quem tem a permissão (grupo do usuário);
 * o banco confere de novo em toda operação.
 * Excluir é exclusão lógica: o membro some das telas e o histórico de presenças é mantido.
 * Editar abre o formulário já com o CPF completo (só para quem tem a permissão membros.cpf).
 */
import { useCallback, useEffect, useState } from "react";
import { Plus } from "lucide-react";
import {
  buscarDadosMembros,
  excluirMembro,
  obterCpfMembro,
  obterQrMembro,
  regenerarQrMembro,
} from "@/modules/membros/services/membros";
import QrMembro from "@/modules/membros/components/QrMembro";
import type { CadastroMusical } from "@/modules/membros/cadastroMusical";
import { obterCadastroMusical } from "@/modules/membros/services/cadastroMusical";
import { urlsDasFotos } from "@/modules/membros/services/fotos";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";
import Button from "@/components/ui/Button";
import ConfirmDialog from "@/components/ui/ConfirmDialog";
import Drawer from "@/components/ui/Drawer";
import Notificacao from "@/components/ui/Notificacao";
import PageHeader from "@/components/ui/PageHeader";
import FormularioMembro from "@/components/FormularioMembro";
import ListaMembros from "@/components/ListaMembros";
import { valoresDistintos } from "@/lib/listas";
import { montarConteudoQr } from "@/lib/qr";
import { gerarImagemQr } from "@/lib/qrImagem";
import type { Cargo, Familia, Instrumento, MembroPublico } from "@/types/database";

// Qual painel lateral está aberto: novo membro, edição ou QR Code de um membro
type PainelMembro =
  | { tipo: "novo" }
  | { tipo: "editar"; membro: MembroPublico; cpfInicial: string | null; cadastro: CadastroMusical | null }
  | { tipo: "qr"; membro: MembroPublico; imagem: string; conteudo: string };

export default function MembrosPage() {
  const { pode } = useSessao();
  const [cargos, setCargos] = useState<Cargo[]>([]);
  const [instrumentos, setInstrumentos] = useState<Instrumento[]>([]);
  const [familias, setFamilias] = useState<Familia[]>([]);
  const [membros, setMembros] = useState<MembroPublico[]>([]);
  const [urlsFotos, setUrlsFotos] = useState<Record<string, string>>({}); // miniaturas da lista
  const [carregando, setCarregando] = useState(true); // inicia true: primeira busca no mount
  const [erro, setErro] = useState<string | null>(null);
  const [painel, setPainel] = useState<PainelMembro | null>(null);
  const [abrindo, setAbrindo] = useState(false); // buscando CPF/QR antes de abrir o painel
  const [erroAcao, setErroAcao] = useState<string | null>(null);
  const [alvoExclusao, setAlvoExclusao] = useState<MembroPublico | null>(null);
  const [excluindo, setExcluindo] = useState(false);
  const [erroExclusao, setErroExclusao] = useState<string | null>(null);
  const [aviso, setAviso] = useState<string | null>(null);

  // Busca cargos, instrumentos, famílias e membros
  const carregar = useCallback(async () => {
    try {
      const dados = await buscarDadosMembros();
      setCargos(dados.cargos);
      setInstrumentos(dados.instrumentos);
      setFamilias(dados.familias);
      setMembros(dados.membros);
      // Miniaturas: um único pedido de URLs temporárias para todas as fotos (se falhar, a lista fica sem miniaturas)
      setUrlsFotos(await urlsDasFotos(dados.membros.flatMap((m) => (m.foto_path ? [m.foto_path] : []))));
      setErro(null);
    } catch (e) {
      // O detalhe técnico vai só para o console; a tela mostra uma mensagem simples
      console.error("Falha ao carregar membros:", e);
      setErro("Não foi possível carregar os dados. Verifique a conexão e tente novamente.");
    } finally {
      setCarregando(false);
    }
  }, []);

  // Primeira carga ao abrir a página
  useEffect(() => {
    // O setState só roda DEPOIS do await (nunca de forma síncrona), então o aviso da regra
    // set-state-in-effect não se aplica a esta carga inicial. Ignorado apenas nesta linha.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    void carregar();
  }, [carregar]);

  // Botão "Tentar novamente": volta ao estado de loading e busca de novo
  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  // Depois de salvar: recarrega a lista, fecha o painel e mostra o aviso de sucesso
  async function aoSalvar(mensagem: string) {
    await carregar();
    setPainel(null);
    setAviso(mensagem);
  }

  // Abre a edição já com o CPF completo (a consulta fica registrada na auditoria)
  async function abrirEdicao(membro: MembroPublico) {
    if (abrindo) return;
    setErroAcao(null);
    setAbrindo(true);
    try {
      const [cpfInicial, cadastro] = await Promise.all([
        pode("membros.cpf") ? obterCpfMembro(membro.id) : Promise.resolve(null),
        obterCadastroMusical(membro.id), // dados pessoais do Cadastro Musical (null = membro antigo, sem cadastro)
      ]);
      setPainel({ tipo: "editar", membro, cpfInicial, cadastro });
    } catch (e) {
      setErroAcao(e instanceof Error ? e.message : "Não foi possível abrir a edição.");
    } finally {
      setAbrindo(false);
    }
  }

  // Busca o código do QR Code no servidor (fica registrado na auditoria) e gera a imagem
  async function montarPainelQr(membro: MembroPublico, codigo: string): Promise<PainelMembro> {
    const conteudo = montarConteudoQr(codigo);
    const imagem = await gerarImagemQr(conteudo);
    return { tipo: "qr", membro, imagem, conteudo };
  }

  async function abrirQr(membro: MembroPublico) {
    if (abrindo) return;
    setErroAcao(null);
    setAbrindo(true);
    try {
      setPainel(await montarPainelQr(membro, await obterQrMembro(membro.id)));
    } catch (e) {
      setErroAcao(e instanceof Error ? e.message : "Não foi possível abrir o QR Code.");
    } finally {
      setAbrindo(false);
    }
  }

  // Substitui o QR Code (o código antigo deixa de funcionar) e mostra a nova imagem
  async function substituirQr(membro: MembroPublico, justificativa: string) {
    const novoCodigo = await regenerarQrMembro(membro.id, justificativa);
    setPainel(await montarPainelQr(membro, novoCodigo));
    setAviso(`QR Code de "${membro.nome}" substituído. O código anterior deixou de funcionar.`);
  }

  // Confirma a exclusão (lógica) do membro escolhido
  async function confirmarExclusao() {
    if (!alvoExclusao) return;
    setExcluindo(true);
    setErroExclusao(null);
    try {
      await excluirMembro(alvoExclusao.id);
      await carregar();
      setAviso(`Membro "${alvoExclusao.nome}" excluído.`);
      setAlvoExclusao(null);
    } catch (e) {
      setErroExclusao(e instanceof Error ? e.message : "Não foi possível excluir o membro.");
    } finally {
      setExcluindo(false);
    }
  }

  // Sugestões dos campos de localidade (valores já cadastrados)
  const sugestoes = {
    congregacoes: valoresDistintos(membros.map((membro) => membro.congregacao)),
    cidades: valoresDistintos(membros.map((membro) => membro.cidade)),
    bairros: valoresDistintos(membros.map((membro) => membro.bairro)),
  };

  const tituloPainel =
    painel?.tipo === "editar"
      ? "Editar cadastro musical"
      : painel?.tipo === "qr"
        ? "QR Code do membro"
        : "Cadastro Musical";

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Músicos / Organistas"
          descricao={`Cadastre e acompanhe os membros do SysMúsica.${
            !carregando && !erro ? ` ${membros.length} cadastrado(s).` : ""
          }`}
          acoes={
            pode("membros.criar") ? (
              <Button icone={Plus} onClick={() => setPainel({ tipo: "novo" })} disabled={carregando || !!erro}>
                Adicionar
              </Button>
            ) : undefined
          }
        />

        {erroAcao && (
          <p role="alert" className="mb-4 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
            {erroAcao}
          </p>
        )}

        <ListaMembros
          membros={membros}
          cargos={cargos}
          instrumentos={instrumentos}
          familias={familias}
          carregando={carregando}
          erro={erro}
          urlsFotos={urlsFotos}
          onTentarNovamente={tentarNovamente}
          podeEditar={pode("membros.editar")}
          podeExcluir={pode("membros.excluir")}
          podeQr={pode("membros.qr")}
          onEditar={(membro) => void abrirEdicao(membro)}
          onQr={(membro) => void abrirQr(membro)}
          onExcluir={(membro) => {
            setErroExclusao(null);
            setAlvoExclusao(membro);
          }}
        />
      </div>

      {/* Painel lateral: novo membro, edição ou QR Code */}
      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={tituloPainel}
        descricao={
          painel?.tipo === "qr"
            ? "Identificação individual e permanente do membro."
            : "Campos com * são obrigatórios. A fotografia identifica a pessoa na portaria."
        }
      >
        {(painel?.tipo === "novo" || painel?.tipo === "editar") && (
          <FormularioMembro
            membro={painel.tipo === "editar" ? painel.membro : null}
            cpfInicial={painel.tipo === "editar" ? painel.cpfInicial : null}
            cadastroInicial={painel.tipo === "editar" ? painel.cadastro : null}
            podeCpf={pode("membros.cpf")}
            podeFoto={pode("membros.foto")}
            cargos={cargos}
            instrumentos={instrumentos}
            familias={familias}
            sugestoes={sugestoes}
            carregando={carregando}
            erro={erro}
            onSalvo={aoSalvar}
          />
        )}
        {painel?.tipo === "qr" && (
          <QrMembro
            nome={painel.membro.nome}
            congregacao={painel.membro.congregacao}
            imagem={painel.imagem}
            conteudo={painel.conteudo}
            podeSubstituir={pode("membros.qr_revogar")}
            onSubstituir={(justificativa) => substituirQr(painel.membro, justificativa)}
          />
        )}
      </Drawer>

      {/* Confirmação de exclusão */}
      <ConfirmDialog
        aberto={alvoExclusao !== null}
        titulo="Excluir membro?"
        mensagem={
          <>
            <p>
              <strong>{alvoExclusao?.nome}</strong> deixará de aparecer nas telas. O histórico de
              presenças dele é mantido. Para apenas impedir novos registros sem excluir, altere a situação
              cadastral para Inativo na edição.
            </p>
            {erroExclusao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroExclusao}
              </p>
            )}
          </>
        }
        textoConfirmar="Excluir"
        processando={excluindo}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExclusao(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

################ src/types/database.ts
/**
 * SysMúsica — Tipos TypeScript das tabelas do banco (Supabase/PostgreSQL)
 *
 * Mantenha este arquivo em sincronia com supabase/schema.sql.
 * Convenções:
 *  - uuid        -> string
 *  - text        -> string
 *  - text[]      -> string[]
 *  - boolean     -> boolean
 *  - date        -> string ("AAAA-MM-DD", sem horário nem fuso)
 *  - timestamptz -> string (ISO 8601, ex: "2026-10-13T19:30:00+00:00")
 *  - colunas que aceitam NULL -> "tipo | null"
 */

// ---------------------------------------------------------------------
// Tabela: cargos
// ---------------------------------------------------------------------
export interface Cargo {
  id: string;
  nome: string;
  permite_instrumento: boolean; // true = o cargo informa instrumento
}

/** Payload para INSERT: apenas o id é gerado pelo banco (permite_instrumento tem padrão false). */
export type CargoInsert = Pick<Cargo, "nome"> & Partial<Pick<Cargo, "id" | "permite_instrumento">>;

// ---------------------------------------------------------------------
// Tabela: familias (Cordas, Madeiras, Metais, Teclas...)
// ---------------------------------------------------------------------
export interface Familia {
  id: string;
  nome: string;
  /** Trechos (minúsculas, sem acento) que classificam um instrumento nesta família. */
  palavras_chave: string[];
}

/** Payload para INSERT: id e palavras_chave são opcionais (o padrão de palavras_chave é lista vazia). */
export type FamiliaInsert = Pick<Familia, "nome"> & Partial<Pick<Familia, "id" | "palavras_chave">>;

// ---------------------------------------------------------------------
// Tabela: instrumentos
// ---------------------------------------------------------------------
export interface Instrumento {
  id: string;
  nome: string; // único (sem diferenciar maiúsculas/minúsculas)
  familia_id: string | null; // FK -> familias.id (null = ainda não classificado)
}

/** Payload para INSERT: o id é gerado pelo banco; se familia_id faltar, o gatilho do banco classifica. */
export type InstrumentoInsert = Pick<Instrumento, "nome"> &
  Partial<Pick<Instrumento, "id" | "familia_id">>;

// ---------------------------------------------------------------------
// Tabela: membros (antiga "musicos")
// ---------------------------------------------------------------------
export interface Membro {
  id: string;
  nome: string;
  congregacao: string;
  cargo_id: string; // FK -> cargos.id
  data_nascimento: string | null; // date (AAAA-MM-DD)
  instrumento_id: string | null; // FK -> instrumentos.id (null = sem instrumento)
  status: string; // ex: "Ativo", "Inativo" (padrão no banco: "Ativo")
  qr_secret: string; // segredo do QR Code; NÃO é legível pela chave anon (ver schema.sql, seção 13)
  data_cadastro: string; // timestamptz, preenchido automaticamente (now())
  excluido_em: string | null; // exclusão lógica: preenchido quando o membro é excluído (some das telas)
  cpf: string | null; // 11 dígitos; NÃO é lido diretamente (use cpf_mascarado ou a função obter_cpf_membro)
  cpf_mascarado: string | null; // "***.***.***-25": coluna gerada pelo banco
  cidade: string | null; // município da congregação
  bairro: string | null; // bairro ou igreja de referência
  atualizado_em: string; // timestamptz, atualizado automaticamente a cada alteração
  casa_oracao_id?: string | null; // FK -> casas_oracao.id: a casa de oração do membro (define quem enxerga o cadastro)
  foto_path?: string | null; // caminho da fotografia no armazenamento privado (exibida por URL temporária)
  categoria?: string | null; // OFICIALIZADO, CULTO_OFICIAL ou RJM (só Músico e Organista; define a linha do painel ao vivo)
}

/**
 * Membro como o frontend consegue LER com a chave anon (sem qr_secret).
 * Use este tipo nas telas que buscam membros. A família do membro é a do seu instrumento.
 */
export type MembroPublico = Omit<Membro, "qr_secret" | "excluido_em" | "cpf">;

/**
 * Colunas liberadas para leitura pública em "membros".
 * Deve ser IGUAL à lista do GRANT SELECT (...) em supabase/schema.sql e às chaves de MembroPublico.
 * Usada no lugar de select("*"), que falha porque inclui o qr_secret.
 */
export const COLUNAS_MEMBRO_PUBLICO =
  "id, nome, congregacao, cargo_id, data_nascimento, instrumento_id, status, data_cadastro, cidade, bairro, atualizado_em, cpf_mascarado, casa_oracao_id, foto_path, categoria" as const;

/** Payload para INSERT: o qr_secret é sempre gerado pelo banco e não pode ser enviado. */
export type MembroInsert = Pick<Membro, "nome" | "congregacao" | "cargo_id"> &
  Partial<
    Pick<
      Membro,
      "id" | "data_nascimento" | "instrumento_id" | "status" | "data_cadastro" | "cidade" | "bairro" | "cpf" | "casa_oracao_id" | "foto_path" | "categoria"
    >
  >;

// ---------------------------------------------------------------------
// Tabela: eventos
// ---------------------------------------------------------------------
export interface Evento {
  id: string;
  tipo: string; // ex: "Ensaio", "Culto"
  data_evento: string; // timestamptz (ISO 8601)
  local: string;
  finalidade_id: string | null; // FK -> finalidades_evento.id (nula em eventos antigos)
  cidade?: string | null; // município do evento (informativo; o setor vem da casa de oração)
  casa_oracao_id?: string | null; // casa de oração do evento: define o setor da portaria e quem enxerga o evento
  cancelado_em?: string | null; // preenchido quando o Administrador cancela o evento
  cancelado_por?: string | null;
  motivo_cancelamento?: string | null;
}

/** Payload para INSERT: o id é gerado pelo banco; a finalidade é opcional (eventos antigos não têm). */
export type EventoInsert = Pick<Evento, "tipo" | "data_evento" | "local"> &
  Partial<Pick<Evento, "id" | "finalidade_id" | "cidade" | "casa_oracao_id">>;

// ---------------------------------------------------------------------
// Tabela: finalidades_evento (catálogo reaproveitável de finalidades)
// ---------------------------------------------------------------------
export interface FinalidadeEvento {
  id: string;
  nome: string; // único (sem diferenciar maiúsculas/minúsculas)
  criado_em: string; // timestamptz
}

/** Payload para INSERT: o id e criado_em são gerados pelo banco. */
export type FinalidadeEventoInsert = Pick<FinalidadeEvento, "nome"> & Partial<Pick<FinalidadeEvento, "id">>;

// ---------------------------------------------------------------------
// Tabela: presencas
// ---------------------------------------------------------------------
export interface Presenca {
  id: string;
  membro_id: string; // FK -> membros.id
  evento_id: string; // FK -> eventos.id
  status: string; // ex: "Presente", "Ausente", "Justificado"
  hora_entrada: string; // timestamptz (padrão no banco: now())
  hora_saida: string | null; // timestamptz; null = entrada ainda em aberto
  portaria_id?: string | null; // FK -> portarias.id (nula em presenças antigas)
  registrado_por?: string | null; // quem registrou (definido pelo banco)
  origem?: string; // "MANUAL" ou "QR" (definida pelo banco)
}

/** Payload para INSERT: hora_entrada, hora_saida e id são opcionais (valores padrão do banco). */
export type PresencaInsert = Pick<Presenca, "membro_id" | "evento_id" | "status"> &
  Partial<Pick<Presenca, "id" | "hora_entrada" | "hora_saida" | "portaria_id">>;

// ---------------------------------------------------------------------
// RPC: registrar_entrada(p_portaria_id, p_membro_id, p_origem) — só em portaria APROVADA
// Uso: supabase.rpc("registrar_entrada", args) -> devolve a Presenca criada
// ---------------------------------------------------------------------
export interface RegistrarEntradaArgs {
  p_portaria_id: string;
  p_membro_id: string;
  p_origem?: "MANUAL" | "QR";
}
