/**
 * SysMúsica — Autorização das rotas de servidor da Administração.
 *
 * A sessão do navegador fica no armazenamento local (não em cookies), então a tela envia o token
 * de login no cabeçalho "Authorization: Bearer ...". Aqui o servidor:
 *  1) cria um cliente "como aquele usuário" (chave pública + o token dele);
 *  2) pergunta ao banco se ele tem a permissão (função tem_permissao).
 * Token inválido/expirado, ou sem a permissão, interrompe a operação ANTES de qualquer ação.
 */
import { createClient } from "@supabase/supabase-js";
import type { CodigoPermissao } from "@/lib/permissoes";
import { ErroHttp } from "./erros";

export async function exigirPermissao(request: Request, permissao: CodigoPermissao): Promise<void> {
  const cabecalho = request.headers.get("authorization") ?? "";
  const token = cabecalho.toLowerCase().startsWith("bearer ") ? cabecalho.slice(7).trim() : "";
  if (!token) throw new ErroHttp(401, "Sessão ausente. Entre novamente.");

  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const chavePublica = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
  if (!url || !chavePublica) {
    console.error("Variáveis públicas do Supabase ausentes no servidor.");
    throw new ErroHttp(500, "Servidor sem a configuração do banco de dados.");
  }

  const clienteDoUsuario = createClient(url, chavePublica, {
    auth: { persistSession: false, autoRefreshToken: false },
    global: { headers: { Authorization: `Bearer ${token}` } },
  });

  const { data, error } = await clienteDoUsuario.rpc("tem_permissao", { p_codigo: permissao });
  if (error) throw new ErroHttp(401, "Sessão inválida ou expirada. Entre novamente.");
  if (data !== true) throw new ErroHttp(403, "Você não tem permissão para esta operação.");
}

/**
 * SysMúsica — Cliente Supabase com a CHAVE SECRETA (service_role). SOMENTE SERVIDOR.
 *
 * Essa chave ignora todas as regras de acesso do banco: ela jamais pode ir para o navegador.
 * O import "server-only" faz o build falhar se algum componente de tela importar este arquivo.
 * A variável SUPABASE_SERVICE_ROLE_KEY (sem NEXT_PUBLIC_) fica só no .env.local do servidor.
 */
import "server-only";
import { createClient } from "@supabase/supabase-js";
import { ErroHttp } from "./erros";

export function clienteAdmin() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const chave = process.env.SUPABASE_SERVICE_ROLE_KEY;

  if (!url || !chave) {
    // O detalhe vai para o log do servidor; a tela recebe uma orientação simples
    console.error("SUPABASE_SERVICE_ROLE_KEY ou NEXT_PUBLIC_SUPABASE_URL ausente no servidor.");
    throw new ErroHttp(
      503,
      "O servidor ainda não está configurado para criar contas. Peça ao responsável técnico para configurar a chave de servidor."
    );
  }

  return createClient(url, chave, {
    auth: { persistSession: false, autoRefreshToken: false },
  });
}

export class ErroHttp extends Error {

/**
 * SysMúsica — Resposta de erro padronizada das rotas de servidor da Administração.
 * Erros esperados (ErroHttp) saem com a própria mensagem; qualquer outro vira um erro genérico
 * e o detalhe técnico fica só no log do servidor.
 */
import { NextResponse } from "next/server";
import { ErroHttp } from "./erros";

export function respostaDeErro(erro: unknown): NextResponse {
  if (erro instanceof ErroHttp) {
    return NextResponse.json({ erro: erro.message }, { status: erro.status });
  }
  console.error("Erro inesperado na rota de administração:", erro);
  return NextResponse.json(
    { erro: "Não foi possível concluir a operação. Tente novamente." },
    { status: 500 }
  );
}

  nome,

  obrigatoria = false,
  onChange,
  disabled = false,
}: CampoFotoProps) {
  const cameraRef = useRef<HTMLInputElement>(null);

    setAviso(mensagem);
  }

  // Abre a edição já com o CPF completo (a consulta fica registrada na auditoria)
  async function abrirEdicao(membro: MembroPublico) {
    if (abrindo) return;
    setErroAcao(null);
    setAbrindo(true);
    try {
      const [cpfInicial, cadastro] = await Promise.all([
        pode("membros.cpf") ? obterCpfMembro(membro.id) : Promise.resolve(null),
        obterCadastroMusical(membro.id), // dados pessoais do Cadastro Musical (null = membro antigo, sem cadastro)
      ]);
      setPainel({ tipo: "editar", membro, cpfInicial, cadastro });
    } catch (e) {
      setErroAcao(e instanceof Error ? e.message : "Não foi possível abrir a edição.");
    } finally {
      setAbrindo(false);
    }
  }

  // Busca o código do QR Code no servidor (fica registrado na auditoria) e gera a imagem
  async function montarPainelQr(membro: MembroPublico, codigo: string): Promise<PainelMembro> {
    const conteudo = montarConteudoQr(codigo);
    const imagem = await gerarImagemQr(conteudo);
    return { tipo: "qr", membro, imagem, conteudo };
  }

  async function abrirQr(membro: MembroPublico) {
    if (abrindo) return;
    setErroAcao(null);
    setAbrindo(true);
    try {
      setPainel(await montarPainelQr(membro, await obterQrMembro(membro.id)));
    } catch (e) {
      setErroAcao(e instanceof Error ? e.message : "Não foi possível abrir o QR Code.");
    } finally {
      setAbrindo(false);
    }
  }

  // Substitui o QR Code (o código antigo deixa de funcionar) e mostra a nova imagem
  async function substituirQr(membro: MembroPublico, justificativa: string) {
    const novoCodigo = await regenerarQrMembro(membro.id, justificativa);
    setPainel(await montarPainelQr(membro, novoCodigo));
    setAviso(`QR Code de "${membro.nome}" substituído. O código anterior deixou de funcionar.`);
  }

  // Confirma a exclusão (lógica) do membro escolhido
  async function confirmarExclusao() {
    if (!alvoExclusao) return;
    setExcluindo(true);
    setErroExclusao(null);
    try {
      await excluirMembro(alvoExclusao.id);
      await carregar();
      setAviso(`Membro "${alvoExclusao.nome}" excluído.`);
      setAlvoExclusao(null);
    } catch (e) {
      setErroExclusao(e instanceof Error ? e.message : "Não foi possível excluir o membro.");
    } finally {
      setExcluindo(false);
    }
  }

  // Sugestões dos campos de localidade (valores já cadastrados)
  const sugestoes = {
    congregacoes: valoresDistintos(membros.map((membro) => membro.congregacao)),
    cidades: valoresDistintos(membros.map((membro) => membro.cidade)),
    bairros: valoresDistintos(membros.map((membro) => membro.bairro)),
  };

  const tituloPainel =
    painel?.tipo === "editar"
      ? "Editar cadastro musical"
      : painel?.tipo === "qr"
        ? "QR Code do membro"
        : "Cadastro Musical";

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Músicos / Organistas"
          descricao={`Cadastre e acompanhe os membros do SysMúsica.${
            !carregando && !erro ? ` ${membros.length} cadastrado(s).` : ""
          }`}
          acoes={
            pode("membros.criar") ? (
              <Button icone={Plus} onClick={() => setPainel({ tipo: "novo" })} disabled={carregando || !!erro}>
                Adicionar
              </Button>
            ) : undefined
          }
        />

        {erroAcao && (
          <p role="alert" className="mb-4 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
            {erroAcao}
          </p>
        )}

        <ListaMembros
          membros={membros}
          cargos={cargos}
          instrumentos={instrumentos}
          familias={familias}
          carregando={carregando}
          erro={erro}
          urlsFotos={urlsFotos}
          onTentarNovamente={tentarNovamente}
          podeEditar={pode("membros.editar")}
          podeExcluir={pode("membros.excluir")}
          podeQr={pode("membros.qr")}
          onEditar={(membro) => void abrirEdicao(membro)}
          onQr={(membro) => void abrirQr(membro)}
          onExcluir={(membro) => {
            setErroExclusao(null);
            setAlvoExclusao(membro);
          }}
        />
      </div>

      {/* Painel lateral: novo membro, edição ou QR Code */}
      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={tituloPainel}
        descricao={
          painel?.tipo === "qr"
            ? "Identificação individual e permanente do membro."
            : "Campos com * são obrigatórios. A fotografia identifica a pessoa na portaria."
        }
      >
        {(painel?.tipo === "novo" || painel?.tipo === "editar") && (
          <FormularioMembro
            membro={painel.tipo === "editar" ? painel.membro : null}
            cpfInicial={painel.tipo === "editar" ? painel.cpfInicial : null}
            cadastroInicial={painel.tipo === "editar" ? painel.cadastro : null}
            podeCpf={pode("membros.cpf")}
            podeFoto={pode("membros.foto")}
            podeDocumento={pode("membros.cpf")}
            cargos={cargos}
            instrumentos={instrumentos}
            familias={familias}
            sugestoes={sugestoes}
            carregando={carregando}
            erro={erro}
            onSalvo={aoSalvar}
          />
        )}
        {painel?.tipo === "qr" && (
          <QrMembro
            nome={painel.membro.nome}
            congregacao={painel.membro.congregacao}
            imagem={painel.imagem}
            conteudo={painel.conteudo}
            podeSubstituir={pode("membros.qr_revogar")}
            onSubstituir={(justificativa) => substituirQr(painel.membro, justificativa)}
          />
        )}
      </Drawer>

      {/* Confirmação de exclusão */}
      <ConfirmDialog
        aberto={alvoExclusao !== null}
        titulo="Excluir membro?"
        mensagem={
          <>
            <p>
              <strong>{alvoExclusao?.nome}</strong> deixará de aparecer nas telas. O histórico de
              presenças dele é mantido. Para apenas impedir novos registros sem excluir, altere a situação
              cadastral para Inativo na edição.
            </p>
            {erroExclusao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroExclusao}
              </p>
            )}
          </>
        }
        textoConfirmar="Excluir"
        processando={excluindo}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExclusao(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

/**
 * SysMúsica — Serviço de dados da organização (Estado › RRM › RA › ADM › Setor › Casa de Oração) e dos vínculos dos usuários.
 * Todas as regras (permissão, nomes repetidos, abrangência) são aplicadas pelo banco (migração 0012);
 * a tela só chama e mostra o resultado, com as mensagens do banco já escritas para o usuário.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import type { NivelOrganizacao, NoOrganizacao, VinculoRotulado, VinculoUsuario } from "../tipos";

interface LinhaNo extends Omit<NoOrganizacao, "qtd_filhos" | "qtd_membros"> {
  qtd_filhos?: number | string;
  qtd_membros?: number | string;
}

/** Árvore completa (tela de Organização e vínculos). */
export async function listarOrganizacao(): Promise<NoOrganizacao[]> {
  const { data, error } = await supabase.rpc("listar_organizacao");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar a organização."));
  return ((data ?? []) as LinhaNo[]).map((linha) => ({
    ...linha,
    qtd_filhos: Number(linha.qtd_filhos ?? 0), // o banco devolve bigint
    qtd_membros: Number(linha.qtd_membros ?? 0),
  }));
}

/** Itens ativos que o usuário pode usar (seletor de casa de oração nos formulários). */
export async function listarOrganizacaoParaSelecao(): Promise<NoOrganizacao[]> {
  const { data, error } = await supabase.rpc("listar_organizacao_para_selecao");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar as casas de oração."));
  return (data ?? []) as NoOrganizacao[];
}

/** Cria (id nulo) ou altera um item da organização. */
export async function salvarNo(dados: {
  nivel: NivelOrganizacao;
  id: string | null;
  paiId: string | null;
  nome: string;
  descricao: string;
  ativo: boolean;
  cidade: string;
  bairro: string;
}): Promise<void> {
  const { error } = await supabase.rpc("salvar_organizacao", {
    p_nivel: dados.nivel,
    p_id: dados.id,
    p_parent: dados.paiId,
    p_nome: dados.nome,
    p_descricao: dados.descricao,
    p_ativo: dados.ativo,
    p_cidade: dados.cidade,
    p_bairro: dados.bairro,
  });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível salvar."));
}

/** Exclui um item VAZIO da organização (o banco confere permissão, filhos, membros, eventos, portarias e vínculos). */
export async function excluirNo(nivel: NivelOrganizacao, id: string): Promise<void> {
  const { error } = await supabase.rpc("excluir_organizacao", { p_nivel: nivel, p_id: id });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível excluir."));
}

/** Muda uma casa de oração de local: o destino é uma ADM (fica sem setor) ou um setor (fica no histórico e na auditoria). */
export async function transferirCasaOracao(casaOracaoId: string, destinoId: string, justificativa: string): Promise<void> {
  const { error } = await supabase.rpc("transferir_casa_oracao", {
    p_casa_oracao: casaOracaoId,
    p_destino: destinoId,
    p_justificativa: justificativa,
  });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível transferir a casa de oração."));
}

/** Vínculos de cada usuário com o rótulo: { idDoUsuario: [{ nivel, alvo_id, rotulo }] }. */
export async function listarVinculosRotulados(): Promise<Record<string, VinculoRotulado[]>> {
  const { data, error } = await supabase.rpc("listar_vinculos_rotulados");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar os vínculos dos usuários."));

  const mapa: Record<string, VinculoRotulado[]> = {};
  for (const linha of (data ?? []) as (VinculoRotulado & { usuario_id: string })[]) {
    mapa[linha.usuario_id] = [
      ...(mapa[linha.usuario_id] ?? []),
      { nivel: linha.nivel, alvo_id: linha.alvo_id, rotulo: linha.rotulo },
    ];
  }
  return mapa;
}

/** Define os vínculos de um usuário, substituindo a lista anterior (exige usuarios.vinculos). */
export async function definirVinculos(usuarioId: string, vinculos: VinculoUsuario[]): Promise<void> {
  const { error } = await supabase.rpc("definir_vinculos", { p_usuario: usuarioId, p_vinculos: vinculos });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível salvar os vínculos."));
}
