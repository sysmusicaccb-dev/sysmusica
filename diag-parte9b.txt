-- SysMúsica — Migração 0038: Cadastro Musical (dados pessoais do músico/organista)
--
--   * Tabela PROTEGIDA membros_cadastro: estado civil, sexo, batismo, selado, telefone, WhatsApp, e-mail,
--     endereço e cartão de identificação. Ninguém lê nem grava a tabela direto (sem permissão para
--     anon/authenticated e sem policy): só as funções abaixo, que conferem a permissão e a abrangência.
--   * obter_cadastro_musical(membro)  -> JSON com os dados (ou null se o membro ainda não tem cadastro)
--   * salvar_cadastro_musical(membro, json) -> grava/atualiza (permissão membros.criar ou membros.editar)
--   * cargos.exige_cartao: define quais cargos exigem o cartão de identificação (configurável por SQL).
--   * Sem permissão nova: criar/editar membros já cobre este cadastro.
--
-- OBS. para futuros administradores: esta tabela ainda NÃO entra no backup/restauração da tela Backup
-- (a lista de tabelas permitidas fica na função backup_aplicar). Incluir numa etapa própria.

-- 1) Cartão de identificação por cargo ---------------------------------------
alter table public.cargos add column if not exists exige_cartao boolean not null default false;
grant select (exige_cartao) on public.cargos to authenticated;

update public.cargos
   set exige_cartao = true
 where public.normalizar_texto(nome) in ('encarregado regional', 'encarregado local', 'examinadora');

-- 2) Tabela protegida ---------------------------------------------------------
create table if not exists public.membros_cadastro (
  membro_id        uuid primary key references public.membros (id) on delete cascade,
  estado_civil     text check (estado_civil in ('Casado(a)', 'Solteiro(a)', 'Viúvo(a)')),
  sexo             text check (sexo in ('Masculino', 'Feminino')),
  data_batismo     date,
  selado           boolean,
  ddd              text check (ddd ~ '^[0-9]{2}$'),
  telefone         text check (telefone ~ '^[0-9]{8,9}$'),
  tem_whatsapp     boolean,
  whatsapp         text check (whatsapp ~ '^[0-9]{10,11}$'),
  email            text,
  cep              text check (cep ~ '^[0-9]{8}$'),
  endereco         text,
  numero           text,
  complemento      text,
  uf               text check (uf ~ '^[A-Z]{2}$'),
  cartao_numero    text,
  cartao_emissao   date,
  cartao_validade  date,
  apresentado_em   date,
  apresentado_por  text,
  observacao       text,
  criado_em        timestamptz not null default now(),
  atualizado_em    timestamptz not null default now(),
  constraint membros_cadastro_cartao_datas check (
    cartao_emissao is null or cartao_validade is null or cartao_emissao <= cartao_validade
  )
);

alter table public.membros_cadastro enable row level security;
revoke all on public.membros_cadastro from public, anon, authenticated;

-- 3) Ler o cadastro ----------------------------------------------------------
create or replace function public.obter_cadastro_musical(p_membro uuid)
returns jsonb
language plpgsql
stable
security definer
set search_path to 'public'
as $$
declare
  v_casa uuid;
  v_res  jsonb;
begin
  if not (public.tem_permissao('membros.editar') or public.tem_permissao('membros.criar')) then
    raise exception 'Você não tem permissão para ver este cadastro.' using errcode = '42501';
  end if;

  select m.casa_oracao_id into v_casa
    from public.membros m
   where m.id = p_membro and m.excluido_em is null;
  if not found then
    raise exception 'Membro não encontrado.' using errcode = 'P0002';
  end if;
  if v_casa is not null and not public.casa_oracao_na_abrangencia(v_casa) then
    raise exception 'Você não tem acesso a este cadastro.' using errcode = '42501';
  end if;

  select to_jsonb(c) - 'membro_id' - 'criado_em' - 'atualizado_em'
    into v_res
    from public.membros_cadastro c
   where c.membro_id = p_membro;
  return v_res; -- null quando o membro ainda não tem o cadastro preenchido
end;
$$;

create or replace function public.salvar_cadastro_musical(p_membro uuid, p_dados jsonb)
returns void
language plpgsql
security definer
set search_path to 'public'
as $$
declare
  v_casa   uuid;
  v_cargo  uuid;
  v_exige  boolean;
  v_estado_civil    text := nullif(btrim(p_dados ->> 'estado_civil'), '');
  v_sexo            text := nullif(btrim(p_dados ->> 'sexo'), '');
  v_data_batismo    date := nullif(btrim(p_dados ->> 'data_batismo'), '')::date;
  v_selado          boolean := (p_dados ->> 'selado')::boolean;
  v_ddd             text := nullif(btrim(p_dados ->> 'ddd'), '');
  v_telefone        text := nullif(btrim(p_dados ->> 'telefone'), '');
  v_tem_whatsapp    boolean := (p_dados ->> 'tem_whatsapp')::boolean;
  v_whatsapp        text := nullif(btrim(p_dados ->> 'whatsapp'), '');
  v_email           text := nullif(btrim(p_dados ->> 'email'), '');
  v_cep             text := nullif(btrim(p_dados ->> 'cep'), '');
  v_endereco        text := nullif(btrim(p_dados ->> 'endereco'), '');
  v_numero          text := nullif(btrim(p_dados ->> 'numero'), '');
  v_complemento     text := nullif(btrim(p_dados ->> 'complemento'), '');
  v_uf              text := nullif(btrim(p_dados ->> 'uf'), '');
  v_cartao_numero   text := nullif(btrim(p_dados ->> 'cartao_numero'), '');
  v_cartao_emissao  date := nullif(btrim(p_dados ->> 'cartao_emissao'), '')::date;
  v_cartao_validade date := nullif(btrim(p_dados ->> 'cartao_validade'), '')::date;
  v_apresentado_em  date := nullif(btrim(p_dados ->> 'apresentado_em'), '')::date;
  v_apresentado_por text := nullif(btrim(p_dados ->> 'apresentado_por'), '');
  v_observacao      text := nullif(btrim(p_dados ->> 'observacao'), '');
begin
  if not (public.tem_permissao('membros.criar') or public.tem_permissao('membros.editar')) then
    raise exception 'Você não tem permissão para salvar este cadastro.' using errcode = '42501';
  end if;

  select m.casa_oracao_id, m.cargo_id into v_casa, v_cargo
    from public.membros m
   where m.id = p_membro and m.excluido_em is null;
  if not found then
    raise exception 'Membro não encontrado.' using errcode = 'P0002';
  end if;
  if v_casa is not null and not public.casa_oracao_na_abrangencia(v_casa) then
    raise exception 'Você não tem acesso a este cadastro.' using errcode = '42501';
  end if;

  -- Validações com mensagem clara (as regras da tabela são a segunda barreira)
  if v_estado_civil is not null and v_estado_civil not in ('Casado(a)', 'Solteiro(a)', 'Viúvo(a)') then
    raise exception 'Estado civil inválido.' using errcode = '22023';
  end if;
  if v_sexo is not null and v_sexo not in ('Masculino', 'Feminino') then
    raise exception 'Sexo inválido.' using errcode = '22023';
  end if;
  if v_ddd is not null and v_ddd !~ '^[0-9]{2}$' then
    raise exception 'O DDD deve ter 2 números.' using errcode = '22023';
  end if;
  if v_telefone is not null and v_telefone !~ '^[0-9]{8,9}$' then
    raise exception 'O telefone deve ter 8 ou 9 números.' using errcode = '22023';
  end if;
  if v_whatsapp is not null and v_whatsapp !~ '^[0-9]{10,11}$' then
    raise exception 'O WhatsApp deve ter o DDD e 8 ou 9 números.' using errcode = '22023';
  end if;
  if v_email is not null and v_email !~ '^[^[:space:]@]+@[^[:space:]@]+\.[^[:space:]@]+$' then
    raise exception 'O e-mail informado não é válido.' using errcode = '22023';
  end if;
  if v_cep is not null and v_cep !~ '^[0-9]{8}$' then
    raise exception 'O CEP deve ter 8 números.' using errcode = '22023';
  end if;
  if v_uf is not null and v_uf !~ '^[A-Z]{2}$' then
    raise exception 'Estado (UF) inválido.' using errcode = '22023';
  end if;
  if v_cartao_emissao is not null and v_cartao_validade is not null and v_cartao_emissao > v_cartao_validade then
    raise exception 'A validade do cartão não pode ser anterior à emissão.' using errcode = '22023';
  end if;

  -- Cargos que exigem o cartão de identificação (coluna cargos.exige_cartao)
  select c.exige_cartao into v_exige from public.cargos c where c.id = v_cargo;
  if coalesce(v_exige, false) and (v_cartao_numero is null or v_cartao_validade is null) then
    raise exception 'Este cargo exige o número e a validade do cartão de identificação.' using errcode = '22023';
  end if;

  -- Sem WhatsApp: não guarda número de WhatsApp
  if v_tem_whatsapp is not true then
    v_whatsapp := null;
  end if;

  insert into public.membros_cadastro as c (
    membro_id, estado_civil, sexo, data_batismo, selado, ddd, telefone, tem_whatsapp, whatsapp, email,
    cep, endereco, numero, complemento, uf, cartao_numero, cartao_emissao, cartao_validade,
    apresentado_em, apresentado_por, observacao
  ) values (
    p_membro, v_estado_civil, v_sexo, v_data_batismo, v_selado, v_ddd, v_telefone, v_tem_whatsapp, v_whatsapp, v_email,
    v_cep, v_endereco, v_numero, v_complemento, v_uf, v_cartao_numero, v_cartao_emissao, v_cartao_validade,
    v_apresentado_em, v_apresentado_por, v_observacao
  )
  on conflict (membro_id) do update set
    estado_civil = excluded.estado_civil, sexo = excluded.sexo, data_batismo = excluded.data_batismo,
    selado = excluded.selado, ddd = excluded.ddd, telefone = excluded.telefone,
    tem_whatsapp = excluded.tem_whatsapp, whatsapp = excluded.whatsapp, email = excluded.email,
    cep = excluded.cep, endereco = excluded.endereco, numero = excluded.numero,
    complemento = excluded.complemento, uf = excluded.uf, cartao_numero = excluded.cartao_numero,
    cartao_emissao = excluded.cartao_emissao, cartao_validade = excluded.cartao_validade,
    apresentado_em = excluded.apresentado_em, apresentado_por = excluded.apresentado_por,
    observacao = excluded.observacao, atualizado_em = now();
end;
$$;

-- SysMúsica — Migração 0040: Cadastro Musical — campos obrigatórios, Batizado, documento e instrumento Órgão
--
--   * Estado civil passa a ser "Casado (a)", "Solteiro (a)", "Viúvo (a)" (os cadastros já gravados são convertidos).
--   * membros_cadastro ganha: batizado (Sim/Não), documento_frente_path e documento_verso_path.
--   * Bucket PRIVADO documentos-membros (frente e verso do documento de identificação):
--       - anexar / trocar / descartar: quem tem membros.criar ou membros.editar;
--       - VER o documento (URL temporária): só quem tem a permissão membros.cpf.
--   * cargos.somente_orgao: cargos que só podem usar o instrumento Órgão (Organista, Examinadora e Instrutora;
--     o Instrutor continua com instrumento livre). O banco recusa outro instrumento (gatilho trg_membros_somente_orgao). Para mudar a lista:
--         update public.cargos set somente_orgao = true|false where nome = '...';
--   * Cargo "Instrutor (a)" vira "Instrutor" e é criado o cargo "Instrutora".
--   * salvar_cadastro_musical passa a EXIGIR: CPF, data de nascimento, estado civil, sexo, batizado (e a data do
--     batismo quando for batizado), instrumento (nos cargos que usam instrumento) e a frente e o verso do documento.
--     (A função só é chamada para o cadastro completo; membros antigos sem cadastro continuam como estão.)
--
-- Tudo roda numa transação só: se der erro em qualquer ponto, NADA é alterado.
-- É seguro rodar mais de uma vez.
-- =============================================================================
begin;

-- 1) Cargos -----------------------------------------------------------------
alter table public.cargos add column if not exists somente_orgao boolean not null default false;
grant select (somente_orgao) on public.cargos to authenticated;

-- "Instrutor (a)" -> "Instrutor" (ninguém usa ainda; se já existir "Instrutor", não mexe)
update public.cargos c
   set nome = 'Instrutor'
 where c.nome = 'Instrutor (a)'
   and not exists (select 1 from public.cargos x where x.nome = 'Instrutor');

-- Novo cargo "Instrutora"
insert into public.cargos (nome, permite_instrumento, exige_instrumento, exige_cartao)
select 'Instrutora', true, true, false
 where not exists (select 1 from public.cargos x where x.nome = 'Instrutora');

-- Cargos travados no Órgão (Organista, Examinadora e Instrutora): o instrumento passa a ser obrigatório
update public.cargos
   set somente_orgao = true,
       permite_instrumento = true,
       exige_instrumento = true
 where public.normalizar_texto(nome) like 'organista%'
    or public.normalizar_texto(nome) like 'examinadora%'
    or public.normalizar_texto(nome) like 'instrutora%';

-- O Instrutor (masculino) NÃO é travado: toca qualquer instrumento (como era antes)
update public.cargos
   set somente_orgao = false,
       exige_instrumento = false
 where public.normalizar_texto(nome) = 'instrutor';

-- Confere os membros que já existem: ninguém pode estar com instrumento diferente de Órgão nesses cargos
do $$
declare v_qtd integer;
begin
  select count(*) into v_qtd
    from public.membros m
    join public.cargos c on c.id = m.cargo_id and c.somente_orgao
    join public.instrumentos i on i.id = m.instrumento_id
   where m.excluido_em is null
     and not public.instrumento_e_orgao(i.nome);
  if v_qtd > 0 then
    raise exception 'Existem % membro(s) em cargo travado no Órgão com outro instrumento. Corrija antes de rodar esta migração.', v_qtd;
  end if;
end $$;

-- Gatilho: o cargo travado só aceita instrumento Órgão (vale também para visitantes)
create or replace function public.membros_somente_orgao()
returns trigger
language plpgsql
security definer
set search_path to 'public'
as $$
declare
  v_somente boolean;
  v_cargo   text;
  v_instr   text;
begin
  -- Restauração de backup: os dados antigos entram como estavam
  if current_setting('sysmusica.restaurando', true) = 'sim' then
    return new;
  end if;
  if new.instrumento_id is null then
    return new; -- a obrigatoriedade do instrumento é conferida por outro gatilho
  end if;
  select c.somente_orgao, c.nome into v_somente, v_cargo from public.cargos c where c.id = new.cargo_id;
  if not coalesce(v_somente, false) then
    return new;
  end if;
  select i.nome into v_instr from public.instrumentos i where i.id = new.instrumento_id;
  if not public.instrumento_e_orgao(v_instr) then
    raise exception 'O cargo % só pode usar o instrumento Órgão.', v_cargo using errcode = '22023';
  end if;
  return new;
end;
$$;

drop trigger if exists trg_membros_somente_orgao on public.membros;
create trigger trg_membros_somente_orgao
  before insert or update of cargo_id, instrumento_id on public.membros
  for each row execute function public.membros_somente_orgao();

-- 4) Gravar o cadastro (agora com as exigências) ---------------------------------
create or replace function public.salvar_cadastro_musical(p_membro uuid, p_dados jsonb)
returns void
language plpgsql
security definer
set search_path to 'public'
as $$
declare
  v_casa   uuid;
  v_cargo  uuid;
  v_exige  boolean;
  v_permite_instr boolean;
  v_cpf    text;
  v_nasc   date;
  v_instr  uuid;
  v_estado_civil    text := nullif(btrim(p_dados ->> 'estado_civil'), '');
  v_sexo            text := nullif(btrim(p_dados ->> 'sexo'), '');
  v_batizado        boolean := (p_dados ->> 'batizado')::boolean;
  v_data_batismo    date := nullif(btrim(p_dados ->> 'data_batismo'), '')::date;
  v_selado          boolean := (p_dados ->> 'selado')::boolean;
  v_ddd             text := nullif(btrim(p_dados ->> 'ddd'), '');
  v_telefone        text := nullif(btrim(p_dados ->> 'telefone'), '');
  v_tem_whatsapp    boolean := (p_dados ->> 'tem_whatsapp')::boolean;
  v_whatsapp        text := nullif(btrim(p_dados ->> 'whatsapp'), '');
  v_email           text := nullif(btrim(p_dados ->> 'email'), '');
  v_cep             text := nullif(btrim(p_dados ->> 'cep'), '');
  v_endereco        text := nullif(btrim(p_dados ->> 'endereco'), '');
  v_numero          text := nullif(btrim(p_dados ->> 'numero'), '');
  v_complemento     text := nullif(btrim(p_dados ->> 'complemento'), '');
  v_uf              text := nullif(btrim(p_dados ->> 'uf'), '');
  v_cartao_numero   text := nullif(btrim(p_dados ->> 'cartao_numero'), '');
  v_cartao_emissao  date := nullif(btrim(p_dados ->> 'cartao_emissao'), '')::date;
  v_cartao_validade date := nullif(btrim(p_dados ->> 'cartao_validade'), '')::date;
  v_apresentado_em  date := nullif(btrim(p_dados ->> 'apresentado_em'), '')::date;
  v_apresentado_por text := nullif(btrim(p_dados ->> 'apresentado_por'), '');
  v_observacao      text := nullif(btrim(p_dados ->> 'observacao'), '');
  v_doc_frente      text := nullif(btrim(p_dados ->> 'documento_frente_path'), '');
  v_doc_verso       text := nullif(btrim(p_dados ->> 'documento_verso_path'), '');
begin
  if not (public.tem_permissao('membros.criar') or public.tem_permissao('membros.editar')) then
    raise exception 'Você não tem permissão para salvar este cadastro.' using errcode = '42501';
  end if;

  select m.casa_oracao_id, m.cargo_id, m.cpf, m.data_nascimento, m.instrumento_id
    into v_casa, v_cargo, v_cpf, v_nasc, v_instr
    from public.membros m
   where m.id = p_membro and m.excluido_em is null;
  if not found then
    raise exception 'Membro não encontrado.' using errcode = 'P0002';
  end if;
  if v_casa is not null and not public.casa_oracao_na_abrangencia(v_casa) then
    raise exception 'Você não tem acesso a este cadastro.' using errcode = '42501';
  end if;

  -- Obrigatórios do Cadastro Musical (a tela confere antes; o banco é a segunda barreira)
  if v_cpf is null then
    raise exception 'O CPF é obrigatório. Quem tem a permissão CPF precisa preenchê-lo.' using errcode = '22023';
  end if;
  if v_nasc is null then
    raise exception 'Informe a data de nascimento.' using errcode = '22023';
  end if;
  if v_estado_civil is null then
    raise exception 'Informe o estado civil.' using errcode = '22023';
  end if;
  if v_sexo is null then
    raise exception 'Informe o sexo.' using errcode = '22023';
  end if;
  if v_batizado is null then
    raise exception 'Informe se a pessoa é batizada.' using errcode = '22023';
  end if;
  if v_batizado and v_data_batismo is null then
    raise exception 'Informe a data do batismo.' using errcode = '22023';
  end if;
  if not v_batizado then
    v_data_batismo := null; -- sem batismo, não guarda data
  end if;
  if v_data_batismo is not null and v_data_batismo < v_nasc then
    raise exception 'A data do batismo não pode ser anterior ao nascimento.' using errcode = '22023';
  end if;

  select c.permite_instrumento into v_permite_instr from public.cargos c where c.id = v_cargo;
  if coalesce(v_permite_instr, false) and v_instr is null then
    raise exception 'Informe o instrumento.' using errcode = '22023';
  end if;

/** O mínimo que precisamos de um cargo (o tipo Cargo do projeto atende a isto). */

  nome: string;

function semAcento(texto: string): string {
  return texto
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .toLowerCase()
    .trim();
}

/**
 * Cargos que têm Cadastro Musical (início do nome, sem acento). Ancião, Diácono, Cooperadores e Secretário da
 * Música não têm mais cadastro: só entram nas contagens. "Candidato(a)" ficou de fora até a CCB confirmar.
 */
export const CARGOS_DO_CADASTRO = [
  "encarregado regional",
  "encarregado local",
  "examinadora",
  "instrutor",
  "musico",
  "organista",
] as const;

/** Cargos que exigem o cartão de identificação enquanto a coluna cargos.exige_cartao não decide. */
const CARGOS_COM_CARTAO_PADRAO = ["encarregado regional", "encarregado local", "examinadora"];

/** Cargos que só aceitam o instrumento Órgão enquanto a coluna cargos.somente_orgao não decide. */
const CARGOS_SOMENTE_ORGAO_PADRAO = ["organista", "examinadora", "instrutor"];

/** O cargo só pode usar o instrumento Órgão? (Organista, Examinadora, Instrutor e Instrutora.) */
export function cargoSomenteOrgao(cargo: CargoCadastro | null | undefined): boolean {
  if (!cargo) return false;
  if (typeof cargo.somente_orgao === "boolean") return cargo.somente_orgao;
  const nome = semAcento(cargo.nome);
  return CARGOS_SOMENTE_ORGAO_PADRAO.some((prefixo) => nome.startsWith(prefixo));
}

/** O instrumento é um Órgão? (mesma regra do banco: o nome tem "orgao", sem acento.) */
export function instrumentoEOrgao(nome: string | null | undefined): boolean {
  return !!nome && semAcento(nome).includes("orgao");
}

/** Instrumentos que o cargo aceita: todos, ou só os Órgãos nos cargos travados. */
export function instrumentosDoCargo<T extends { nome: string }>(instrumentos: T[], somenteOrgao: boolean): T[] {
  return somenteOrgao ? instrumentos.filter((instrumento) => instrumentoEOrgao(instrumento.nome)) : instrumentos;
}

export function cargoEntraNoCadastro(nomeCargo: string | null | undefined): boolean {
  if (!nomeCargo) return false;
  const nome = semAcento(nomeCargo);
  return CARGOS_DO_CADASTRO.some((prefixo) => nome.startsWith(prefixo));
}

/**
 * Cargos oferecidos no formulário. Na edição, o cargo atual do membro continua na lista mesmo que seja
 * um cargo antigo (ex.: Ancião), para não apagar nem trocar o cargo sem querer.
 */
export function cargosDoCadastro<T extends CargoCadastro>(cargos: T[], cargoAtualId?: string | null): T[] {
  return cargos.filter((cargo) => cargoEntraNoCadastro(cargo.nome) || cargo.id === cargoAtualId);
}

/** O cargo exige o cartão de identificação (número, validade...)? */
export function cargoExigeCartao(cargo: CargoCadastro | null | undefined): boolean {
  if (!cargo) return false;
  if (typeof cargo.exige_cartao === "boolean") return cargo.exige_cartao;
  const nome = semAcento(cargo.nome);
  return CARGOS_COM_CARTAO_PADRAO.some((prefixo) => nome.startsWith(prefixo));
}

// ---------- Formatos de digitação ----------

export function somenteDigitos(texto: string, maximo?: number): string {
  const digitos = texto.replace(/\D/g, "");
  return maximo ? digitos.slice(0, maximo) : digitos;
}

/** "40000-000" (aceita digitação parcial). */
export function formatarCep(texto: string): string {
  const d = somenteDigitos(texto, 8);
  return d.length > 5 ? `${d.slice(0, 5)}-${d.slice(5)}` : d;
}

/** Telefone sem DDD: "99999-9999" (9 dígitos) ou "9999-9999" (8 dígitos). */
export function formatarTelefone(texto: string): string {
  const d = somenteDigitos(texto, 9);
  if (d.length > 8) return `${d.slice(0, 5)}-${d.slice(5)}`;
  return d.length > 4 ? `${d.slice(0, 4)}-${d.slice(4)}` : d;
}

/** WhatsApp com DDD: "(71) 99999-9999" (aceita digitação parcial). */
export function formatarWhatsapp(texto: string): string {
  const d = somenteDigitos(texto, 11);
  if (d.length <= 2) return d.length ? `(${d}` : "";
  const resto = d.slice(2);
  const corte = resto.length > 8 ? 5 : 4;
  return `(${d.slice(0, 2)}) ${resto.length > corte ? `${resto.slice(0, corte)}-${resto.slice(corte)}` : resto}`;
}

/** Confere o cadastro. Devolve a mensagem do primeiro problema ou null se estiver tudo certo. */
export function validarCadastro(dados: CadastroMusical, opcoes: OpcoesValidacao): string | null {
  const { completo, exigeCartao, cidade, bairro } = opcoes;
  const vazio = (texto: string) => texto.trim() === "";

  if (completo) {
    if (opcoes.cpfObrigatorio && vazio(opcoes.cpf)) return "Informe o CPF.";
    if (vazio(opcoes.dataNascimento)) return "Informe a data de nascimento.";
    if (vazio(dados.estado_civil)) return "Informe o estado civil.";
    if (vazio(dados.sexo)) return "Informe o sexo.";
    if (opcoes.instrumentoObrigatorio && vazio(opcoes.instrumentoId)) return "Informe o instrumento.";
    if (dados.batizado === "") return "Informe se a pessoa é batizada.";
    if (dados.batizado === "sim" && vazio(dados.data_batismo)) return "Informe a data do batismo.";
    if (dados.selado === "") return "Informe se a pessoa é selada.";
    if (vazio(dados.ddd) || vazio(dados.telefone)) return "Informe o DDD e o telefone.";
    if (dados.tem_whatsapp === "") return "Informe se o telefone tem WhatsApp.";
    if (vazio(dados.cep)) return "Informe o CEP.";
    if (vazio(dados.endereco)) return "Informe o endereço.";
    if (vazio(dados.numero)) return "Informe o número do endereço.";
    if (vazio(dados.uf)) return "Informe o estado (UF).";
    if (vazio(cidade)) return "Informe a cidade.";
    if (vazio(bairro)) return "Informe o bairro.";
    if (!opcoes.temDocumentoFrente) return "Anexe a frente do documento.";
    if (!opcoes.temDocumentoVerso) return "Anexe o verso do documento.";
  }

  if (
    dados.batizado === "sim" &&
    !vazio(dados.data_batismo) &&
    !vazio(opcoes.dataNascimento) &&
    dados.data_batismo < opcoes.dataNascimento
  ) {
    return "A data do batismo não pode ser anterior ao nascimento.";
  }

  if (!vazio(dados.ddd) && !/^\d{2}$/.test(dados.ddd)) return "O DDD deve ter 2 números.";
  if (!vazio(dados.telefone) && !/^\d{8,9}$/.test(dados.telefone)) return "O telefone deve ter 8 ou 9 números.";
  if (dados.tem_whatsapp === "sim" && !vazio(dados.whatsapp) && !/^\d{10,11}$/.test(dados.whatsapp)) {
    return "O WhatsApp deve ter o DDD e 8 ou 9 números.";
  }
  if (!vazio(dados.email) && !emailValido(dados.email)) return "O e-mail informado não é válido.";
  if (!vazio(dados.cep) && !/^\d{8}$/.test(dados.cep)) return "O CEP deve ter 8 números.";
  if (!vazio(dados.uf) && !(UFS as readonly string[]).includes(dados.uf)) return "Escolha um estado (UF) da lista.";

  if (exigeCartao) {
    if (vazio(dados.cartao_numero)) return "Informe o número do cartão de identificação.";
    if (vazio(dados.cartao_validade)) return "Informe a validade do cartão de identificação.";
  }
  if (!vazio(dados.cartao_emissao) && !vazio(dados.cartao_validade) && dados.cartao_emissao > dados.cartao_validade) {
    return "A validade do cartão não pode ser anterior à emissão.";
  }
  return null;
}

/** O que o CEP preenche no formulário (campo vazio = o serviço não trouxe aquele dado). */

/**
 * Resposta do ViaCEP (https://viacep.com.br/ws/CEP/json/) -> dados do formulário.
 * Devolve null quando o CEP não existe (o serviço responde {"erro": true}) ou a resposta é inválida.
 * CEP "geral" de cidade pequena vem sem rua e sem bairro: o que faltar o usuário digita.
 */
export function enderecoDoViaCep(bruto: unknown): EnderecoDoCep | null {
  if (!bruto || typeof bruto !== "object") return null;
  const b = bruto as Bruto;
  if (b.erro === true || b.erro === "true") return null;
  const uf = texto(b.uf).trim().toUpperCase();
  const cidade = texto(b.localidade).trim();
  if (!(UFS as readonly string[]).includes(uf) && cidade === "") return null; // resposta sem nada útil
  return {

// Guarda as URLs temporárias por alguns minutos para não pedir a mesma foto várias vezes
const cacheUrls = new Map<string, { url: string; expira: number }>();
const VALIDADE_URL_SEGUNDOS = 600;

/** URL temporária de uma foto (ou null se não for possível gerar). */
export async function urlDaFoto(caminho: string): Promise<string | null> {
  const agora = Date.now();
  const guardada = cacheUrls.get(caminho);
  if (guardada && guardada.expira > agora) return guardada.url;

  const { data, error } = await supabase.storage
    .from(BUCKET_FOTOS)
    .createSignedUrl(caminho, VALIDADE_URL_SEGUNDOS);
  if (error || !data) return null;

  cacheUrls.set(caminho, { url: data.signedUrl, expira: agora + (VALIDADE_URL_SEGUNDOS - 60) * 1000 });
  return data.signedUrl;
}

/** URLs temporárias de várias fotos de uma vez: { caminho: url }. */
export async function urlsDasFotos(caminhos: string[]): Promise<Record<string, string>> {
  if (caminhos.length === 0) return {};
  const { data, error } = await supabase.storage
    .from(BUCKET_FOTOS)
    .createSignedUrls(caminhos, VALIDADE_URL_SEGUNDOS);
  if (error || !data) return {};

  const mapa: Record<string, string> = {};
  for (const item of data) {
    if (item.path && item.signedUrl) mapa[item.path] = item.signedUrl;
  }
  return mapa;
}

/** Envia a foto (já reduzida) e a vincula ao membro; apaga o arquivo anterior. */

 * A conversão da resposta fica em cadastroMusical.ts (enderecoDoViaCep), onde é testada.
 */

export type ResultadoCep =
  | { tipo: "ok"; endereco: EnderecoDoCep }
  | { tipo: "nao_encontrado" } // o CEP não existe
  | { tipo: "indisponivel" }; // sem internet, serviço fora do ar ou demorou demais

const TEMPO_MAXIMO_MS = 8000;

/** Busca o endereço de um CEP de 8 números. Nunca lança erro: devolve "indisponivel" se algo falhar. */
export async function buscarEnderecoPorCep(cep: string): Promise<ResultadoCep> {
  if (!/^\d{8}$/.test(cep)) return { tipo: "nao_encontrado" };
  const controle = new AbortController();
  const relogio = setTimeout(() => controle.abort(), TEMPO_MAXIMO_MS);
  try {
    const resposta = await fetch(`https://viacep.com.br/ws/${cep}/json/`, { signal: controle.signal });
    if (resposta.status === 400 || resposta.status === 404) return { tipo: "nao_encontrado" };
    if (!resposta.ok) return { tipo: "indisponivel" };
    const endereco = enderecoDoViaCep(await resposta.json());
    return endereco ? { tipo: "ok", endereco } : { tipo: "nao_encontrado" };
  } catch {
    return { tipo: "indisponivel" };
  } finally {
    clearTimeout(relogio);
  }
}

/**
 * SysMúsica — Preparo da fotografia no navegador antes do envio.
 * Reduz a imagem (lado maior de 800 px) e converte para JPEG: a foto fica leve (poucas centenas de KB)
 * e o envio é rápido, inclusive em rede móvel. A conferência de tipo e tamanho final também é feita
 * pelo armazenamento do banco (limite de 2 MB; JPEG, PNG ou WebP).
 */

const LADO_MAXIMO = 800;
const QUALIDADE_JPEG = 0.82;
const TAMANHO_MAXIMO_ORIGINAL = 15 * 1024 * 1024; // 15 MB de entrada (a saída é bem menor)

export async function prepararFoto(arquivo: File): Promise<Blob> {
  if (!arquivo.type.startsWith("image/")) {
    throw new Error("Escolha um arquivo de imagem (JPG, PNG ou WebP).");
  }
  if (arquivo.size > TAMANHO_MAXIMO_ORIGINAL) {
    throw new Error("A imagem é grande demais (máximo de 15 MB).");
  }

  // "from-image" respeita a rotação gravada pela câmera do celular; se o navegador não aceitar a opção, tenta sem ela
  let bitmap: ImageBitmap;
  try {
    bitmap = await createImageBitmap(arquivo, { imageOrientation: "from-image" });
  } catch {
    try {
      bitmap = await createImageBitmap(arquivo);
    } catch {
      throw new Error("Não foi possível ler esta imagem. Tente outra foto.");
    }
  }

  const escala = Math.min(1, LADO_MAXIMO / Math.max(bitmap.width, bitmap.height));
  const largura = Math.max(1, Math.round(bitmap.width * escala));
  const altura = Math.max(1, Math.round(bitmap.height * escala));

  const canvas = document.createElement("canvas");
  canvas.width = largura;
  canvas.height = altura;
  const contexto = canvas.getContext("2d");
  if (!contexto) {
    bitmap.close();
    throw new Error("Não foi possível processar a imagem neste navegador.");
  }
  contexto.drawImage(bitmap, 0, 0, largura, altura);
  bitmap.close();

  const imagem = await new Promise<Blob | null>((resolver) =>
    canvas.toBlob(resolver, "image/jpeg", QUALIDADE_JPEG)
  );
  if (!imagem) throw new Error("Não foi possível processar a imagem.");
  return imagem;
}

      setErroForm(e instanceof Error ? e.message : "Não foi possível salvar o membro.");

      setSalvando(false);
    }
  }

  // ---------- Estados da tela ----------

  if (carregando) {
    return <p className="text-sm text-slate-600">Carregando os dados do formulário...</p>;
  }

  if (erro) {
    return (
      <p className="text-sm text-slate-600">
        Formulário indisponível porque os dados não puderam ser carregados.
      </p>
    );
  }

  // Sem cargos: a migração do banco ainda não foi aplicada
  if (cargos.length === 0) {
    return (
      <p className="text-sm text-slate-600">
        Nenhum cargo encontrado. Aplique as migrações do banco (pasta supabase/migrations) para criar
        os cargos padrão.
      </p>
    );
  }

  // Data máxima do campo de nascimento = hoje (formato AAAA-MM-DD, no fuso local)
  const hoje = new Date().toLocaleDateString("sv-SE");
  const rotuloSimNao = (obrigatorio: boolean) => (obrigatorio ? "Selecione" : "—");

  return (
    <form onSubmit={salvar} className="space-y-6">
      {/* ---------- Fotografia ---------- */}
      {/* Obrigatória no cadastro novo; na edição só troca/remove quem tem a permissão de fotografia */}
      {!editando || podeFoto ? (
        <div className="space-y-1.5">
          <span className="text-sm font-medium text-slate-700">
            Fotografia{" "}
            {editando ? (
              <span className="font-normal text-slate-400">(opcional na edição)</span>
            ) : (
              <span className="font-normal text-red-600">(obrigatória)</span>
            )}
          </span>
          <CampoFoto
            nome={nome || "Membro"}
            fotoAtualCaminho={membro?.foto_path ?? null}
            obrigatoria={!editando}
            onChange={setFoto}
            disabled={salvando}
          />
          {editando && membro?.foto_path && (
            <div className="pt-1">
              {confirmandoRemocaoFoto ? (
                <div className="flex flex-wrap items-center gap-2 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-800">
                  <span>Remover a fotografia? Na portaria ele aparecerá sem foto.</span>
                  <button
                    type="button"
                    onClick={() => void removerFotoAtual()}
                    disabled={salvando}
                    className="rounded-lg bg-red-600 px-3 py-1 text-xs font-semibold text-white hover:bg-red-500 disabled:opacity-60"
                  >
                    Sim, remover
                  </button>
                  <button
                    type="button"
                    onClick={() => setConfirmandoRemocaoFoto(false)}
                    disabled={salvando}
                    className="rounded-lg border border-slate-300 bg-white px-3 py-1 text-xs font-medium text-slate-700 hover:bg-slate-50 disabled:opacity-60"
                  >
                    Cancelar
                  </button>
                </div>
              ) : (
                <button
                  type="button"
                  onClick={() => setConfirmandoRemocaoFoto(true)}
                  disabled={salvando}
                  className="text-xs font-semibold text-red-600 hover:underline disabled:opacity-60"
                >
                  Remover a fotografia atual
                </button>
              )}
            </div>
          )}
        </div>
      ) : (
        <div className="space-y-1.5">
          <span className="text-sm font-medium text-slate-700">Fotografia</span>
          <FotoMembro caminho={membro?.foto_path ?? null} nome={nome || "Membro"} tamanho="grande" />
          <p className="text-xs text-slate-500">A fotografia só pode ser trocada por quem tem a permissão Fotografia.</p>
        </div>
      )}

      {/* ---------- Identificação ---------- */}
      <Secao titulo="Identificação">
        <Campo id="nome" rotulo="Nome completo" obrigatorio>
          <input
            id="nome"
            type="text"
            required
            value={nome}
            onChange={(e) => setNome(e.target.value)}
            disabled={salvando}
            placeholder="Ex: João da Silva"
            className={classeInput}
          />
        </Campo>

        {/* CPF: só para quem tem a permissão membros.cpf */}
        {podeCpf && (
          <Campo
            id="cpf"
            rotulo="CPF"
            obrigatorio={completo}
            dica="Evita cadastros duplicados. O CPF é protegido: só quem tem permissão o vê por completo."
          >
            <input
              id="cpf"
              type="text"
              inputMode="numeric"
              autoComplete="off"
              value={cpf}
              onChange={(e) => setCpf(formatarCpf(e.target.value))}
              disabled={salvando}
              placeholder="000.000.000-00"
              className={classeInput}
            />
          </Campo>
        )}

        {/* Sem a permissão CPF não dá para preencher o CPF obrigatório de um cadastro novo */}
        {!podeCpf && !editando && (
          <p role="alert" className="rounded-lg bg-amber-50 px-3 py-2 text-xs text-amber-800">
            O CPF é obrigatório no cadastro e o seu perfil não tem a permissão CPF. Peça a um administrador para liberá-la
            em Administração &gt; Grupos de acesso.
          </p>
        )}

        <Campo id="data_nascimento" rotulo="Data de nascimento" obrigatorio={completo}>
          <input
            id="data_nascimento"
            type="date"
            max={hoje}
            value={dataNascimento}
            onChange={(e) => setDataNascimento(e.target.value)}
            disabled={salvando}
            className={classeInput}
          />
        </Campo>

        <div className="grid grid-cols-2 gap-3">
          <Campo id="estado_civil" rotulo="Estado civil" obrigatorio={completo}>
            <select
              id="estado_civil"
              value={cadastro.estado_civil}
              onChange={(e) => mudar("estado_civil", e.target.value)}
              disabled={salvando}
              className={classeInput}
            >
              <option value="">{rotuloSimNao(completo)}</option>
              {ESTADOS_CIVIS.map((item) => (
                <option key={item} value={item}>
                  {item}
                </option>
              ))}
            </select>
          </Campo>
          <Campo id="sexo" rotulo="Sexo" obrigatorio={completo}>
            <select
              id="sexo"
              value={cadastro.sexo}
              onChange={(e) => mudar("sexo", e.target.value)}
              disabled={salvando}
              className={classeInput}
            >
              <option value="">{rotuloSimNao(completo)}</option>
              {SEXOS.map((item) => (
                <option key={item} value={item}>
                  {item}
                </option>
              ))}
            </select>
          </Campo>
        </div>
      </Secao>

      {/* ---------- Vínculo e função ---------- */}
      <Secao titulo="Casa de Oração e cargo">
        {/* Casa de oração (lista do banco) */}
        <Campo
          id="casa-oracao-membro"
          rotulo="Casa de Oração"
          obrigatorio
          dica="A casa de oração define o setor e quem enxerga o cadastro. Se a casa de oração não aparecer, peça a um administrador para cadastrá-la em Administração > Organização."
        >
          <SeletorCasaOracao
            id="casa-oracao-membro"
            valor={casaOracaoId}
            onChange={setCasaOracaoId}
            onEscolherNome={setCongregacao}
            disabled={salvando}
            className={classeInput}
          />
        </Campo>

        <Campo id="cargo" rotulo="Cargo" obrigatorio>
          <select
            id="cargo"
            required
            value={cargoId}
            onChange={(e) => trocarCargo(e.target.value)}
            disabled={salvando}
            className={classeInput}
          >
            <option value="">Selecione um cargo</option>
            {cargosDisponiveis.map((cargo) => (
              <option key={cargo.id} value={cargo.id}>
                {cargo.nome}
              </option>
            ))}
          </select>
        </Campo>

        {/* Status do cargo (Categoria): só Músico e Organista informam; os demais cargos são Oficializado */}
        {tipoCategoria ? (
          <Campo id="categoria" rotulo="Status do cargo" obrigatorio>
            <select
              id="categoria"
              required
              value={categoria}
              onChange={(e) => setCategoria(e.target.value)}
              disabled={salvando}
              className={classeInput}
            >
              <option value="">Selecione o status</option>
              {CATEGORIAS.map((item) => (
                <option key={item.valor} value={item.valor}>
                  {item[tipoCategoria]}
                </option>
              ))}
            </select>
          </Campo>
        ) : (
          cargoSelecionado && (
            <p className="rounded-lg bg-slate-50 px-3 py-2 text-xs text-slate-600">
              Status do cargo: <strong>Oficializado</strong> (automático para este cargo).
            </p>
          )
        )}

        {/* Instrumento (lista por família; obrigatório para Músico) + Família automática */}
        {permiteInstrumento && (
          <>
            <Campo
              id="instrumento"
              rotulo="Instrumento"
              obrigatorio={instrumentoObrigatorio}
              dica={
                somenteOrgao
                  ? instrumentosDisponiveis.length === 0
                    ? "Este cargo só usa o instrumento Órgão, mas nenhum Órgão está cadastrado. Cadastre em Instrumentos e famílias."
                    : "Este cargo só usa o instrumento Órgão (escolhido automaticamente)."
                  : "Escolha na lista. Para incluir um instrumento novo, cadastre em Instrumentos e famílias."
              }
            >
              <SeletorInstrumento
                id="instrumento"
                valor={instrumentoId}
                onChange={setInstrumentoId}
                instrumentos={instrumentosDisponiveis}
                familias={familias}
                obrigatorio={instrumentoObrigatorio}
                disabled={salvando || (somenteOrgao && instrumentosDisponiveis.length === 1)}
                className={classeInput}
              />
            </Campo>

            <FamiliaAutomatica instrumentoId={instrumentoId} instrumentos={instrumentos} familias={familias} />
          </>
        )}

        <div className="grid grid-cols-2 gap-3">
          <Campo id="batizado" rotulo="Batizado(a)?" obrigatorio={completo}>
            <select
              id="batizado"
              value={cadastro.batizado}
              onChange={(e) => escolherBatizado(e.target.value as SimNao)}
              disabled={salvando}
              className={classeInput}
            >
              <option value="">{rotuloSimNao(completo)}</option>
              <option value="sim">Sim</option>
              <option value="nao">Não</option>
            </select>
          </Campo>
          <Campo id="selado" rotulo="Selado(a)?" obrigatorio={completo}>
            <select
              id="selado"
              value={cadastro.selado}
              onChange={(e) => mudar("selado", e.target.value as SimNao)}
              disabled={salvando}
              className={classeInput}
            >
              <option value="">{rotuloSimNao(completo)}</option>
              <option value="sim">Sim</option>
              <option value="nao">Não</option>
            </select>
          </Campo>
        </div>

        {/* Data do batismo: só aparece (e é obrigatória) para quem é batizado */}
        {cadastro.batizado === "sim" && (
          <Campo id="data_batismo" rotulo="Data do batismo" obrigatorio>
            <input
              id="data_batismo"
              type="date"
              max={hoje}
              value={cadastro.data_batismo}
              onChange={(e) => mudar("data_batismo", e.target.value)}
              disabled={salvando}
              className={classeInput}
            />
          </Campo>
        )}
      </Secao>

      {/* ---------- Cartão de identificação (só para os cargos marcados em cargos.exige_cartao) ---------- */}
      {exigeCartao && (
        <Secao titulo="Cartão de identificação">
          <Campo id="cartao_numero" rotulo="Número do cartão" obrigatorio>
            <input
              id="cartao_numero"
              type="text"
              autoComplete="off"
              maxLength={40}
              value={cadastro.cartao_numero}
              onChange={(e) => mudar("cartao_numero", e.target.value)}
              disabled={salvando}
              className={classeInput}
            />
          </Campo>
          <div className="grid grid-cols-2 gap-3">
            <Campo id="cartao_emissao" rotulo="Emissão do cartão">
              <input
                id="cartao_emissao"
                type="date"
                value={cadastro.cartao_emissao}
                onChange={(e) => mudar("cartao_emissao", e.target.value)}
                disabled={salvando}
                className={classeInput}
              />
            </Campo>
            <Campo id="cartao_validade" rotulo="Validade do cartão" obrigatorio>
              <input
                id="cartao_validade"
                type="date"
                value={cadastro.cartao_validade}
                onChange={(e) => mudar("cartao_validade", e.target.value)}
                disabled={salvando}
                className={classeInput}
              />
            </Campo>
          </div>
          <div className="grid grid-cols-2 gap-3">
            <Campo id="apresentado_em" rotulo="Apresentado em">
              <input
                id="apresentado_em"
                type="date"
                value={cadastro.apresentado_em}
                onChange={(e) => mudar("apresentado_em", e.target.value)}
                disabled={salvando}
                className={classeInput}
              />
            </Campo>
            <Campo id="apresentado_por" rotulo="Apresentado por" dica="Nome do ancião.">
              <input
                id="apresentado_por"
                type="text"
                autoComplete="off"
                maxLength={120}
                value={cadastro.apresentado_por}
                onChange={(e) => mudar("apresentado_por", e.target.value)}
                disabled={salvando}
                className={classeInput}
              />
            </Campo>
          </div>
        </Secao>
      )}

      {/* ---------- Documento de identificação (frente e verso; armazenamento privado) ---------- */}
      <Secao titulo="Documento de identificação">
        <p className="text-xs text-slate-500">
          Anexe uma foto da frente e uma do verso do documento. Os arquivos ficam em armazenamento privado: só quem tem a
          permissão CPF consegue vê-los.
        </p>
        <CampoDocumento
          id="doc-frente"
          rotulo="Frente do documento"
          obrigatorio={completo}
          caminhoAtual={cadastro.documento_frente_path || null}
          podeVer={podeDocumento}
          onChange={setDocFrente}
          disabled={salvando}
        />
        <CampoDocumento
          id="doc-verso"
          rotulo="Verso do documento"
          obrigatorio={completo}
          caminhoAtual={cadastro.documento_verso_path || null}
          podeVer={podeDocumento}
          onChange={setDocVerso}
          disabled={salvando}
        />
      </Secao>

      {/* ---------- Contato ---------- */}
      <Secao titulo="Contato">
        <div className="grid grid-cols-3 gap-3">
          <Campo id="ddd" rotulo="DDD" obrigatorio={completo}>
            <input
              id="ddd"
              type="text"
              inputMode="numeric"
              autoComplete="off"
              value={cadastro.ddd}
              onChange={(e) => mudar("ddd", somenteDigitos(e.target.value, 2))}
              disabled={salvando}
              placeholder="71"
              className={classeInput}
            />
          </Campo>
          <div className="col-span-2">
            <Campo id="telefone" rotulo="Telefone" obrigatorio={completo}>
              <input
                id="telefone"
                type="text"
                inputMode="numeric"
                autoComplete="off"
                value={formatarTelefone(cadastro.telefone)}
                onChange={(e) => mudar("telefone", somenteDigitos(e.target.value, 9))}
                disabled={salvando}
                placeholder="99999-9999"
                className={classeInput}
              />
            </Campo>
          </div>
        </div>

        <Campo id="tem_whatsapp" rotulo="Tem WhatsApp?" obrigatorio={completo}>
          <select
            id="tem_whatsapp"
            value={cadastro.tem_whatsapp}
            onChange={(e) => escolherWhatsapp(e.target.value as SimNao)}
            disabled={salvando}
            className={classeInput}
          >
            <option value="">{rotuloSimNao(completo)}</option>
            <option value="sim">Sim</option>
            <option value="nao">Não</option>
          </select>
        </Campo>

        {cadastro.tem_whatsapp === "sim" && (
          <Campo id="whatsapp" rotulo="Número do WhatsApp" dica="Já vem com o telefone acima; troque se for outro número.">
            <input
              id="whatsapp"
              type="text"
              inputMode="numeric"
              autoComplete="off"
              value={formatarWhatsapp(cadastro.whatsapp)}
              onChange={(e) => mudar("whatsapp", somenteDigitos(e.target.value, 11))}
              disabled={salvando}
              placeholder="(71) 99999-9999"
              className={classeInput}
            />
          </Campo>
        )}

        <Campo id="email" rotulo="E-mail">
          <input
            id="email"
            type="email"
            autoComplete="off"
            maxLength={160}
            value={cadastro.email}
            onChange={(e) => mudar("email", e.target.value)}
            disabled={salvando}
            placeholder="nome@exemplo.com"
            className={classeInput}
          />
        </Campo>
      </Secao>

      {/* ---------- Endereço ---------- */}
      <Secao titulo="Endereço">
        <div className="grid grid-cols-2 gap-3">
          <Campo id="cep" rotulo="CEP" obrigatorio={completo} dica={textoDoStatusCep(statusCep)}>
            <input
              id="cep"
              type="text"
              inputMode="numeric"
              autoComplete="off"
              value={formatarCep(cadastro.cep)}
              onChange={(e) => void mudarCep(e.target.value)}
              disabled={salvando}
              placeholder="00000-000"
              className={classeInput}
            />
          </Campo>
          <Campo id="numero" rotulo="Número" obrigatorio={completo}>
            <input
              id="numero"
              type="text"
              autoComplete="off"
              maxLength={20}
              value={cadastro.numero}
              onChange={(e) => mudar("numero", e.target.value)}
              disabled={salvando}
              className={classeInput}
            />
          </Campo>
        </div>

        <Campo id="endereco" rotulo="Endereço" obrigatorio={completo}>
          <input
            id="endereco"
            type="text"
            autoComplete="off"
            maxLength={160}
            value={cadastro.endereco}
            onChange={(e) => mudar("endereco", e.target.value)}
            disabled={salvando}
            placeholder="Rua, avenida..."
            className={classeInput}
          />
        </Campo>

        <Campo id="complemento" rotulo="Complemento">
          <input
            id="complemento"
            type="text"
            autoComplete="off"
            maxLength={80}
            value={cadastro.complemento}
            onChange={(e) => mudar("complemento", e.target.value)}
            disabled={salvando}
            className={classeInput}
          />
        </Campo>

        <div className="grid grid-cols-3 gap-3">
          <Campo id="uf" rotulo="Estado" obrigatorio={completo}>
            <select
              id="uf"
              value={cadastro.uf}
              onChange={(e) => mudar("uf", e.target.value)}
              disabled={salvando}
              className={classeInput}
            >
              <option value="">UF</option>
              {UFS.map((sigla) => (
                <option key={sigla} value={sigla}>
                  {sigla}
                </option>
              ))}
            </select>
          </Campo>
          <div className="col-span-2">
            <Campo id="cidade" rotulo="Cidade" obrigatorio={completo}>
              <input
                id="cidade"
                type="text"
                list="sugestoes-cidade"
                autoComplete="off"
                value={cidade}
                onChange={(e) => setCidade(e.target.value)}
                disabled={salvando}
                placeholder="Ex: Salvador, Lauro de Freitas"
                className={classeInput}
              />
              <datalist id="sugestoes-cidade">
                {sugestoes.cidades.map((valor) => (
                  <option key={valor} value={valor} />
                ))}
              </datalist>
            </Campo>
          </div>
        </div>

        <Campo id="bairro" rotulo="Bairro" obrigatorio={completo}>
          <input
            id="bairro"
            type="text"
            list="sugestoes-bairro"
            autoComplete="off"
            value={bairro}
            onChange={(e) => setBairro(e.target.value)}
            disabled={salvando}
            placeholder="Ex: Itinga"
            className={classeInput}
          />
          <datalist id="sugestoes-bairro">
            {sugestoes.bairros.map((valor) => (
              <option key={valor} value={valor} />
            ))}
          </datalist>
        </Campo>
      </Secao>

      {/* ---------- Observação e situação ---------- */}
      <Secao titulo="Outras informações">
        <Campo id="observacao" rotulo="Observação">
          <textarea
            id="observacao"
            rows={3}
            maxLength={500}
            value={cadastro.observacao}
            onChange={(e) => mudar("observacao", e.target.value)}
            disabled={salvando}
            className={classeInput}
          />
        </Campo>

        {/* Situação: só na edição */}
        {editando && (
          <Campo id="status" rotulo="Situação cadastral" obrigatorio>
            <select
              id="status"
              value={status}
              onChange={(e) => setStatus(e.target.value)}
              disabled={salvando}
              className={classeInput}
            >
              <option value="Ativo">Ativo</option>
              <option value="Inativo">Inativo</option>
            </select>
          </Campo>
        )}
      </Secao>

      {erroForm && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erroForm}
        </p>
      )}

      <Button type="submit" disabled={salvando} className="w-full">
        {salvando ? "Salvando..." : editando ? "Salvar alterações" : "Cadastrar membro"}
      </Button>
    </form>
  );
}

/** Andamento da consulta de endereço pelo CEP ("" = nada a mostrar). */
type StatusCep = "" | "buscando" | "encontrado" | "nao_encontrado" | "indisponivel";

function textoDoStatusCep(status: StatusCep): string | undefined {
  switch (status) {
    case "buscando":
      return "Buscando o endereço...";
    case "encontrado":
      return "Endereço preenchido pelo CEP. Confira e complete o número.";
    case "nao_encontrado":
      return "CEP não encontrado. Confira os números ou preencha o endereço à mão.";
    case "indisponivel":
      return "Não foi possível consultar o CEP agora. Preencha o endereço à mão.";
    default:
      return undefined;
  }
}

/** Bloco do formulário com título (agrupa os campos por assunto). */
function Secao({ titulo, children }: { titulo: string; children: ReactNode }) {
  return (
    <section className="space-y-4">
      <h3 className="border-b border-slate-200 pb-1 text-xs font-semibold uppercase tracking-wide text-slate-500">
        {titulo}
      </h3>
      {children}
    </section>
  );
}

/** Campo com rótulo, marcação de obrigatório (*) ou (opcional) e dica opcional. */
function Campo({
  id,
  rotulo,
  obrigatorio = false,
  dica,
  children,
}: {
  id: string;
  rotulo: string;
  obrigatorio?: boolean;
  dica?: string;
  children: ReactNode;
}) {
  return (
    <div className="space-y-1.5">
      <label htmlFor={id} className="text-sm font-medium text-slate-700">
        {rotulo}{" "}
        {obrigatorio ? (
          <span className="font-normal text-red-600" aria-hidden="true">
            *
          </span>
        ) : (
          <span className="font-normal text-slate-400">(opcional)</span>
        )}

  type Familia,

  type MembroInsert,
  type MembroPublico,

const MSG_CPF_DUPLICADO = "Já existe um membro cadastrado com este CPF.";

export interface DadosMembros {
  cargos: Cargo[];
  instrumentos: Instrumento[];
  familias: Familia[];
  membros: MembroPublico[];
}

/** Busca cargos, instrumentos, famílias e membros em paralelo. Lança erro se qualquer consulta falhar. */
export async function buscarDadosMembros(): Promise<DadosMembros> {
  const [resCargos, resInstrumentos, resFamilias, resMembros] = await Promise.all([
    supabase.from("cargos").select("*").order("nome", { ascending: true }),
    supabase.from("instrumentos").select("*").order("nome", { ascending: true }),
    supabase.from("familias").select("*").order("nome", { ascending: true }),
    supabase.from("membros").select(COLUNAS_MEMBRO_PUBLICO).order("nome", { ascending: true }),
  ]);

  const erroBanco = resCargos.error ?? resInstrumentos.error ?? resFamilias.error ?? resMembros.error;
  if (erroBanco) throw new Error(erroBanco.message);

  return {
    cargos: (resCargos.data ?? []) as Cargo[],
    instrumentos: (resInstrumentos.data ?? []) as Instrumento[],
    familias: (resFamilias.data ?? []) as Familia[],
    // Cast duplo: com lista de colunas, o cliente (sem tipagem gerada) não infere a linha
    membros: (resMembros.data ?? []) as unknown as MembroPublico[],
  };
}

import type { Session } from "@supabase/supabase-js";

export interface Conta {
  nome: string;
  email: string;
  grupo: string | null;
  ativo: boolean;
  permissoes: string[];
}

interface ContextoSessao {
  carregando: boolean; // ainda verificando a sessão salva e as permissões
  sessao: Session | null;
  conta: Conta | null;
  erroConta: string | null;
  pode: (permissao: CodigoPermissao) => boolean;
  sair: () => Promise<void>;
}

const Contexto = createContext<ContextoSessao | null>(null);

/** Acesso à sessão e às permissões. Só funciona dentro de <SessaoProvider>. */
export function useSessao(): ContextoSessao {
  const contexto = useContext(Contexto);
  if (!contexto) throw new Error("useSessao precisa estar dentro de <SessaoProvider>.");
  return contexto;
}

export default function SessaoProvider({ children }: { children: ReactNode }) {
  const [sessao, setSessao] = useState<Session | null>(null);
  const [conta, setConta] = useState<Conta | null>(null);
  const [erroConta, setErroConta] = useState<string | null>(null);
  const [sessaoPronta, setSessaoPronta] = useState(false); // já sabemos se existe sessão salva?
  const [contaPronta, setContaPronta] = useState(false); // já buscamos o grupo e as permissões?

  // Busca nome, grupo e permissões do usuário logado (função minha_conta do banco)
  const carregarConta = useCallback(async () => {
    try {
      const { data, error } = await supabase.rpc("minha_conta").maybeSingle();
      if (error) throw new Error(error.message);
      setConta((data as Conta | null) ?? null);
      setErroConta(null);
    } catch (e) {
      console.error("Falha ao carregar a conta:", e);
      setConta(null);
      setErroConta("Não foi possível carregar suas permissões. Tente novamente.");
    } finally {
      setContaPronta(true);
    }
  }, []);

  // Acompanha a sessão: o Supabase avisa a sessão salva no navegador (evento INITIAL_SESSION)
  // logo ao se inscrever e, depois, cada login, logout e renovação de token.
  useEffect(() => {
    const { data: assinatura } = supabase.auth.onAuthStateChange((_evento, novaSessao) => {
      setSessao(novaSessao);
      setSessaoPronta(true);

      if (!novaSessao) {
        // Saiu da conta: limpa tudo (o próximo login volta a mostrar "carregando")
        setConta(null);
        setContaPronta(false);
        return;
      }

      // Não se deve chamar o Supabase de dentro deste callback (pode travar): adia para o próximo ciclo
      window.setTimeout(() => {
        void carregarConta();
      }, 0);
    });
    return () => assinatura.subscription.unsubscribe();
  }, [carregarConta]);

  // Conjunto de permissões (vazio quando a conta está inativa)
  const permissoes = useMemo(() => new Set<string>(conta?.ativo ? conta.permissoes : []), [conta]);
  const pode = useCallback((permissao: CodigoPermissao) => permissoes.has(permissao), [permissoes]);

  const sair = useCallback(async () => {
    await supabase.auth.signOut(); // o callback acima limpa a sessão e a conta
  }, []);

  const carregando = !sessaoPronta || (sessao !== null && !contaPronta);

  const valor = useMemo(
    () => ({ carregando, sessao, conta, erroConta, pode, sair }),
    [carregando, sessao, conta, erroConta, pode, sair]
  );

  return (
    <Contexto.Provider value={valor}>
      {/* Aparência do sistema (modo, cor e tamanho da letra); o botão de Configurações fica na barra do topo */}
      <PreferenciasProvider logado={sessao !== null && contaPronta}>
        {children}
        
      </PreferenciasProvider>
    </Contexto.Provider>
  );
}

  // Sem login: vai para a tela de entrada

    if (!carregando && !sessao) router.replace("/login");
  }, [carregando, sessao, router]);

  if (carregando || !sessao) return <TelaCarregando />;

  const botaoSair = (
    <Button variante="secundario" icone={LogOut} onClick={() => void sair()}>
      Sair da conta
    </Button>
  );

  if (erroConta) {
    return (
      <TelaAvisoDeAcesso
        titulo="Não foi possível verificar seu acesso"
        mensagem={erroConta}
        acoes={
          <>
            <Button onClick={() => window.location.reload()}>Tentar novamente</Button>
            {botaoSair}
          </>
        }
      />
    );
  }

  if (!conta || !conta.ativo || conta.permissoes.length === 0) {
    return (
      <TelaAvisoDeAcesso
        titulo="Sua conta ainda não tem acesso"
        mensagem="Peça a um administrador para liberar o seu acesso, definindo o grupo ao qual você pertence."
        acoes={botaoSair}
      />
    );
  }

  return <>{children}</>;
}

=================== src/app/api/admin/usuarios/route.ts
/**
 * SysMúsica — POST /api/admin/usuarios: cria uma conta de login e a coloca em um grupo.
 *
 * Exige a permissão usuarios.criar (conferida no banco). A senha só trafega nesta requisição
 * (HTTPS) e é guardada pelo Supabase com hash; ela nunca é registrada em log.
 * Fluxo: valida entrada -> confere o grupo -> cria a conta (já confirmada) -> o gatilho do banco
 * cria a linha em "usuarios" -> vincula o grupo. Se o vínculo falhar, a conta é removida.
 */
import { NextResponse } from "next/server";
import { validarNovoUsuario } from "@/modules/administracao/regras";
import { exigirPermissao } from "@/modules/administracao/server/autorizacao";
import { clienteAdmin } from "@/modules/administracao/server/clienteAdmin";
import { ErroHttp } from "@/modules/administracao/server/erros";
import { respostaDeErro } from "@/modules/administracao/server/resposta";

export async function POST(request: Request) {
  try {
    await exigirPermissao(request, "usuarios.criar");

    const validacao = validarNovoUsuario(await request.json().catch(() => null));
    if (!validacao.ok) throw new ErroHttp(400, validacao.erro);
    const { nome, email, senha, grupoId } = validacao.dados;

    const admin = clienteAdmin();

    // O grupo precisa existir e estar ativo
    const { data: grupo, error: erroGrupo } = await admin
      .from("grupos")
      .select("id, ativo")
      .eq("id", grupoId)
      .maybeSingle();
    if (erroGrupo) throw erroGrupo;
    if (!grupo || !grupo.ativo) throw new ErroHttp(400, "Escolha um grupo ativo.");

    // Cria a conta de login (já confirmada: o administrador informa a senha ao usuário)
    const { data: criado, error: erroCriacao } = await admin.auth.admin.createUser({
      email,
      password: senha,
      email_confirm: true,
      user_metadata: { nome },
    });
    if (erroCriacao || !criado.user) {
      if (erroCriacao?.code === "email_exists") {
        throw new ErroHttp(409, "Já existe um usuário com este e-mail.");
      }
      throw erroCriacao ?? new Error("Conta não criada.");
    }

    // Vincula ao grupo (a linha em "usuarios" foi criada pelo gatilho do banco)
    const { error: erroVinculo } = await admin
      .from("usuarios")
      .update({ nome, grupo_id: grupoId, ativo: true })
      .eq("id", criado.user.id);
    if (erroVinculo) {
      // Desfaz: não deixa uma conta sem grupo para trás
      await admin.auth.admin.deleteUser(criado.user.id);
      throw erroVinculo;
    }

    return NextResponse.json({ id: criado.user.id }, { status: 201 });
  } catch (erro) {
    return respostaDeErro(erro);
  }
}

/**
 * SysMúsica — Cliente Supabase (conexão com o banco de dados)
 *
 * Único ponto de criação do cliente: todos os módulos devem importar

      "NEXT_PUBLIC_SUPABASE_ANON_KEY no arquivo .env.local e reinicie o servidor."
