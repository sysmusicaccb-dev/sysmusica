export interface ReferenciaTabela {
  coluna: string; // coluna que aponta para o "pai"
  tabela: NomeTabela; // tabela "pai" (a coluna aponta para o campo id dela)
}

export interface DefinicaoTabela {
  nome: NomeTabela;
  rotulo: string;
  grupo: GrupoDados;
  pk: string[];
  exclusaoLogica: boolean; // tem a coluna excluido_em
  somenteInserir: boolean; // histórico: só se devolve o que foi apagado, nunca se altera
  referencias: ReferenciaTabela[];
  sensiveis: string[]; // colunas que NUNCA vão para o navegador (aparecem como "(protegido)")
}

export interface DefinicaoGrupo {
  id: GrupoDados;
  rotulo: string;
  descricao: string;
  tabelas: NomeTabela[];
}

function tabela(
  nome: NomeTabela,
  rotulo: string,
  grupo: GrupoDados,
  extras: Partial<Omit<DefinicaoTabela, "nome" | "rotulo" | "grupo">> = {}
): DefinicaoTabela {
  return {
    nome,
    rotulo,
    grupo,
    pk: ["id"],
    exclusaoLogica: false,
    somenteInserir: false,
    referencias: [],
    sensiveis: [],
    ...extras,
  };
}

const INDICE = new Map<NomeTabela, number>(TABELAS.map((item, indice) => [item.nome, indice] as const));

/** Dados de uma tabela (erro claro se o nome não existir). */
export function tabelaPorNome(nome: NomeTabela): DefinicaoTabela {
  const encontrada = TABELAS.find((item) => item.nome === nome);
  if (!encontrada) throw new Error(`Tabela desconhecida no backup: ${nome}`);
  return encontrada;
}

/** Tabelas de um grupo de dados, em ordem de dependência. */
export function tabelasDoGrupo(grupo: GrupoDados): DefinicaoTabela[] {
  return TABELAS.filter((item) => item.grupo === grupo);
}

/** Ordem para GRAVAR: pais primeiro. Remove repetidas e não altera a lista original. */
export function ordenarParaGravar(nomes: NomeTabela[]): NomeTabela[] {
  return [...new Set(nomes)].sort((a, b) => (INDICE.get(a) ?? 0) - (INDICE.get(b) ?? 0));
}

/** Ordem para APAGAR: filhos primeiro (o inverso da ordem de gravação). */
export function ordenarParaApagar(nomes: NomeTabela[]): NomeTabela[] {
  return ordenarParaGravar(nomes).reverse();
}

/** Identificador de uma linha pela chave primária (ex.: "uuid" ou "grupo|permissao"). */
export function chaveDaLinha(definicao: DefinicaoTabela, linha: Record<string, unknown>): string {
  return definicao.pk.map((coluna) => String(linha[coluna] ?? "")).join("|");
}

/**
 * SysMúsica — Regras puras do seletor de instrumento (sem banco). Testadas em selecao.test.ts.
 * O mesmo seletor é usado no cadastro de membro e no cadastro de visitante, para a escolha ser sempre
 * igual: lista suspensa agrupada por família, e a família aparece sozinha pelo instrumento escolhido.
 */

export interface InstrumentoOpcao {
  id: string;
  nome: string;
  familia_id: string | null;
}

export interface FamiliaOpcao {
  id: string;
  nome: string;
}

export interface GrupoInstrumentos {
  familiaId: string | null; // nulo = instrumentos sem família
  rotulo: string;
  instrumentos: InstrumentoOpcao[];
}

const ROTULO_SEM_FAMILIA = "Sem família definida";

// Ordem natural em português
const porNome = (a: { nome: string }, b: { nome: string }) => a.nome.localeCompare(b.nome, "pt-BR", { numeric: true });

/** Agrupa os instrumentos por família (famílias em ordem de nome; os sem família vão para o fim). */
export function agruparInstrumentosPorFamilia(
  instrumentos: InstrumentoOpcao[],
  familias: FamiliaOpcao[]
): GrupoInstrumentos[] {
  const conhecidas = new Set(familias.map((familia) => familia.id));
  const porFamilia = new Map<string, InstrumentoOpcao[]>();
  const semFamilia: InstrumentoOpcao[] = [];

  for (const instrumento of instrumentos) {
    if (instrumento.familia_id && conhecidas.has(instrumento.familia_id)) {
      porFamilia.set(instrumento.familia_id, [...(porFamilia.get(instrumento.familia_id) ?? []), instrumento]);
    } else {
      semFamilia.push(instrumento);
    }
  }

  const grupos: GrupoInstrumentos[] = [...familias]
    .filter((familia) => porFamilia.has(familia.id))
    .sort(porNome)
    .map((familia) => ({
      familiaId: familia.id,
      rotulo: familia.nome,
      instrumentos: [...(porFamilia.get(familia.id) ?? [])].sort(porNome),
    }));

  if (semFamilia.length > 0) {
    grupos.push({ familiaId: null, rotulo: ROTULO_SEM_FAMILIA, instrumentos: [...semFamilia].sort(porNome) });
  }
  return grupos;
}

/** Nome da família do instrumento escolhido; nulo se não houver instrumento ou se ele estiver sem família. */
export function nomeDaFamiliaDoInstrumento(
  instrumentoId: string,
  instrumentos: InstrumentoOpcao[],
  familias: FamiliaOpcao[]
): string | null {
  const instrumento = instrumentos.find((item) => item.id === instrumentoId);
  if (!instrumento?.familia_id) return null;
  return familias.find((familia) => familia.id === instrumento.familia_id)?.nome ?? null;
}

/**
 * Regra do instrumento obrigatório: devolve a mensagem de erro ou null.
 * (O banco confere de novo: a tela é só a resposta rápida.)
 */
export function validarInstrumentoDoCargo(
  exige: boolean,
  instrumentoId: string,
  nomeCargo: string | undefined
): string | null {
  if (exige && !instrumentoId) {
    return `O cargo ${nomeCargo ?? "escolhido"} exige o instrumento. Escolha o instrumento na lista.`;
  }
  return null;
}

/**
 * SysMúsica — Seletor de instrumento (lista suspensa agrupada por FAMÍLIA) e campo "Família (automática)".
 * É o único jeito de escolher instrumento no sistema (cadastro de membro e de visitante), igual aos
 * demais seletores (Cargo, Comum). Só aparecem os instrumentos cadastrados no banco; instrumento novo
 * se cadastra em Instrumentos e famílias, onde a família também é definida.
 */
import { useMemo } from "react";
import {
  agruparInstrumentosPorFamilia,
  nomeDaFamiliaDoInstrumento,
  type FamiliaOpcao,
  type InstrumentoOpcao,
} from "../selecao";

interface SeletorInstrumentoProps {
  id?: string;
  valor: string; // id do instrumento escolhido ("" = nenhum)
  onChange: (instrumentoId: string) => void;
  instrumentos: InstrumentoOpcao[];
  familias: FamiliaOpcao[];
  obrigatorio?: boolean;
  disabled?: boolean;
  className?: string;
}

export default function SeletorInstrumento({
  id,
  valor,
  onChange,
  instrumentos,
  familias,
  obrigatorio,
  disabled,
  className,
}: SeletorInstrumentoProps) {
  const grupos = useMemo(() => agruparInstrumentosPorFamilia(instrumentos, familias), [instrumentos, familias]);

  return (
    <select
      id={id}
      value={valor}
      onChange={(e) => onChange(e.target.value)}
      required={obrigatorio}
      disabled={disabled}
      className={className}
    >
      <option value="">{obrigatorio ? "Selecione o instrumento" : "Sem instrumento / não informado"}</option>
      {grupos.map((grupo) => (
        <optgroup key={grupo.familiaId ?? "sem-familia"} label={grupo.rotulo}>
          {grupo.instrumentos.map((instrumento) => (
            <option key={instrumento.id} value={instrumento.id}>
              {instrumento.nome}
            </option>
          ))}
        </optgroup>
      ))}
    </select>
  );
}

interface FamiliaAutomaticaProps {
  instrumentoId: string;
  instrumentos: InstrumentoOpcao[];
  familias: FamiliaOpcao[];
}

/** Campo somente leitura: a família sai do instrumento escolhido (cadastro do banco). */
export function FamiliaAutomatica({ instrumentoId, instrumentos, familias }: FamiliaAutomaticaProps) {
  const escolhido = instrumentoId !== "" && instrumentos.some((instrumento) => instrumento.id === instrumentoId);
  const nome = nomeDaFamiliaDoInstrumento(instrumentoId, instrumentos, familias);
  const texto = !escolhido ? "" : (nome ?? "Sem família definida (classifique em Instrumentos e famílias)");

  return (
    <div className="space-y-1.5">
      <label htmlFor="familia-instrumento" className="text-sm font-medium text-slate-700">
        Família <span className="font-normal text-slate-400">(automática)</span>
      </label>
      <input
        id="familia-instrumento"
        type="text"
        readOnly
        tabIndex={-1}
        value={texto}
        placeholder="Definida automaticamente pelo instrumento"
        className="w-full cursor-default rounded-lg border border-slate-200 bg-slate-50 px-3 py-2 text-sm text-slate-700 placeholder:text-slate-400 focus:outline-none"
      />
    </div>
  );
}

        const periodo = `${aplicados.inicio || "início"} a ${aplicados.fim || "hoje"}`;
        const aberta = imprimirTabela({
          titulo: "Auditoria do sistema",
          subtitulo: `Período: ${periodo} · emitido por ${conta?.nome ?? "usuário"}`,
          colunas,
          linhas: dados,
          orientacao: "paisagem",
        });
        if (!aberta) {
          setErroSaida("O navegador bloqueou a janela de impressão. Libere os pop-ups deste site e tente de novo.");
          return;
        }
      }

      setAviso(
        totalGeral > todas.length
          ? `Foram incluídos os ${todas.length} registros mais recentes de ${totalGeral}. Refine o período para ver o restante.`
          : "Pronto. Esta ação ficou registrada na auditoria."
      );
    } catch (e) {
      setErroSaida(e instanceof Error ? e.message : "Não foi possível gerar a saída.");
    } finally {
      setSaindo(false);
    }
  }

  const totalPaginas = Math.max(1, Math.ceil(total / TAMANHO_PAGINA));

  const entidadesOrdenadas = [...(opcoes?.entidades ?? [])].sort((a, b) =>
    rotuloEntidade(a).localeCompare(rotuloEntidade(b), "pt-BR")
  );
  const acoesOrdenadas = [...(opcoes?.acoes ?? [])].sort((a, b) => rotuloAcao(a).localeCompare(rotuloAcao(b), "pt-BR"));
  const usuariosOrdenados = [...(opcoes?.usuarios ?? [])].sort((a, b) => a.nome.localeCompare(b.nome, "pt-BR"));

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Auditoria"
          descricao={`Tudo o que foi feito no sistema: quem, quando e o quê.${
            !carregando && !erro ? ` ${total} registro(s) neste filtro.` : ""
          }`}
          acoes={
            <>
              <button type="button" className={classeBotao} onClick={() => void gerarSaida("impressao")} disabled={saindo || carregando || !!erro}>
                <Printer className="h-4 w-4" aria-hidden="true" />
                Imprimir
              </button>
              <button type="button" className={classeBotao} onClick={() => void gerarSaida("csv")} disabled={saindo || carregando || !!erro}>
                <Download className="h-4 w-4" aria-hidden="true" />
                Exportar CSV
              </button>
            </>
          }
        />

        {/* Filtros */}
        <form onSubmit={aplicarFiltros} className="mb-6 rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
          <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
            <div className="space-y-1.5">
              <label htmlFor="filtro-inicio" className="text-xs font-medium text-slate-600">De</label>
              <input
                id="filtro-inicio"
                type="date"
                value={formulario.inicio}
                onChange={(e) => setFormulario({ ...formulario, inicio: e.target.value })}
                className={classeCampo}
              />
            </div>
            <div className="space-y-1.5">
              <label htmlFor="filtro-fim" className="text-xs font-medium text-slate-600">Até</label>
              <input
                id="filtro-fim"
                type="date"
                value={formulario.fim}
                onChange={(e) => setFormulario({ ...formulario, fim: e.target.value })}
                className={classeCampo}
              />
            </div>
            <div className="space-y-1.5">
              <label htmlFor="filtro-usuario" className="text-xs font-medium text-slate-600">Usuário</label>
              <select
                id="filtro-usuario"
                value={formulario.usuarioId}
                onChange={(e) => setFormulario({ ...formulario, usuarioId: e.target.value })}
                className={classeCampo}
              >
                <option value="">Todos</option>
                {usuariosOrdenados.map((usuario) => (
                  <option key={usuario.id} value={usuario.id}>{usuario.nome}</option>
                ))}
              </select>
            </div>
            <div className="space-y-1.5">
              <label htmlFor="filtro-entidade" className="text-xs font-medium text-slate-600">Tipo de item</label>
              <select
                id="filtro-entidade"
                value={formulario.entidade}
                onChange={(e) => setFormulario({ ...formulario, entidade: e.target.value })}
                className={classeCampo}
              >
                <option value="">Todos</option>
                {entidadesOrdenadas.map((entidade) => (
                  <option key={entidade} value={entidade}>{rotuloEntidade(entidade)}</option>
                ))}
              </select>
            </div>
            <div className="space-y-1.5">
              <label htmlFor="filtro-acao" className="text-xs font-medium text-slate-600">Ação</label>
              <select
                id="filtro-acao"
                value={formulario.acao}
                onChange={(e) => setFormulario({ ...formulario, acao: e.target.value })}
                className={classeCampo}
              >
                <option value="">Todas</option>
                {acoesOrdenadas.map((acao) => (
                  <option key={acao} value={acao}>{rotuloAcao(acao)}</option>
                ))}
              </select>
            </div>
            <div className="space-y-1.5">
              <label htmlFor="filtro-busca" className="text-xs font-medium text-slate-600">Buscar no texto</label>
              <input
                id="filtro-busca"
                type="search"
                autoComplete="off"
                value={formulario.busca}
                onChange={(e) => setFormulario({ ...formulario, busca: e.target.value })}
                placeholder="Nome, valor alterado..."
                className={classeCampo}
              />
            </div>
          </div>

          {erroFiltro && (
            <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">{erroFiltro}</p>
          )}

          <div className="mt-4 flex flex-wrap gap-2">
            <button
              type="submit"
              className="rounded-lg bg-brand-600 px-4 py-2 text-sm font-semibold text-white shadow-sm transition hover:bg-brand-700"
            >
              Aplicar filtros
            </button>
            <button type="button" className={classeBotao} onClick={limparFiltros}>
              Últimos 7 dias
            </button>
          </div>
        </form>

        {erroSaida && (
          <p role="alert" className="mb-4 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">{erroSaida}</p>
        )}

        {carregando ? (
          <div role="status" aria-live="polite" className="space-y-3">
            <span className="sr-only">Carregando a auditoria...</span>
            {[0, 1, 2, 3, 4].map((i) => (
              <Skeleton key={i} className="h-12" />
            ))}
          </div>
        ) : erro ? (
          <ErrorState
            titulo="Não foi possível carregar a auditoria."
            mensagem={erro}
            onTentarNovamente={() => {
              setCarregando(true);
              void carregar();
            }}
          />
        ) : linhas.length === 0 ? (
          <EmptyState titulo="Nenhum registro neste filtro." descricao="Mude o período ou os filtros e aplique de novo." />
        ) : (
          <>
            <TabelaAuditoria linhas={linhas} onDetalhes={setSelecionada} />

            <div className="mt-4 flex flex-wrap items-center justify-between gap-3 text-sm text-slate-600">
              <span>
                Página {pagina + 1} de {totalPaginas}
              </span>
              <div className="flex gap-2">
                <button type="button" className={classeBotao} onClick={() => irParaPagina(pagina - 1)} disabled={pagina === 0}>
                  Anterior
                </button>
                <button
                  type="button"
                  className={classeBotao}
                  onClick={() => irParaPagina(pagina + 1)}
                  disabled={pagina + 1 >= totalPaginas}
                >
                  Próxima
                </button>
              </div>
            </div>
          </>
        )}
      </div>

      <Drawer
        aberto={selecionada !== null}
        onFechar={() => setSelecionada(null)}
        titulo="Detalhes do registro"
        descricao={selecionada ? rotuloAcao(selecionada.acao) : undefined}
      >
        {selecionada && <DetalheAuditoria linha={selecionada} />}
      </Drawer>

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

import { readdirSync, readFileSync } from "node:fs";

// Junta o texto de TODAS as migrações (as permissões são criadas em mais de um arquivo)
function lerMigracoes(): string {
  const pasta = path.resolve(process.cwd(), "supabase/migrations");
  return readdirSync(pasta)
    .filter((arquivo) => arquivo.endsWith(".sql"))
    .sort()
    .map((arquivo) => readFileSync(path.join(pasta, arquivo), "utf-8"))
    .join("\n");
}

describe("CATALOGO_PERMISSOES", () => {
  it("não tem códigos repetidos", () => {
    const codigos = CATALOGO_PERMISSOES.map((permissao) => permissao.codigo);

  });

  it("todo código segue o formato modulo.acao em minúsculas", () => {

      expect(codigo).toMatch(/^[a-z]+\.[a-z_]+$/);
    }
  });

  it("todo módulo tem a permissão 'visualizar' (as demais ações dependem dela)", () => {
    const modulos = new Set(CATALOGO_PERMISSOES.map((permissao) => permissao.codigo.split(".")[0]));
    const codigos = CATALOGO_PERMISSOES.map((permissao) => permissao.codigo as string);

/**
 * SysMúsica — Regras puras da organização (sem banco). Testadas em regras.test.ts.
 * As regras de verdade (permissão, nomes repetidos, abrangência) ficam no banco; aqui há
 * montagem de listas para a tela e conferências rápidas.
 */
import type { NivelOrganizacao, NoOrganizacao, VinculoRotulado, VinculoUsuario } from "./tipos";

export const ROTULO_NIVEL: Record<NivelOrganizacao, string> = {
  ESTADO: "Estado",
  REGIAO: "Região",
  SETOR: "Setor",
  COMUM: "Comum",
};

/** Nível que fica logo abaixo de cada nível. */
export const FILHO_DE: Record<NivelOrganizacao, NivelOrganizacao | null> = {
  ESTADO: "REGIAO",
  REGIAO: "SETOR",
  SETOR: "COMUM",
  COMUM: null,
};

const NIVEIS: NivelOrganizacao[] = ["ESTADO", "REGIAO", "SETOR", "COMUM"];

// Ordem natural em português ("Setor 2" antes de "Setor 10")
const porNome = (a: NoOrganizacao, b: NoOrganizacao) => a.nome.localeCompare(b.nome, "pt-BR", { numeric: true });

/** Itens de um nível que pertencem a um pai (pai nulo = os do topo / sem pai), em ordem de nome. */
export function filhosDe(nos: NoOrganizacao[], pai: string | null, nivel: NivelOrganizacao): NoOrganizacao[] {
  return nos.filter((no) => no.nivel === nivel && no.parent_id === pai).sort(porNome);
}

/**
 * Se o item ainda tem algo dentro (itens filhos ou membros), explica por que não pode ser excluído;
 * senão devolve null. É só uma resposta rápida da tela: o banco confere tudo de novo, inclusive eventos,
 * portarias e usuários vinculados.
 */
export function motivoBloqueioExclusao(no: NoOrganizacao): string | null {
  const filhos = no.qtd_filhos ?? 0;
  const membros = no.qtd_membros ?? 0;

  if (no.nivel === "ESTADO" && filhos > 0) {
    return `"${no.nome}" ainda tem ${filhos} ${filhos === 1 ? "região" : "regiões"}. Exclua as regiões antes.`;
  }
  if (no.nivel === "REGIAO" && filhos > 0) {
    return `"${no.nome}" ainda tem ${filhos} ${filhos === 1 ? "setor" : "setores"}. Exclua os setores antes.`;
  }
  if (no.nivel === "SETOR" && filhos > 0) {
    return `"${no.nome}" ainda tem ${filhos} ${filhos === 1 ? "comum" : "comuns"}. Transfira as comuns para outro setor antes.`;
  }
  if (no.nivel === "COMUM" && membros > 0) {
    return `"${no.nome}" ainda tem ${membros} ${membros === 1 ? "membro" : "membros"}. Mude os membros para outra comum (ou exclua-os) antes.`;
  }
  return null;
}

/** Valida o nome de um item da organização; devolve a mensagem de erro ou null. */
export function validarNomeNo(nome: string): string | null {
  const tamanho = nome.trim().replace(/\s+/g, " ").length;
  return tamanho < 2 || tamanho > 80 ? "O nome deve ter entre 2 e 80 caracteres." : null;
}

export interface GrupoComuns {
  rotulo: string; // ex.: "Salvador › Setor 1"
  comuns: NoOrganizacao[];
}

/** Agrupa as comuns por "Região › Setor" (com o estado na frente se houver mais de um). Comuns sem setor vão para o fim. */
export function agruparComunsParaSelecao(nos: NoOrganizacao[]): GrupoComuns[] {
  const porChave = new Map(nos.map((no) => [`${no.nivel}:${no.id}`, no] as const));
  const estados = nos.filter((no) => no.nivel === "ESTADO");
  const grupos = new Map<string, NoOrganizacao[]>();
  const semSetor: NoOrganizacao[] = [];

  for (const comum of nos.filter((no) => no.nivel === "COMUM")) {
    const setor = comum.parent_id ? porChave.get(`SETOR:${comum.parent_id}`) : undefined;
    const regiao = setor?.parent_id ? porChave.get(`REGIAO:${setor.parent_id}`) : undefined;
    if (!setor || !regiao) {
      semSetor.push(comum);
      continue;
    }
    const estado = regiao.parent_id ? porChave.get(`ESTADO:${regiao.parent_id}`) : undefined;
    const prefixo = estados.length > 1 && estado ? `${estado.nome} › ` : "";
    const rotulo = `${prefixo}${regiao.nome} › ${setor.nome}`;
    grupos.set(rotulo, [...(grupos.get(rotulo) ?? []), comum]);
  }

  const lista: GrupoComuns[] = [...grupos.entries()]
    .sort(([a], [b]) => a.localeCompare(b, "pt-BR", { numeric: true }))
    .map(([rotulo, comuns]) => ({ rotulo, comuns: [...comuns].sort(porNome) }));
  if (semSetor.length > 0) lista.push({ rotulo: "Sem setor definido", comuns: [...semSetor].sort(porNome) });
  return lista;
}

export interface GrupoSetores {
  rotulo: string; // ex.: "Salvador" (ou "Bahia › Salvador")
  setores: NoOrganizacao[];
}

/** Agrupa os setores ATIVOS por região (para escolher o destino de uma transferência). */
export function agruparSetoresParaSelecao(nos: NoOrganizacao[]): GrupoSetores[] {
  const porChave = new Map(nos.map((no) => [`${no.nivel}:${no.id}`, no] as const));
  const estados = nos.filter((no) => no.nivel === "ESTADO");
  const grupos = new Map<string, NoOrganizacao[]>();

  for (const setor of nos.filter((no) => no.nivel === "SETOR" && no.ativo !== false)) {
    const regiao = setor.parent_id ? porChave.get(`REGIAO:${setor.parent_id}`) : undefined;
    if (!regiao || regiao.ativo === false) continue;
    const estado = regiao.parent_id ? porChave.get(`ESTADO:${regiao.parent_id}`) : undefined;
    const rotulo = `${estados.length > 1 && estado ? `${estado.nome} › ` : ""}${regiao.nome}`;
    grupos.set(rotulo, [...(grupos.get(rotulo) ?? []), setor]);
  }

  return [...grupos.entries()]
    .sort(([a], [b]) => a.localeCompare(b, "pt-BR", { numeric: true }))
    .map(([rotulo, setores]) => ({ rotulo, setores: [...setores].sort(porNome) }));
}

/** Chave única de um vínculo, para guardar em conjuntos (ex.: "SETOR:abc"). */
export function chaveVinculo(nivel: NivelOrganizacao, id: string): string {
  return `${nivel}:${id}`;
}

/** Desfaz a chave de um vínculo; devolve nulo se estiver malformada. */
export function lerChaveVinculo(chave: string): VinculoUsuario | null {
  const posicao = chave.indexOf(":");
  if (posicao < 0) return null;
  const nivel = chave.slice(0, posicao);
  const alvo = chave.slice(posicao + 1);
  if (!NIVEIS.includes(nivel as NivelOrganizacao) || alvo === "") return null;
  return { nivel: nivel as NivelOrganizacao, alvo_id: alvo };
}

/** De { usuario: [vínculos com rótulo] } para { usuario: ["rótulo", ...] } (coluna da tabela de usuários). */
export function rotulosPorUsuario(mapa: Record<string, VinculoRotulado[]>): Record<string, string[]> {
  const resultado: Record<string, string[]> = {};
  for (const [usuarioId, vinculos] of Object.entries(mapa)) {
    resultado[usuarioId] = vinculos.map((vinculo) => vinculo.rotulo);
  }
  return resultado;
}
