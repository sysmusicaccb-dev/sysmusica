=================== src/modules/eventos/camposEvento.ts
/**
 * SysMúsica — Campos obrigatórios do evento (regra pura, sem banco).
 * Todos os campos do formulário de evento são obrigatórios. Devolve os NOMES (como aparecem na tela) dos
 * campos que estão vazios, na ordem do formulário, para a tela marcar cada um e mostrar o alerta.
 */
export interface CamposEvento {
  tipo: string; // "Descrição do evento" na tela
  dataEvento: string; // valor do campo data e hora (vazio = não preenchido)
  cidade: string;
  casaOracaoId: string;
}

export const ROTULO_TIPO = "Descrição do evento";
export const ROTULO_DATA = "Data e hora";
export const ROTULO_CIDADE = "Cidade";
export const ROTULO_CASA_ORACAO = "Casa de Oração";

export function camposFaltando(campos: CamposEvento): string[] {
  const faltando: string[] = [];
  // Ordem do formulário: Descrição do evento, Data e hora, Casa de Oração, Cidade
  if (!campos.tipo.trim()) faltando.push(ROTULO_TIPO);
  if (!campos.dataEvento.trim()) faltando.push(ROTULO_DATA);
  if (!campos.casaOracaoId.trim()) faltando.push(ROTULO_CASA_ORACAO);
  if (!campos.cidade.trim()) faltando.push(ROTULO_CIDADE);
  return faltando;
}

/** Mensagem do alerta: "Preencha todos os campos obrigatórios: Local, Cidade." */
export function mensagemCamposFaltando(faltando: string[]): string {
  return `Preencha todos os campos obrigatórios: ${faltando.join(", ")}.`;
}

=================== src/modules/eventos/casaDoEvento.ts
/**
 * SysMúsica — Regra pura: cidade do evento a partir da Casa de Oração escolhida (sem banco).
 * A cidade vem do cadastro da casa e fica travada na tela. Se a casa não tem cidade cadastrada
 * (ou o cadastro não pôde ser lido), o campo fica livre para digitar e um aviso orienta a completar o cadastro.
 */
export interface DadosDaCasa {
  nome: string;
  cidade: string | null;
}

export interface CidadeDaCasa {
  cidade: string; // texto para o campo Cidade ("" = a pessoa precisa informar)
  travada: boolean; // true = veio do cadastro da casa (campo somente leitura)
  aviso: string | null;
}

export const AVISO_SEM_CIDADE =
  "Esta Casa de Oração ainda não tem cidade cadastrada. Informe a cidade abaixo e depois complete o cadastro dela em Organização.";

export function cidadeDaCasa(casa: DadosDaCasa | null): CidadeDaCasa {
  const cidade = casa?.cidade?.trim() ?? "";
  if (cidade) return { cidade, travada: true, aviso: null };
  return { cidade: "", travada: false, aviso: AVISO_SEM_CIDADE };
}

=================== src/modules/eventos/ciclo.ts
/**
 * SysMúsica — Ciclo de vida do evento (regras puras, sem banco).
 *
 * A situação é CALCULADA pela data (ninguém precisa "finalizar" o evento na mão):
 *   Agendado     -> ainda falta mais de 2h para começar
 *   Em andamento -> de 2h antes até 3h depois do início (a mesma janela da portaria)
 *   Encerrado    -> passou de 3h depois do início (somente leitura; só o Administrador corrige)
 *   Arquivado    -> encerrado há mais de 30 dias (continua consultável, na aba Arquivo)
 *   Cancelado    -> só o Administrador cancela, sempre com motivo; o histórico fica guardado
 * O banco aplica as mesmas regras (migração 0034); aqui ficam só os cálculos para a tela.
 */
import type { VarianteBadge } from "@/components/ui/Badge";
import { janelaDaPortaria } from "@/modules/portarias/janela";

export type SituacaoEvento = "AGENDADO" | "EM_ANDAMENTO" | "ENCERRADO" | "ARQUIVADO" | "CANCELADO";
export type AbaEvento = "proximos" | "encerrados" | "arquivo";

/** Dias depois do encerramento em que o evento passa para o Arquivo. */
export const DIAS_ARQUIVO = 30;
const MS_DIA = 24 * 60 * 60 * 1000;

export const ROTULO_SITUACAO_EVENTO: Record<SituacaoEvento, string> = {
  AGENDADO: "Agendado",
  EM_ANDAMENTO: "Em andamento",
  ENCERRADO: "Encerrado",
  ARQUIVADO: "Arquivado",
  CANCELADO: "Cancelado",
};

export const MOTIVO_CANCELAMENTO_MINIMO = 5;
export const MOTIVO_CANCELAMENTO_MAXIMO = 250;

/** Situação do evento no instante "agora" (ms), ignorando o cancelamento. */
function situacaoPelaData(dataEvento: string, agora: number): Exclude<SituacaoEvento, "CANCELADO"> {
  const janela = janelaDaPortaria(dataEvento, agora);
  if (!janela) return "AGENDADO"; // data inválida: mantém visível para alguém corrigir
  if (janela.situacao === "antes") return "AGENDADO";
  if (janela.situacao === "aberta") return "EM_ANDAMENTO";
  return agora - janela.fecha.getTime() > DIAS_ARQUIVO * MS_DIA ? "ARQUIVADO" : "ENCERRADO";
}

/** Situação do evento: cancelado vence a data. */
export function situacaoDoEvento(dataEvento: string, cancelado: boolean, agora: number): SituacaoEvento {
  if (cancelado) return "CANCELADO";
  return situacaoPelaData(dataEvento, agora);
}

/** Em qual aba da tela de Eventos o evento aparece. Cancelado nunca fica em "Próximos". */
export function abaDoEvento(dataEvento: string, cancelado: boolean, agora: number): AbaEvento {
  const pelaData = situacaoPelaData(dataEvento, agora);
  if (pelaData === "ARQUIVADO") return "arquivo";
  if (cancelado || pelaData === "ENCERRADO") return "encerrados";
  return "proximos";
}

/** O evento ainda está aberto (agendado ou em andamento, e não cancelado)? */
export function eventoAberto(dataEvento: string, cancelado: boolean, agora: number): boolean {
  const situacao = situacaoDoEvento(dataEvento, cancelado, agora);
  return situacao === "AGENDADO" || situacao === "EM_ANDAMENTO";
}

/** Cor do selo de cada situação. */
export function varianteSituacaoEvento(situacao: SituacaoEvento): VarianteBadge {
  if (situacao === "EM_ANDAMENTO") return "sucesso";
  if (situacao === "CANCELADO") return "perigo";
  return "neutro";
}

/** Valida o motivo do cancelamento: devolve a mensagem de erro ou null se estiver certo. */
export function validarMotivoCancelamento(texto: string): string | null {
  const tamanho = texto.trim().length;
  if (tamanho < MOTIVO_CANCELAMENTO_MINIMO) {
    return `Informe o motivo do cancelamento (mínimo de ${MOTIVO_CANCELAMENTO_MINIMO} caracteres).`;
  }
  if (tamanho > MOTIVO_CANCELAMENTO_MAXIMO) {
    return `O motivo pode ter no máximo ${MOTIVO_CANCELAMENTO_MAXIMO} caracteres.`;
  }
  return null;
}

"use client";

/**
 * SysMúsica — Cancelamento de evento (só o Administrador). O motivo é obrigatório e fica no histórico.
 * As portarias pendentes ou aprovadas do evento são rejeitadas automaticamente pelo banco.
 */
import { useState, type FormEvent } from "react";
import Button from "@/components/ui/Button";
import { formatarDataHora } from "@/lib/datas";
import { cancelarEvento } from "@/modules/eventos/services/eventos";
import { MOTIVO_CANCELAMENTO_MAXIMO, validarMotivoCancelamento } from "@/modules/eventos/ciclo";
import type { Evento } from "@/types/database";

interface FormularioCancelamentoProps {
  evento: Evento;
  onSalvo: (mensagem: string) => void | Promise<void>;
}

export default function FormularioCancelamento({ evento, onSalvo }: FormularioCancelamentoProps) {
  const [motivo, setMotivo] = useState("");
  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  async function enviar(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const invalido = validarMotivoCancelamento(motivo);
    if (invalido) {
      setErro(invalido);
      return;
    }
    setSalvando(true);
    setErro(null);
    try {
      await cancelarEvento(evento.id, motivo.trim());
      await onSalvo("Evento cancelado. As portarias dele foram rejeitadas e o histórico foi mantido.");
    } catch (falha) {
      setErro(falha instanceof Error ? falha.message : "Não foi possível cancelar o evento.");
      setSalvando(false);
    }
  }

  return (
    <form onSubmit={(e) => void enviar(e)} className="space-y-5">
      <div className="rounded-lg bg-slate-50 px-3 py-2 text-sm">
        <p className="font-medium text-slate-900">{formatarDataHora(evento.data_evento)}</p>
        <p className="text-slate-600">{[evento.tipo, evento.local, evento.cidade].filter(Boolean).join(" · ")}</p>
      </div>

      <div className="space-y-1.5">
        <label htmlFor="motivo-cancelamento" className="text-sm font-medium text-slate-700">
          Motivo do cancelamento
        </label>
        <textarea
          id="motivo-cancelamento"
          rows={4}
          required
          maxLength={MOTIVO_CANCELAMENTO_MAXIMO}
          value={motivo}
          onChange={(e) => setMotivo(e.target.value)}
          disabled={salvando}
          placeholder="Explique por que o evento foi cancelado"
          className="w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 placeholder:text-slate-400 shadow-sm focus:border-brand-500 focus:outline-none focus:ring-2 focus:ring-brand-200 disabled:opacity-60"
        />
        <p className="text-xs text-slate-500">
          O evento sai de “Próximos” e vai para o histórico. Portarias pendentes ou aprovadas dele são rejeitadas. A
          decisão não pode ser desfeita.
        </p>
      </div>

      {erro && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erro}
        </p>
      )}

      <Button type="submit" variante="perigo" disabled={salvando} className="w-full">
        {salvando ? "Cancelando..." : "Cancelar evento"}
      </Button>
    </form>
  );
}

/**
 * SysMúsica — Regra da data do evento (pura, sem banco): não se agenda evento em data e hora que já passaram.
 * Uma tolerância de 5 minutos evita falso alarme (o campo de data só tem precisão de minutos e o usuário leva
 * alguns segundos para salvar). Ao EDITAR, só vale se a data/hora for alterada: um evento antigo continua
 * podendo ter outros dados corrigidos. A mesma regra é aplicada no banco (migração 0025).
 */
export const TOLERANCIA_MS = 5 * 60 * 1000;

export const MENSAGEM_DATA_PASSADA =
  "Não é possível agendar um evento em uma data e hora que já passaram. Escolha uma data e hora futuras.";

/**
 * Devolve a mensagem de erro ou null se a data está correta.
 * dataOriginal = data já gravada do evento (null ao cadastrar um evento novo).
 */
export function validarDataDoEvento(dataIso: string, agora: number, dataOriginal: string | null): string | null {
  const nova = new Date(dataIso).getTime();
  if (Number.isNaN(nova)) return "Informe uma data e hora válidas.";

  // Edição sem mudar a data e a hora (no minuto): não reavalia
  if (dataOriginal !== null) {
    const original = new Date(dataOriginal).getTime();
    if (!Number.isNaN(original) && Math.floor(original / 60000) === Math.floor(nova / 60000)) return null;
  }

  return nova < agora - TOLERANCIA_MS ? MENSAGEM_DATA_PASSADA : null;
}

/**

export const FINALIDADE_MINIMO = 3;
export const FINALIDADE_MAXIMO = 150;

/** Ajusta a finalidade antes de salvar: tira espaços extras e põe a 1ª letra em maiúscula. */
export function formatarNomeFinalidade(texto: string): string {
  const limpo = texto.trim().replace(/\s+/g, " ");
  return limpo.charAt(0).toUpperCase() + limpo.slice(1);
}

/** Valida o texto da finalidade; devolve a mensagem de erro ou null quando está tudo certo. */
export function validarFinalidade(texto: string): string | null {
  const tamanho = texto.trim().replace(/\s+/g, " ").length;
  if (tamanho < FINALIDADE_MINIMO) {
    return `Informe a finalidade do evento (mínimo de ${FINALIDADE_MINIMO} caracteres).`;
  }
  if (tamanho > FINALIDADE_MAXIMO) {
    return `A finalidade pode ter no máximo ${FINALIDADE_MAXIMO} caracteres.`;
  }
  return null;
}

/** Busca eventos (do mais recente para o mais antigo) e finalidades. Lança erro se a consulta falhar. */

  eventos: Evento[];
  finalidades: FinalidadeEvento[];
}> {
  const [resEventos, resFinalidades] = await Promise.all([
    supabase.from("eventos").select("*").order("data_evento", { ascending: false }),
    supabase.from("finalidades_evento").select("*").order("nome", { ascending: true }),

  const erroBanco = resEventos.error ?? resFinalidades.error;
  if (erroBanco) throw new Error(erroBanco.message);

  return {
    eventos: (resEventos.data ?? []) as Evento[],
    finalidades: (resFinalidades.data ?? []) as FinalidadeEvento[],
  };
}

/**
 * SysMúsica — Serviço do catálogo de finalidades de evento.
 * O formulário de evento usa obterOuCriarFinalidade: se a pessoa digita uma finalidade que ainda
 * não existe, ela é salva e fica disponível para os próximos eventos.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import { normalizarTexto } from "@/lib/texto";
import type { FinalidadeEvento, FinalidadeEventoInsert } from "@/types/database";
import { formatarNomeFinalidade } from "../regras";

export interface ResultadoFinalidade {
  id: string;
  nome: string;
  criada: boolean; // true = acabou de ser criada
}

/**
 * Devolve a finalidade digitada, criando-a se ainda não existir.
 * A comparação ignora acento, maiúsculas e espaços repetidos.
 */
export async function obterOuCriarFinalidade(
  texto: string,
  existentes: FinalidadeEvento[]
): Promise<ResultadoFinalidade> {
  const chave = normalizarTexto(texto);

  // 1) Já existe na lista carregada? Então só reaproveita o id
  const existente = existentes.find((finalidade) => normalizarTexto(finalidade.nome) === chave);
  if (existente) return { id: existente.id, nome: existente.nome, criada: false };

  // 2) Não existe: salva e resgata o registro
  const nova: FinalidadeEventoInsert = { nome: formatarNomeFinalidade(texto) };
  const { data, error } = await supabase.from("finalidades_evento").insert(nova).select().single();
  if (!error && data) {
    const criada = data as FinalidadeEvento;
    return { id: criada.id, nome: criada.nome, criada: true };
  }

  // 3) 23505 = repetida: outra pessoa criou a mesma finalidade agora há pouco. Busca de novo.
  if (error?.code === "23505") {
    const { data: lista, error: erroBusca } = await supabase.from("finalidades_evento").select("*");
    if (erroBusca) throw new Error(erroBusca.message);
    const achada = ((lista ?? []) as FinalidadeEvento[]).find(
      (finalidade) => normalizarTexto(finalidade.nome) === chave
    );
    if (achada) return { id: achada.id, nome: achada.nome, criada: false };
  }

  throw new Error(
    error
      ? mensagemDoErro(error, "Não foi possível salvar a finalidade do evento.")
      : "Não foi possível salvar a finalidade do evento."
  );
}

/**
 * SysMúsica — Formulário de edição de uma presença da portaria (conteúdo do painel lateral).
 * O membro e o evento são só informativos (não mudam: para outro membro ou evento, exclua e

import type { Presenca } from "@/types/database";

  presenca: Presenca;
  nomeMembro: string;
  descricaoEvento: string;
  /** Chamado após salvar com sucesso, com a mensagem para o aviso da tela. */

// [color-scheme:light] mantém o ícone do seletor de data visível mesmo com o navegador em tema escuro.
const classeInput =
  "w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 " +
  "shadow-sm focus:border-brand-500 focus:outline-none focus:ring-2 focus:ring-brand-200 " +

export default function FormularioEdicaoPresenca({
  presenca,
  nomeMembro,
  descricaoEvento,
  onSalvo,
}: FormularioEdicaoPresencaProps) {
  const [status, setStatus] = useState(presenca.status);
  const [entrada, setEntrada] = useState(isoParaDatetimeLocal(presenca.hora_entrada));
  const [saida, setSaida] = useState(presenca.hora_saida ? isoParaDatetimeLocal(presenca.hora_saida) : "");
  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  // Os horários só importam para quem esteve presente
  const mostrarHorarios = status === "Presente";

  async function salvar(submissao: FormEvent<HTMLFormElement>) {
    submissao.preventDefault();
    if (salvando) return; // evita envio duplicado
    setErro(null);

    if (mostrarHorarios) {
      const erroHorarios = validarHorarios(entrada, saida);
      if (erroHorarios) return setErro(erroHorarios);

    setSalvando(true);

      // Se não é "Presente", mantém os horários como estavam (não foram editados)

        hora_saida: mostrarHorarios
          ? saida.trim() === ""
            ? null
            : datetimeLocalParaIso(saida)
          : presenca.hora_saida,
      });

    } catch (e) {
      setErro(e instanceof Error ? e.message : "Não foi possível salvar a presença.");
    } finally {
      setSalvando(false);
    }
  }

  return (
    <form onSubmit={salvar} className="space-y-5">
      {/* Identificação (somente leitura) */}
      <div className="rounded-xl bg-slate-50 px-4 py-3 text-sm">
        <p className="font-medium text-slate-900">{nomeMembro}</p>
        <p className="mt-0.5 text-slate-600">{descricaoEvento}</p>
      </div>

      <div className="space-y-1.5">
        <label htmlFor="status-presenca" className="text-sm font-medium text-slate-700">

        </label>
        <select
          id="status-presenca"
          value={status}
          onChange={(e) => setStatus(e.target.value)}
          disabled={salvando}
          className={classeInput}
        >
          {STATUS_PRESENCA.map((opcao) => (
            <option key={opcao} value={opcao}>
              {opcao}
            </option>
          ))}
        </select>
      </div>

      {mostrarHorarios && (
        <>
          <div className="space-y-1.5">
            <label htmlFor="hora-entrada" className="text-sm font-medium text-slate-700">
              Hora de entrada
            </label>
            <input
              id="hora-entrada"
              type="datetime-local"
              required
              value={entrada}
              onChange={(e) => setEntrada(e.target.value)}
              disabled={salvando}
              className={classeInput}
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="hora-saida" className="text-sm font-medium text-slate-700">
              Hora de saída <span className="font-normal text-slate-400">(opcional)</span>
            </label>
            <input
              id="hora-saida"
              type="datetime-local"
              value={saida}
              onChange={(e) => setSaida(e.target.value)}
              disabled={salvando}
              className={classeInput}
            />
            <p className="text-xs text-slate-500">Deixe em branco se o membro ainda não saiu.</p>
          </div>
        </>
      )}

      {erro && (
        <p role="alert" className="rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
          {erro}
        </p>
      )}

      <Button type="submit" disabled={salvando} className="w-full">
        {salvando ? "Salvando..." : "Salvar alterações"}
      </Button>
    </form>
  );
}

/**
 * SysMúsica — Regras puras da Portaria (sem acesso ao banco). Testadas em regras.test.ts.
 */
import { datetimeLocalParaIso } from "@/lib/dataLocal";

/** Status possíveis de uma presença. */
export const STATUS_PRESENCA = ["Presente", "Ausente", "Justificado"] as const;

/**
 * Valida os horários do formulário de edição (valores do campo datetime-local):
 * a entrada é obrigatória e válida; a saída é opcional, mas não pode ser anterior à entrada.
 * Devolve a mensagem de erro ou null quando está tudo certo.
 */
export function validarHorarios(entradaLocal: string, saidaLocal: string): string | null {
  const entrada = datetimeLocalParaIso(entradaLocal);
  if (!entrada) return "Informe uma hora de entrada válida.";

  if (saidaLocal.trim() === "") return null; // saída em branco = ainda não saiu

  const saida = datetimeLocalParaIso(saidaLocal);
  if (!saida) return "Informe uma hora de saída válida.";
  if (new Date(saida).getTime() < new Date(entrada).getTime()) {
    return "A hora de saída não pode ser anterior à hora de entrada.";
  }
  return null;
}

/**
 * SysMúsica — Serviço de dados da Portaria (registro de entrada e histórico).
 * Em "membros" só se lêem colunas liberadas (nunca "*": qr_secret e CPF completo são protegidos).
 * As portarias APROVADAS que o usuário pode usar vêm de listarPortarias (o banco já filtra pela
 * permissão e pela abrangência). Excluir é EXCLUSÃO LÓGICA. Editar altera só status e horários.

import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import { listarPortarias } from "@/modules/portarias/services/portarias";
import type { PortariaDetalhada } from "@/modules/portarias/tipos";
import {
  COLUNAS_MEMBRO_PUBLICO,
  type Evento,
  type Instrumento,
  type MembroPublico,
  type Presenca,
} from "@/types/database";

export interface DadosPortaria {
  eventos: Evento[];
  membros: MembroPublico[];
  instrumentos: Instrumento[];
  presencas: Presenca[];
  portarias: PortariaDetalhada[]; // as que o usuário pode ver (a tela usa só as APROVADAS)
}

/** Busca eventos, membros, instrumentos, presenças e portarias em paralelo. Lança erro se algo falhar. */
export async function buscarDadosPortaria(): Promise<DadosPortaria> {
  const [resEventos, resMembros, resPresencas, resInstrumentos, portarias] = await Promise.all([
    supabase.from("eventos").select("*").order("data_evento", { ascending: false }),
    supabase.from("membros").select(COLUNAS_MEMBRO_PUBLICO).order("nome", { ascending: true }),
    supabase.from("presencas").select("*"),
    supabase.from("instrumentos").select("*"),
    listarPortarias(),
  ]);

  const erroBanco = resEventos.error ?? resMembros.error ?? resPresencas.error ?? resInstrumentos.error;
  if (erroBanco) throw new Error(erroBanco.message);

  return {
    eventos: (resEventos.data ?? []) as Evento[],
    // Cast duplo: com lista de colunas, o cliente (sem tipagem gerada) não infere a linha
    membros: (resMembros.data ?? []) as unknown as MembroPublico[],
    instrumentos: (resInstrumentos.data ?? []) as Instrumento[],
    presencas: (resPresencas.data ?? []) as Presenca[],
    portarias,
  };
}

/** Campos que podem ser alterados na edição de uma presença. */
export interface DadosAtualizacaoPresenca {
  status: string;
  hora_entrada: string; // ISO (UTC)
  hora_saida: string | null; // ISO (UTC) ou null = ainda não saiu
}

export async function atualizarPresenca(id: string, dados: DadosAtualizacaoPresenca): Promise<void> {
  // .select("id") devolve as linhas alteradas: se o RLS negar, o update não dá erro, só não altera nada
  const { data, error } = await supabase.from("presencas").update(dados).eq("id", id).select("id");
  if (error) {
    // 23514 = regra do banco "saída depois da entrada" (a mensagem nativa é técnica)
    if (error.code === "23514") {
      throw new Error("A hora de saída não pode ser anterior à hora de entrada.");
    }
    throw new Error(
      mensagemDoErro(
        error,
        "Não foi possível salvar as alterações da presença.",
        "Este membro já tem uma entrada em aberto neste evento."
      )
    );
  }
  if (!data || data.length === 0) {
    throw new Error("Não foi possível salvar: você não tem permissão ou o registro não existe mais.");
  }
}

/** Exclusão lógica (função do banco, que confere a permissão portaria.excluir). */

  if (error) throw new Error(mensagemDoErro(error, "Não foi possível excluir o registro."));
}

=================== src/modules/portarias/janela.ts
/**
 * SysMúsica — Janela de atendimento da portaria (regras puras, sem banco).
 * A portaria abre 2 horas ANTES do início do evento e encerra 3 horas DEPOIS do início (horário de Salvador).
 * Ex.: ensaio às 14h -> aberta das 12h às 17h. A regra de verdade é aplicada pelo banco (migração 0024);
 * aqui ficam só os cálculos para a tela mostrar quando abre, até quando vale e se está liberada.
 */

export const HORAS_ANTES = 2;
export const HORAS_DEPOIS = 3; // encerra 3 horas depois do início (igual ao banco, migração 0027)
const MS_HORA = 60 * 60 * 1000;
const FUSO = "America/Sao_Paulo";

export type SituacaoJanela = "antes" | "aberta" | "encerrada";

export interface JanelaPortaria {
  abre: Date;
  fecha: Date;
  situacao: SituacaoJanela;
}

/** Janela do evento no instante "agora" (ms). Nulo se o evento não tem data válida. Os limites são inclusivos. */
export function janelaDaPortaria(dataEvento: string | null, agora: number): JanelaPortaria | null {
  if (!dataEvento) return null;
  const inicio = new Date(dataEvento).getTime();
  if (Number.isNaN(inicio)) return null;

  const abre = inicio - HORAS_ANTES * MS_HORA;
  const fecha = inicio + HORAS_DEPOIS * MS_HORA;
  const situacao: SituacaoJanela = agora < abre ? "antes" : agora > fecha ? "encerrada" : "aberta";
  return { abre: new Date(abre), fecha: new Date(fecha), situacao };
}

// Dia (AAAA-MM-DD) e hora no fuso de Salvador
function diaSalvador(data: Date): string {
  return new Intl.DateTimeFormat("en-CA", { timeZone: FUSO, year: "numeric", month: "2-digit", day: "2-digit" }).format(data);
}

function horaSalvador(data: Date): string {
  const [h, m] = new Intl.DateTimeFormat("pt-BR", { timeZone: FUSO, hour: "2-digit", minute: "2-digit", hour12: false })
    .format(data)
    .split(":");
  return `${h}h${m}`;
}

/** "12h00" quando é hoje; "19/10 às 12h00" quando é em outro dia. */
export function rotuloDoHorario(data: Date, agora: number): string {
  if (diaSalvador(data) === diaSalvador(new Date(agora))) return horaSalvador(data);
  const [, mes, dia] = diaSalvador(data).split("-");
  return `${dia}/${mes} às ${horaSalvador(data)}`;
}

/** Texto curto para a tela: "Abre às 12h00", "Aberta até 16h00" ou "Encerrada às 16h00" (com a data se não for hoje). */
export function descreverJanela(janela: JanelaPortaria, agora: number): string {
  if (janela.situacao === "antes") return `Abre ${comPreposicao(janela.abre, agora)}`;
  if (janela.situacao === "aberta") return `Aberta até ${rotuloDoHorario(janela.fecha, agora)}`;
  return `Encerrada ${comPreposicao(janela.fecha, agora)}`;
}

// "às 12h00" (hoje) ou "19/10 às 12h00" (outro dia)
function comPreposicao(data: Date, agora: number): string {
  const rotulo = rotuloDoHorario(data, agora);
  return rotulo.includes("às") ? rotulo : `às ${rotulo}`;
}

/**
 * SysMúsica — Regras puras da Gestão de portarias (sem banco). As regras de verdade (cota,
 * aprovação, abrangência) ficam no banco; aqui só há textos e conferências auxiliares da tela.
 */
import type { VarianteBadge } from "@/components/ui/Badge";
import { normalizarTexto } from "@/lib/texto";
import type { CotaPortarias, StatusPortaria } from "./tipos";

export const ROTULO_STATUS: Record<StatusPortaria, string> = {
  PENDENTE_APROVACAO: "Pendente de aprovação",
  APROVADA: "Aprovada",
  REJEITADA: "Rejeitada",
};

export const ROTULO_ACAO: Record<string, string> = {
  CRIADA: "Criada",
  APROVADA: "Aprovada",
  REJEITADA: "Rejeitada",
  EXCLUIDA: "Excluída",
};

/** Cor do selo de cada situação. */
export function varianteStatus(status: StatusPortaria): VarianteBadge {
  if (status === "APROVADA") return "sucesso";
  if (status === "REJEITADA") return "perigo";
  return "alerta";
}

export const MOTIVO_MINIMO = 5;
export const MOTIVO_MAXIMO = 300;

/** Valida o motivo da rejeição: devolve a mensagem de erro ou null se estiver certo. */
export function validarMotivoRejeicao(texto: string): string | null {
  const tamanho = texto.trim().length;
  if (tamanho < MOTIVO_MINIMO) return `Informe o motivo da rejeição (mínimo de ${MOTIVO_MINIMO} caracteres).`;
  if (tamanho > MOTIVO_MAXIMO) return `O motivo pode ter no máximo ${MOTIVO_MAXIMO} caracteres.`;
  return null;
}

/** Texto da cota do dia: "Hoje: 2 de 4 portarias criadas · restam 2". */
export function descreverCota(cota: CotaPortarias): string {
  // Controle total: não há limite diário
  if (cota.ilimitada) return `Hoje: ${cota.usadas} portarias criadas · sem limite (controle total)`;
  const sobra =
    cota.restantes === 0 ? "limite atingido" : cota.restantes === 1 ? "resta 1" : `restam ${cota.restantes}`;
  return `Hoje: ${cota.usadas} de ${cota.limite} portarias criadas · ${sobra}`;
}

/** A cidade está na lista? (ignora acento e maiúsculas, como o banco) */
export function cidadeNaAbrangencia(cidade: string | null, cidades: string[]): boolean {
  if (!cidade) return false;
  const chave = normalizarTexto(cidade);
  return cidades.some((item) => normalizarTexto(item) === chave);
}

/**
 * SysMúsica — Serviço de dados da Gestão de portarias.
 * Todas as regras (permissão, cota de 4 por dia, abrangência, nunca decidir a própria portaria,
 * motivo da rejeição) são aplicadas pelas funções do banco; a tela só chama e mostra o resultado.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import type { CotaPortarias, EventoParaPortaria, HistoricoPortaria, PortariaDetalhada } from "../tipos";

/** Portarias que o usuário pode ver, da mais recente para a mais antiga. */
export async function listarPortarias(): Promise<PortariaDetalhada[]> {
  const { data, error } = await supabase.rpc("listar_portarias");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar as portarias."));
  return (data ?? []) as PortariaDetalhada[];
}

/** Eventos para escolher ao criar uma portaria (os mais recentes primeiro). */
export async function buscarEventosParaPortaria(): Promise<EventoParaPortaria[]> {
  // O banco já devolve só os eventos das casas de oração da abrangência do usuário ("cidade" = Região › Setor)
  const { data, error } = await supabase.rpc("listar_eventos_para_portaria");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar os eventos."));
  return (data ?? []) as EventoParaPortaria[];
}

/** (antiga) Eventos direto da tabela, sem filtro por setor. A tela não usa mais. */
export async function buscarEventosParaPortariaSemSetor(): Promise<EventoParaPortaria[]> {
  const { data, error } = await supabase
    .from("eventos")
    .select("id, tipo, data_evento, local, cidade")
    .order("data_evento", { ascending: false })
    .limit(300);
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar os eventos."));
  return (data ?? []) as EventoParaPortaria[];
}

/** Cota do dia de quem está logado (exige a permissão de criar). */
export async function minhaCota(): Promise<CotaPortarias> {
  const { data, error } = await supabase.rpc("minha_cota_portarias");
  const linha = ((data ?? []) as CotaPortarias[])[0];
  if (error || !linha) {
    throw new Error(error ? mensagemDoErro(error, "Não foi possível consultar a cota do dia.") : "Cota indisponível.");
  }
  return linha;
}

/** Cidades da abrangência de quem está logado. */
export async function minhasCidades(): Promise<string[]> {
  const { data, error } = await supabase.rpc("minhas_cidades");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível consultar a sua abrangência."));
  return (data ?? []) as string[];
}

/** Cria uma portaria PENDENTE para o evento (o banco confere permissão, cota e duplicidade). */
export async function criarPortaria(eventoId: string): Promise<void> {
  const { error } = await supabase.rpc("criar_portaria", { p_evento_id: eventoId });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível criar a portaria."));
}

/** Aprova ou rejeita (a rejeição exige motivo). O banco confere permissão, abrangência e autoria. */
export async function decidirPortaria(id: string, aprovar: boolean, motivo: string | null): Promise<void> {
  const { error } = await supabase.rpc("decidir_portaria", { p_id: id, p_aprovar: aprovar, p_motivo: motivo });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível registrar a decisão."));
}

/** Exclusão lógica (a cota do dia NÃO é devolvida). */
export async function excluirPortaria(id: string): Promise<void> {
  const { error } = await supabase.rpc("excluir_portaria", { p_id: id });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível excluir a portaria."));
}

/** Histórico de uma portaria (criação, decisões e exclusão). */
export async function historicoDaPortaria(id: string): Promise<HistoricoPortaria[]> {
  const { data, error } = await supabase.rpc("historico_portaria", { p_id: id });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar o histórico."));
  return (data ?? []) as HistoricoPortaria[];
}
