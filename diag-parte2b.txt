"use client";

/**
 * SysMúsica — Seletor de CASA DE ORAÇÃO (igreja/salão), agrupado por "Região › Setor".
 * Mostra só as casas de oração que o usuário pode usar (o banco filtra pela abrangência dele; quem tem
 * acesso total vê todas). Usado nos eventos e, em seguida, no cadastro de membros.
 */
import { useEffect, useMemo, useState } from "react";
import { agruparCasasOracaoParaSelecao } from "../regras";
import { listarOrganizacaoParaSelecao } from "../services/organizacao";
import type { NoOrganizacao } from "../tipos";

interface SeletorCasaOracaoProps {
  id?: string;
  valor: string; // id da casa de oração escolhida ("" = nenhuma)
  onChange: (casaOracaoId: string) => void;
  /** Opcional: devolve também o nome da casa de oração escolhida (vazio quando nenhuma). */
  onEscolherNome?: (nome: string) => void;
  disabled?: boolean;
  className?: string;
}

export default function SeletorCasaOracao({
  id,
  valor,
  onChange,
  onEscolherNome,
  disabled,
  className,
}: SeletorCasaOracaoProps) {
  const [nos, setNos] = useState<NoOrganizacao[] | null>(null); // nulo = ainda carregando
  const [erro, setErro] = useState<string | null>(null);

  /* eslint-disable react-hooks/set-state-in-effect */
  useEffect(() => {
    let cancelado = false;
    void listarOrganizacaoParaSelecao()
      .then((lista) => {
        if (!cancelado) setNos(lista);
      })
      .catch((e: unknown) => {
        if (cancelado) return;
        setNos([]);
        setErro(e instanceof Error ? e.message : "Não foi possível carregar as casas de oração.");
      });
    return () => {
      cancelado = true;
    };
  }, []);
  /* eslint-enable react-hooks/set-state-in-effect */

  const grupos = useMemo(() => agruparCasasOracaoParaSelecao(nos ?? []), [nos]);
  const existe = grupos.some((grupo) => grupo.casasOracao.some((casaOracao) => casaOracao.id === valor));

  return (
    <>
      <select
        id={id}
        value={valor}
        onChange={(e) => {
          const novoId = e.target.value;
          onChange(novoId);
          onEscolherNome?.((nos ?? []).find((no) => no.nivel === "CASA_ORACAO" && no.id === novoId)?.nome ?? "");
        }}
        disabled={disabled || nos === null}
        className={className}
      >
        <option value="">{nos === null ? "Carregando..." : "Nenhuma (definir depois)"}</option>
        {/* A casa de oração atual do registro pode estar fora da abrangência de quem edita: continua visível, sem trocar sozinha */}
        {nos !== null && valor !== "" && !existe && <option value={valor}>Casa de Oração atual (fora da sua abrangência)</option>}
        {grupos.map((grupo) => (
          <optgroup key={grupo.rotulo} label={grupo.rotulo}>
            {grupo.casasOracao.map((casaOracao) => (
              <option key={casaOracao.id} value={casaOracao.id}>
                {casaOracao.nome}
              </option>
            ))}
          </optgroup>
        ))}
      </select>
      {erro && (
        <p role="alert" className="mt-1 text-xs text-red-700">
          {erro}
        </p>
      )}
    </>
  );
}

/**
 * SysMúsica — Serviço de dados da organização (Estado > Região > Setor > Casa de Oração) e dos vínculos dos usuários.
 * Todas as regras (permissão, nomes repetidos, abrangência) são aplicadas pelo banco (migração 0012);
 * a tela só chama e mostra o resultado, com as mensagens do banco já escritas para o usuário.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import type { NivelOrganizacao, NoOrganizacao, VinculoRotulado, VinculoUsuario } from "../tipos";

interface LinhaNo extends Omit<NoOrganizacao, "qtd_filhos" | "qtd_membros"> {
  qtd_filhos?: number | string;
  qtd_membros?: number | string;
}

/** Árvore completa (tela de Organização e vínculos). */
export async function listarOrganizacao(): Promise<NoOrganizacao[]> {
  const { data, error } = await supabase.rpc("listar_organizacao");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar a organização."));
  return ((data ?? []) as LinhaNo[]).map((linha) => ({
    ...linha,
    qtd_filhos: Number(linha.qtd_filhos ?? 0), // o banco devolve bigint
    qtd_membros: Number(linha.qtd_membros ?? 0),
  }));
}

/** Itens ativos que o usuário pode usar (seletor de casa de oração nos formulários). */
export async function listarOrganizacaoParaSelecao(): Promise<NoOrganizacao[]> {
  const { data, error } = await supabase.rpc("listar_organizacao_para_selecao");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar as casas de oração."));
  return (data ?? []) as NoOrganizacao[];
}

/** Cria (id nulo) ou altera um item da organização. */
export async function salvarNo(dados: {
  nivel: NivelOrganizacao;
  id: string | null;
  paiId: string | null;
  nome: string;
  descricao: string;
  ativo: boolean;
  cidade: string;
  bairro: string;
}): Promise<void> {
  const { error } = await supabase.rpc("salvar_organizacao", {
    p_nivel: dados.nivel,
    p_id: dados.id,
    p_parent: dados.paiId,
    p_nome: dados.nome,
    p_descricao: dados.descricao,
    p_ativo: dados.ativo,
    p_cidade: dados.cidade,
    p_bairro: dados.bairro,
  });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível salvar."));
}

/** Exclui um item VAZIO da organização (o banco confere permissão, filhos, membros, eventos, portarias e vínculos). */
export async function excluirNo(nivel: NivelOrganizacao, id: string): Promise<void> {
  const { error } = await supabase.rpc("excluir_organizacao", { p_nivel: nivel, p_id: id });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível excluir."));
}

/** Muda uma casa de oração de setor (fica no histórico e na auditoria). */
export async function transferirCasaOracao(casaOracaoId: string, setorId: string, justificativa: string): Promise<void> {
  const { error } = await supabase.rpc("transferir_casa_oracao", {
    p_casa_oracao: casaOracaoId,
    p_setor: setorId,
    p_justificativa: justificativa,
  });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível transferir a casa de oração."));
}

/** Vínculos de cada usuário com o rótulo: { idDoUsuario: [{ nivel, alvo_id, rotulo }] }. */
export async function listarVinculosRotulados(): Promise<Record<string, VinculoRotulado[]>> {
  const { data, error } = await supabase.rpc("listar_vinculos_rotulados");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível carregar os vínculos dos usuários."));

  const mapa: Record<string, VinculoRotulado[]> = {};
  for (const linha of (data ?? []) as (VinculoRotulado & { usuario_id: string })[]) {
    mapa[linha.usuario_id] = [
      ...(mapa[linha.usuario_id] ?? []),
      { nivel: linha.nivel, alvo_id: linha.alvo_id, rotulo: linha.rotulo },
    ];
  }
  return mapa;
}

/** Define os vínculos de um usuário, substituindo a lista anterior (exige usuarios.vinculos). */
export async function definirVinculos(usuarioId: string, vinculos: VinculoUsuario[]): Promise<void> {
  const { error } = await supabase.rpc("definir_vinculos", { p_usuario: usuarioId, p_vinculos: vinculos });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível salvar os vínculos."));
}

/** SysMúsica — Tipos da organização (Estado > Região > Setor > Casa de Oração). Campos = colunas das funções do banco (0012). */

export type NivelOrganizacao = "ESTADO" | "REGIAO" | "SETOR" | "CASA_ORACAO";

/** Um item da árvore. Os campos opcionais só vêm na consulta completa (tela de Organização). */
export interface NoOrganizacao {
  nivel: NivelOrganizacao;
  id: string;
  parent_id: string | null; // estado -> nulo; região -> estado; setor -> região; casa de oração -> setor (nulo = sem setor)
  nome: string;
  descricao?: string | null;
  ativo?: boolean;
  cidade?: string | null; // só casas de oração
  bairro?: string | null; // só casas de oração
  qtd_filhos?: number; // regiões do estado, setores da região, casas de oração do setor
  qtd_membros?: number; // só casas de oração
}

/** Vínculo de um usuário com um nível da organização (sem o rótulo). */
export interface VinculoUsuario {
  nivel: NivelOrganizacao;
  alvo_id: string;
}

/** Vínculo com o texto para exibir (ex.: "Salvador › Setor 1"). */
export interface VinculoRotulado extends VinculoUsuario {
  rotulo: string;
}
