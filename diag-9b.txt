/**
 * SysMúsica — Página inicial "/".
 * Quem tem permissão vê o dashboard. Quem não tem (ex: grupo Portaria) vai direto para o
 * primeiro módulo que pode acessar.

/**
 * SysMúsica — Cálculos dos gráficos do dashboard (funções PURAS, sem acesso ao banco).
 * Ficam separados do serviço de dados para poderem ser testados (agregacoes.test.ts).
 */
import type { ItemGrafico } from "@/components/charts/tipos";
import type { Cargo, Familia, Instrumento } from "@/types/database";

/** Quantidade máxima de barras antes de agrupar o restante em "Outros". */
const LIMITE_BARRAS = 7;

/** Campos de "membros" usados nas distribuições (sem dados sensíveis). */
export interface MembroResumo {
  id: string;
  cargo_id: string;
  instrumento_id: string | null;
  congregacao: string;
  status: string;
}

/** Conta itens agrupando pela chave devolvida; ordena do maior para o menor (empate: ordem alfabética). */
export function contarPor<T>(itens: T[], chave: (item: T) => string): ItemGrafico[] {
  const contagem = new Map<string, number>();
  for (const item of itens) {
    const rotulo = chave(item);
    contagem.set(rotulo, (contagem.get(rotulo) ?? 0) + 1);
  }
  return [...contagem.entries()]
    .map(([rotulo, valor]) => ({ rotulo, valor }))
    .sort((a, b) => b.valor - a.valor || a.rotulo.localeCompare(b.rotulo, "pt-BR"));
}

/** Mantém os N maiores e soma o restante em "Outros". */
export function topComOutros(itens: ItemGrafico[], limite: number): ItemGrafico[] {
  if (itens.length <= limite) return itens;
  const principais = itens.slice(0, limite);
  const resto = itens.slice(limite).reduce((soma, item) => soma + item.valor, 0);
  return [...principais, { rotulo: "Outros", valor: resto }];
}

export interface Distribuicoes {
  porCargo: ItemGrafico[];
  porFamilia: ItemGrafico[];
  porInstrumento: ItemGrafico[];
  porCongregacao: ItemGrafico[];
}

/** Monta as distribuições de membros por cargo, família, instrumento e congregação. */
export function montarDistribuicoes(entrada: {
  membros: MembroResumo[];
  cargos: Cargo[];
  instrumentos: Instrumento[];
  familias: Familia[];
}): Distribuicoes {
  const { membros, cargos, instrumentos, familias } = entrada;
  const cargoPorId = new Map(cargos.map((cargo): [string, Cargo] => [cargo.id, cargo]));
  const familiaPorId = new Map(familias.map((familia): [string, Familia] => [familia.id, familia]));
  const instrumentoPorId = new Map(
    instrumentos.map((instrumento): [string, Instrumento] => [instrumento.id, instrumento])
  );

  const porCargo = topComOutros(
    contarPor(membros, (membro) => cargoPorId.get(membro.cargo_id)?.nome ?? "Cargo não encontrado"),
    LIMITE_BARRAS + 1
  );

  // Família do membro = família do instrumento dele
  const porFamilia = contarPor(membros, (membro) => {
    if (!membro.instrumento_id) return "Sem instrumento";
    const instrumento = instrumentoPorId.get(membro.instrumento_id);
    if (!instrumento) return "Sem instrumento";
    if (!instrumento.familia_id) return "Sem família";
    return familiaPorId.get(instrumento.familia_id)?.nome ?? "Sem família";
  });

  // Instrumentos mais usados (só membros que têm instrumento)
  const nomesDeInstrumento = membros.flatMap((membro) => {
    const nome = membro.instrumento_id ? instrumentoPorId.get(membro.instrumento_id)?.nome : undefined;
    return nome ? [nome] : [];
  });
  const porInstrumento = topComOutros(contarPor(nomesDeInstrumento, (nome) => nome), LIMITE_BARRAS);

  const porCongregacao = topComOutros(
    contarPor(membros, (membro) => membro.congregacao.trim() || "Sem congregação"),
    LIMITE_BARRAS
  );

  return { porCargo, porFamilia, porInstrumento, porCongregacao };
}

export interface FrequenciaEvento {
  eventoId: string;
  rotulo: string; // "dd/mm"
  presentes: number;
  ausentes: number;
  justificados: number;
  total: number;
}

/**
 * Agrupa as presenças por evento. Os eventos chegam do mais recente para o mais antigo;
 * o resultado sai em ordem cronológica (mais antigo primeiro), como o gráfico exibe.
 */
export function agruparFrequencia(
  eventos: { id: string; data_evento: string }[],
  presencas: { evento_id: string; status: string }[]
): FrequenciaEvento[] {
  const ordenados = [...eventos].reverse();
  const porEvento = new Map<string, FrequenciaEvento>();

  for (const evento of ordenados) {
    porEvento.set(evento.id, {
      eventoId: evento.id,
      rotulo: new Date(evento.data_evento).toLocaleDateString("pt-BR", {
        day: "2-digit",
        month: "2-digit",
      }),
      presentes: 0,
      ausentes: 0,
      justificados: 0,
      total: 0,
    });
  }

  for (const presenca of presencas) {
    const frequencia = porEvento.get(presenca.evento_id);
    if (!frequencia) continue;
    frequencia.total += 1;
    if (presenca.status === "Presente") frequencia.presentes += 1;
    else if (presenca.status === "Ausente") frequencia.ausentes += 1;
    else if (presenca.status === "Justificado") frequencia.justificados += 1;
  }

  return ordenados.flatMap((evento) => {
    const frequencia = porEvento.get(evento.id);
    return frequencia ? [frequencia] : [];
  });
}

/** Percentual de presenças "Presente" sobre todos os registros (null quando não há registros). */
export function calcularFrequenciaMedia(frequencias: FrequenciaEvento[]): number | null {
  const total = frequencias.reduce((soma, item) => soma + item.total, 0);
  if (total === 0) return null;
  const presentes = frequencias.reduce((soma, item) => soma + item.presentes, 0);
  return Math.round((presentes / total) * 100);
}

  frequenciaMedia: number | null;
}

/** Presenças dos eventos informados, lidas em páginas de 1000 para não truncar. */
async function buscarPresencasDosEventos(
  ids: string[]
): Promise<{ evento_id: string; status: string }[]> {
  if (ids.length === 0) return [];

  const linhas: { evento_id: string; status: string }[] = [];
  for (let pagina = 0; pagina < MAXIMO_PAGINAS; pagina += 1) {
    const inicio = pagina * TAMANHO_PAGINA;
    const { data, error } = await supabase
      .from("presencas")
      .select("evento_id, status")
      .in("evento_id", ids)
      .order("id") // ordem estável entre as páginas
      .range(inicio, inicio + TAMANHO_PAGINA - 1);
    if (error) throw new Error(error.message);

    const trecho = (data ?? []) as { evento_id: string; status: string }[];
    linhas.push(...trecho);
    if (trecho.length < TAMANHO_PAGINA) break;
  }
  return linhas;
}

/** Busca e calcula todos os dados dos gráficos. Lança erro se qualquer consulta falhar. */
export async function buscarDadosGraficos(): Promise<DadosGraficos> {
  const agora = new Date().toISOString();
  const contarStatus = (status: string) =>
    supabase.from("presencas").select("id", { count: "exact", head: true }).eq("status", status);

  const [
    resMembros,
    resCargos,
    resInstrumentos,
    resFamilias,
    resEventos,
    resPresente,
    resAusente,
    resJustificado,
  ] = await Promise.all([
    supabase.from("membros").select("id, cargo_id, instrumento_id, congregacao, status"),
    supabase.from("cargos").select("*"),
    supabase.from("instrumentos").select("*"),
    supabase.from("familias").select("*"),
    supabase
      .from("eventos")
      .select("id, data_evento")
      .lte("data_evento", agora) // só eventos que já aconteceram
      .order("data_evento", { ascending: false })
      .limit(EVENTOS_NO_GRAFICO),
    contarStatus("Presente"),
    contarStatus("Ausente"),
    contarStatus("Justificado"),
  ]);

  const respostas = [
    resMembros,
    resCargos,
    resInstrumentos,
    resFamilias,
    resEventos,
    resPresente,
    resAusente,
    resJustificado,
  ];
  const erroBanco = respostas.find((resposta) => resposta.error)?.error;
  if (erroBanco) throw new Error(erroBanco.message);

  const distribuicoes = montarDistribuicoes({
    membros: (resMembros.data ?? []) as MembroResumo[],
    cargos: (resCargos.data ?? []) as Cargo[],
    instrumentos: (resInstrumentos.data ?? []) as Instrumento[],
    familias: (resFamilias.data ?? []) as Familia[],
  });

  const eventos = (resEventos.data ?? []) as { id: string; data_evento: string }[];
  const presencas = await buscarPresencasDosEventos(eventos.map((evento) => evento.id));
  const frequencia = agruparFrequencia(eventos, presencas);

  return {
    ...distribuicoes,
    presencasPorStatus: [
      { rotulo: "Presente", valor: resPresente.count ?? 0, cor: COR_PRESENTE },
      { rotulo: "Ausente", valor: resAusente.count ?? 0, cor: COR_AUSENTE },
      { rotulo: "Justificado", valor: resJustificado.count ?? 0, cor: COR_JUSTIFICADO },
    ],
    frequencia,
    frequenciaMedia: calcularFrequenciaMedia(frequencia),
  };
}

/**
 * SysMúsica — Serviço do dashboard (acesso a dados).
 *
 * Fica fora do componente para a tela não misturar interface com consultas ao banco.
 * As contagens usam { count: "exact", head: true }: o banco só devolve o número, sem baixar linhas.
 * Atenção: em "membros" sempre se seleciona "id" (nunca "*"), pois o qr_secret é protegido.
 */
import { supabase } from "@/lib/supabase";
import type { Evento } from "@/types/database";

export interface Indicadores {
  membros: number;
  membrosAtivos: number;
  eventos: number;
  eventosFuturos: number;
  presencas: number;
  instrumentos: number;
  instrumentosSemFamilia: number;
}

/** Busca todos os totais do dashboard. Lança erro se qualquer consulta falhar. */
export async function buscarIndicadores(): Promise<Indicadores> {
  const agora = new Date().toISOString();

  const resultados = await Promise.all([
    supabase.from("membros").select("id", { count: "exact", head: true }),
    supabase.from("membros").select("id", { count: "exact", head: true }).eq("status", "Ativo"),
    supabase.from("eventos").select("id", { count: "exact", head: true }),
    supabase.from("eventos").select("id", { count: "exact", head: true }).gte("data_evento", agora),
    supabase.from("presencas").select("id", { count: "exact", head: true }),
    supabase.from("instrumentos").select("id", { count: "exact", head: true }),
    supabase.from("instrumentos").select("id", { count: "exact", head: true }).is("familia_id", null),
  ]);

  // Se qualquer consulta falhar, interrompe com o primeiro erro encontrado
  const erroBanco = resultados.find((resultado) => resultado.error)?.error;
  if (erroBanco) throw new Error(erroBanco.message);

  const [membros, membrosAtivos, eventos, eventosFuturos, presencas, instrumentos, semFamilia] =
    resultados.map((resultado) => resultado.count ?? 0);

  return {
    membros,
    membrosAtivos,
    eventos,
    eventosFuturos,
    presencas,
    instrumentos,
    instrumentosSemFamilia: semFamilia,
  };
}

/** Próximos eventos (a partir de agora), do mais próximo para o mais distante. */
export async function buscarProximosEventos(limite = 5): Promise<Evento[]> {
  const { data, error } = await supabase
    .from("eventos")
    .select("*")
    .gte("data_evento", new Date().toISOString())
    .order("data_evento", { ascending: true })
    .limit(limite);

  if (error) throw new Error(error.message);
  return (data ?? []) as Evento[];
}
