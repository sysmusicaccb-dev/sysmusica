"use client";

import SeletorCasaOracao from "@/modules/organizacao/components/SeletorCasaOracao";

/**
 * SysMúsica — Formulário de evento (cadastro e edição) — conteúdo do painel lateral.
 * evento = null -> cadastrar; preenchido -> editar.
 *
 * Finalidade (combobox): sugere as finalidades já cadastradas. Se a pessoa digita uma que ainda não
 * existe, ela é salva automaticamente e passa a aparecer como sugestão nos próximos eventos.
 */

import Button from "@/components/ui/Button";
import { datetimeLocalParaIso, isoParaDatetimeLocal } from "@/lib/dataLocal";

interface FormularioEventoProps {
  evento: Evento | null;
  finalidades: FinalidadeEvento[];
  sugestoesCidade: string[]; // cidades já cadastradas em outros eventos
  /** Chamado após salvar com sucesso, com a mensagem para o aviso da tela. */
  onSalvo: (mensagem: string) => void | Promise<void>;
}

// Mesma identidade visual dos demais formulários.
// [color-scheme:light] mantém o ícone do seletor de data visível mesmo com o navegador em tema escuro.
const classeInput =
  "w-full rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 " +
  "placeholder:text-slate-400 shadow-sm focus:border-brand-500 focus:outline-none " +
  "focus:ring-2 focus:ring-brand-200 disabled:cursor-not-allowed disabled:opacity-60 " +
  "[color-scheme:light]";

export default function FormularioEvento({
  evento,
  finalidades,
  sugestoesCidade,
  onSalvo,
}: FormularioEventoProps) {
  const editando = evento !== null;

  const finalidadeInicial = evento?.finalidade_id
    ? (finalidades.find((finalidade) => finalidade.id === evento.finalidade_id)?.nome ?? "")
    : "";

  const [tipo, setTipo] = useState(evento?.tipo ?? "");
  const [finalidadeTexto, setFinalidadeTexto] = useState(finalidadeInicial);
  const [dataEvento, setDataEvento] = useState(evento ? isoParaDatetimeLocal(evento.data_evento) : "");
  const [local, setLocal] = useState(evento?.local ?? "");
  const [cidade, setCidade] = useState(evento?.cidade ?? "");
  const [casaOracaoId, setCasaOracaoId] = useState(evento?.casa_oracao_id ?? "");
  const [salvando, setSalvando] = useState(false);
  const [erro, setErro] = useState<string | null>(null);
  const [faltando, setFaltando] = useState<string[]>([]); // campos obrigatórios vazios (borda vermelha)

  // Borda vermelha no campo obrigatório que ficou vazio
  const classeCampo = (rotulo: string) =>
    classeInput + (faltando.includes(rotulo) ? " !border-red-500 !ring-2 !ring-red-200" : "");

  // A finalidade digitada já existe? (comparação sem acento e sem diferenciar maiúsculas)
  const finalidadeDigitada = finalidadeTexto.trim();
  const finalidadeExistente = finalidadeDigitada
    ? finalidades.find(
        (finalidade) => normalizarTexto(finalidade.nome) === normalizarTexto(finalidadeDigitada)
      )
    : undefined;

  async function salvar(submissao: FormEvent<HTMLFormElement>) {
    submissao.preventDefault();
    if (salvando) return; // evita envio duplicado
    setErro(null);

    // Todos os campos são obrigatórios: o alerta lista os que faltam e eles ficam com borda vermelha
    const vazios = camposFaltando({ tipo, finalidade: finalidadeTexto, dataEvento, local, cidade, casaOracaoId });
    setFaltando(vazios);
    if (vazios.length > 0) return setErro(mensagemCamposFaltando(vazios));

    const erroFinalidade = validarFinalidade(finalidadeTexto);
    if (erroFinalidade) return setErro(erroFinalidade);

    // O datetime-local não traz fuso: new Date() o lê como hora LOCAL do navegador e a conversão
    // para ISO (UTC) é o que o campo timestamptz do banco espera.
    const dataIso = datetimeLocalParaIso(dataEvento);
    if (!dataIso) return setErro("Informe uma data e hora válidas.");

    // Não se agenda evento em data e hora que já passaram (ao editar, só vale se a data for alterada)
    const erroData = validarDataDoEvento(dataIso, Date.now(), evento?.data_evento ?? null);
    if (erroData) return setErro(erroData);

    setSalvando(true);
    try {
      // Passo 1: finalidade (usa a existente ou salva uma nova para os próximos eventos)
      const finalidade = await obterOuCriarFinalidade(finalidadeTexto, finalidades);

      // Passo 2: grava o evento
      const dados = {
        tipo: tipo.trim(),
        data_evento: dataIso,
        local: local.trim(),
        finalidade_id: finalidade.id,
        cidade: cidade.trim() || null,
        casa_oracao_id: casaOracaoId || null,
      };
      if (evento) await atualizarEvento(evento.id, dados);
      else await criarEvento(dados);

      const complemento = finalidade.criada
        ? ` A finalidade "${finalidade.nome}" foi salva para os próximos eventos.`
        : "";
      await onSalvo(
        editando ? `Evento atualizado.${complemento}` : `Evento cadastrado com sucesso!${complemento}`
      );
    } catch (e) {
      setErro(e instanceof Error ? e.message : "Não foi possível salvar o evento.");
    } finally {
      setSalvando(false);
    }
  }

  return (
    <form onSubmit={salvar} noValidate className="space-y-5">
      <p className="text-xs text-slate-500">Todos os campos são obrigatórios (marcados com *).</p>

      <div className="space-y-1.5">
        <label htmlFor="tipo" className="text-sm font-medium text-slate-700">
          Tipo <span className="text-red-600"> *</span>
        </label>
        {/* Campo livre com sugestões (datalist): aceita Ensaio, Culto ou qualquer outro tipo */}
        <input
          id="tipo"
          type="text"
          list="sugestoes-tipo-evento"
          required
          value={tipo}
          onChange={(e) => setTipo(e.target.value)}
          disabled={salvando}
          placeholder="Ex: Ensaio, Culto"
          className={classeCampo(ROTULO_TIPO)}
        />
        <datalist id="sugestoes-tipo-evento">
          <option value="Ensaio" />
          <option value="Culto" />
        </datalist>
      </div>

      {/* Finalidade (combobox): escolha uma cadastrada ou digite uma nova */}
      <div className="space-y-1.5">
        <label htmlFor="finalidade" className="text-sm font-medium text-slate-700">
          Finalidade do evento <span className="text-red-600"> *</span>
        </label>
        <input
          id="finalidade"
          type="text"
          list="lista-finalidades"
          autoComplete="off"
          required
          maxLength={FINALIDADE_MAXIMO}
          value={finalidadeTexto}
          onChange={(e) => setFinalidadeTexto(e.target.value)}
          disabled={salvando}
          placeholder="Digite ou escolha uma finalidade"
          className={classeCampo(ROTULO_FINALIDADE)}
        />
        <datalist id="lista-finalidades">
          {finalidades.map((finalidade) => (
            <option key={finalidade.id} value={finalidade.nome} />
          ))}
        </datalist>
        {/* Feedback ao vivo: finalidade já cadastrada ou nova (será salva para os próximos eventos) */}
        {finalidadeDigitada && (
          <p className={"text-xs " + (finalidadeExistente ? "text-slate-500" : "text-brand-700")}>
            {finalidadeExistente
              ? "Finalidade já cadastrada."
              : `Nova finalidade: "${formatarNomeFinalidade(finalidadeDigitada)}" será salva para os próximos eventos.`}
          </p>
        )}
      </div>

      <div className="space-y-1.5">
        <label htmlFor="data_evento" className="text-sm font-medium text-slate-700">
          Data e hora <span className="text-red-600"> *</span>
        </label>
        <input
          id="data_evento"
          type="datetime-local"
          required
          value={dataEvento}
          onChange={(e) => setDataEvento(e.target.value)}
          disabled={salvando}
          className={classeCampo(ROTULO_DATA)}
        />
      </div>

      <div className="space-y-1.5">
        <label htmlFor="local" className="text-sm font-medium text-slate-700">
          Local <span className="text-red-600"> *</span>
        </label>
        <input
          id="local"
          type="text"
          required
          value={local}
          onChange={(e) => setLocal(e.target.value)}
          disabled={salvando}
          placeholder="Ex: Congregação Central"
          className={classeCampo(ROTULO_LOCAL)}
        />
      </div>

      <div className="space-y-1.5">
        <label htmlFor="cidade-evento" className="text-sm font-medium text-slate-700">
          Cidade <span className="text-red-600"> *</span>
        </label>
        <datalist id="sugestoes-cidade-evento">
          {sugestoesCidade.map((sugestao) => (
            <option key={sugestao} value={sugestao} />
          ))}
        </datalist>
        <input
          id="cidade-evento"
          type="text"
          list="sugestoes-cidade-evento"
          autoComplete="off"
          value={cidade}
          onChange={(e) => setCidade(e.target.value)}
          disabled={salvando}
          placeholder="Ex: Lauro de Freitas"
          className={classeCampo(ROTULO_CIDADE)}
        />
      </div>

      <div className="space-y-1.5">
        <label htmlFor="casa-oracao-evento" className="text-sm font-medium text-slate-700">

        {salvando ? "Salvando..." : editando ? "Salvar alterações" : "Cadastrar evento"}
      </Button>
    </form>
  );
}

################ src/modules/eventos/camposEvento.ts
/**
 * SysMúsica — Campos obrigatórios do evento (regra pura, sem banco).
 * Todos os campos do formulário de evento são obrigatórios. Devolve os NOMES (como aparecem na tela) dos
 * campos que estão vazios, na ordem do formulário, para a tela marcar cada um e mostrar o alerta.
 */
export interface CamposEvento {
  tipo: string;
  finalidade: string;
  dataEvento: string; // valor do campo data e hora (vazio = não preenchido)
  local: string;
  cidade: string;
  casaOracaoId: string;
}

export const ROTULO_TIPO = "Tipo";
export const ROTULO_FINALIDADE = "Finalidade do evento";
export const ROTULO_DATA = "Data e hora";
export const ROTULO_LOCAL = "Local";
export const ROTULO_CIDADE = "Cidade";
export const ROTULO_CASA_ORACAO = "Casa de Oração";

export function camposFaltando(campos: CamposEvento): string[] {
  const faltando: string[] = [];
  if (!campos.tipo.trim()) faltando.push(ROTULO_TIPO);
  if (!campos.finalidade.trim()) faltando.push(ROTULO_FINALIDADE);
  if (!campos.dataEvento.trim()) faltando.push(ROTULO_DATA);
  if (!campos.local.trim()) faltando.push(ROTULO_LOCAL);
  if (!campos.cidade.trim()) faltando.push(ROTULO_CIDADE);
  if (!campos.casaOracaoId.trim()) faltando.push(ROTULO_CASA_ORACAO);
  return faltando;
}

/** Mensagem do alerta: "Preencha todos os campos obrigatórios: Local, Cidade." */
export function mensagemCamposFaltando(faltando: string[]): string {
  return `Preencha todos os campos obrigatórios: ${faltando.join(", ")}.`;
}

/**
 * SysMúsica — Serviço de dados de eventos.
 * Excluir é EXCLUSÃO LÓGICA: o evento some das telas e as presenças registradas são mantidas.
 */
import { supabase } from "@/lib/supabase";
import { mensagemDoErro } from "@/lib/erros";
import type { Evento, EventoInsert, FinalidadeEvento } from "@/types/database";

/** Busca eventos (do mais recente para o mais antigo) e finalidades. Lança erro se a consulta falhar. */
export async function buscarEventosEFinalidades(): Promise<{
  eventos: Evento[];
  finalidades: FinalidadeEvento[];
}> {
  const [resEventos, resFinalidades] = await Promise.all([
    supabase.from("eventos").select("*").order("data_evento", { ascending: false }),
    supabase.from("finalidades_evento").select("*").order("nome", { ascending: true }),
  ]);

  const erroBanco = resEventos.error ?? resFinalidades.error;
  if (erroBanco) throw new Error(erroBanco.message);

  return {
    eventos: (resEventos.data ?? []) as Evento[],
    finalidades: (resFinalidades.data ?? []) as FinalidadeEvento[],
  };
}

export async function criarEvento(dados: EventoInsert): Promise<void> {
  const { error } = await supabase.from("eventos").insert(dados);
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível cadastrar o evento."));
}

/** Campos que podem ser alterados na edição (o banco libera só estas colunas). */
export interface DadosAtualizacaoEvento {
  tipo: string;
  data_evento: string;
  local: string;
  finalidade_id: string | null;
  cidade: string | null;
  casa_oracao_id?: string | null;
}

export async function atualizarEvento(id: string, dados: DadosAtualizacaoEvento): Promise<void> {
  // .select("id") devolve as linhas alteradas: se o RLS negar, o update não dá erro, só não altera nada
  const { data, error } = await supabase.from("eventos").update(dados).eq("id", id).select("id");
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível salvar as alterações do evento."));
  if (!data || data.length === 0) {
    throw new Error("Não foi possível salvar: você não tem permissão ou o evento não existe mais.");
  }
}

/** Cancela o evento com motivo (função do banco, que confere a permissão eventos.cancelar: só o Administrador). */
export async function cancelarEvento(id: string, motivo: string): Promise<void> {
  const { error } = await supabase.rpc("cancelar_evento", { p_id: id, p_motivo: motivo });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível cancelar o evento."));
}

/** Exclusão lógica (função do banco, que confere a permissão eventos.excluir). */
export async function excluirEvento(id: string): Promise<void> {
  const { error } = await supabase.rpc("excluir_evento", { p_id: id });
  if (error) throw new Error(mensagemDoErro(error, "Não foi possível excluir o evento."));
}

          />
        </div>
      </div>

      <div className="overflow-x-auto">
        <table className="min-w-full divide-y divide-slate-200 text-sm">
          <thead className="bg-slate-50">

              </th>

              </th>

              </th>

                </th>

                const cancelado = Boolean(evento.cancelado_em);
                const situacao = situacaoDoEvento(evento.data_evento, cancelado, agora);
                // Evento encerrado ou cancelado é só leitura; o Administrador (podeCancelar) ainda corrige encerrados
                const podeAlterar = !cancelado && (eventoAberto(evento.data_evento, cancelado, agora) || podeCancelar);

                return (
                  <tr key={evento.id} className="transition hover:bg-slate-50">
                    <td className="px-4 py-3">
                      <div className="flex items-center gap-3">
                        {/* Bloco de data: dia e mês abreviado */}
                        <div className="flex h-11 w-11 shrink-0 flex-col items-center justify-center rounded-xl bg-brand-50 text-brand-700">
                          <span className="text-sm font-bold leading-none">{dia}</span>
                          <span className="mt-0.5 text-[10px] font-medium uppercase">{mes}</span>
                        </div>
                        <div className="min-w-0">
                          <p className="text-slate-700">{formatarDataHora(evento.data_evento)}</p>
                          {/* Telas pequenas: finalidade e local aparecem aqui */}
                          <p className="line-clamp-2 text-xs text-slate-500 md:hidden">
                            {[finalidade, evento.local, evento.cidade].filter(Boolean).join(" · ")}
                          </p>
                        </div>
                      </div>
                    </td>
                    <td className="px-4 py-3">
                      <Badge variante={varianteTipo(evento.tipo)}>{evento.tipo}</Badge>
                    </td>
                    <td className="px-4 py-3">
                      <Badge variante={varianteSituacaoEvento(situacao)}>{ROTULO_SITUACAO_EVENTO[situacao]}</Badge>
                      {cancelado && evento.motivo_cancelamento && (
                        <span className="mt-1 block max-w-xs text-xs text-slate-500">
                          Motivo: {evento.motivo_cancelamento}
                        </span>
                      )}
                    </td>
                    <td className="hidden max-w-sm px-4 py-3 text-slate-700 md:table-cell">
                      {finalidade ?? <span className="text-slate-400">—</span>}
                    </td>
                    <td className="hidden px-4 py-3 font-medium text-slate-900 md:table-cell">
                      {evento.local}
                        {evento.cidade && (
                          <span className="block text-xs font-normal text-slate-500">{evento.cidade}</span>
                        )}
                    </td>
                    {mostrarAcoes && (
                      <td className="px-4 py-3">
                        <div className="flex justify-end gap-1">
                          {podeEditar && podeAlterar && (
                            <BotaoIcone
                              icone={Pencil}
                              rotulo={`Editar o evento de ${formatarDataHora(evento.data_evento)}`}
                              onClick={() => onEditar(evento)}
                            />
                          )}
                          {podeExcluir && podeAlterar && (
                            <BotaoIcone
                              icone={Trash2}
                              variante="perigo"
                              rotulo={`Excluir o evento de ${formatarDataHora(evento.data_evento)}`}
                              onClick={() => onExcluir(evento)}
                            />
                          )}
                          {podeCancelar && !cancelado && (
                            <BotaoIcone
                              icone={Ban}
                              variante="perigo"
                              rotulo={`Cancelar o evento de ${formatarDataHora(evento.data_evento)}`}
                              onClick={() => onCancelar(evento)}
                            />
                          )}
                        </div>
                      </td>
                    )}
                  </tr>
                );
              })
            )}
          </tbody>
        </table>
      </div>

      <div className="border-t border-slate-200 bg-slate-50 px-4 py-3 text-xs text-slate-500">
        Exibindo {filtrados.length} de {eventos.length} evento(s)
      </div>
    </div>
  );
}
