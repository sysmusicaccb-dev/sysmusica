 * SysMúsica — Configuração do menu e dos breadcrumbs (fonte única).
 *
 * Para incluir uma tela no menu, basta adicionar um item aqui: o menu lateral e os
 * breadcrumbs são gerados a partir desta lista. Só devem entrar telas que EXISTEM.
 * Cada item informa a permissão necessária: quem não a tem não vê o item no menu
 * (e a própria tela também é protegida por RequerPermissao).
 */

export interface ItemMenu {

  permissao: CodigoPermissao; // quem não tem esta permissão não vê o item

export interface GrupoMenu {

/** O usuário pode ver o item? (a permissão principal ou qualquer uma das alternativas) */
function itemPermitido(item: ItemMenu, pode: (permissao: CodigoPermissao) => boolean): boolean {
  return pode(item.permissao) || (item.qualquer ?? []).some((permissao) => pode(permissao));
}

/** Filtra o menu pelas permissões do usuário, removendo também os grupos que ficarem vazios. */
export function filtrarMenu(
  grupos: GrupoMenu[],
  pode: (permissao: CodigoPermissao) => boolean
): GrupoMenu[] {
  return grupos
    .map((grupo) => ({ ...grupo, itens: grupo.itens.filter((item) => itemPermitido(item, pode)) }))
    .filter((grupo) => grupo.itens.length > 0);
}

/** Primeira tela (fora do dashboard) que o usuário pode acessar; null se não puder nenhuma. */
export function primeiraRotaPermitida(pode: (permissao: CodigoPermissao) => boolean): string | null {
  for (const grupo of GRUPOS_MENU) {
    for (const item of grupo.itens) {
      if (item.href !== "/" && itemPermitido(item, pode)) return item.href;
    }
  }
  return null;
}

/** O item está ativo na rota atual? A raiz "/" só é ativa exatamente em "/". */
export function itemAtivo(href: string, caminho: string): boolean {
  if (href === "/") return caminho === "/";
  return caminho === href || caminho.startsWith(`${href}/`);
}

export interface Migalha {
  rotulo: string;
  href?: string; // sem href = apenas texto (grupo ou página atual)
}

/** Monta o caminho exibido nos breadcrumbs: Início > Grupo > Página. */
export function montarMigalhas(caminho: string): Migalha[] {
  for (const grupo of GRUPOS_MENU) {
    for (const item of grupo.itens) {
      if (!itemAtivo(item.href, caminho)) continue;

      // Na raiz, o próprio Dashboard é o único nível
      if (item.href === "/") return [{ rotulo: item.rotulo }];

      const migalhas: Migalha[] = [{ rotulo: "Início", href: "/" }];
      if (grupo.titulo) migalhas.push({ rotulo: grupo.titulo });
      migalhas.push({ rotulo: item.rotulo });
      return migalhas;
    }
  }
  // Rota fora do menu: mostra só o caminho de volta ao início
  return [{ rotulo: "Início", href: "/" }];
}

/**
 * Remove dos breadcrumbs o link "Início" para quem não acessa o dashboard (ex: grupo Portaria):
 * para essa pessoa o menu mostra somente o que o grupo permite, e nada além disso.
 */
export function migalhasVisiveis(
  migalhas: Migalha[],
  pode: (permissao: CodigoPermissao) => boolean
): Migalha[] {
  return migalhas.filter((migalha) => migalha.href !== "/" || pode("dashboard.visualizar"));
}

  return (permissao: CodigoPermissao) => permitidas.includes(permissao);

describe("itemAtivo", () => {
  it("a raiz só fica ativa exatamente em '/'", () => {
    expect(itemAtivo("/", "/")).toBe(true);
    expect(itemAtivo("/", "/membros")).toBe(false);
  });

  it("um item fica ativo na própria rota e nas subrotas", () => {
    expect(itemAtivo("/membros", "/membros")).toBe(true);
    expect(itemAtivo("/membros", "/membros/123")).toBe(true);
  });

  it("não confunde rotas com o mesmo prefixo", () => {
    expect(itemAtivo("/membros", "/membrosX")).toBe(false);
  });
});

describe("montarMigalhas", () => {
  it("mostra só 'Dashboard' na raiz", () => {
    expect(montarMigalhas("/")).toEqual([{ rotulo: "Dashboard" }]);

  it("monta Início > Grupo > Página", () => {
    expect(montarMigalhas("/membros")).toEqual([
      { rotulo: "Início", href: "/" },
      { rotulo: "Cadastros" },
      { rotulo: "Músicos / Organistas" },

    expect(montarMigalhas("/portaria")).toEqual([
      { rotulo: "Início", href: "/" },
      { rotulo: "Operações" },
      { rotulo: "Portaria" },
    ]);
  });

  it("subrotas usam os mesmos breadcrumbs da página", () => {
    expect(montarMigalhas("/membros/123")).toEqual(montarMigalhas("/membros"));
  });

  it("rota desconhecida volta ao início", () => {
    expect(montarMigalhas("/inexistente")).toEqual([{ rotulo: "Início", href: "/" }]);
  });
});

describe("filtrarMenu", () => {
  it("o grupo Portaria enxerga apenas o item Portaria", () => {
    const menu = filtrarMenu(GRUPOS_MENU, criarPode(["portaria.visualizar", "portaria.registrar"]));
    expect(menu.map((grupo) => grupo.titulo)).toEqual(["Operações"]);
    expect(menu[0].itens.map((item) => item.rotulo)).toEqual(["Portaria"]);

  it("remove os grupos que ficam vazios", () => {
    const menu = filtrarMenu(GRUPOS_MENU, criarPode(["dashboard.visualizar"]));
    expect(menu.map((grupo) => grupo.titulo)).toEqual([null]);

  it("sem permissões, o menu fica vazio", () => {
    expect(filtrarMenu(GRUPOS_MENU, criarPode([]))).toEqual([]);

"use client";

/**
 * SysMúsica — Abas simples para separar uma lista (ex.: Próximos / Encerrados / Arquivo), com contagem.
 * Só apresentação: quem usa decide o que cada aba mostra.
 */
export interface AbaDeLista<Id extends string> {
  id: Id;
  rotulo: string;
  total: number;
}

interface AbasDeListaProps<Id extends string> {
  abas: AbaDeLista<Id>[];
  atual: Id;
  onTrocar: (id: Id) => void;
  descricao?: string;
}

export default function AbasDeLista<Id extends string>({ abas, atual, onTrocar, descricao }: AbasDeListaProps<Id>) {
  return (
    <div role="tablist" aria-label={descricao} className="mb-4 flex gap-1 overflow-x-auto border-b border-slate-200">
      {abas.map((aba) => {
        const ativa = aba.id === atual;
        return (
          <button
            key={aba.id}
            type="button"
            role="tab"
            aria-selected={ativa}
            onClick={() => onTrocar(aba.id)}
            className={
              "-mb-px flex shrink-0 items-center gap-2 border-b-2 px-4 py-2.5 text-sm font-medium transition " +
              (ativa
                ? "border-brand-600 text-brand-700"
                : "border-transparent text-slate-500 hover:border-slate-300 hover:text-slate-700")
            }
          >
            {aba.rotulo}
            <span
              className={
                "rounded-full px-2 py-0.5 text-xs " + (ativa ? "bg-brand-50 text-brand-700" : "bg-slate-100 text-slate-600")
              }
            >
              {aba.total}
            </span>
          </button>
        );
      })}
    </div>
  );
}

"use client";

/**
 * SysMúsica — Estrutura do painel: menu lateral + topo + área de conteúdo.
 *
 * Guarda o estado do menu:
 *  - recolhida: no desktop, o menu fica só com ícones;
 *  - menuAberto: no celular/tablet, o menu vira uma gaveta sobre a tela.
 * Guarda também se a janela de Configurações (aparência) está aberta: ela fica aqui, fora da barra do
 * topo, e o botão que a abre fica na barra.
 * O fundo claro fica neste wrapper (e não no <body>) para a interface continuar legível
 * mesmo com o navegador em tema escuro.
 * As páginas internas trazem o próprio <main>, por isso a área de conteúdo é um <div>.
 */
import { useEffect, useState, type ReactNode } from "react";
import { cn } from "@/lib/cn";
import JanelaConfiguracoes from "@/modules/preferencias/components/JanelaConfiguracoes";
import Sidebar from "./Sidebar";
import Topbar from "./Topbar";

export default function AppShell({ children }: { children: ReactNode }) {
  const [recolhida, setRecolhida] = useState(false);
  const [menuAberto, setMenuAberto] = useState(false);
  const [configuracoesAbertas, setConfiguracoesAbertas] = useState(false);

  // A tecla Esc fecha a gaveta do menu (só enquanto ela está aberta)
  useEffect(() => {
    if (!menuAberto) return;
    function aoPressionar(evento: KeyboardEvent) {
      if (evento.key === "Escape") setMenuAberto(false);
    }
    window.addEventListener("keydown", aoPressionar);
    return () => window.removeEventListener("keydown", aoPressionar);
  }, [menuAberto]);

  return (
    <div className="min-h-screen bg-slate-50 text-slate-900">
      {/* Atalho de acessibilidade: aparece ao navegar com Tab e pula o menu */}
      <a
        href="#conteudo"
        className="sr-only focus:not-sr-only focus:fixed focus:left-4 focus:top-4 focus:z-[60] focus:rounded-lg focus:bg-white focus:px-4 focus:py-2 focus:text-sm focus:font-medium focus:text-brand-700 focus:shadow-lg"
      >
        Pular para o conteúdo
      </a>

      <Sidebar recolhida={recolhida} aberta={menuAberto} onFechar={() => setMenuAberto(false)} />

      {/* O espaço à esquerda acompanha a largura do menu fixo (só no desktop) */}
      <div
        className={cn(
          "flex min-h-screen flex-col transition-[padding] duration-200",
          recolhida ? "lg:pl-[4.5rem]" : "lg:pl-64"
        )}
      >
        <Topbar
          recolhida={recolhida}
          menuAberto={menuAberto}
          onAbrirMenu={() => setMenuAberto(true)}
          onAlternarRecolhimento={() => setRecolhida((atual) => !atual)}
          onAbrirConfiguracoes={() => setConfiguracoesAbertas(true)}
        />
        <div id="conteudo" className="flex-1">
          {children}
        </div>
      </div>

      <JanelaConfiguracoes aberto={configuracoesAbertas} onFechar={() => setConfiguracoesAbertas(false)} />
    </div>
  );
}

"use client";

/**
 * SysMúsica — Breadcrumbs (indicam onde o usuário está): Início > Grupo > Página.
 * Gerados pela rota atual a partir de ./navegacao.ts.
 */
import Link from "next/link";
import { usePathname } from "next/navigation";
import { ChevronRight } from "lucide-react";
import { cn } from "@/lib/cn";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";
import { migalhasVisiveis, montarMigalhas } from "./navegacao";

export default function Breadcrumbs() {
  const caminho = usePathname() ?? "";
  const { pode } = useSessao();
  // Quem não acessa o dashboard (ex: grupo Portaria) não vê o "Início" nos breadcrumbs
  const migalhas = migalhasVisiveis(montarMigalhas(caminho), pode);

  return (
    <nav aria-label="Você está em" className="min-w-0">
      <ol className="flex items-center gap-1.5 text-sm text-slate-500">
        {migalhas.map((migalha, indice) => {
          const ultima = indice === migalhas.length - 1;

          return (
            <li key={`${migalha.rotulo}-${indice}`} className="flex min-w-0 items-center gap-1.5">
              {indice > 0 && (

/**
 * SysMúsica — Menu do usuário (topo da tela): identificação, grupo de acesso e botão Sair.
 * Só tem ações que existem: por enquanto, encerrar a sessão.
 */
import { useEffect, useRef, useState } from "react";
import { ChevronDown, LogOut } from "lucide-react";
import Badge from "@/components/ui/Badge";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";

export default function UserMenu() {
  const { conta, sessao, sair } = useSessao();
  const [aberto, setAberto] = useState(false);
  const raizRef = useRef<HTMLDivElement>(null);

  const nome = conta?.nome ?? sessao?.user.email ?? "Usuário";
  const email = conta?.email ?? sessao?.user.email ?? "";

  // O menu fecha ao clicar fora ou apertar Esc (só enquanto está aberto)
  useEffect(() => {
    if (!aberto) return;

    function aoClicarFora(evento: MouseEvent) {
      if (raizRef.current && !raizRef.current.contains(evento.target as Node)) setAberto(false);
    }
    function aoPressionar(evento: KeyboardEvent) {
      if (evento.key === "Escape") setAberto(false);
    }

    document.addEventListener("mousedown", aoClicarFora);
    document.addEventListener("keydown", aoPressionar);
    return () => {
      document.removeEventListener("mousedown", aoClicarFora);
      document.removeEventListener("keydown", aoPressionar);
    };
  }, [aberto]);

  return (
    <div ref={raizRef} className="relative">
      <button
        type="button"
        onClick={() => setAberto((atual) => !atual)}
        aria-haspopup="menu"
        aria-expanded={aberto}
        className="flex items-center gap-2 rounded-lg px-2 py-1.5 text-left transition hover:bg-slate-100 focus:outline-none focus-visible:ring-2 focus-visible:ring-brand-300"
      >
        <span
          aria-hidden="true"
          className="flex h-8 w-8 items-center justify-center rounded-full bg-brand-100 text-sm font-bold text-brand-700"
        >
          {nome.charAt(0).toUpperCase()}
        </span>
        <span className="hidden min-w-0 sm:block">
          <span className="block max-w-40 truncate text-sm font-medium text-slate-900">{nome}</span>
          {conta?.grupo && (
            <span className="block max-w-40 truncate text-xs text-slate-500">{conta.grupo}</span>
          )}
        </span>
        <ChevronDown className="h-4 w-4 text-slate-500" aria-hidden="true" />
      </button>

      {aberto && (
        <div
          role="menu"
          className="absolute right-0 z-50 mt-2 w-64 rounded-xl border border-slate-200 bg-white p-2 shadow-lg"
        >
          <div className="px-3 py-2">
            <p className="truncate text-sm font-medium text-slate-900">{nome}</p>
            <p className="truncate text-xs text-slate-500">{email}</p>
            {conta?.grupo && (
              <div className="mt-2">
                <Badge variante="marca">{conta.grupo}</Badge>
              </div>
            )}
          </div>
          <div className="my-1 border-t border-slate-100" />
          <button
            type="button"
            role="menuitem"
            onClick={() => {
              setAberto(false);
              void sair();
            }}
            className="flex w-full items-center gap-2 rounded-lg px-3 py-2 text-sm text-slate-700 transition hover:bg-slate-100 hover:text-slate-900"
          >
            <LogOut className="h-4 w-4" aria-hidden="true" />
            Sair da conta
          </button>
        </div>
      )}
    </div>
  );
}
