"use client";

/**
 * SysMúsica — Página /administracao/organizacao: estados, regiões, setores e casas de oração.
 * Cadastro pelo sistema (sem código fixo), com ativar/inativar e transferência de casa de oração entre setores.
 * Cada botão só aparece para quem tem a permissão; o banco confere de novo em toda operação.
 */
import { useCallback, useEffect, useState } from "react";
import { Plus } from "lucide-react";
import { useSessao } from "@/modules/autenticacao/SessaoProvider";
import ArvoreOrganizacao from "@/modules/organizacao/components/ArvoreOrganizacao";
import FormularioNo, { type ModoNo } from "@/modules/organizacao/components/FormularioNo";
import FormularioTransferencia from "@/modules/organizacao/components/FormularioTransferencia";
import { motivoBloqueioExclusao, ROTULO_NIVEL } from "@/modules/organizacao/regras";
import { excluirNo, listarOrganizacao, salvarNo } from "@/modules/organizacao/services/organizacao";
import type { NivelOrganizacao, NoOrganizacao } from "@/modules/organizacao/tipos";
import Button from "@/components/ui/Button";
import ConfirmDialog from "@/components/ui/ConfirmDialog";
import Drawer from "@/components/ui/Drawer";

import ErrorState from "@/components/ui/ErrorState";
import Notificacao from "@/components/ui/Notificacao";
import PageHeader from "@/components/ui/PageHeader";
import Skeleton from "@/components/ui/Skeleton";

// Qual painel lateral está aberto
type Painel = { tipo: "no"; modo: ModoNo } | { tipo: "transferir"; casaOracao: NoOrganizacao };

// Título do painel lateral
function tituloDoPainel(painel: Painel | null): string {
  if (!painel) return "";
  if (painel.tipo === "transferir") return "Transferir casa de oração";
  const modo = painel.modo;
  return modo.tipo === "editar"
    ? `Editar ${ROTULO_NIVEL[modo.no.nivel].toLowerCase()}`
    : `Novo(a) ${ROTULO_NIVEL[modo.nivel].toLowerCase()}`;
}

export default function OrganizacaoPage() {
  const { pode } = useSessao();
  const podeCriar = pode("organizacao.criar");
  const podeEditar = pode("organizacao.editar");
  const podeTransferir = pode("organizacao.transferir");
  const podeExcluir = pode("organizacao.excluir");

  const [nos, setNos] = useState<NoOrganizacao[]>([]);
  const [carregando, setCarregando] = useState(true); // inicia true: primeira busca no mount
  const [erro, setErro] = useState<string | null>(null);
  const [painel, setPainel] = useState<Painel | null>(null);
  const [alvoAlternar, setAlvoAlternar] = useState<NoOrganizacao | null>(null);
  const [alvoExcluir, setAlvoExcluir] = useState<NoOrganizacao | null>(null);
  const [erroAcao, setErroAcao] = useState<string | null>(null);
  const [processando, setProcessando] = useState(false);
  const [erroConfirmacao, setErroConfirmacao] = useState<string | null>(null);
  const [aviso, setAviso] = useState<string | null>(null);

  const carregar = useCallback(async () => {
    try {
      setNos(await listarOrganizacao());
      setErro(null);
    } catch (e) {
      console.error("Falha ao carregar a organização:", e);
      setErro(e instanceof Error ? e.message : "Não foi possível carregar a organização.");
    } finally {
      setCarregando(false);
    }
  }, []);

  // Primeira carga ao abrir a página
  useEffect(() => {
    // O setState só roda DEPOIS do await (nunca de forma síncrona). Ignorado apenas nesta linha.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    void carregar();
  }, [carregar]);

  function tentarNovamente() {
    setCarregando(true);
    void carregar();
  }

  // Depois de salvar: recarrega, fecha o painel e mostra o aviso
  async function aoSalvar(mensagem: string) {
    await carregar();
    setPainel(null);
    setAviso(mensagem);
  }

  // Ativa ou inativa o item (reenvia os dados atuais com a situação trocada)
  async function confirmarAlternar() {
    if (!alvoAlternar) return;
    setProcessando(true);
    setErroConfirmacao(null);
    try {
      await salvarNo({
        nivel: alvoAlternar.nivel,
        id: alvoAlternar.id,
        paiId: alvoAlternar.parent_id,
        nome: alvoAlternar.nome,
        descricao: alvoAlternar.descricao ?? "",
        ativo: alvoAlternar.ativo === false,
        cidade: alvoAlternar.cidade ?? "",
        bairro: alvoAlternar.bairro ?? "",
      });
      await carregar();
      setAviso(`"${alvoAlternar.nome}" ${alvoAlternar.ativo === false ? "ativado(a)" : "inativado(a)"}.`);
      setAlvoAlternar(null);
    } catch (e) {
      setErroConfirmacao(e instanceof Error ? e.message : "Não foi possível alterar a situação.");
    } finally {
      setProcessando(false);
    }
  }

  // Pede a exclusão: se o item ainda tiver algo dentro, explica o motivo; senão abre a confirmação
  function pedirExclusao(no: NoOrganizacao) {
    const motivo = motivoBloqueioExclusao(no);
    if (motivo) {
      setErroAcao(motivo);
      return;
    }
    setErroAcao(null);
    setErroConfirmacao(null);
    setAlvoExcluir(no);
  }

  // Exclui de vez o item (o banco confere de novo e informa se houver eventos, portarias ou usuários ligados)
  async function confirmarExclusao() {
    if (!alvoExcluir) return;
    setProcessando(true);
    setErroConfirmacao(null);
    try {
      await excluirNo(alvoExcluir.nivel, alvoExcluir.id);
      await carregar();
      setAviso(`"${alvoExcluir.nome}" excluído(a).`);
      setAlvoExcluir(null);
    } catch (e) {
      setErroConfirmacao(e instanceof Error ? e.message : "Não foi possível excluir.");
    } finally {
      setProcessando(false);
    }
  }

  function abrirNovo(nivel: NivelOrganizacao, pai: NoOrganizacao | null) {
    setPainel({ tipo: "no", modo: { tipo: "novo", nivel, pai } });
  }

  return (
    <main className="text-slate-900">
      <div className="mx-auto max-w-5xl px-4 py-8 sm:px-6 lg:px-8">
        <PageHeader
          titulo="Organização"
          descricao="Estados, regiões, setores e casas de oração. O acesso dos usuários aos dados segue esta estrutura."
          acoes={
            podeCriar ? (
              <Button icone={Plus} onClick={() => abrirNovo("ESTADO", null)} disabled={carregando || !!erro}>
                Adicionar
              </Button>
            ) : undefined
          }
        />

        {erroAcao && (
          <p role="alert" className="mb-4 rounded-lg bg-red-50 px-3 py-2 text-sm text-red-700">
            {erroAcao}
          </p>
        )}

        {carregando ? (
          <div role="status" aria-live="polite" className="space-y-3">
            <span className="sr-only">Carregando a organização...</span>
            {[0, 1, 2, 3, 4].map((i) => (
              <Skeleton key={i} className="h-12" />
            ))}
          </div>
        ) : erro ? (
          <ErrorState titulo="Não foi possível carregar a organização." mensagem={erro} onTentarNovamente={tentarNovamente} />
        ) : nos.length === 0 ? (
          <EmptyState
            titulo="Nenhum item cadastrado."
            descricao={podeCriar ? "Clique em “Adicionar” para começar." : "A estrutura ainda não foi cadastrada."}
          />
        ) : (
          <ArvoreOrganizacao
            nos={nos}
            podeCriar={podeCriar}
            podeEditar={podeEditar}
            podeTransferir={podeTransferir}
            podeExcluir={podeExcluir}
            onExcluir={pedirExclusao}
            onNovo={abrirNovo}
            onEditar={(no) => setPainel({ tipo: "no", modo: { tipo: "editar", no } })}
            onAlternar={(no) => {
              setErroConfirmacao(null);
              setAlvoAlternar(no);
            }}
            onTransferir={(casaOracao) => setPainel({ tipo: "transferir", casaOracao })}
          />
        )}
      </div>

      <Drawer
        aberto={painel !== null}
        onFechar={() => setPainel(null)}
        titulo={tituloDoPainel(painel)}
        descricao={
          painel?.tipo === "transferir"
            ? "A casa de oração e as portarias dela passam a valer para o setor escolhido."
            : "Os nomes não podem se repetir dentro do mesmo item pai."
        }
      >
        {painel?.tipo === "no" && <FormularioNo modo={painel.modo} onSalvo={aoSalvar} />}
        {painel?.tipo === "transferir" && (
          <FormularioTransferencia casaOracao={painel.casaOracao} nos={nos} onSalvo={aoSalvar} />
        )}
      </Drawer>

      <ConfirmDialog
        aberto={alvoAlternar !== null}
        titulo={alvoAlternar?.ativo === false ? "Ativar item?" : "Inativar item?"}
        mensagem={
          <>
            <p>
              {alvoAlternar?.ativo === false ? (
                <>
                  <strong>{alvoAlternar?.nome}</strong> volta a aparecer nas escolhas e a valer para o acesso.
                </>
              ) : (
                <>
                  <strong>{alvoAlternar?.nome}</strong> deixa de aparecer nas escolhas e quem tem acesso por esse
                  item (e pelos itens abaixo dele) perde o acesso aos dados dele. Quem tem acesso total não é
                  afetado.
                </>
              )}
            </p>
            {erroConfirmacao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroConfirmacao}
              </p>
            )}
          </>
        }
        textoConfirmar={alvoAlternar?.ativo === false ? "Ativar" : "Inativar"}
        processando={processando}
        onConfirmar={() => void confirmarAlternar()}
        onCancelar={() => setAlvoAlternar(null)}
      />

      <ConfirmDialog
        aberto={alvoExcluir !== null}
        titulo="Excluir item?"
        mensagem={
          <>
            <p>
              <strong>{alvoExcluir?.nome}</strong> será excluído(a) de vez. Esta ação não pode ser desfeita. Se
              preferir só esconder, use Inativar.
            </p>
            {erroConfirmacao && (
              <p role="alert" className="mt-3 rounded-lg bg-red-50 px-3 py-2 text-red-700">
                {erroConfirmacao}
              </p>
            )}
          </>
        }
        textoConfirmar="Excluir"
        processando={processando}
        onConfirmar={() => void confirmarExclusao()}
        onCancelar={() => setAlvoExcluir(null)}
      />

      <Notificacao mensagem={aviso} onFechar={() => setAviso(null)} />
    </main>
  );
}

  onNovo: (nivel: NivelOrganizacao, pai: NoOrganizacao | null) => void;
  onEditar: (no: NoOrganizacao) => void;
  onAlternar: (no: NoOrganizacao) => void;
  onTransferir: (no: NoOrganizacao) => void;
  onExcluir: (no: NoOrganizacao) => void;
}

// Texto de contagem de cada item (ex.: "5 setores")
function contagem(no: NoOrganizacao): string {
  const n = no.qtd_filhos ?? 0;
  if (no.nivel === "ESTADO") return `${n} ${n === 1 ? "região" : "regiões"}`;
  if (no.nivel === "REGIAO") return `${n} ${n === 1 ? "setor" : "setores"}`;
  if (no.nivel === "SETOR") return `${n} ${n === 1 ? "casa de oração" : "casas de oração"}`;
  const m = no.qtd_membros ?? 0;
  return `${m} ${m === 1 ? "membro" : "membros"}`;
}

export default function ArvoreOrganizacao({
  nos,
  podeCriar,
  podeEditar,
  podeTransferir,
  podeExcluir,
  onNovo,
  onEditar,
  onAlternar,
  onTransferir,
  onExcluir,
}: ArvoreOrganizacaoProps) {
  // Começa com estados e regiões abertos
  const [abertos, setAbertos] = useState<Set<string>>(
    () => new Set(nos.filter((no) => no.nivel === "ESTADO" || no.nivel === "REGIAO").map((no) => no.id))
  );

  function alternarAberto(id: string) {
    setAbertos((anterior) => {
      const novos = new Set(anterior);
      if (novos.has(id)) novos.delete(id);
      else novos.add(id);
      return novos;
    });
  }

  // Linha de um item e, se estiver aberto, os filhos dele (função comum e não componente: é recursiva)
  function renderLinha(no: NoOrganizacao, profundidade: number): ReactNode {
    const nivelFilho = FILHO_DE[no.nivel];
    const filhos = nivelFilho ? filhosDe(nos, no.id, nivelFilho) : [];
    const aberto = abertos.has(no.id);
    const inativo = no.ativo === false;

    return (
      <li key={`${no.nivel}:${no.id}`}>
        <div
          className="flex items-center gap-2 rounded-lg py-1.5 pr-2 hover:bg-slate-50"
          style={{ paddingLeft: `${profundidade * 1.25 + 0.25}rem` }}
        >
          {filhos.length > 0 ? (
            <button
              type="button"
              onClick={() => alternarAberto(no.id)}
              aria-expanded={aberto}
              aria-label={`${aberto ? "Fechar" : "Abrir"} ${no.nome}`}
              className="rounded p-0.5 text-slate-500 hover:bg-slate-100"
            >
              {aberto ? <ChevronDown className="h-4 w-4" aria-hidden="true" /> : <ChevronRight className="h-4 w-4" aria-hidden="true" />}
            </button>
          ) : (
            <span className="inline-block w-5" aria-hidden="true" />
          )}

          <span className="min-w-0 flex-1">
            <span className="mr-2 text-xs font-semibold uppercase tracking-wide text-slate-400">
              {ROTULO_NIVEL[no.nivel]}
            </span>
            <span className={inativo ? "font-medium text-slate-400" : "font-medium text-slate-900"}>{no.nome}</span>
            <span className="ml-2 text-xs text-slate-500">
              {contagem(no)}
              {no.nivel === "CASA_ORACAO" && (no.bairro || no.cidade)
                ? ` · ${[no.bairro, no.cidade].filter(Boolean).join(", ")}`
                : ""}
            </span>
          </span>

          <Badge variante={inativo ? "perigo" : "sucesso"}>{inativo ? "Inativo" : "Ativo"}</Badge>

          <div className="flex shrink-0 gap-1">
            {podeCriar && nivelFilho && (
              <BotaoIcone
                icone={Plus}
                rotulo={`Adicionar ${ROTULO_NIVEL[nivelFilho].toLowerCase()} em ${no.nome}`}
                onClick={() => onNovo(nivelFilho, no)}
              />
            )}
            {podeTransferir && no.nivel === "CASA_ORACAO" && (
              <BotaoIcone icone={ArrowRightLeft} rotulo={`Transferir ${no.nome} para outro setor`} onClick={() => onTransferir(no)} />
            )}
            {podeEditar && (
              <>
                <BotaoIcone icone={Pencil} rotulo={`Editar ${no.nome}`} onClick={() => onEditar(no)} />
                <BotaoIcone
                  icone={Power}
                  rotulo={`${inativo ? "Ativar" : "Inativar"} ${no.nome}`}
                  onClick={() => onAlternar(no)}
                />
              </>
            )}
            {podeExcluir && (
              <>
                <BotaoIcone
                  icone={Trash2}
                  variante="perigo"
                  rotulo={`Excluir ${no.nome}`}
                  onClick={() => onExcluir(no)}
                />
              </>
            )}
          </div>
        </div>

        {aberto && filhos.length > 0 && <ul>{filhos.map((filho) => renderLinha(filho, profundidade + 1))}</ul>}
      </li>
    );
  }

  const estados = filhosDe(nos, null, "ESTADO");
  const semSetor = filhosDe(nos, null, "CASA_ORACAO");

  return (
    <div className="space-y-6">
      <div className="rounded-2xl border border-slate-200 bg-white p-3 shadow-sm">
        {estados.length === 0 ? (
          <p className="px-2 py-6 text-center text-sm text-slate-500">Nenhum estado cadastrado.</p>
        ) : (
          <ul>{estados.map((estado) => renderLinha(estado, 0))}</ul>
        )}
      </div>

      {semSetor.length > 0 && (
        <section className="rounded-2xl border border-amber-200 bg-amber-50/50 p-3 shadow-sm">
          <h2 className="px-2 pt-1 text-sm font-semibold text-slate-900">Casas de Oração sem setor ({semSetor.length})</h2>
          <p className="px-2 pb-2 text-xs text-slate-600">
            Enquanto uma casa de oração estiver sem setor, quem tem acesso restrito não vê nada dela (membros, eventos e
            portarias). Use o botão de transferir para vinculá-la a um setor.
          </p>
          <ul>{semSetor.map((casaOracao) => renderLinha(casaOracao, 0))}</ul>
        </section>
      )}
    </div>
  );
}

/**
 * SysMúsica — Regras puras da organização (sem banco). Testadas em regras.test.ts.
 * As regras de verdade (permissão, nomes repetidos, abrangência) ficam no banco; aqui há
 * montagem de listas para a tela e conferências rápidas.
 */
import type { NivelOrganizacao, NoOrganizacao, VinculoRotulado, VinculoUsuario } from "./tipos";

export const ROTULO_NIVEL: Record<NivelOrganizacao, string> = {
  ESTADO: "Estado",
  REGIAO: "Região",
  SETOR: "Setor",
  CASA_ORACAO: "Casa de Oração",
};

/** Nível que fica logo abaixo de cada nível. */
export const FILHO_DE: Record<NivelOrganizacao, NivelOrganizacao | null> = {
  ESTADO: "REGIAO",
  REGIAO: "SETOR",
  SETOR: "CASA_ORACAO",
  CASA_ORACAO: null,
};

const NIVEIS: NivelOrganizacao[] = ["ESTADO", "REGIAO", "SETOR", "CASA_ORACAO"];

// Ordem natural em português ("Setor 2" antes de "Setor 10")
const porNome = (a: NoOrganizacao, b: NoOrganizacao) => a.nome.localeCompare(b.nome, "pt-BR", { numeric: true });

/** Itens de um nível que pertencem a um pai (pai nulo = os do topo / sem pai), em ordem de nome. */
export function filhosDe(nos: NoOrganizacao[], pai: string | null, nivel: NivelOrganizacao): NoOrganizacao[] {
  return nos.filter((no) => no.nivel === nivel && no.parent_id === pai).sort(porNome);
}

/**
 * Se o item ainda tem algo dentro (itens filhos ou membros), explica por que não pode ser excluído;
 * senão devolve null. É só uma resposta rápida da tela: o banco confere tudo de novo, inclusive eventos,
 * portarias e usuários vinculados.
 */
export function motivoBloqueioExclusao(no: NoOrganizacao): string | null {
  const filhos = no.qtd_filhos ?? 0;
  const membros = no.qtd_membros ?? 0;

  if (no.nivel === "ESTADO" && filhos > 0) {
    return `"${no.nome}" ainda tem ${filhos} ${filhos === 1 ? "região" : "regiões"}. Exclua as regiões antes.`;
  }
  if (no.nivel === "REGIAO" && filhos > 0) {
    return `"${no.nome}" ainda tem ${filhos} ${filhos === 1 ? "setor" : "setores"}. Exclua os setores antes.`;
  }
  if (no.nivel === "SETOR" && filhos > 0) {
    return `"${no.nome}" ainda tem ${filhos} ${filhos === 1 ? "casa de oração" : "casas de oração"}. Transfira as casas de oração para outro setor antes.`;
  }
  if (no.nivel === "CASA_ORACAO" && membros > 0) {
    return `"${no.nome}" ainda tem ${membros} ${membros === 1 ? "membro" : "membros"}. Mude os membros para outra casa de oração (ou exclua-os) antes.`;
  }
  return null;
}

/** Valida o nome de um item da organização; devolve a mensagem de erro ou null. */
export function validarNomeNo(nome: string): string | null {
  const tamanho = nome.trim().replace(/\s+/g, " ").length;
  return tamanho < 2 || tamanho > 80 ? "O nome deve ter entre 2 e 80 caracteres." : null;
}

export interface GrupoCasasOracao {
  rotulo: string; // ex.: "Salvador › Setor 1"
  casasOracao: NoOrganizacao[];
}

/** Agrupa as casas de oração por "Região › Setor" (com o estado na frente se houver mais de um). Casas de Oração sem setor vão para o fim. */
export function agruparCasasOracaoParaSelecao(nos: NoOrganizacao[]): GrupoCasasOracao[] {
  const porChave = new Map(nos.map((no) => [`${no.nivel}:${no.id}`, no] as const));
  const estados = nos.filter((no) => no.nivel === "ESTADO");
  const grupos = new Map<string, NoOrganizacao[]>();
  const semSetor: NoOrganizacao[] = [];

  for (const casaOracao of nos.filter((no) => no.nivel === "CASA_ORACAO")) {
    const setor = casaOracao.parent_id ? porChave.get(`SETOR:${casaOracao.parent_id}`) : undefined;
    const regiao = setor?.parent_id ? porChave.get(`REGIAO:${setor.parent_id}`) : undefined;
    if (!setor || !regiao) {
      semSetor.push(casaOracao);
      continue;
    }
    const estado = regiao.parent_id ? porChave.get(`ESTADO:${regiao.parent_id}`) : undefined;
    const prefixo = estados.length > 1 && estado ? `${estado.nome} › ` : "";
    const rotulo = `${prefixo}${regiao.nome} › ${setor.nome}`;
    grupos.set(rotulo, [...(grupos.get(rotulo) ?? []), casaOracao]);
  }

  const lista: GrupoCasasOracao[] = [...grupos.entries()]
    .sort(([a], [b]) => a.localeCompare(b, "pt-BR", { numeric: true }))
    .map(([rotulo, casasOracao]) => ({ rotulo, casasOracao: [...casasOracao].sort(porNome) }));
  if (semSetor.length > 0) lista.push({ rotulo: "Sem setor definido", casasOracao: [...semSetor].sort(porNome) });
  return lista;
}

export interface GrupoSetores {
  rotulo: string; // ex.: "Salvador" (ou "Bahia › Salvador")
  setores: NoOrganizacao[];
}

/** Agrupa os setores ATIVOS por região (para escolher o destino de uma transferência). */
export function agruparSetoresParaSelecao(nos: NoOrganizacao[]): GrupoSetores[] {
  const porChave = new Map(nos.map((no) => [`${no.nivel}:${no.id}`, no] as const));
  const estados = nos.filter((no) => no.nivel === "ESTADO");
  const grupos = new Map<string, NoOrganizacao[]>();

  for (const setor of nos.filter((no) => no.nivel === "SETOR" && no.ativo !== false)) {
    const regiao = setor.parent_id ? porChave.get(`REGIAO:${setor.parent_id}`) : undefined;
    if (!regiao || regiao.ativo === false) continue;
    const estado = regiao.parent_id ? porChave.get(`ESTADO:${regiao.parent_id}`) : undefined;
    const rotulo = `${estados.length > 1 && estado ? `${estado.nome} › ` : ""}${regiao.nome}`;
    grupos.set(rotulo, [...(grupos.get(rotulo) ?? []), setor]);
  }

  return [...grupos.entries()]
    .sort(([a], [b]) => a.localeCompare(b, "pt-BR", { numeric: true }))
    .map(([rotulo, setores]) => ({ rotulo, setores: [...setores].sort(porNome) }));
}

/** Chave única de um vínculo, para guardar em conjuntos (ex.: "SETOR:abc"). */
export function chaveVinculo(nivel: NivelOrganizacao, id: string): string {
  return `${nivel}:${id}`;
}

/** Desfaz a chave de um vínculo; devolve nulo se estiver malformada. */
export function lerChaveVinculo(chave: string): VinculoUsuario | null {
  const posicao = chave.indexOf(":");
  if (posicao < 0) return null;
  const nivel = chave.slice(0, posicao);
  const alvo = chave.slice(posicao + 1);
  if (!NIVEIS.includes(nivel as NivelOrganizacao) || alvo === "") return null;
  return { nivel: nivel as NivelOrganizacao, alvo_id: alvo };
}

/** De { usuario: [vínculos com rótulo] } para { usuario: ["rótulo", ...] } (coluna da tabela de usuários). */
export function rotulosPorUsuario(mapa: Record<string, VinculoRotulado[]>): Record<string, string[]> {
  const resultado: Record<string, string[]> = {};
  for (const [usuarioId, vinculos] of Object.entries(mapa)) {
    resultado[usuarioId] = vinculos.map((vinculo) => vinculo.rotulo);
  }
  return resultado;
}

-------- definir_abrangencia  (0007_portarias.sql)
create or replace function public.definir_abrangencia(p_usuario_id uuid, p_cidades text[])
returns void language plpgsql security definer set search_path = public as $$
declare
  v_cidade text;
  v_limpa  text;
  v_vistas text[] := '{}';
begin
  perform public.exigir_permissao('usuarios.editar');
  if not exists (select 1 from public.usuarios u where u.id = p_usuario_id) then
    raise exception 'Usuário não encontrado.';
  end if;

  delete from public.usuario_cidades uc where uc.usuario_id = p_usuario_id;

  foreach v_cidade in array coalesce(p_cidades, '{}'::text[]) loop
    v_limpa := btrim(regexp_replace(coalesce(v_cidade, ''), '\s+', ' ', 'g'));
    if v_limpa = '' then continue; end if;
    if char_length(v_limpa) > 80 then
      raise exception 'O nome da cidade pode ter no máximo 80 caracteres.';
    end if;
    -- ignora repetidas (sem acento e sem diferenciar maiúsculas)
    if public.normalizar_texto(v_limpa) = any (v_vistas) then continue; end if;
    v_vistas := v_vistas || public.normalizar_texto(v_limpa);
    insert into public.usuario_cidades (usuario_id, cidade) values (p_usuario_id, v_limpa);
  end loop;

  perform public.registrar_auditoria('abrangencia_definida', 'usuario', p_usuario_id,
            jsonb_build_object('cidades', to_jsonb(coalesce(p_cidades, '{}'::text[]))));
end;
$$;

-------- definir_vinculos  (0031_casa_de_oracao.sql)
CREATE OR REPLACE FUNCTION public.definir_vinculos(p_usuario uuid, p_vinculos jsonb)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_item  jsonb;
  v_nivel text;
  v_alvo  uuid;
  v_lista jsonb := coalesce(p_vinculos, '[]'::jsonb);
  v_ok    boolean;
begin
  perform public.exigir_permissao('usuarios.vinculos');

  if not exists (select 1 from public.usuarios u where u.id = p_usuario) then
    raise exception 'Usuário não encontrado.';
  end if;
  if jsonb_typeof(v_lista) <> 'array' then
    raise exception 'Vínculos inválidos.';
  end if;

  delete from public.usuario_vinculos where usuario_id = p_usuario;

  for v_item in select * from jsonb_array_elements(v_lista) loop
    v_nivel := v_item ->> 'nivel';
    v_alvo  := (v_item ->> 'alvo_id')::uuid;
    v_ok := case v_nivel
      when 'ESTADO' then exists (select 1 from public.estados x where x.id = v_alvo)
      when 'REGIAO' then exists (select 1 from public.regioes x where x.id = v_alvo)
      when 'SETOR'  then exists (select 1 from public.setores x where x.id = v_alvo)
      when 'CASA_ORACAO'  then exists (select 1 from public.casas_oracao x where x.id = v_alvo)
      else false
    end;
    if not v_ok then
      raise exception 'Vínculo inválido: o cadastro escolhido não existe mais.';
    end if;
    insert into public.usuario_vinculos (usuario_id, nivel, alvo_id) values (p_usuario, v_nivel, v_alvo)
    on conflict do nothing;
  end loop;

  perform public.registrar_auditoria('vinculos_definidos', 'usuario', p_usuario,
            jsonb_build_object('vinculos', v_lista));
end;
$function$;

CREATE OR REPLACE FUNCTION public.eventos_valida_casa_oracao()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  if current_setting('sysmusica.org_excluindo', true) = 'sim' then
    return new;
  end if;

  if (select auth.uid()) is null then
    return new;   -- SQL Editor / rotinas internas: sem restrição
  end if;

  -- Quem tem acesso por região precisa escolher a casa de oração (senão o evento sumiria da própria lista)
  if new.casa_oracao_id is null and not public.tem_acesso_total_organizacao()
     and (tg_op = 'INSERT' or old.casa_oracao_id is not null) then
    raise exception 'Escolha a casa de oração do evento.';
  end if;

  if tg_op = 'INSERT' then
    if new.casa_oracao_id is not null and not public.casa_oracao_na_abrangencia(new.casa_oracao_id) then
      raise exception 'Esta casa de oração está fora da sua abrangência.' using errcode = '42501';
    end if;
  elsif new.casa_oracao_id is distinct from old.casa_oracao_id then
    if (new.casa_oracao_id is not null and not public.casa_oracao_na_abrangencia(new.casa_oracao_id))
       or (old.casa_oracao_id is not null and not public.casa_oracao_na_abrangencia(old.casa_oracao_id)) then
      raise exception 'Você não pode mudar a casa de oração deste evento: ela está fora da sua abrangência.'
        using errcode = '42501';
    end if;
    update public.portarias
       set casa_oracao_id = new.casa_oracao_id, cidade = public.rotulo_setor(new.casa_oracao_id)
     where evento_id = new.id and excluido_em is null;
    perform public.registrar_auditoria('evento_casa_oracao_alterada', 'evento', new.id,
              jsonb_build_object('de', old.casa_oracao_id, 'para', new.casa_oracao_id));
  end if;
  return new;

$function$;

begin
  perform public.exigir_permissao('organizacao.excluir');

  if p_nivel is null or p_nivel not in ('ESTADO', 'REGIAO', 'SETOR', 'CASA_ORACAO') then
    raise exception 'Nível inválido.';
  end if;

  -- Trava o item e lê o nome (para as mensagens e a auditoria)
  if p_nivel = 'ESTADO' then
    select e.nome into v_nome from public.estados e where e.id = p_id for update;
  elsif p_nivel = 'REGIAO' then
    select r.nome into v_nome from public.regioes r where r.id = p_id for update;
  elsif p_nivel = 'SETOR' then
    select s.nome into v_nome from public.setores s where s.id = p_id for update;
  else
    select c.nome into v_nome from public.casas_oracao c where c.id = p_id for update;
  end if;
  if v_nome is null then
    raise exception 'Cadastro não encontrado.';
  end if;

  -- Usuários vinculados a este item perderiam o acesso sem aviso: a exclusão é bloqueada
  select count(*) into v_qtd from public.usuario_vinculos v where v.nivel = p_nivel and v.alvo_id = p_id;
  if v_qtd > 0 then
    raise exception '"%" ainda tem % usuário(s) vinculado(s). Retire o vínculo em Administração > Usuários antes de excluir.',
      v_nome, v_qtd;
  end if;

  if p_nivel = 'ESTADO' then
    select count(*) into v_qtd from public.regioes x where x.estado_id = p_id;
    if v_qtd > 0 then
      raise exception 'O estado "%" ainda tem % região(ões). Exclua as regiões antes.', v_nome, v_qtd;
    end if;
    delete from public.estados where id = p_id;

  elsif p_nivel = 'REGIAO' then
    select count(*) into v_qtd from public.setores x where x.regiao_id = p_id;
    if v_qtd > 0 then
      raise exception 'A região "%" ainda tem % setor(es). Exclua os setores antes.', v_nome, v_qtd;
    end if;
    delete from public.regioes where id = p_id;

  elsif p_nivel = 'SETOR' then
    select count(*) into v_qtd from public.casas_oracao x where x.setor_id = p_id;
    if v_qtd > 0 then
      raise exception 'O setor "%" ainda tem % casa(s) de oração. Transfira as casas de oração para outro setor antes de excluir.', v_nome, v_qtd;
    end if;
    delete from public.setores where id = p_id;

  else  -- CASA_ORACAO
    select count(*) into v_qtd from public.membros x where x.casa_oracao_id = p_id and x.excluido_em is null;
    if v_qtd > 0 then
      raise exception 'A casa de oração "%" ainda tem % membro(s). Mude os membros para outra casa de oração (ou exclua-os) antes.', v_nome, v_qtd;
    end if;
    select count(*) into v_qtd from public.eventos x where x.casa_oracao_id = p_id and x.excluido_em is null;
    if v_qtd > 0 then
      raise exception 'A casa de oração "%" ainda tem % evento(s). Mude a casa de oração dos eventos (ou exclua-os) antes.', v_nome, v_qtd;
    end if;
    select count(*) into v_qtd from public.portarias x where x.casa_oracao_id = p_id;
    if v_qtd > 0 then
      raise exception 'A casa de oração "%" tem % portaria(s) no histórico e não pode ser excluída. Use Inativar.', v_nome, v_qtd;
    end if;

    -- Membros e eventos já excluídos (exclusão lógica) deixam de apontar para a casa de oração
    perform set_config('sysmusica.org_excluindo', 'sim', true);
    update public.membros set casa_oracao_id = null where casa_oracao_id = p_id;
    update public.eventos set casa_oracao_id = null where casa_oracao_id = p_id;
    perform set_config('sysmusica.org_excluindo', '', true);

    delete from public.casas_oracao where id = p_id;
  end if;

  perform public.registrar_auditoria('organizacao_excluida', 'organizacao', p_id,
            jsonb_build_object('nivel', p_nivel, 'nome', v_nome));
end;
$function$;

-------- listar_organizacao_para_selecao  (0031_casa_de_oracao.sql)
CREATE OR REPLACE FUNCTION public.listar_organizacao_para_selecao()
 RETURNS TABLE(nivel text, id uuid, parent_id uuid, nome text)
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_total boolean;
begin
  if not public.tem_acesso() then
    raise exception 'Sem acesso.' using errcode = '42501';
  end if;
  v_total := public.tem_acesso_total_organizacao();

  return query
    with ok as (
      select c.id as casa_oracao_id, c.nome as casa_oracao_nome, c.setor_id as setor_id
      from public.casas_oracao c
      where c.ativo and (v_total or public.casa_oracao_na_minha_abrangencia(c.id))
    ),
    sset as (
      select s.id as setor_id, s.regiao_id as regiao_id, s.nome as setor_nome
      from public.setores s
      where s.ativo and (v_total or s.id in (select ok.setor_id from ok))
    ),
    rset as (
      select r.id as regiao_id, r.estado_id as estado_id, r.nome as regiao_nome
      from public.regioes r
      where r.ativo and (v_total or r.id in (select sset.regiao_id from sset))
    ),
    eset as (
      select e.id as estado_id, e.nome as estado_nome
      from public.estados e
      where e.ativo and (v_total or e.id in (select rset.estado_id from rset))
    )
    select 'ESTADO'::text, eset.estado_id, null::uuid, eset.estado_nome from eset
    union all
    select 'REGIAO'::text, rset.regiao_id, rset.estado_id, rset.regiao_nome from rset
    union all
    select 'SETOR'::text, sset.setor_id, sset.regiao_id, sset.setor_nome from sset
    union all
    select 'CASA_ORACAO'::text, ok.casa_oracao_id, ok.setor_id, ok.casa_oracao_nome from ok;
end;
$function$;

-------- membros_sincroniza_casa_oracao  (0031_casa_de_oracao.sql)
CREATE OR REPLACE FUNCTION public.membros_sincroniza_casa_oracao()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_id  uuid;
  v_qtd int;
begin
  if current_setting('sysmusica.org_excluindo', true) = 'sim' then
    return new;
  end if;

  if new.casa_oracao_id is not null and (tg_op = 'INSERT' or new.casa_oracao_id is distinct from old.casa_oracao_id) then
    select c.nome into new.congregacao from public.casas_oracao c where c.id = new.casa_oracao_id;
    if not found then
      raise exception 'Casa de Oração não encontrada.';
    end if;
  elsif tg_op = 'INSERT' or new.congregacao is distinct from old.congregacao then
    select (array_agg(c.id))[1], count(*) into v_id, v_qtd
    from public.casas_oracao c
    where public.normalizar_texto(c.nome) = public.normalizar_texto(btrim(coalesce(new.congregacao, '')));
    new.casa_oracao_id := case when v_qtd = 1 then v_id else null end;
  end if;

  -- Quem tem acesso por região precisa ligar o membro a uma casa de oração (senão ele sumiria da própria lista)
  if new.casa_oracao_id is null and (select auth.uid()) is not null and not public.tem_acesso_total_organizacao() then
    raise exception 'Não encontrei a casa de oração "%". Use o nome exato de uma casa de oração cadastrada em Administração > Organização.',
      coalesce(new.congregacao, '');
  end if;
  return new;
end;
$function$;

  select r.nome || ' › ' || s.nome
  from public.casas_oracao c

-------- rotulo_vinculo  (0031_casa_de_oracao.sql)
CREATE OR REPLACE FUNCTION public.rotulo_vinculo(p_nivel text, p_alvo uuid)
 RETURNS text
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select case p_nivel
    when 'ESTADO' then (select e.nome from public.estados e where e.id = p_alvo)
    when 'REGIAO' then (select e.nome || ' › ' || r.nome
                          from public.regioes r join public.estados e on e.id = r.estado_id where r.id = p_alvo)
    when 'SETOR'  then (select r.nome || ' › ' || s.nome
                          from public.setores s join public.regioes r on r.id = s.regiao_id where s.id = p_alvo)
    when 'CASA_ORACAO'  then (select coalesce(r.nome || ' › ' || s.nome || ' › ', '') || c.nome
                          from public.casas_oracao c
                          left join public.setores s on s.id = c.setor_id
                          left join public.regioes r on r.id = s.regiao_id
                         where c.id = p_alvo)
  end
$function$;

-------- salvar_organizacao  (0031_casa_de_oracao.sql)
CREATE OR REPLACE FUNCTION public.salvar_organizacao(p_nivel text, p_id uuid, p_parent uuid, p_nome text, p_descricao text, p_ativo boolean, p_cidade text, p_bairro text)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_novo   boolean := p_id is null;
  v_nome   text    := btrim(regexp_replace(coalesce(p_nome, ''), '\s+', ' ', 'g'));
  v_desc   text    := nullif(btrim(coalesce(p_descricao, '')), '');
  v_cidade text    := nullif(btrim(coalesce(p_cidade, '')), '');
  v_bairro text    := nullif(btrim(coalesce(p_bairro, '')), '');
  v_id     uuid;
  v_pai    uuid;
begin
  if p_nivel is null or p_nivel not in ('ESTADO', 'REGIAO', 'SETOR', 'CASA_ORACAO') then
    raise exception 'Nível inválido.';
  end if;
  if v_novo then
    perform public.exigir_permissao('organizacao.criar');
  else
    perform public.exigir_permissao('organizacao.editar');
  end if;

  if char_length(v_nome) < 2 or char_length(v_nome) > 80 then
    raise exception 'O nome deve ter entre 2 e 80 caracteres.';
  end if;
  if v_desc is not null and char_length(v_desc) > 200 then
    raise exception 'A descrição pode ter no máximo 200 caracteres.';
  end if;
  if (v_cidade is not null and char_length(v_cidade) > 80) or (v_bairro is not null and char_length(v_bairro) > 80) then
    raise exception 'Cidade e bairro podem ter no máximo 80 caracteres.';
  end if;

  if p_nivel = 'ESTADO' then
    if exists (select 1 from public.estados x
                where x.id is distinct from p_id
                  and public.normalizar_texto(x.nome) = public.normalizar_texto(v_nome)) then
      raise exception 'Já existe um estado com este nome.';
    end if;
    if v_novo then
      insert into public.estados (nome) values (v_nome) returning id into v_id;
    else
      update public.estados set nome = v_nome, ativo = coalesce(p_ativo, ativo) where id = p_id returning id into v_id;
    end if;

  elsif p_nivel = 'REGIAO' then
    v_pai := case when v_novo then p_parent else (select x.estado_id from public.regioes x where x.id = p_id) end;
    if v_pai is null or not exists (select 1 from public.estados e where e.id = v_pai) then
      raise exception 'Escolha o estado da região.';
    end if;
    if exists (select 1 from public.regioes x
                where x.estado_id = v_pai and x.id is distinct from p_id
                  and public.normalizar_texto(x.nome) = public.normalizar_texto(v_nome)) then
      raise exception 'Já existe uma região com este nome neste estado.';
    end if;
    if v_novo then
      insert into public.regioes (estado_id, nome, descricao) values (v_pai, v_nome, v_desc) returning id into v_id;
    else
      update public.regioes set nome = v_nome, descricao = v_desc, ativo = coalesce(p_ativo, ativo)
       where id = p_id returning id into v_id;
    end if;

  elsif p_nivel = 'SETOR' then
    v_pai := case when v_novo then p_parent else (select x.regiao_id from public.setores x where x.id = p_id) end;
    if v_pai is null or not exists (select 1 from public.regioes r where r.id = v_pai) then
      raise exception 'Escolha a região do setor.';
    end if;
    if exists (select 1 from public.setores x
                where x.regiao_id = v_pai and x.id is distinct from p_id
                  and public.normalizar_texto(x.nome) = public.normalizar_texto(v_nome)) then
      raise exception 'Já existe um setor com este nome nesta região.';
    end if;
    if v_novo then
      insert into public.setores (regiao_id, nome, descricao) values (v_pai, v_nome, v_desc) returning id into v_id;
    else
      update public.setores set nome = v_nome, descricao = v_desc, ativo = coalesce(p_ativo, ativo)
       where id = p_id returning id into v_id;
    end if;

  else  -- CASA_ORACAO
    v_pai := case when v_novo then p_parent else (select x.setor_id from public.casas_oracao x where x.id = p_id) end;
    if v_novo and (v_pai is null or not exists (select 1 from public.setores s where s.id = v_pai)) then
      raise exception 'Escolha o setor da casa de oração.';
    end if;
    if exists (select 1 from public.casas_oracao x
                where x.setor_id is not distinct from v_pai and x.id is distinct from p_id
                  and public.normalizar_texto(x.nome) = public.normalizar_texto(v_nome)) then
      raise exception 'Já existe uma casa de oração com este nome neste setor.';
    end if;
    if v_novo then
      insert into public.casas_oracao (setor_id, nome, cidade, bairro) values (v_pai, v_nome, v_cidade, v_bairro)
      returning id into v_id;
    else
      update public.casas_oracao set nome = v_nome, cidade = v_cidade, bairro = v_bairro, ativo = coalesce(p_ativo, ativo)
       where id = p_id returning id into v_id;
    end if;
  end if;

  if v_id is null then
    raise exception 'Cadastro não encontrado.';
  end if;

  perform public.registrar_auditoria(
    case when v_novo then 'organizacao_criada' else 'organizacao_alterada' end,
    'organizacao', v_id,
    jsonb_build_object('nivel', p_nivel, 'nome', v_nome, 'ativo', p_ativo));
  return v_id;
end;
$function$;

-------- transferir_casa_oracao  (0031_casa_de_oracao.sql)
CREATE OR REPLACE FUNCTION public.transferir_casa_oracao(p_casa_oracao uuid, p_setor uuid, p_justificativa text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_c    public.casas_oracao;
  v_just text := nullif(btrim(coalesce(p_justificativa, '')), '');
begin
  perform public.exigir_permissao('organizacao.transferir');

  select * into v_c from public.casas_oracao c where c.id = p_casa_oracao for update;
  if not found then
    raise exception 'Casa de Oração não encontrada.';
  end if;
  if not exists (select 1 from public.setores s where s.id = p_setor and s.ativo) then
    raise exception 'Escolha um setor ativo de destino.';
  end if;
  if v_c.setor_id is not distinct from p_setor then
    raise exception 'A casa de oração já pertence a este setor.';
  end if;
  if v_just is not null and char_length(v_just) > 300 then
    raise exception 'A justificativa pode ter no máximo 300 caracteres.';
  end if;
  if exists (select 1 from public.casas_oracao x
              where x.setor_id = p_setor and x.id <> p_casa_oracao
                and public.normalizar_texto(x.nome) = public.normalizar_texto(v_c.nome)) then
    raise exception 'O setor de destino já tem uma casa de oração com este nome.';
  end if;

  update public.casas_oracao set setor_id = p_setor where id = p_casa_oracao;
  -- As portarias seguem a casa de oração do evento: só o rótulo precisa ser atualizado
  update public.portarias p set cidade = public.rotulo_setor(p.casa_oracao_id)
   where p.casa_oracao_id = p_casa_oracao and p.excluido_em is null;

  insert into public.casas_oracao_historico (casa_oracao_id, setor_anterior, setor_novo, usuario_id, justificativa)
  values (p_casa_oracao, v_c.setor_id, p_setor, (select auth.uid()), v_just);

  perform public.registrar_auditoria('casa_oracao_transferida', 'casa_oracao', p_casa_oracao,
            jsonb_build_object('setor_anterior', v_c.setor_id, 'setor_novo', p_setor, 'justificativa', v_just));
end;
$function$;

-------- transferir_comum  (0012_hierarquia_organizacional.sql)
create or replace function public.transferir_comum(p_comum uuid, p_setor uuid, p_justificativa text)
returns void language plpgsql security definer set search_path = public as $$
declare
  v_c    public.comuns;
  v_just text := nullif(btrim(coalesce(p_justificativa, '')), '');
begin
  perform public.exigir_permissao('organizacao.transferir');

  select * into v_c from public.comuns c where c.id = p_comum for update;
  if not found then
    raise exception 'Comum não encontrada.';
  end if;
  if not exists (select 1 from public.setores s where s.id = p_setor and s.ativo) then
    raise exception 'Escolha um setor ativo de destino.';
  end if;
  if v_c.setor_id is not distinct from p_setor then
    raise exception 'A comum já pertence a este setor.';
  end if;
  if v_just is not null and char_length(v_just) > 300 then
    raise exception 'A justificativa pode ter no máximo 300 caracteres.';
  end if;
  if exists (select 1 from public.comuns x
              where x.setor_id = p_setor and x.id <> p_comum
                and public.normalizar_texto(x.nome) = public.normalizar_texto(v_c.nome)) then
    raise exception 'O setor de destino já tem uma comum com este nome.';
  end if;

  update public.comuns set setor_id = p_setor where id = p_comum;
  -- As portarias seguem a comum do evento: só o rótulo precisa ser atualizado
  update public.portarias p set cidade = public.rotulo_setor(p.comum_id)
   where p.comum_id = p_comum and p.excluido_em is null;

  insert into public.comuns_historico (comum_id, setor_anterior, setor_novo, usuario_id, justificativa)
  values (p_comum, v_c.setor_id, p_setor, (select auth.uid()), v_just);

  perform public.registrar_auditoria('comum_transferida', 'comum', p_comum,
            jsonb_build_object('setor_anterior', v_c.setor_id, 'setor_novo', p_setor, 'justificativa', v_just));
end;
$$;
